## Introduction

Session guarantees are a set of consistency properties that apply within a client session, providing better user experience without requiring system-wide strong consistency. The most important is Read-Your-Writes consistency, which ensures that users immediately see the effects of their own actions. These guarantees bridge the gap between eventual consistency and strong consistency for user-facing applications.

## The Four Session Guarantees

### 1. Read-Your-Writes (RYW)

**Guarantee:** A client always reads its own writes.

```csharp
public class ReadYourWritesStore
{
    private readonly IDistributedStore _store;
    private readonly Dictionary<string, long> _sessionWriteVersions = new();
    
    public async Task WriteAsync(string sessionId, string key, string value)
    {
        // Write to store and get version
        var version = await _store.WriteAsync(key, value);
        
        // Track this write in the session
        _sessionWriteVersions[$"{sessionId}:{key}"] = version;
        
        _logger.LogInformation(
            "Session {SessionId} wrote {Key} at version {Version}",
            sessionId, key, version);
    }
    
    public async Task<string> ReadAsync(string sessionId, string key)
    {
        // Check if this session has written this key
        var sessionKey = $"{sessionId}:{key}";
        var minVersion = _sessionWriteVersions.GetValueOrDefault(sessionKey, 0);
        
        // Read from store, but ensure we get at least our own writes
        var (value, version) = await _store.ReadWithVersionAsync(key);
        
        if (version < minVersion)
        {
            // Store is behind our writes - wait or read from primary
            (value, version) = await _store.ReadFromPrimaryAsync(key);
            
            if (version < minVersion)
            {
                throw new ConsistencyException(
                    $"Cannot satisfy Read-Your-Writes guarantee for session {sessionId}");
            }
        }
        
        return value;
    }
}
```

### 2. Monotonic Reads (MR)

**Guarantee:** Once a client reads a value, subsequent reads never return an older value.

```csharp
public class MonotonicReadsStore
{
    private readonly IDistributedStore _store;
    private readonly Dictionary<string, Dictionary<string, long>> _sessionReadVersions = new();
    
    public async Task<string> ReadAsync(string sessionId, string key)
    {
        // Initialize session tracking if needed
        if (!_sessionReadVersions.ContainsKey(sessionId))
        {
            _sessionReadVersions[sessionId] = new Dictionary<string, long>();
        }
        
        var sessionVersions = _sessionReadVersions[sessionId];
        var minVersion = sessionVersions.GetValueOrDefault(key, 0);
        
        // Read from store
        var (value, version) = await _store.ReadWithVersionAsync(key);
        
        if (version < minVersion)
        {
            // This would violate monotonic reads!
            // Read from a different replica or wait
            (value, version) = await _store.ReadAtLeastVersionAsync(key, minVersion);
        }
        
        // Update session's read version
        sessionVersions[key] = version;
        
        _logger.LogInformation(
            "Session {SessionId} read {Key} at version {Version}",
            sessionId, key, version);
        
        return value;
    }
}
```

### 3. Monotonic Writes (MW)

**Guarantee:** A client's writes are applied in the order they were issued.

```csharp
public class MonotonicWritesStore
{
    private readonly IDistributedStore _store;
    private readonly Dictionary<string, long> _sessionWriteSequence = new();
    
    public async Task WriteAsync(string sessionId, string key, string value)
    {
        // Get next sequence number for this session
        var sequence = _sessionWriteSequence.GetValueOrDefault(sessionId, 0) + 1;
        _sessionWriteSequence[sessionId] = sequence;
        
        // Write with sequence number
        await _store.WriteWithSequenceAsync(key, value, sessionId, sequence);
        
        _logger.LogInformation(
            "Session {SessionId} write #{Sequence}: {Key}={Value}",
            sessionId, sequence, key, value);
    }
    
    // On replica receiving write:
    public async Task ReplicateWriteAsync(
        string key,
        string value,
        string sessionId,
        long sequence)
    {
        // Ensure we've applied all previous writes from this session
        var lastApplied = await GetLastAppliedSequenceAsync(sessionId);
        
        if (sequence != lastApplied + 1)
        {
            // Out of order! Buffer this write
            await BufferWriteAsync(sessionId, sequence, key, value);
            return;
        }
        
        // Apply in order
        await ApplyWriteAsync(key, value);
        await SetLastAppliedSequenceAsync(sessionId, sequence);
        
        // Check if we can now apply buffered writes
        await ProcessBufferedWritesAsync(sessionId);
    }
}
```

### 4. Writes-Follow-Reads (WFR)

**Guarantee:** A write is guaranteed to be applied after any reads that causally precede it.

```csharp
public class WritesFollowReadsStore
{
    private readonly IDistributedStore _store;
    private readonly Dictionary<string, long> _sessionReadVersions = new();
    
    public async Task<string> ReadAsync(string sessionId, string key)
    {
        var (value, version) = await _store.ReadWithVersionAsync(key);
        
        // Track the version this session has read
        if (!_sessionReadVersions.TryGetValue(sessionId, out var currentVersion) ||
            version > currentVersion)
        {
            _sessionReadVersions[sessionId] = version;
        }
        
        return value;
    }
    
    public async Task WriteAsync(string sessionId, string key, string value)
    {
        // Get the version this session has read up to
        var readVersion = _sessionReadVersions.GetValueOrDefault(sessionId, 0);
        
        // Ensure write happens after the reads
        // by including read version as dependency
        await _store.WriteWithDependencyAsync(key, value, readVersion);
        
        _logger.LogInformation(
            "Session {SessionId} write {Key} with dependency version {Version}",
            sessionId, key, readVersion);
    }
}
```

## Implementing Session Tokens

### Azure Cosmos DB Session Token Pattern

```csharp
public class CosmosDbSessionTokenExample
{
    private readonly Container _container;
    private readonly Dictionary<string, string> _sessionTokens = new();
    
    public async Task<T> ReadAsync<T>(string sessionId, string id, string partitionKey)
    {
        var requestOptions = new ItemRequestOptions();
        
        // Use session token if available
        if (_sessionTokens.TryGetValue(sessionId, out var sessionToken))
        {
            requestOptions.SessionToken = sessionToken;
        }
        
        var response = await _container.ReadItemAsync<T>(
            id,
            new PartitionKey(partitionKey),
            requestOptions);
        
        // Update session token
        _sessionTokens[sessionId] = response.Headers.Session;
        
        return response.Resource;
    }
    
    public async Task WriteAsync<T>(
        string sessionId,
        T item,
        string id,
        string partitionKey)
    {
        var requestOptions = new ItemRequestOptions();
        
        // Include current session token
        if (_sessionTokens.TryGetValue(sessionId, out var sessionToken))
        {
            requestOptions.SessionToken = sessionToken;
        }
        
        var response = await _container.UpsertItemAsync(
            item,
            new PartitionKey(partitionKey),
            requestOptions);
        
        // Update session token for future operations
        _sessionTokens[sessionId] = response.Headers.Session;
    }
}
```

### Custom Session Token Implementation

```csharp
public class CustomSessionToken
{
    // Session token = version vector of what this session has seen
    public Dictionary<string, long> Versions { get; set; } = new();
    public string SessionId { get; set; }
    public DateTime LastActivity { get; set; }
    
    public string Serialize()
    {
        var json = JsonSerializer.Serialize(new
        {
            SessionId,
            Versions,
            LastActivity
        });
        
        return Convert.ToBase64String(Encoding.UTF8.GetBytes(json));
    }
    
    public static CustomSessionToken Deserialize(string token)
    {
        var json = Encoding.UTF8.GetString(Convert.FromBase64String(token));
        var data = JsonSerializer.Deserialize<CustomSessionToken>(json);
        return data;
    }
}

public class SessionTokenStore
{
    private readonly IDistributedStore _store;
    
    public async Task<(T Value, string NewToken)> ReadAsync<T>(
        string key,
        string? sessionToken = null)
    {
        // Parse session token
        var token = sessionToken != null
            ? CustomSessionToken.Deserialize(sessionToken)
            : new CustomSessionToken { SessionId = Guid.NewGuid().ToString() };
        
        // Read with minimum version requirements
        var (value, version) = await _store.ReadAtLeastVersionsAsync<T>(
            key,
            token.Versions);
        
        // Update token with new version info
        token.Versions[key] = version;
        token.LastActivity = DateTime.UtcNow;
        
        return (value, token.Serialize());
    }
    
    public async Task<string> WriteAsync<T>(
        string key,
        T value,
        string? sessionToken = null)
    {
        var token = sessionToken != null
            ? CustomSessionToken.Deserialize(sessionToken)
            : new CustomSessionToken { SessionId = Guid.NewGuid().ToString() };
        
        // Write with session dependencies
        var version = await _store.WriteWithDependenciesAsync(
            key,
            value,
            token.Versions);
        
        // Update token
        token.Versions[key] = version;
        token.LastActivity = DateTime.UtcNow;
        
        return token.Serialize();
    }
}
```

## Session Affinity (Sticky Sessions)

### Load Balancer Configuration

```csharp
public class SessionAffinityMiddleware
{
    private readonly RequestDelegate _next;
    private readonly IDistributedCache _cache;
    
    public async Task InvokeAsync(HttpContext context)
    {
        // Get or create session ID
        var sessionId = context.Request.Cookies["SessionId"];
        if (string.IsNullOrEmpty(sessionId))
        {
            sessionId = Guid.NewGuid().ToString();
            context.Response.Cookies.Append("SessionId", sessionId, new CookieOptions
            {
                HttpOnly = true,
                Secure = true,
                SameSite = SameSiteMode.Strict,
                MaxAge = TimeSpan.FromHours(24)
            });
        }
        
        // Store session affinity - which server this session should use
        var serverId = Environment.MachineName;
        await _cache.SetStringAsync(
            $"session:{sessionId}:server",
            serverId,
            new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow = TimeSpan.FromHours(24)
            });
        
        // Add session context to request
        context.Items["SessionId"] = sessionId;
        context.Items["ServerId"] = serverId;
        
        await _next(context);
    }
}
```

### Routing Based on Session

```csharp
public class SessionAwareRepository
{
    private readonly IDatabase _primaryDb;
    private readonly IReadOnlyList<IDatabase> _replicasDbs;
    private readonly IDistributedCache _sessionCache;
    
    public async Task<User> GetUserAsync(string userId, string sessionId)
    {
        // Check which replica this session last read from
        var lastReplicaId = await _sessionCache.GetStringAsync(
            $"session:{sessionId}:replica");
        
        IDatabase database;
        
        if (lastReplicaId != null)
        {
            // Route to same replica for monotonic reads
            database = _replicasDbs.FirstOrDefault(r => r.Id == lastReplicaId)
                ?? _replicasDbs[0];
        }
        else
        {
            // First read - pick any replica
            database = _replicasDbs[Random.Shared.Next(_replicasDbs.Count)];
            
            // Remember for next time
            await _sessionCache.SetStringAsync(
                $"session:{sessionId}:replica",
                database.Id);
        }
        
        return await database.GetUserAsync(userId);
    }
    
    public async Task UpdateUserAsync(User user, string sessionId)
    {
        // Writes always go to primary
        await _primaryDb.UpdateUserAsync(user);
        
        // Invalidate session's replica affinity
        // Next read should go to primary or wait for replication
        await _sessionCache.RemoveAsync($"session:{sessionId}:replica");
    }
}
```

## Practical User Experience Patterns

### 1. Profile Updates with Immediate Feedback

```csharp
public class UserProfileService
{
    private readonly ISessionStore _store;
    
    public async Task<ProfileUpdateResult> UpdateProfileAsync(
        string userId,
        ProfileUpdate update,
        string sessionToken)
    {
        // Write with session guarantee
        var newToken = await _store.WriteAsync(
            $"user:{userId}:profile",
            update,
            sessionToken);
        
        // Immediately read back - guaranteed to see the update
        var (updatedProfile, _) = await _store.ReadAsync<Profile>(
            $"user:{userId}:profile",
            newToken);
        
        return new ProfileUpdateResult
        {
            Profile = updatedProfile,
            SessionToken = newToken,
            Message = "Profile updated successfully"
        };

        var (profile, newToken) = await _store.ReadAsync<Profile>(
            $"user:{userId}:profile",
            sessionToken);
        
        return new ProfileView
        {
            Profile = profile,
            SessionToken = newToken,
            IsFresh = sessionToken != null, // Indicates if user has session context
            LastModified = profile.LastModified
        };
    }
}
```

### 2. Shopping Cart Consistency

```csharp
public class ShoppingCartService
{
    private readonly ISessionStore _store;
    private readonly ILogger _logger;
    
    public async Task<CartOperationResult> AddToCartAsync(
        string userId,
        CartItem item,
        string sessionToken)
    {
        var cartKey = $"cart:{userId}";
        
        try
        {
            // Read current cart (monotonic reads ensure we see our previous adds)
            var (cart, readToken) = await _store.ReadAsync<Cart>(
                cartKey,
                sessionToken);
            
            // Add item
            cart.Items.Add(item);
            cart.LastModified = DateTime.UtcNow;
            cart.Version++;
            
            // Write back (writes-follow-reads ensures causal ordering)
            var writeToken = await _store.WriteAsync(
                cartKey,
                cart,
                readToken);
            
            _logger.LogInformation(
                "Added item {ProductId} to cart for user {UserId}",
                item.ProductId, userId);
            
            // User immediately sees their addition (Read-Your-Writes)
            return new CartOperationResult
            {
                Success = true,
                Cart = cart,
                SessionToken = writeToken
            };
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to add item to cart");
            return new CartOperationResult
            {
                Success = false,
                ErrorMessage = "Failed to add item to cart. Please try again."
            };
        }
    }
    
    public async Task<Cart> GetCartAsync(string userId, string sessionToken)
    {
        var (cart, newToken) = await _store.ReadAsync<Cart>(
            $"cart:{userId}",
            sessionToken);
        
        // Monotonic reads guarantee: 
        // User never sees items disappear unexpectedly
        return cart;
    }
}
```

### 3. Comment Threading with Causality

```csharp
public class CommentService
{
    private readonly ISessionStore _store;
    
    public async Task<Comment> PostCommentAsync(
        string postId,
        string content,
        string? parentCommentId,
        string userId,
        string sessionToken)
    {
        var comment = new Comment
        {
            Id = Guid.NewGuid().ToString(),
            PostId = postId,
            ParentCommentId = parentCommentId,
            Content = content,
            AuthorId = userId,
            CreatedAt = DateTime.UtcNow
        };
        
        string writeToken = sessionToken;
        
        // If replying to a comment, ensure we've seen it
        if (parentCommentId != null)
        {
            var (parent, parentToken) = await _store.ReadAsync<Comment>(
                $"comment:{parentCommentId}",
                sessionToken);
            
            // Writes-Follow-Reads: Our reply depends on reading parent
            writeToken = parentToken;
        }
        
        // Write comment with causal dependencies
        var newToken = await _store.WriteAsync(
            $"comment:{comment.Id}",
            comment,
            writeToken);
        
        // Add to post's comment list
        await AddCommentToPostAsync(postId, comment.Id, newToken);
        
        return comment;
    }
    
    public async Task<List<Comment>> GetCommentsAsync(
        string postId,
        string sessionToken)
    {
        var (commentIds, token) = await _store.ReadAsync<List<string>>(
            $"post:{postId}:comments",
            sessionToken);
        
        var comments = new List<Comment>();
        var currentToken = token;
        
        foreach (var commentId in commentIds)
        {
            var (comment, newToken) = await _store.ReadAsync<Comment>(
                $"comment:{commentId}",
                currentToken);
            
            comments.Add(comment);
            currentToken = newToken; // Accumulate session knowledge
        }
        
        // Session guarantees ensure:
        // 1. User sees their own comments immediately (Read-Your-Writes)
        // 2. Replies appear after parent comments (causal ordering)
        // 3. Comments don't disappear (Monotonic Reads)
        
        return comments;
    }
}
```

### 4. Collaborative Document Editing

```csharp
public class DocumentEditService
{
    private readonly ISessionStore _store;
    
    public async Task<EditResult> ApplyEditAsync(
        string documentId,
        string userId,
        TextEdit edit,
        string sessionToken)
    {
        try
        {
            // Read document (Monotonic Reads - see all previous edits)
            var (document, readToken) = await _store.ReadAsync<Document>(

            if (edit.BaseVersion != document.Version)

            var writeToken = await _store.WriteAsync(
                documentId,
                document,
                readToken);
            
            return new EditResult
            {
                Success = true,
                Document = document,
                SessionToken = writeToken
            };

        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to apply edit");
            return new EditResult
            {
                Success = false,
                ErrorMessage = "Failed to apply edit. Please try again."
            };
        }
    }
}
```

```csharp
public class OrderWizardService
{
    private readonly ISessionStore _store;

    public async Task<string> SaveItemSelectionAsync(

            throw new InvalidOperationException(

        // Update order
        order.ShippingAddress = address;
        order.Status = OrderStatus.ShippingAddressAdded;
        order.LastModified = DateTime.UtcNow;
        
        // Write depends on previous read (Writes-Follow-Reads)
        var token = await _store.WriteAsync(
            $"order:{orderId}",
            order,
            readToken);
        
        return token;
    }
    
    // Step 3: Add payment (depends on step 2)
    public async Task<string> SavePaymentAsync(
        string orderId,
        PaymentMethod payment,
        string sessionToken)
    {
        var (order, readToken) = await _store.ReadAsync<Order>(
            $"order:{orderId}",
            sessionToken);
        
        // Session consistency ensures we see all previous steps
        if (order.Status != OrderStatus.ShippingAddressAdded)
        {
            throw new InvalidOperationException(
                "Must add shipping address before payment");
        }
        
        order.PaymentMethod = payment;
        order.Status = OrderStatus.ReadyToSubmit;
        order.LastModified = DateTime.UtcNow;
        
        var token = await _store.WriteAsync(
            $"order:{orderId}",
            order,
            readToken);
        
        return token;
    }
    
    // Final step: Submit order
    public async Task<OrderResult> SubmitOrderAsync(
        string orderId,
        string sessionToken)
    {
        var (order, _) = await _store.ReadAsync<Order>(
            $"order:{orderId}",
            sessionToken);
        
        // Session guarantees ensure user sees all their wizard steps
        if (order.Status != OrderStatus.ReadyToSubmit)
        {
            return new OrderResult
            {
                Success = false,
                ErrorMessage = "Please complete all wizard steps"
            };
        }
        
        // Submit order...
        order.Status = OrderStatus.Submitted;
        order.SubmittedAt = DateTime.UtcNow;

        return new OrderResult
        {
            Success = true,
            OrderId = orderId,
```

## Session Management Best Practices

### 1. Session Expiration

```csharp
public class SessionManager
{
    private readonly IDistributedCache _cache;
    private readonly TimeSpan _sessionTimeout = TimeSpan.FromHours(24);
    
    public async Task<bool> IsSessionValidAsync(string sessionToken)
    {
        var session = await _cache.GetStringAsync($"session:{sessionToken}");
        
        if (session == null)
        {
            return false;
        }
        
        var sessionData = JsonSerializer.Deserialize<SessionData>(session);
        var age = DateTime.UtcNow - sessionData.LastActivity;
        
        if (age > _sessionTimeout)
        {
            await _cache.RemoveAsync($"session:{sessionToken}");
            return false;
        }
        
        // Extend session
        sessionData.LastActivity = DateTime.UtcNow;
        await _cache.SetStringAsync(
            $"session:{sessionToken}",
            JsonSerializer.Serialize(sessionData),
            new DistributedCacheEntryOptions
            {
                AbsoluteExpirationRelativeToNow = _sessionTimeout
            });
        
        return true;
    }
}
```

### 2. Session Token Size Management

```csharp
public class CompactSessionToken
{
    // Use bloomfilter or compact representation
    // to avoid unbounded growth
    
    public byte[] ToCompactFormat()
    {
        // Instead of full version vector,
        // use compact representation
        using var ms = new MemoryStream();
        using var writer = new BinaryWriter(ms);
        
        // Write only recent versions (bounded size)
        var recentVersions = _versions
            .OrderByDescending(kvp => kvp.Value)
            .Take(100) // Limit to 100 most recent
            .ToList();
        
        writer.Write(recentVersions.Count);
        foreach (var (key, version) in recentVersions)
        {
            writer.Write(key);
            writer.Write(version);
        }
        
        return ms.ToArray();
    }
}
```

## Key Takeaways

- Session guarantees provide better UX without system-wide strong consistency
- Read-Your-Writes ensures users see their own actions immediately
- Monotonic Reads prevents "time travel" where data appears to revert
- Monotonic Writes ensures causally ordered updates from the same client
- Writes-Follow-Reads maintains causality between reads and subsequent writes
- Session tokens track what a session has seen (version vectors)
- Can be implemented with sticky sessions, session tokens, or both
- Azure Cosmos DB provides session consistency as a built-in option
- Lower cost than strong consistency, better UX than eventual consistency
