# Lesson 2: CRUD Operations - Update and Delete

## Learning Objectives

By the end of this lesson, you will be able to:
- Update documents using updateOne() and updateMany()
- Use update operators like $set, $inc, $push, and $pull
- Delete documents using deleteOne() and deleteMany()
- Understand replace operations and upserts

---

## Introduction

Updating and deleting documents are crucial operations in any database system. MongoDB provides powerful update operators that let you modify specific fields without rewriting entire documents. This lesson covers the update and delete operations, including advanced patterns like upserts and atomic updates.

## Key Concepts Summary

### Update Operations

**updateOne() - Update First Matching Document:**

```javascript
// Basic update with $set
db.users.updateOne(
  { name: "Alice Johnson" },  // Filter
  { $set: { age: 31 } }        // Update
)

// Result
{
  acknowledged: true,
  matchedCount: 1,
  modifiedCount: 1
}
```

**updateMany() - Update All Matching Documents:**

```javascript
// Update all inactive users
db.users.updateMany(
  { status: "inactive" },
  { $set: { status: "archived", archivedDate: new Date() } }
)

// Result
{
  acknowledged: true,
  matchedCount: 15,
  modifiedCount: 15
}
```

### Update Operators

**$set - Set Field Values:**

```javascript
// Set single field
db.users.updateOne(
  { _id: ObjectId("...") },
  { $set: { email: "newemail@example.com" } }
)

// Set multiple fields
db.users.updateOne(
  { _id: ObjectId("...") },
  { 
    $set: { 
      email: "newemail@example.com",
      lastModified: new Date(),
      "profile.bio": "Updated bio"
    } 
  }
)

// Set nested fields
db.users.updateOne(
  { _id: ObjectId("...") },
  { $set: { "address.city": "New York" } }
)
```

**$unset - Remove Fields:**

```javascript
// Remove field from document
db.users.updateOne(
  { _id: ObjectId("...") },
  { $unset: { temporaryField: "" } }
)

// Remove multiple fields
db.users.updateOne(
  { _id: ObjectId("...") },
  { $unset: { field1: "", field2: "", "nested.field": "" } }
)
```

**$inc - Increment Numeric Values:**

```javascript
// Increment by 1
db.products.updateOne(
  { sku: "WIDGET-001" },
  { $inc: { views: 1 } }
)

// Increment/decrement by any amount
db.users.updateOne(
  { _id: ObjectId("...") },
  { 
    $inc: { 
      loginCount: 1,
      points: 10,
      balance: -50  // Decrement
    } 
  }
)

// Creates field if it doesn't exist
db.stats.updateOne(
  { _id: "daily" },
  { $inc: { pageViews: 1 } }  // Creates pageViews: 1 if missing
)
```

**$mul - Multiply Values:**

```javascript
// Multiply price by 1.1 (10% increase)
db.products.updateOne(
  { sku: "WIDGET-001" },
  { $mul: { price: 1.1 } }
)
```

**$rename - Rename Fields:**

```javascript
// Rename field
db.users.updateMany(
  {},
  { $rename: { "userName": "username" } }
)
```

**$currentDate - Set to Current Date:**

```javascript
// Set to current date
db.users.updateOne(
  { _id: ObjectId("...") },
  { 
    $currentDate: { 
      lastModified: true,
      "audit.timestamp": { $type: "timestamp" }
    } 
  }
)
```

### Array Update Operators

**$push - Add Elements to Array:**

```javascript
// Add single element
db.users.updateOne(
  { _id: ObjectId("...") },
  { $push: { tags: "mongodb" } }
)

// Add multiple elements with $each
db.users.updateOne(
  { _id: ObjectId("...") },
  { 
    $push: { 
      tags: { 
        $each: ["mongodb", "database", "nosql"] 
      } 
    } 
  }
)

// Push with sorting and limiting
db.users.updateOne(
  { _id: ObjectId("...") },
  { 
    $push: { 
      scores: { 
        $each: [85, 92, 78],
        $sort: -1,  // Sort descending
        $slice: 5   // Keep only top 5
      } 
    } 
  }
)
```

**$addToSet - Add if Not Exists:**

```javascript
// Add only if value doesn't exist (no duplicates)
db.users.updateOne(
  { _id: ObjectId("...") },
  { $addToSet: { tags: "mongodb" } }
)

// Add multiple unique values
db.users.updateOne(
  { _id: ObjectId("...") },
  { 
    $addToSet: { 
      tags: { $each: ["mongodb", "database"] } 
    } 
  }
)
```

**$pull - Remove Elements from Array:**

```javascript
// Remove specific value
db.users.updateOne(
  { _id: ObjectId("...") },
  { $pull: { tags: "oldtag" } }
)

// Remove matching condition
db.users.updateOne(
  { _id: ObjectId("...") },
  { $pull: { scores: { $lt: 60 } } }  // Remove scores < 60
)
```

**$pop - Remove First or Last Element:**

```javascript
// Remove last element
db.users.updateOne(
  { _id: ObjectId("...") },
  { $pop: { tags: 1 } }
)

// Remove first element
db.users.updateOne(
  { _id: ObjectId("...") },
  { $pop: { tags: -1 } }
)
```

**$pullAll - Remove Multiple Values:**

```javascript
// Remove multiple specific values
db.users.updateOne(
  { _id: ObjectId("...") },
  { $pullAll: { tags: ["tag1", "tag2", "tag3"] } }
)
```

### Replace Operations

**replaceOne() - Replace Entire Document:**

```javascript
// Replace entire document (except _id)
db.users.replaceOne(
  { _id: ObjectId("...") },
  {
    name: "Alice Johnson",
    email: "alice@example.com",
    age: 31,
    status: "active"
  }
)

// Note: All fields not in replacement are removed
// Cannot use update operators with replaceOne()
```

### Upsert Operations

**Upsert - Update or Insert:**

```javascript
// Update if exists, insert if not
db.users.updateOne(
  { email: "alice@example.com" },
  { 
    $set: { 
      name: "Alice Johnson",
      lastLogin: new Date()
    } 
  },
  { upsert: true }  // Insert if no match found
)

// With $setOnInsert (only set on insert, not update)
db.users.updateOne(
  { email: "alice@example.com" },
  { 
    $set: { lastLogin: new Date() },
    $setOnInsert: { 
      createdAt: new Date(),
      status: "active"
    }
  },
  { upsert: true }
)
```

### Delete Operations

**deleteOne() - Delete First Matching Document:**

```javascript
// Delete single document
db.users.deleteOne({ _id: ObjectId("...") })

// Delete by filter
db.users.deleteOne({ status: "inactive" })

// Result
{
  acknowledged: true,
  deletedCount: 1
}
```

**deleteMany() - Delete All Matching Documents:**

```javascript
// Delete all inactive users
db.users.deleteMany({ status: "inactive" })

// Delete all documents (be careful!)
db.temporaryData.deleteMany({})

// Result
{
  acknowledged: true,
  deletedCount: 23
}
```

**findOneAndDelete() - Find and Delete:**

```javascript
// Find, delete, and return deleted document
const deletedUser = db.users.findOneAndDelete(
  { email: "alice@example.com" }
)

// Returns the deleted document
```

### Practical Examples

**Example 1: User Profile Update**

```javascript
// Update user profile
db.users.updateOne(
  { _id: ObjectId("...") },
  { 
    $set: { 
      "profile.bio": "Software engineer passionate about databases",
      "profile.avatar": "https://example.com/avatar.jpg",
      lastModified: new Date()
    },
    $inc: { "profile.views": 1 }
  }
)
```

**Example 2: E-commerce Order**

```javascript
// Add item to order
db.orders.updateOne(
  { _id: ObjectId("...") },
  { 
    $push: { 
      items: {
        productId: "WIDGET-001",
        quantity: 2,
        price: 99.99
      }
    },
    $inc: { totalItems: 2, total: 199.98 }
  }
)

// Remove item from order
db.orders.updateOne(
  { _id: ObjectId("...") },
  { 
    $pull: { items: { productId: "WIDGET-001" } },
    $inc: { totalItems: -2, total: -199.98 }
  }
)
```

**Example 3: Social Media Post**

```javascript
// Like a post
db.posts.updateOne(
  { _id: ObjectId("...") },
  { 
    $addToSet: { likes: "user123" },  // Add user if not already liked
    $inc: { likeCount: 1 }
  }
)

// Unlike a post
db.posts.updateOne(
  { _id: ObjectId("...") },
  { 
    $pull: { likes: "user123" },
    $inc: { likeCount: -1 }
  }
)
```

**Example 4: Archiving Old Records**

```javascript
// Archive old inactive accounts
db.users.updateMany(
  { 
    status: "inactive",
    lastLogin: { $lt: new Date("2024-01-01") }
  },
  { 
    $set: { 
      status: "archived",
      archivedDate: new Date()
    }
  }
)
```

### Best Practices

**1. Use Specific Updates:**
```javascript
// GOOD: Update specific fields
db.users.updateOne(
  { _id: ObjectId("...") },
  { $set: { email: "new@example.com" } }
)

// AVOID: Replacing entire document
db.users.replaceOne(
  { _id: ObjectId("...") },
  entireNewDocument
)
```

**2. Always Include Filters:**
```javascript
// GOOD: Specific filter
db.users.deleteOne({ _id: ObjectId("...") })

// DANGEROUS: Could delete any document
db.users.deleteOne({})
```

**3. Check Return Values:**
```javascript
const result = db.users.updateOne(
  { _id: ObjectId("...") },
  { $set: { status: "active" } }
)

if (result.matchedCount === 0) {
  print("User not found")
} else if (result.modifiedCount === 0) {
  print("User already had that status")
}
```

**4. Use Atomic Operators:**
```javascript
// GOOD: Atomic increment
db.products.updateOne(
  { _id: ObjectId("...") },
  { $inc: { stock: -1 } }
)

// AVOID: Read-modify-write (race condition)
const product = db.products.findOne({ _id: ObjectId("...") })
db.products.updateOne(
  { _id: ObjectId("...") },
  { $set: { stock: product.stock - 1 } }
)
```
