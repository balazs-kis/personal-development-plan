---

## Introduction

Phil Karlton famously said: "There are only two hard things in Computer Science: cache invalidation and naming things." Cache invalidation determines when and how to remove or refresh cached data. Getting this wrong leads to serving stale data, while being too aggressive with invalidation defeats the purpose of caching. This lesson explores various invalidation strategies and their trade-offs.

## Key Concepts Summary

### Cache Invalidation Strategies

**1. Time-Based (TTL - Time To Live)**
```csharp
// Absolute expiration - expires at specific time
var options = new DistributedCacheEntryOptions
{
    AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(10)
};

// Sliding expiration - resets timer on each access
var options = new DistributedCacheEntryOptions
{
    SlidingExpiration = TimeSpan.FromMinutes(5)
};

// Combination - expires after 1 hour OR 10 minutes of inactivity
var options = new DistributedCacheEntryOptions
{
    AbsoluteExpirationRelativeToNow = TimeSpan.FromHours(1),
    SlidingExpiration = TimeSpan.FromMinutes(10)
};
```

**2. Event-Based Invalidation**
```csharp
public class ProductService
{
    private readonly ICacheService _cache;
    private readonly IMediator _mediator;
    
    public async Task UpdateProductAsync(Product product)
    {
        await _dbContext.SaveChangesAsync();
        
        // Invalidate cache when data changes
        await _cache.RemoveAsync($"product:{product.Id}");
        
        // Publish event for other services
        await _mediator.Publish(new ProductUpdatedEvent(product.Id));
    }
}

// Event handler in another service
public class ProductCacheInvalidationHandler 
    : INotificationHandler<ProductUpdatedEvent>
{
    public async Task Handle(ProductUpdatedEvent notification, CancellationToken ct)
    {
        await _cache.RemoveAsync($"product:{notification.ProductId}");
        await _cache.RemoveByPrefixAsync($"category:{notification.CategoryId}");
    }
}
```

**3. Write-Through Invalidation**
```csharp
public async Task UpdateProductAsync(Product product)
{
    // Update database
    _dbContext.Products.Update(product);
    await _dbContext.SaveChangesAsync();
    
    // Update cache immediately with new value
    await _cache.SetAsync(
        $"product:{product.Id}",
        product,
        TimeSpan.FromMinutes(10));
}
```

**4. Lazy/Reactive Invalidation**
```csharp
// Use versioning in cache keys
public async Task<Product> GetProductAsync(int id)
{
    var version = await GetProductVersionAsync(id);
    var cacheKey = $"product:{id}:v{version}";
    
    return await _cache.GetOrSetAsync(
        cacheKey,
        () => FetchProductAsync(id));
}

// When product updates, increment version
public async Task UpdateProductAsync(Product product)
{
    await _dbContext.SaveChangesAsync();
    await IncrementProductVersionAsync(product.Id);
    // Old cached versions become irrelevant
}
```

### Choosing TTL Values

```csharp
// Data characteristics determine TTL
public static class CacheDurations
{
    // Static reference data that rarely changes
    public static TimeSpan StaticData = TimeSpan.FromHours(24);
    
    // User profiles - medium change frequency
    public static TimeSpan UserProfile = TimeSpan.FromMinutes(30);
    
    // Shopping cart - changes frequently but okay to be slightly stale
    public static TimeSpan ShoppingCart = TimeSpan.FromMinutes(5);
    
    // Real-time data - very short cache
    public static TimeSpan RealtimeData = TimeSpan.FromSeconds(30);
    
    // API rate limit counters - precise timing needed
    public static TimeSpan RateLimit = TimeSpan.FromMinutes(1);
}
```

### Cache Stampede Prevention with Soft Expiration

```csharp
public class CacheEntry<T>
{
    public T Value { get; set; }
    public DateTime HardExpiration { get; set; }
    public DateTime SoftExpiration { get; set; }  // Refresh trigger
}

public async Task<T> GetWithSoftExpirationAsync<T>(
    string key,
    Func<Task<T>> factory,
    TimeSpan softExpiration,
    TimeSpan hardExpiration)
{
    var cached = await _cache.GetAsync<CacheEntry<T>>(key);
    
    if (cached != null)
    {
        // Still valid
        if (DateTime.UtcNow < cached.HardExpiration)
        {
            // Refresh in background if past soft expiration
            if (DateTime.UtcNow > cached.SoftExpiration)
            {
                _ = Task.Run(async () =>
                {
                    var fresh = await factory();
                    await SetWithSoftExpirationAsync(key, fresh, softExpiration, hardExpiration);
                });
            }
            
            return cached.Value;
        }
    }
    
    // Cache miss or hard expired
    var value = await factory();
    await SetWithSoftExpirationAsync(key, value, softExpiration, hardExpiration);
    return value;
}
```

### Eviction Policies

```csharp
public class CacheOptions
{
    // Maximum items in cache
    public int SizeLimit { get; set; } = 1000;
    
    // Compact cache when it reaches this percentage
    public double CompactionPercentage { get; set; } = 0.25;

// Configure memory cache with size limits
builder.Services.AddMemoryCache(options =>

    options.SizeLimit = 1000;
    options.CompactionPercentage = 0.25;
});

// Specify size for entries
_cache.Set("key", value, new MemoryCacheEntryOptions
{
    Size = 1,  // Contributes 1 to the size limit
```

### Cache Invalidation Patterns by Data Type

```csharp
// Immutable data - long cache, never invalidate
public async Task<Country> GetCountryAsync(string code)
{
    return await _cache.GetOrSetAsync(
        $"country:{code}",
        () => FetchCountryAsync(code),
        TimeSpan.FromDays(30));
}

// User data - invalidate on user actions
public async Task<UserProfile> GetUserProfileAsync(int userId)
{
    return await _cache.GetOrSetAsync(
        $"user:{userId}:profile",
        () => FetchUserProfileAsync(userId),
        TimeSpan.FromMinutes(15));
}

// Aggregate data - scheduled invalidation
public async Task<DashboardStats> GetDashboardStatsAsync()
{
    return await _cache.GetOrSetAsync(
        "dashboard:stats",
        () => CalculateDashboardStatsAsync(),
        TimeSpan.FromMinutes(5));  // Recalculate every 5 minutes
}
```

### Distributed Invalidation

```csharp
// Using Redis Pub/Sub for cache invalidation across instances
public class RedisCacheInvalidationService
{
    private readonly IConnectionMultiplexer _redis;
    private readonly IDistributedCache _cache;
    
    public async Task InvalidateAcrossInstancesAsync(string key)
    {
        // Remove from local cache
        await _cache.RemoveAsync(key);
        
        // Publish invalidation message
        var subscriber = _redis.GetSubscriber();
        await subscriber.PublishAsync("cache:invalidate", key);
    }
    
    public void SubscribeToInvalidations()
    {
        var subscriber = _redis.GetSubscriber();
        subscriber.Subscribe("cache:invalidate", async (channel, key) =>
        {
            await _cache.RemoveAsync(key);
        });
    }
}
```
