# Lesson 4: Object Pooling and Resource Management

## Learning Objectives
- Implement object pooling patterns
- Use ArrayPool and ObjectPool in .NET
- Manage expensive resources efficiently
- Understand when to use pooling

## Introduction

Object Pooling is a creational pattern where expensive-to-create objects are reused from a pool rather than created and destroyed repeatedly. This improves performance and reduces garbage collection pressure.

## Key Concepts Summary

### Why Object Pooling?

**Problem:**
```csharp
// Creating expensive objects repeatedly
for (int i = 0; i < 1000; i++)
{
    var buffer = new byte[8192];  // Allocates 8KB each time
    // Use buffer
    // Buffer becomes garbage
}
// Result: 8MB of garbage, frequent GC collections
```

**Solution with ArrayPool:**
```csharp
var pool = ArrayPool<byte>.Shared;

for (int i = 0; i < 1000; i++)
{
    var buffer = pool.Rent(8192);  // Reuse from pool
    try
    {
        // Use buffer
    }
    finally
    {
        pool.Return(buffer);  // Return to pool
    }
}
// Result: Reused buffers, minimal allocations
```

### ArrayPool<T>
```csharp
public class FileProcessor
{
    public async Task ProcessLargeFile(string path)
    {
        var pool = ArrayPool<byte>.Shared;
        var buffer = pool.Rent(4096);
        
        try
        {
            using var stream = File.OpenRead(path);
            int bytesRead;
            
            while ((bytesRead = await stream.ReadAsync(buffer, 0, buffer.Length)) > 0)
            {
                // Process buffer
                ProcessBytes(buffer.AsSpan(0, bytesRead));
            }
        }
        finally
        {
            pool.Return(buffer, clearArray: true);  // Clear sensitive data
        }
    }
}
```

### ObjectPool<T>
```csharp
// Custom pooled object
public class StringBuilder Pool
{
    public class StringBuilderPooledObjectPolicy : IPooledObjectPolicy<StringBuilder>
    {
        public StringBuilder Create() => new StringBuilder();
        
        public bool Return(StringBuilder obj)
        {
            obj.Clear();  // Reset state before returning
            return obj.Capacity <= 4096;  // Don't pool if too large
        }
    }
}

// Usage
var pool = new DefaultObjectPool<StringBuilder>(new StringBuilderPooledObjectPolicy());

var sb = pool.Get();
try
{
    sb.Append("Hello, ");
    sb.Append("World!");
    return sb.ToString();
}
finally
{
    pool.Return(sb);
}
```

### Custom Object Pool
```csharp
public class DatabaseConnectionPool
{
    private readonly ConcurrentBag<DbConnection> _connections = new();
    private readonly string _connectionString;
    private readonly int _maxPoolSize;
    
    public DatabaseConnectionPool(string connectionString, int maxPoolSize = 10)
    {
        _connectionString = connectionString;
        _maxPoolSize = maxPoolSize;
    }
    
    public DbConnection Rent()
    {
        if (_connections.TryTake(out var connection))
        {
            if (connection.State == ConnectionState.Open)
                return connection;
            
            connection.Dispose();
        }
        
        return CreateNewConnection();
    }
    
    public void Return(DbConnection connection)
    {
        if (connection.State != ConnectionState.Open || _connections.Count >= _maxPoolSize)
        {
            connection.Dispose();
            return;
        }
        
        _connections.Add(connection);
    }
    
    private DbConnection CreateNewConnection()
    {
        var connection = new SqlConnection(_connectionString);
        connection.Open();
        return connection;
    }
}
```

### Generic Object Pool Implementation

```csharp
public interface IPooledObjectPolicy<T>
{
    T Create();
    bool Return(T obj);
}

public class DefaultObjectPool<T> : ObjectPool<T> where T : class
{
    private readonly ConcurrentBag<T> _items = new();
    private readonly IPooledObjectPolicy<T> _policy;
    private readonly int _maxCapacity;
    
    public DefaultObjectPool(IPooledObjectPolicy<T> policy, int maxCapacity = 100)
    {
        _policy = policy;
        _maxCapacity = maxCapacity;
    }
    
    public override T Get()
    {
        if (_items.TryTake(out var item))
            return item;
        
        return _policy.Create();
    }
    
    public override void Return(T obj)
    {
        if (_items.Count < _maxCapacity && _policy.Return(obj))
        {
            _items.Add(obj);
        }
    }
}

// Example policy for StringBuilder
public class StringBuilderPoolPolicy : IPooledObjectPolicy<StringBuilder>
{
    private readonly int _initialCapacity;
    private readonly int _maxCapacity;
    
    public StringBuilderPoolPolicy(int initialCapacity = 256, int maxCapacity = 4096)
    {
        _initialCapacity = initialCapacity;
        _maxCapacity = maxCapacity;
    }
    
    public StringBuilder Create()
    {
        return new StringBuilder(_initialCapacity);
    }
    
    public bool Return(StringBuilder obj)
    {
        if (obj.Capacity > _maxCapacity)
            return false;
        
        obj.Clear();
        return true;
    }
}

// Usage
var pool = new DefaultObjectPool<StringBuilder>(
    new StringBuilderPoolPolicy(initialCapacity: 512, maxCapacity: 8192));

var sb = pool.Get();
try
{
    sb.Append("Hello ");
    sb.Append("World!");
    var result = sb.ToString();
}
finally
{
    pool.Return(sb);
}
```

### Async-Friendly Pool with Disposable Wrapper

```csharp
public class AsyncObjectPool<T> where T : class
{
    private readonly ConcurrentBag<T> _items = new();
    private readonly Func<Task<T>> _factory;
    private readonly Action<T> _reset;
    private readonly int _maxSize;
    
    public AsyncObjectPool(
        Func<Task<T>> factory,
        Action<T> reset,
        int maxSize = 100)
    {
        _factory = factory;
        _reset = reset;
        _maxSize = maxSize;
    }
    
    public async Task<PooledItem<T>> RentAsync()
    {
        if (_items.TryTake(out var item))
            return new PooledItem<T>(item, this);
        
        item = await _factory();
        return new PooledItem<T>(item, this);
    }
    
    internal void Return(T item)
    {
        if (_items.Count < _maxSize)
        {
            _reset?.Invoke(item);
            _items.Add(item);
        }
        else if (item is IDisposable disposable)
        {
            disposable.Dispose();
        }
    }
}

public readonly struct PooledItem<T> : IDisposable where T : class
{
    private readonly T _item;
    private readonly AsyncObjectPool<T> _pool;
    
    internal PooledItem(T item, AsyncObjectPool<T> pool)
    {
        _item = item;
        _pool = pool;
    }
    
    public T Item => _item;
    
    public void Dispose()
    {
        _pool.Return(_item);
    }
}

// Usage
var pool = new AsyncObjectPool<HttpClient>(
    factory: async () => 
    {
        var client = new HttpClient();
        client.BaseAddress = new Uri("https://api.example.com");
        await Task.CompletedTask;
        return client;
    },
    reset: client => 
    {
        client.DefaultRequestHeaders.Clear();
    },
    maxSize: 20
);

using var pooledClient = await pool.RentAsync();
var response = await pooledClient.Item.GetAsync("/endpoint");
```

### When to Use Object Pooling

| Use When | Don't Use When | Example |
|----------|----------------|---------|
| Object creation is expensive | Objects are cheap to create | Buffer pools vs simple DTOs |
| High allocation rate | Low allocation rate | Hot path operations vs one-time setup |
| Large objects | Small objects | 8KB+ buffers vs small structs |
| Known maximum needed | Unbounded usage | Connection pools vs ad-hoc requests |
| Thread-safe objects | Stateful objects requiring cleanup | Immutable formatters vs user sessions |

### Common Mistakes

1. **Not Clearing Sensitive Data**
```csharp
// BAD
pool.Return(buffer);

// GOOD
Array.Clear(buffer, 0, buffer.Length);
pool.Return(buffer, clearArray: true);
```

2. **Using After Return**
```csharp
// BAD
var buffer = pool.Rent(1024);
pool.Return(buffer);
ProcessData(buffer);  // Buffer may be reused!

// GOOD
var buffer = pool.Rent(1024);
try
{
    ProcessData(buffer);
}
finally
{
    pool.Return(buffer);
}
```

3. **Over-Pooling Small Objects**
```csharp
// BAD - pooling is overkill for 24-byte objects
var pool = new ObjectPool<SmallDto>();

// GOOD - just allocate
var dto = new SmallDto();
```

## Practice Exercise

After completing the materials (10 minutes max):

**Task:** Create a pooled JSON serializer service

**Requirements:**
1. Create a pool for `MemoryStream` objects
2. Implement a `JsonSerializerPool` that reuses streams
3. Add methods:
   - `SerializeAsync<T>(T obj)` returns byte[]
   - `DeserializeAsync<T>(byte[] data)` returns T
4. Ensure streams are reset before returning to pool
5. Set maximum pool size to 50
6. Add metrics tracking (total rents, returns, creates)

**Solution Template:**
```csharp
public class JsonSerializerPool
{
    private readonly ConcurrentBag<MemoryStream> _streams = new();
    private readonly int _maxPoolSize;
    private int _totalCreated;
    
    public JsonSerializerPool(int maxPoolSize = 50)
    {
        _maxPoolSize = maxPoolSize;
    }
    
    public async Task<byte[]> SerializeAsync<T>(T obj)
    {
        var stream = RentStream();
        try
        {
            await JsonSerializer.SerializeAsync(stream, obj);
            return stream.ToArray();
        }
        finally
        {
            ReturnStream(stream);
        }
    }
    
    public async Task<T> DeserializeAsync<T>(byte[] data)
    {
        var stream = RentStream();
        try
        {
            await stream.WriteAsync(data);
            stream.Position = 0;
            return await JsonSerializer.DeserializeAsync<T>(stream);
        }
        finally
        {
            ReturnStream(stream);
        }
    }
    
    private MemoryStream RentStream()
    {
        if (_streams.TryTake(out var stream))
            return stream;
        
        Interlocked.Increment(ref _totalCreated);
        return new MemoryStream();
    }
    
    private void ReturnStream(MemoryStream stream)
    {
        stream.SetLength(0);
        stream.Position = 0;
        
        if (_streams.Count < _maxPoolSize)
        {
            _streams.Add(stream);
        }
        else
        {
            stream.Dispose();
        }
    }
    
    public int PoolSize => _streams.Count;
    public int TotalCreated => _totalCreated;
}
```

## Self-Check Questions

1. **When should you use object pooling?**
   - Object creation is expensive (CPU/memory)
   - High allocation rate causing GC pressure
   - Measurable performance benefit

2. **What's the difference between ArrayPool and ObjectPool?**
   - ArrayPool: Specialized for arrays, shared default instance
   - ObjectPool: Generic for any reference type, custom policies

3. **Why is it important to clear/reset pooled objects before returning them?**
   - Prevents data leaks between consumers
   - Ensures consistent initial state
   - Critical for security with sensitive data

4. **What are the trade-offs of object pooling?**
   - Benefits: Reduced allocations, less GC pressure, better performance
   - Costs: Added complexity, memory overhead, potential bugs if misused

5. **How does IHttpClientFactory use pooling?**
   - Pools HttpMessageHandler instances (not HttpClient)
   - Manages handler lifetime and rotation
   - Prevents socket exhaustion

6. **What's the captive reference problem in pools?**
   - When pooled object holds references to large objects
   - Prevents garbage collection
   - Solution: Clear references when returning to pool
