## Introduction

The Decorator pattern attaches additional responsibilities to an object dynamically. It provides a flexible alternative to subclassing for extending functionality. In .NET, this pattern is widely used in middleware pipelines and cross-cutting concerns.

---

## Key Concepts Summary

### Problem: Adding Functionality Without Inheritance

```csharp
// BAD - Inheritance explosion
public class BasicNotifier { }
public class EmailNotifier : BasicNotifier { }
public class SmsNotifier : BasicNotifier { }
public class EmailAndSmsNotifier : BasicNotifier { }
public class EmailAndSlackNotifier : BasicNotifier { }
// ... combinatorial explosion!
```

### Solution: Decorator Pattern

```csharp
// Component interface
public interface INotifier
{
    void Send(string message);
}

// Concrete component
public class BasicNotifier : INotifier
{
    public void Send(string message)
    {
        Console.WriteLine($"Basic notification: {message}");
    }
}

// Base decorator
public abstract class NotifierDecorator : INotifier
{
    protected readonly INotifier _wrapped;
    
    protected NotifierDecorator(INotifier wrapped)
    {
        _wrapped = wrapped;
    }
    
    public virtual void Send(string message)
    {
        _wrapped.Send(message);
    }
}

// Concrete decorators
public class EmailDecorator : NotifierDecorator
{
    public EmailDecorator(INotifier wrapped) : base(wrapped) { }
    
    public override void Send(string message)
    {
        base.Send(message);  // Call wrapped component
        SendEmail(message);   // Add email functionality
    }
    
    private void SendEmail(string message)
    {
        Console.WriteLine($"Sending email: {message}");
    }
}

public class SmsDecorator : NotifierDecorator
{
    public SmsDecorator(INotifier wrapped) : base(wrapped) { }
    
    public override void Send(string message)
    {
        base.Send(message);
        SendSms(message);
    }
    
    private void SendSms(string message)
    {
        Console.WriteLine($"Sending SMS: {message}");
    }
}

public class SlackDecorator : NotifierDecorator
{
    public SlackDecorator(INotifier wrapped) : base(wrapped) { }
    
    public override void Send(string message)
    {
        base.Send(message);
        SendSlack(message);
    }
    
    private void SendSlack(string message)
    {
        Console.WriteLine($"Sending Slack message: {message}");
    }
}

// Usage - compose decorators dynamically
INotifier notifier = new BasicNotifier();
notifier = new EmailDecorator(notifier);
notifier = new SmsDecorator(notifier);
notifier = new SlackDecorator(notifier);

notifier.Send("Server is down!");
// Output:
// Basic notification: Server is down!
// Sending email: Server is down!
// Sending SMS: Server is down!
// Sending Slack message: Server is down!
```

### Decorator with Dependency Injection

```csharp
// Service interface
public interface ICacheService
{
    Task<T> GetAsync<T>(string key);
    Task SetAsync<T>(string key, T value);
}

// Core implementation
public class RedisCacheService : ICacheService
{
    public async Task<T> GetAsync<T>(string key)
    {
        // Redis implementation
        return await Task.FromResult(default(T));
    }
    
    public async Task SetAsync<T>(string key, T value)
    {
        // Redis implementation
        await Task.CompletedTask;
    }
}

// Logging decorator
public class LoggingCacheDecorator : ICacheService
{
    private readonly ICacheService _inner;
    private readonly ILogger<LoggingCacheDecorator> _logger;
    
    public LoggingCacheDecorator(ICacheService inner, ILogger<LoggingCacheDecorator> logger)
    {
        _inner = inner;
        _logger = logger;
    }
    
    public async Task<T> GetAsync<T>(string key)
    {
        _logger.LogInformation("Getting cache key: {Key}", key);
        var result = await _inner.GetAsync<T>(key);
        _logger.LogInformation("Cache {Status} for key: {Key}", 
            result != null ? "HIT" : "MISS", key);
        return result;
    }
    
    public async Task SetAsync<T>(string key, T value)
    {
        _logger.LogInformation("Setting cache key: {Key}", key);
        await _inner.SetAsync(key, value);
    }
}

// Registration
services.AddSingleton<RedisCacheService>();
services.AddSingleton<ICacheService>(sp =>
{
    var redis = sp.GetRequiredService<RedisCacheService>();
    var logger = sp.GetRequiredService<ILogger<LoggingCacheDecorator>>();
    return new LoggingCacheDecorator(redis, logger);
});
```

### ASP.NET Core Middleware (Decorator Pattern)

```csharp
// Middleware is decorator pattern in action!
public class RequestLoggingMiddleware
{
    private readonly RequestDelegate _next;  // Wrapped "component"
    private readonly ILogger<RequestLoggingMiddleware> _logger;
    
    public RequestLoggingMiddleware(RequestDelegate next, ILogger<RequestLoggingMiddleware> logger)
    {
        _next = next;
        _logger = logger;
    }
    
    public async Task InvokeAsync(HttpContext context)
    {
        _logger.LogInformation("Request: {Method} {Path}", 
            context.Request.Method, 
            context.Request.Path);
        
        var sw = Stopwatch.StartNew();
        
        await _next(context);  // Call next middleware
        
        sw.Stop();
        _logger.LogInformation("Response: {StatusCode} in {ElapsedMs}ms",
            context.Response.StatusCode,
            sw.ElapsedMilliseconds);
    }
}

// Pipeline configuration
app.UseMiddleware<RequestLoggingMiddleware>();
app.UseAuthentication();
app.UseAuthorization();
app.UseEndpoints(...);
```

### Decorator for Repository Pattern

```csharp
// Core repository
public interface IProductRepository
{
    Task<Product> GetByIdAsync(Guid id);
    Task AddAsync(Product product);
}

public class ProductRepository : IProductRepository
{
    private readonly DbContext _context;
    
    public async Task<Product> GetByIdAsync(Guid id)
    {
        return await _context.Set<Product>().FindAsync(id);
    }
    
    public async Task AddAsync(Product product)
    {
        await _context.Set<Product>().AddAsync(product);
        await _context.SaveChangesAsync();
    }
}

// Caching decorator
public class CachedProductRepository : IProductRepository
{
    private readonly IProductRepository _inner;
    private readonly IMemoryCache _cache;
    
    public CachedProductRepository(IProductRepository inner, IMemoryCache cache)
    {
        _inner = inner;
        _cache = cache;
    }
    
    public async Task<Product> GetByIdAsync(Guid id)
    {
        var cacheKey = $"product_{id}";
        
        if (_cache.TryGetValue(cacheKey, out Product cachedProduct))
            return cachedProduct;
        
        var product = await _inner.GetByIdAsync(id);
        
        if (product != null)
        {
            _cache.Set(cacheKey, product, TimeSpan.FromMinutes(5));
        }
        
        return product;
    }
    
    public async Task AddAsync(Product product)
    {
        await _inner.AddAsync(product);
        _cache.Remove($"product_{product.Id}");  // Invalidate cache
    }
}
```

### Multiple Decorators Chaining

```csharp
// Service interface
public interface IOrderService
{
    Task<Order> CreateOrderAsync(CreateOrderRequest request);
    Task<Order> GetOrderAsync(Guid orderId);
}

// Base implementation
public class OrderService : IOrderService
{
    private readonly IOrderRepository _repository;
    
    public OrderService(IOrderRepository repository)
    {
        _repository = repository;
    }
    
    public async Task<Order> CreateOrderAsync(CreateOrderRequest request)
    {
        var order = new Order(request);
        await _repository.AddAsync(order);
        return order;
    }
    
    public async Task<Order> GetOrderAsync(Guid orderId)
    {
        return await _repository.GetByIdAsync(orderId);
    }
}

// Validation decorator
public class ValidationOrderServiceDecorator : IOrderService
{
    private readonly IOrderService _inner;
    private readonly IValidator<CreateOrderRequest> _validator;
    
    public ValidationOrderServiceDecorator(
        IOrderService inner,
        IValidator<CreateOrderRequest> validator)
    {
        _inner = inner;
        _validator = validator;
    }
    
    public async Task<Order> CreateOrderAsync(CreateOrderRequest request)
    {
        var validationResult = await _validator.ValidateAsync(request);
        if (!validationResult.IsValid)
        {
            throw new ValidationException(validationResult.Errors);
        }
        
        return await _inner.CreateOrderAsync(request);
    }
    
    public Task<Order> GetOrderAsync(Guid orderId)
    {
        return _inner.GetOrderAsync(orderId);
    }
}

// Logging decorator
public class LoggingOrderServiceDecorator : IOrderService
{
    private readonly IOrderService _inner;
    private readonly ILogger<LoggingOrderServiceDecorator> _logger;
    
    public LoggingOrderServiceDecorator(
        IOrderService inner,
        ILogger<LoggingOrderServiceDecorator> logger)
    {
        _inner = inner;
        _logger = logger;
    }
    
    public async Task<Order> CreateOrderAsync(CreateOrderRequest request)
    {
        _logger.LogInformation("Creating order for customer {CustomerId}", request.CustomerId);
        
        var stopwatch = Stopwatch.StartNew();
        try
        {
            var order = await _inner.CreateOrderAsync(request);
            _logger.LogInformation(
                "Order {OrderId} created successfully in {ElapsedMs}ms",
                order.Id,
                stopwatch.ElapsedMilliseconds);
            return order;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to create order for customer {CustomerId}", request.CustomerId);
            throw;
        }
    }
    
    public async Task<Order> GetOrderAsync(Guid orderId)
    {
        _logger.LogDebug("Fetching order {OrderId}", orderId);
        return await _inner.GetOrderAsync(orderId);
    }
}

// Caching decorator
public class CachingOrderServiceDecorator : IOrderService
{
    private readonly IOrderService _inner;
    private readonly IDistributedCache _cache;
    
    public CachingOrderServiceDecorator(
        IOrderService inner,
        IDistributedCache cache)
    {
        _inner = inner;
        _cache = cache;
    }
    
    public Task<Order> CreateOrderAsync(CreateOrderRequest request)
    {
        // Don't cache creates
        return _inner.CreateOrderAsync(request);
    }
    
    public async Task<Order> GetOrderAsync(Guid orderId)
    {
        var cacheKey = $"order:{orderId}";
        var cached = await _cache.GetStringAsync(cacheKey);
        
        if (cached != null)
        {
            return JsonSerializer.Deserialize<Order>(cached);
        }
        
        var order = await _inner.GetOrderAsync(orderId);
        
        if (order != null)
        {
            await _cache.SetStringAsync(
                cacheKey,
                JsonSerializer.Serialize(order),
                new DistributedCacheEntryOptions
                {
                    AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(10)
                });
        }
        
        return order;
    }
}

// Registration - decorators applied in order
services.AddScoped<OrderService>();
services.AddScoped<IOrderService>(provider =>
{
    var repository = provider.GetRequiredService<IOrderRepository>();
    var validator = provider.GetRequiredService<IValidator<CreateOrderRequest>>();
    var logger = provider.GetRequiredService<ILogger<LoggingOrderServiceDecorator>>();
    var cache = provider.GetRequiredService<IDistributedCache>();
    
    IOrderService service = new OrderService(repository);
    service = new ValidationOrderServiceDecorator(service, validator);
    service = new LoggingOrderServiceDecorator(service, logger);
    service = new CachingOrderServiceDecorator(service, cache);
    
    return service;
});
```

### Decorator for Retry Logic

```csharp
public class RetryDecorator<T> : DispatchProxy
{
    private T _decorated;
    private int _maxRetries;
    
    protected override object Invoke(MethodInfo targetMethod, object[] args)
    {
        int retryCount = 0;
        while (true)
        {
            try
            {
                return targetMethod.Invoke(_decorated, args);
            }
            catch (Exception ex)
            {
                if (++retryCount > _maxRetries)
                    throw;
                
                Thread.Sleep(TimeSpan.FromSeconds(Math.Pow(2, retryCount)));
            }
        }
    }
    
    public static T Create(T decorated, int maxRetries)
    {
        object proxy = Create<T, RetryDecorator<T>>();
        ((RetryDecorator<T>)proxy).SetParameters(decorated, maxRetries);
        return (T)proxy;
    }
    
    private void SetParameters(T decorated, int maxRetries)
    {
        _decorated = decorated;
        _maxRetries = maxRetries;
    }
}

// Usage
var service = RetryDecorator<IExternalApiService>.Create(
    new ExternalApiService(),
    maxRetries: 3);
```

### Decorator for Performance Monitoring

```csharp
public class PerformanceMonitoringDecorator<T> : IOrderService
{
    private readonly IOrderService _inner;
    private readonly IMetricsCollector _metrics;
    private readonly string _serviceName;
    
    public PerformanceMonitoringDecorator(
        IOrderService inner,
        IMetricsCollector metrics)
    {
        _inner = inner;
        _metrics = metrics;
        _serviceName = typeof(T).Name;
    }
    
    public async Task<Order> CreateOrderAsync(CreateOrderRequest request)
    {
        var stopwatch = Stopwatch.StartNew();
        try
        {
            var result = await _inner.CreateOrderAsync(request);
            
            _metrics.RecordDuration(
                $"{_serviceName}.CreateOrder",
                stopwatch.Elapsed,
                new Dictionary<string, string>
                {
                    ["status"] = "success"
                });
            
            return result;
        }
        catch (Exception)
        {
            _metrics.RecordDuration(
                $"{_serviceName}.CreateOrder",
                stopwatch.Elapsed,
                new Dictionary<string, string>
                {
                    ["status"] = "error"
                });
            throw;
        }
    }
    
    public Task<Order> GetOrderAsync(Guid orderId)
    {
        return _inner.GetOrderAsync(orderId);
    }
}
```

### Decorator vs Other Patterns

| Pattern | Purpose | Structure |
|---------|---------|-----------|
| **Decorator** | Add responsibilities dynamically | Wraps and enhances |
| **Adapter** | Make incompatible interfaces work | Converts interface |
| **Proxy** | Control access to object | Same interface, different purpose |
| **Chain of Responsibility** | Pass request along chain | Handlers may not call next |

### When to Use Decorator

**Use when:**
- Need to add responsibilities to objects dynamically
- Want to add/remove functionality at runtime
- Have cross-cutting concerns (logging, caching, validation)
- Subclassing would create too many classes

**Don't use when:**
- Simple behavior modification (use inheritance)
- Need to change object's interface (use Adapter)
- Need to limit access (use Proxy)
- Order of decorators matters rigidly (reconsider design)

## Real-World Scenario

**HTTP Client with Decorators**

Build a resilient HTTP client using decorator pattern:

```csharp
public interface IHttpClient
{
    Task<HttpResponseMessage> SendAsync(HttpRequestMessage request);
}

// Base implementation
public class BaseHttpClient : IHttpClient
{
    private readonly HttpClient _httpClient;
    
    public BaseHttpClient(HttpClient httpClient)
    {
        _httpClient = httpClient;
    }
    
    public Task<HttpResponseMessage> SendAsync(HttpRequestMessage request)
    {
        return _httpClient.SendAsync(request);
    }
}

// Retry decorator
public class RetryHttpClientDecorator : IHttpClient
{
    private readonly IHttpClient _inner;
    private readonly int _maxRetries;
    
    public RetryHttpClientDecorator(IHttpClient inner, int maxRetries = 3)
    {
        _inner = inner;
        _maxRetries = maxRetries;
    }
    
    public async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request)
    {
        for (int i = 0; i < _maxRetries; i++)
        {
            try
            {
                var response = await _inner.SendAsync(request);
                if (response.IsSuccessStatusCode || i == _maxRetries - 1)
                    return response;
                
                await Task.Delay(TimeSpan.FromSeconds(Math.Pow(2, i)));
            }
            catch when (i < _maxRetries - 1)
            {
                await Task.Delay(TimeSpan.FromSeconds(Math.Pow(2, i)));
            }
        }
        
        throw new HttpRequestException("Max retries exceeded");
    }
}

// Logging decorator
public class LoggingHttpClientDecorator : IHttpClient
{
    private readonly IHttpClient _inner;
    private readonly ILogger _logger;
    
    public async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request)
    {
        _logger.LogInformation("Sending {Method} request to {Uri}",
            request.Method, request.RequestUri);
        
        var response = await _inner.SendAsync(request);
        
        _logger.LogInformation("Received {StatusCode} from {Uri}",
            (int)response.StatusCode, request.RequestUri);
        
        return response;
    }
}

// Usage
IHttpClient client = new BaseHttpClient(new HttpClient());
client = new RetryHttpClientDecorator(client, maxRetries: 3);
client = new LoggingHttpClientDecorator(client, logger);

var response = await client.SendAsync(new HttpRequestMessage(
    HttpMethod.Get,
    "https://api.example.com/data"));
```

## Self-Check Questions

1. **How does Decorator differ from inheritance for adding functionality?**
   - Decorator: Runtime composition, flexible, can combine multiple decorators
   - Inheritance: Compile-time, rigid, creates class explosion with combinations

2. **Why is the Decorator pattern useful for cross-cutting concerns?**
   - Separates cross-cutting logic from business logic
   - Allows adding/removing concerns without modifying core code
   - Can apply multiple concerns in any order

3. **How does ASP.NET Core middleware implement the Decorator pattern?**
   - Each middleware wraps the next one in the pipeline
   - Calls `_next(context)` to invoke the wrapped component
   - Can execute logic before and after calling next

4. **When would you choose Decorator over Proxy?**
   - Decorator: Adding functionality, stacking behaviors
   - Proxy: Controlling access, lazy loading, representing remote objects

5. **What's the difference between Decorator and Chain of Responsibility?**
   - Decorator: All decorators execute, wrapping each other
   - Chain of Responsibility: Request handled by first capable handler

6. **How do you prevent decorator ordering issues?**
   - Make decorators independent when possible
   - Document required ordering
   - Use builder pattern to enforce correct order
   - Consider if design needs refactoring
