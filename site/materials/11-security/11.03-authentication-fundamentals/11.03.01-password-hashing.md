## Introduction

Password hashing is a one-way cryptographic function that transforms passwords into fixed-length strings that cannot be reversed. Unlike encryption (which is reversible), hashing ensures that even if attackers access your database, they cannot easily obtain plaintext passwords. Modern hashing algorithms like bcrypt, Argon2, and PBKDF2 are specifically designed to be slow and computationally expensive, making brute-force attacks impractical.

### Password Hashing Algorithms

```csharp
// OWASP Recommendations (in order of preference):

// 1. Argon2id - PREFERRED (Winner of Password Hashing Competition 2015)
// Configuration: 19 MiB memory, 2 iterations, 1 degree of parallelism
public class Argon2PasswordHasher
{
    public string HashPassword(string password)
    {
        var argon2 = new Argon2id(Encoding.UTF8.GetBytes(password))
        {
            DegreeOfParallelism = 1,
            MemorySize = 19456, // 19 MiB
            Iterations = 2,
            Salt = GenerateSalt(16) // 16 bytes
        };
        
        return Convert.ToBase64String(argon2.GetBytes(32)); // 32-byte hash
    }
}

// 2. bcrypt - GOOD (widely supported, battle-tested)
// Configuration: Work factor of 10+ (ASP.NET Core Identity default)
public class BcryptPasswordHasher
{
    public string HashPassword(string password)
    {
        // Work factor 12 means 2^12 iterations
        int workFactor = 12;
        
        // BCrypt automatically generates and includes salt
        return BCrypt.Net.BCrypt.HashPassword(password, workFactor);
    }
    
    public bool VerifyPassword(string password, string hash)
    {
        return BCrypt.Net.BCrypt.Verify(password, hash);
    }
}

// 3. scrypt - ACCEPTABLE
// Configuration: n=2^17, r=8, p=1, 64 bytes salt

// 4. PBKDF2 - ACCEPTABLE (when FIPS compliance required)
// Configuration: 600,000 iterations minimum
public class Pbkdf2PasswordHasher
{
    public string HashPassword(string password)
    {
        int iterations = 600000; // OWASP minimum
        byte[] salt = GenerateSalt(16);
        
        using var pbkdf2 = new Rfc2898DeriveBytes(
            password,
            salt,
            iterations,
            HashAlgorithmName.SHA256);
        
        byte[] hash = pbkdf2.GetBytes(32);
        
        // Store salt with hash
        byte[] hashBytes = new byte[48]; // 16 salt + 32 hash
        Array.Copy(salt, 0, hashBytes, 0, 16);
        Array.Copy(hash, 0, hashBytes, 16, 32);
        
        return Convert.ToBase64String(hashBytes);
    }
    
    public bool VerifyPassword(string password, string storedHash)
    {
        byte[] hashBytes = Convert.FromBase64String(storedHash);
        byte[] salt = new byte[16];
        Array.Copy(hashBytes, 0, salt, 0, 16);
        
        using var pbkdf2 = new Rfc2898DeriveBytes(
            password,
            salt,
            600000,
            HashAlgorithmName.SHA256);
        
        byte[] hash = pbkdf2.GetBytes(32);
        
        // Compare hashes
        for (int i = 0; i < 32; i++)
        {
            if (hashBytes[i + 16] != hash[i])
                return false;
        }
        
        return true;
    }
}
```

### Understanding Salt

```csharp
// Salt is random data added to passwords before hashing
// Purpose: Prevents rainbow table attacks and ensures identical passwords have different hashes

// BAD: No salt
string hash1 = SHA256("password123"); // Always produces same hash
string hash2 = SHA256("password123"); // Identical to hash1
// Rainbow tables can precompute hashes for common passwords

// GOOD: With salt
string salt1 = GenerateRandomSalt();
string hash1 = SHA256(salt1 + "password123"); // Unique hash
string salt2 = GenerateRandomSalt();
string hash2 = SHA256(salt2 + "password123"); // Different hash!
// Rainbow tables are useless - attacker must crack each hash individually

public byte[] GenerateSalt(int size = 16)
{
    using var rng = RandomNumberGenerator.Create();
    byte[] salt = new byte[size];
    rng.GetBytes(salt);
    return salt;
}

// Modern algorithms like bcrypt include salt automatically
// No need to manage it separately
```

### Work Factor / Cost Factor

```csharp
// Work factor determines how many iterations the algorithm performs
// Higher work factor = more secure but slower

// bcrypt work factor examples
int workFactor = 10; // 2^10 = 1,024 iterations    (~0.1 seconds)
int workFactor = 12; // 2^12 = 4,096 iterations    (~0.3 seconds) RECOMMENDED
int workFactor = 14; // 2^14 = 16,384 iterations   (~1.2 seconds)

// Why increase work factor?
// - Moore's Law: Computers get faster over time
// - GPUs can crack passwords faster
// - Slow down brute force attacks

// Benchmark your work factor
public async Task<TimeSpan> BenchmarkWorkFactor(int workFactor)
{
    var stopwatch = Stopwatch.StartNew();
    
    BCrypt.Net.BCrypt.HashPassword("test-password", workFactor);
    
    stopwatch.Stop();
    return stopwatch.Elapsed;
}

// Target: 250-500ms per hash on your production hardware
// Acceptable range: 100ms - 1000ms
```

### ASP.NET Core Identity Password Hashing

```csharp
// ASP.NET Core Identity uses PBKDF2 by default
// Can be customized with different hashers

// Default configuration
public class Startup
{
    public void ConfigureServices(IServiceCollection services)
    {
        services.AddIdentity<ApplicationUser, IdentityRole>(options =>
        {
            // Password requirements
            options.Password.RequireDigit = true;
            options.Password.RequireLowercase = true;
            options.Password.RequireNonAlphanumeric = true;
            options.Password.RequireUppercase = true;
            options.Password.RequiredLength = 8;
            options.Password.RequiredUniqueChars = 4;
        })
        .AddEntityFrameworkStores<ApplicationDbContext>()
        .AddDefaultTokenProviders();
    }
}

// Custom password hasher using bcrypt
public class BcryptPasswordHasher<TUser> : IPasswordHasher<TUser> 
    where TUser : class
{
    private readonly int _workFactor;
    
    public BcryptPasswordHasher(int workFactor = 12)
    {
        _workFactor = workFactor;
    }
    
    public string HashPassword(TUser user, string password)
    {
        return BCrypt.Net.BCrypt.HashPassword(password, _workFactor);
    }
    
    public PasswordVerificationResult VerifyHashedPassword(
        TUser user, 
        string hashedPassword, 
        string providedPassword)
    {
        if (BCrypt.Net.BCrypt.Verify(providedPassword, hashedPassword))
        {
            // Check if rehashing is needed (work factor changed)
            if (BCrypt.Net.BCrypt.PasswordNeedsRehash(hashedPassword, _workFactor))
            {
                return PasswordVerificationResult.SuccessRehashNeeded;
            }
            
            return PasswordVerificationResult.Success;
        }
        
        return PasswordVerificationResult.Failed;
    }
}

// Register custom hasher
services.AddScoped<IPasswordHasher<ApplicationUser>, 
    BcryptPasswordHasher<ApplicationUser>>();
```

### Password Peppering (Optional Extra Layer)

```csharp
// Pepper: Secret value added before hashing, stored separately from database
// Provides extra protection if database is compromised but application secrets are not

public class PepperedPasswordHasher
{
    private readonly string _pepper;
    
    public PepperedPasswordHasher(IConfiguration config)
    {
        // Store pepper in environment variable or Key Vault
        _pepper = config["Security:PasswordPepper"] 
            ?? throw new InvalidOperationException("Pepper not configured");
    }
    
    public string HashPassword(string password)
    {
        // Combine password with pepper before hashing
        string pepperedPassword = password + _pepper;
        
        return BCrypt.Net.BCrypt.HashPassword(pepperedPassword, 12);
    }
    
    public bool VerifyPassword(string password, string hash)
    {
        string pepperedPassword = password + _pepper;
        return BCrypt.Net.BCrypt.Verify(pepperedPassword, hash);
    }
}

// Note: If pepper is compromised, must rehash all passwords
// If pepper is lost, all passwords become unverifiable
// Use with caution - adds complexity
```

### Password Strength Requirements

```csharp
// Implement strong password policies
public class PasswordValidator
{
    public ValidationResult ValidatePassword(string password)
    {
        var errors = new List<string>();
        
        // Minimum length
        if (password.Length < 12)
        {
            errors.Add("Password must be at least 12 characters long");
        }
        
        // Complexity requirements
        if (!password.Any(char.IsUpper))
        {
            errors.Add("Password must contain at least one uppercase letter");
        }
        
        if (!password.Any(char.IsLower))
        {
            errors.Add("Password must contain at least one lowercase letter");
        }
        
        if (!password.Any(char.IsDigit))
        {
            errors.Add("Password must contain at least one number");
        }
        
        if (!password.Any(ch => !char.IsLetterOrDigit(ch)))
        {
            errors.Add("Password must contain at least one special character");
        }
        
        // Check against common passwords
        if (IsCommonPassword(password))
        {
            errors.Add("Password is too common, please choose a stronger password");
        }
        
        // Check for sequential characters
        if (HasSequentialCharacters(password))
        {
            errors.Add("Password contains sequential characters");
        }
        
        return errors.Any() 
            ? ValidationResult.Failure(errors) 
            : ValidationResult.Success();
    }
    
    private bool IsCommonPassword(string password)
    {
        // Check against list of 10,000 most common passwords
        var commonPasswords = LoadCommonPasswords();
        return commonPasswords.Contains(password.ToLower());
    }
    
    private bool HasSequentialCharacters(string password)
    {
        for (int i = 0; i < password.Length - 2; i++)
        {
            if (password[i] + 1 == password[i + 1] && 
                password[i + 1] + 1 == password[i + 2])
            {
                return true; // "abc", "123", etc.
            }
        }
        
        return false;
    }
}
```

### Migration from Weak Hashing

```csharp
// If upgrading from legacy system with weak hashing (MD5, SHA-1)
// Use progressive enhancement approach

public class PasswordMigrationService
{
    public async Task<LoginResult> LoginAndMigrateAsync(
        string username, 
        string password)
    {
        var user = await _userRepository.GetByUsernameAsync(username);
        
        if (user == null)
            return LoginResult.Failed();
        
        // Check which hashing algorithm was used
        if (user.PasswordHashVersion == 1) // Legacy MD5
        {
            // Verify with old algorithm
            string legacyHash = ComputeMD5Hash(password);
            
            if (legacyHash != user.PasswordHash)
                return LoginResult.Failed();
            
            // Successful login - upgrade to bcrypt
            user.PasswordHash = BCrypt.Net.BCrypt.HashPassword(password, 12);
            user.PasswordHashVersion = 2; // Mark as bcrypt
            await _userRepository.UpdateAsync(user);
            
            _logger.LogInformation($"Migrated password hash for user {username}");
        }
        else if (user.PasswordHashVersion == 2) // bcrypt
        {
            if (!BCrypt.Net.BCrypt.Verify(password, user.PasswordHash))
                return LoginResult.Failed();
        }
        
        return LoginResult.Success(user);
    }
}
```
