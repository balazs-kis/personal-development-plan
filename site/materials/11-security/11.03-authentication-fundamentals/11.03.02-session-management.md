## Introduction

Session management maintains user state across multiple HTTP requests. Since HTTP is stateless, applications use sessions (typically implemented with cookies) to remember authenticated users. Insecure session management is a top security vulnerability—attackers can hijack sessions, fixate session IDs, or exploit weak session handling to impersonate users and access sensitive data.

## Key Concepts Summary

### Session Lifecycle

```csharp
/*
Session Lifecycle:

1. User visits application (anonymous)
2. User logs in (authentication)
   └─> Server creates session
   └─> Server generates unique session ID
   └─> Server stores session ID in cookie
   └─> Browser sends cookie with each request
3. User interacts with application
   └─> Server validates session ID
   └─> Server loads session data
   └─> Server extends session timeout (sliding expiration)
4. User logs out OR session times out
   └─> Server invalidates session
   └─> Browser clears cookie
*/
```

### Secure Cookie Configuration

```csharp
// ASP.NET Core Identity cookie configuration
public class Startup
{
    public void ConfigureServices(IServiceCollection services)
    {
        services.ConfigureApplicationCookie(options =>
        {
            // HttpOnly: Prevents JavaScript from accessing the cookie
            // Protects against XSS attacks stealing session cookies
            options.Cookie.HttpOnly = true;
            
            // Secure: Cookie only sent over HTTPS
            // Prevents man-in-the-middle attacks from stealing cookies
            options.Cookie.SecurePolicy = CookieSecurePolicy.Always;
            
            // SameSite: Prevents CSRF attacks
            // Strict: Cookie not sent on cross-site requests
            // Lax: Cookie sent on top-level navigation (links)
            // None: Cookie sent on all requests (requires Secure)
            options.Cookie.SameSite = SameSiteMode.Strict;
            
            // Cookie name (avoid revealing framework)
            options.Cookie.Name = "AppSession";
            
            // Session timeout
            options.ExpireTimeSpan = TimeSpan.FromMinutes(30);
            
            // Sliding expiration: Extends timeout on each request
            options.SlidingExpiration = true;
            
            // Login/logout paths
            options.LoginPath = "/Account/Login";
            options.LogoutPath = "/Account/Logout";
            options.AccessDeniedPath = "/Account/AccessDenied";
        });
    }
}

// For APIs or SPAs without cookies
public class JwtConfiguration
{

                options.RequireHttpsMetadata = true; // Enforce HTTPS
```

### Session Timeouts

```csharp
// Implement both absolute and idle timeouts
public class SessionTimeoutMiddleware
{
    private readonly RequestDelegate _next;
    private readonly TimeSpan _absoluteTimeout;
    private readonly TimeSpan _idleTimeout;
    
    public SessionTimeoutMiddleware(
        RequestDelegate next,
        TimeSpan absoluteTimeout,
        TimeSpan idleTimeout)
    {
        _next = next;
        _absoluteTimeout = absoluteTimeout;
        _idleTimeout = idleTimeout;
    }
    
    public async Task InvokeAsync(HttpContext context)
    {
        if (context.User.Identity?.IsAuthenticated ?? false)
        {
            var session = context.Session;
            
            // Check absolute timeout (max session duration)
            var sessionStart = session.GetString("SessionStart");
            if (sessionStart != null)
            {
                var startTime = DateTime.Parse(sessionStart);
                if (DateTime.UtcNow - startTime > _absoluteTimeout)
                {
                    await SignOutAsync(context);
                    context.Response.Redirect("/Account/SessionExpired");
                    return;
                }
            }
            else
            {
                session.SetString("SessionStart", DateTime.UtcNow.ToString("O"));
            }
            
            // Check idle timeout (inactivity)
            var lastActivity = session.GetString("LastActivity");
            if (lastActivity != null)
            {
                var lastTime = DateTime.Parse(lastActivity);
                if (DateTime.UtcNow - lastTime > _idleTimeout)
                {
                    await SignOutAsync(context);
                    context.Response.Redirect("/Account/SessionExpired");
                    return;
                }
            }
            
            // Update last activity
            session.SetString("LastActivity", DateTime.UtcNow.ToString("O"));
        }
        
        await _next(context);
    }
    
    private async Task SignOutAsync(HttpContext context)
    {
        context.Session.Clear();
        // Additional sign-out logic
    }
}

// Usage
app.UseMiddleware<SessionTimeoutMiddleware>(
    absoluteTimeout: TimeSpan.FromHours(8),
    idleTimeout: TimeSpan.FromMinutes(30)
);
```

### Preventing Session Fixation

```csharp
// Session fixation: Attacker sets victim's session ID before login
// Prevention: Regenerate session ID after authentication

public class AccountController : Controller
{
    [HttpPost]
    public async Task<IActionResult> Login(LoginViewModel model)
    {
        if (!ModelState.IsValid)
            return View(model);
        
        // Verify credentials
        var result = await _signInManager.PasswordSignInAsync(
            model.Email,
            model.Password,
            model.RememberMe,
            lockoutOnFailure: true);
        
        if (result.Succeeded)
        {
            // CRITICAL: Regenerate session ID after successful login
            // This prevents session fixation attacks
            await HttpContext.Session.LoadAsync();
            
            // Clear old session data
            var oldSessionData = new Dictionary<string, string>();
            foreach (var key in HttpContext.Session.Keys)
            {
                oldSessionData[key] = HttpContext.Session.GetString(key);
            }
            
            // Create new session
            HttpContext.Session.Clear();
            await HttpContext.SignOutAsync(
                CookieAuthenticationDefaults.AuthenticationScheme);
            
            // Sign in with new session
            var user = await _userManager.FindByEmailAsync(model.Email);
            await _signInManager.SignInAsync(user, model.RememberMe);
            
            // Restore necessary session data (if any)
            foreach (var kvp in oldSessionData)
            {
                if (IsSessionDataSafe(kvp.Key))
                {
                    HttpContext.Session.SetString(kvp.Key, kvp.Value);
                }
            }
            
            _logger.LogInformation($"User {model.Email} logged in with new session");
            
            return RedirectToAction("Index", "Home");
        }
        
        ModelState.AddModelError("", "Invalid login attempt");
        return View(model);
    }
}
```

### Session Security Monitoring

```csharp
// Monitor suspicious session activity
public class SessionMonitoringMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ILogger<SessionMonitoringMiddleware> _logger;
    
    public async Task InvokeAsync(HttpContext context)
    {
        if (context.User.Identity?.IsAuthenticated ?? false)
        {
            var userId = context.User.GetUserId();
            var sessionId = context.Session.Id;
            var ipAddress = context.Connection.RemoteIpAddress?.ToString();
            var userAgent = context.Request.Headers["User-Agent"].ToString();
            
            // Check for suspicious activity
            var lastIpAddress = context.Session.GetString("LastIpAddress");
            var lastUserAgent = context.Session.GetString("LastUserAgent");
            
            if (lastIpAddress != null && lastIpAddress != ipAddress)
            {
                // IP address changed - possible session hijacking
                _logger.LogWarning(
                    "IP address change detected for user {UserId}. " +
                    "Old: {OldIp}, New: {NewIp}",
                    userId, lastIpAddress, ipAddress);
                
                // Optional: Force re-authentication
                await context.SignOutAsync();
                context.Response.Redirect("/Account/SecurityAlert");
                return;
            }
            
            if (lastUserAgent != null && lastUserAgent != userAgent)
            {
                // User agent changed - possible session hijacking
                _logger.LogWarning(
                    "User agent change detected for user {UserId}",
                    userId);
            }
            
            // Store current values
            context.Session.SetString("LastIpAddress", ipAddress ?? "");
            context.Session.SetString("LastUserAgent", userAgent);
            
            // Track concurrent sessions (optional)
            await TrackActiveSessions(userId, sessionId);
        }
        
        await _next(context);
    }
    
    private async Task TrackActiveSessions(string userId, string sessionId)
    {
        // Store in Redis or database
        var key = $"user:{userId}:sessions";
        
        await _cache.SetAsync(key, new ActiveSession
        {
            SessionId = sessionId,
            LastActivity = DateTime.UtcNow,
            IpAddress = _httpContextAccessor.HttpContext.Connection.RemoteIpAddress?.ToString()
        }, TimeSpan.FromHours(1));
    }
}
```

### Logout Implementation

```csharp
// Proper logout clears all session data
public class AccountController : Controller
{
    [HttpPost]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> Logout()
    {
        var userId = User.GetUserId();
        
        // 1. Sign out from authentication system
        await _signInManager.SignOutAsync();
        
        // 2. Clear session data
        HttpContext.Session.Clear();
        
        // 3. Clear any application-specific state
        await _userStateService.ClearUserStateAsync(userId);
        
        // 4. Invalidate refresh tokens (if using)
        await _tokenService.RevokeAllTokensAsync(userId);
        
        // 5. Log the event
        _logger.LogInformation($"User {userId} logged out");
        
        // 6. Redirect to public page
        return RedirectToAction("Index", "Home");
    }
}

// Client-side: Clear any cached data
// JavaScript
function logout() {
    // Clear localStorage
    localStorage.clear();
    
    // Clear sessionStorage
    sessionStorage.clear();
    
    // Submit logout form
```

### Session Data Encryption

```csharp
// Encrypt sensitive data stored in session
public class SecureSessionService
{
    private readonly IDataProtector _protector;
    
    public SecureSessionService(IDataProtectionProvider provider)
    {
        _protector = provider.CreateProtector("SessionData");
    }
    
    public void SetSecureString(ISession session, string key, string value)
    {
        var encrypted = _protector.Protect(value);
        session.SetString(key, encrypted);
    }
    
    public string GetSecureString(ISession session, string key)
    {
        var encrypted = session.GetString(key);
        if (encrypted == null)
            return null;
        
        try
        {
            return _protector.Unprotect(encrypted);
        }
        catch (CryptographicException)
        {
            // Key rotated or data tampered
            session.Remove(key);
            return null;
        }
    }
    
    public void SetSecureObject<T>(ISession session, string key, T value)
    {
        var json = JsonSerializer.Serialize(value);
        var encrypted = _protector.Protect(json);
        session.SetString(key, encrypted);
    }
    
    public T GetSecureObject<T>(ISession session, string key)
    {
        var encrypted = session.GetString(key);
        if (encrypted == null)
            return default;
        
        try
        {
            var json = _protector.Unprotect(encrypted);
            return JsonSerializer.Deserialize<T>(json);
        }
        catch
        {
            session.Remove(key);
            return default;
        }
    }
}
```
