## Introduction

Brute force attacks attempt to guess passwords through automated trial and error. Without protection, attackers can try thousands of password combinations per second. Effective brute force protection combines multiple strategies: rate limiting, account lockout, CAPTCHA, and monitoring. The goal is to make attacks impractically slow while minimizing impact on legitimate users.

### Rate Limiting Implementation

```csharp
// IP-based rate limiting for login attempts
public class LoginRateLimitMiddleware
{
    private readonly RequestDelegate _next;
    private readonly IMemoryCache _cache;
    private readonly ILogger<LoginRateLimitMiddleware> _logger;
    
    // Rate limit: 5 attempts per 15 minutes per IP
    private const int MaxAttempts = 5;
    private static readonly TimeSpan LockoutDuration = TimeSpan.FromMinutes(15);
    
    public async Task InvokeAsync(HttpContext context)
    {
        // Only apply to login endpoint
        if (context.Request.Path.StartsWithSegments("/Account/Login") &&
            context.Request.Method == "POST")
        {
            var ipAddress = context.Connection.RemoteIpAddress?.ToString();
            var key = $"login_attempts:{ipAddress}";
            
            var attempts = _cache.Get<LoginAttempts>(key) ?? new LoginAttempts();
            
            if (attempts.Count >= MaxAttempts)
            {
                var timeRemaining = attempts.LockoutEnd - DateTime.UtcNow;
                
                if (timeRemaining > TimeSpan.Zero)
                {
                    context.Response.StatusCode = 429; // Too Many Requests
                    await context.Response.WriteAsJsonAsync(new
                    {
                        error = "Too many login attempts",
                        retryAfter = (int)timeRemaining.TotalSeconds
                    });
                    
                    _logger.LogWarning(
                        $"Rate limit exceeded for IP {ipAddress}. " +
                        $"{timeRemaining.TotalMinutes:F1} minutes remaining");
                    
                    return;
                }
                
                // Lockout expired, reset counter
                attempts = new LoginAttempts();
            }
            
            // Increment attempt counter
            attempts.Count++;
            attempts.LockoutEnd = DateTime.UtcNow.Add(LockoutDuration);
            
            _cache.Set(key, attempts, LockoutDuration);
        }
        
        await _next(context);
    }
}

public class LoginAttempts
{
    public int Count { get; set; }
    public DateTime LockoutEnd { get; set; }
}

// Register middleware
app.UseMiddleware<LoginRateLimitMiddleware>();
```

### Account Lockout Configuration

```csharp
// Configure ASP.NET Core Identity lockout
public class Startup
{
    public void ConfigureServices(IServiceCollection services)
    {
        services.AddIdentity<ApplicationUser, IdentityRole>(options =>
        {
            // Lockout settings
            options.Lockout.DefaultLockoutTimeSpan = TimeSpan.FromMinutes(15);
            options.Lockout.MaxFailedAccessAttempts = 5;
            options.Lockout.AllowedForNewUsers = true;
        })
        .AddEntityFrameworkStores<ApplicationDbContext>()
        .AddDefaultTokenProviders();
    }
}

// Login with lockout
[HttpPost]
public async Task<IActionResult> Login(LoginViewModel model)
{
    if (!ModelState.IsValid)
        return View(model);
    
    // lockoutOnFailure: true enables account lockout
    var result = await _signInManager.PasswordSignInAsync(
        model.Email,
        model.Password,
        model.RememberMe,
        lockoutOnFailure: true);
    
    if (result.Succeeded)
    {
        // Clear failed attempt counter on successful login
        var user = await _userManager.FindByEmailAsync(model.Email);
        await _userManager.ResetAccessFailedCountAsync(user);
        
        _logger.LogInformation($"User {model.Email} logged in successfully");
        return RedirectToAction("Index", "Home");
    }
    
    if (result.IsLockedOut)
    {
        var user = await _userManager.FindByEmailAsync(model.Email);
        var lockoutEnd = await _userManager.GetLockoutEndDateAsync(user);
        
        _logger.LogWarning($"User {model.Email} account locked until {lockoutEnd}");
        
        return View("Lockout", new LockoutViewModel
        {
            LockoutEnd = lockoutEnd?.UtcDateTime
        });
    }
    
    if (result.RequiresTwoFactor)
    {
        return RedirectToAction("LoginWith2fa", new
        {
            ReturnUrl = model.ReturnUrl,
            RememberMe = model.RememberMe
        });
    }
    
    // Failed login attempt
    var failedUser = await _userManager.FindByEmailAsync(model.Email);
    if (failedUser != null)
    {
        var failedCount = await _userManager.GetAccessFailedCountAsync(failedUser);
        
        _logger.LogWarning(
            $"Failed login attempt for {model.Email}. " +
            $"Attempt {failedCount + 1} of {_userManager.Options.Lockout.MaxFailedAccessAttempts}");
    }
    
    ModelState.AddModelError("", "Invalid login attempt");
    return View(model);
}
```

### Progressive Delay

```csharp
// Implement progressive delays after failed attempts
public class ProgressiveDelayService
{
    private readonly IMemoryCache _cache;
    
    public async Task ApplyDelayAsync(string identifier)
    {
        var key = $"delay:{identifier}";
        var failedAttempts = _cache.Get<int>(key);
        
        // Progressive delay: 2^attempts seconds (capped at 60 seconds)
        var delaySeconds = Math.Min(Math.Pow(2, failedAttempts), 60);
        
        await Task.Delay(TimeSpan.FromSeconds(delaySeconds));
        
        // Increment counter
        _cache.Set(key, failedAttempts + 1, TimeSpan.FromHours(1));
    }
    
    public void ResetDelay(string identifier)
    {
        var key = $"delay:{identifier}";
        _cache.Remove(key);
    }
}

// Usage in login
[HttpPost]
public async Task<IActionResult> Login(LoginViewModel model)
{
    var identifier = $"{model.Email}:{Request.HttpContext.Connection.RemoteIpAddress}";
    
    var result = await _signInManager.PasswordSignInAsync(
        model.Email,
        model.Password,
        model.RememberMe,
        lockoutOnFailure: true);
    
    if (result.Succeeded)
    {
        _progressiveDelayService.ResetDelay(identifier);
        return RedirectToAction("Index", "Home");
    }
    
    // Apply progressive delay on failure
    await _progressiveDelayService.ApplyDelayAsync(identifier);
    
    ModelState.AddModelError("", "Invalid login attempt");
    return View(model);
}

// Delay progression:
// 1st attempt: 2 seconds
// 2nd attempt: 4 seconds
// 3rd attempt: 8 seconds
// 4th attempt: 16 seconds
// 5th attempt: 32 seconds
// 6th+ attempts: 60 seconds (cap)
```

### CAPTCHA Integration

```csharp
// Add Google reCAPTCHA after failed attempts
public class CaptchaService
{
    private readonly HttpClient _httpClient;
    private readonly string _secretKey;
    
    public CaptchaService(HttpClient httpClient, IConfiguration config)
    {
        _httpClient = httpClient;
        _secretKey = config["ReCaptcha:SecretKey"];
    }
    
    public async Task<bool> VerifyTokenAsync(string token, string ipAddress)
    {
        var response = await _httpClient.PostAsync(
            "https://www.google.com/recaptcha/api/siteverify",
            new FormUrlEncodedContent(new Dictionary<string, string>
            {
                ["secret"] = _secretKey,
                ["response"] = token,
                ["remoteip"] = ipAddress
            }));
        
        var result = await response.Content.ReadFromJsonAsync<ReCaptchaResponse>();
        
        return result?.Success ?? false;
    }
}

// Login with CAPTCHA requirement
[HttpPost]
public async Task<IActionResult> Login(LoginViewModel model)
{
    if (!ModelState.IsValid)
        return View(model);
    
    // Check if CAPTCHA is required
    var ipAddress = HttpContext.Connection.RemoteIpAddress?.ToString();
    var failedAttempts = await GetFailedAttemptsAsync(ipAddress);
    
    if (failedAttempts >= 3)
    {
        // Require CAPTCHA after 3 failed attempts
        if (string.IsNullOrEmpty(model.CaptchaToken))
        {
            ModelState.AddModelError("", "Please complete the CAPTCHA");
            return View(model);

            model.CaptchaToken,
            ipAddress);

        {
            ModelState.AddModelError("", "CAPTCHA validation failed");
            return View(model);
        }
    }
    
    // Proceed with normal login
    var result = await _signInManager.PasswordSignInAsync(
        model.Email,
        model.Password,
        model.RememberMe,
        lockoutOnFailure: true);
    
    // ... rest of login logic
}
```

### Distributed Rate Limiting with Redis

```csharp
// Scale rate limiting across multiple servers
public class DistributedRateLimiter
{
    private readonly IConnectionMultiplexer _redis;
    private readonly ILogger<DistributedRateLimiter> _logger;
    
    public async Task<bool> IsAllowedAsync(
        string key,
        int maxAttempts,
        TimeSpan window)
    {
        var db = _redis.GetDatabase();
        var now = DateTimeOffset.UtcNow.ToUnixTimeSeconds();
        var windowStart = now - (long)window.TotalSeconds;
        
        // Use Redis sorted set to track attempts
        var transaction = db.CreateTransaction();
        
        // Remove old attempts outside the window
        var removeTask = transaction.SortedSetRemoveRangeByScoreAsync(
            key,
            double.NegativeInfinity,
            windowStart);
        
        // Add current attempt
        var addTask = transaction.SortedSetAddAsync(key, now, now);
        
        // Set expiration
        var expireTask = transaction.KeyExpireAsync(key, window);
        
        await transaction.ExecuteAsync();
        
        // Count attempts in current window
        var count = await db.SortedSetLengthAsync(key);
        
        if (count > maxAttempts)
        {
            _logger.LogWarning($"Rate limit exceeded for key: {key}");
            return false;
        }
        
        return true;
    }
}

// Usage
public class LoginController : Controller
{
    [HttpPost]
    public async Task<IActionResult> Login(LoginViewModel model)
    {
        var rateLimitKey = $"login:{model.Email}:{HttpContext.Connection.RemoteIpAddress}";
        
        var allowed = await _rateLimiter.IsAllowedAsync(
            rateLimitKey,
            maxAttempts: 5,
            window: TimeSpan.FromMinutes(15));
        
        if (!allowed)
        {
            return StatusCode(429, new { error = "Too many login attempts" });
        }
        
        // Proceed with login...
    }
}
```

### Monitoring and Alerting

```csharp
// Monitor for suspicious patterns
public class LoginMonitoringService
{
    private readonly ILogger<LoginMonitoringService> _logger;
    private readonly INotificationService _notifications;
    
    public async Task RecordFailedLoginAsync(
        string email,
        string ipAddress,
        string userAgent)
    {
        var attempt = new FailedLoginAttempt
        {
            Email = email,
            IpAddress = ipAddress,
            UserAgent = userAgent,
            Timestamp = DateTime.UtcNow
        };
        
        await _repository.SaveAsync(attempt);
        
        // Check for suspicious patterns
        await DetectBruteForcePatternAsync(email, ipAddress);
    }
    
    private async Task DetectBruteForcePatternAsync(string email, string ipAddress)
    {
        var recentAttempts = await _repository.GetRecentAttemptsAsync(
            email,
            TimeSpan.FromMinutes(30));
        
        // Alert on high volume of failures
        if (recentAttempts.Count >= 20)
        {
            await _notifications.SendSecurityAlertAsync(
                $"Possible brute force attack detected. " +
                $"Email: {email}, IP: {ipAddress}, " +
                $"Attempts: {recentAttempts.Count}");
        }
        
        // Check for distributed attack (multiple IPs)
        var uniqueIps = recentAttempts.Select(a => a.IpAddress).Distinct().Count();
        if (uniqueIps >= 10)
        {
            await _notifications.SendSecurityAlertAsync(
                $"Distributed brute force attack detected. " +
                $"Email: {email}, Unique IPs: {uniqueIps}");
        }
    }
}
```
