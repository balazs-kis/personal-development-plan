# Lesson 2: Security Misconfiguration

## Learning Objectives

By the end of this lesson, you will be able to:
- Identify common security misconfigurations in applications
- Configure secure defaults for web applications and APIs
- Implement security headers and harden application settings
- Understand why misconfiguration is the #2 vulnerability in 2025

---

## Introduction

Security misconfiguration has jumped to #2 in the OWASP Top 10:2025, with 100% of tested applications showing some form of misconfiguration. This category encompasses insecure default settings, incomplete configurations, exposed sensitive information, and missing security hardening. As applications become more configurable, the attack surface from misconfiguration grows larger.

## Key Concepts Summary

### Why Misconfiguration is #2

**Statistics:**
- 100% of tested applications have some misconfiguration
- 3.00% average incidence rate
- 719,000+ occurrences across CWEs
- Moving from #5 (2021) to #2 (2025)

**Root cause:** Modern software is highly configurable, creating more opportunities for misconfigurations.

### Common Misconfigurations

```csharp
// 1. BAD: Default credentials still enabled
// appsettings.json
{
  "AdminAccount": {
    "Username": "admin",
    "Password": "admin123" // Default password!
  }
}

// GOOD: Force password change on first login
public class DefaultPasswordService
{
    public async Task<bool> RequiresPasswordChange(User user)
    {
        return user.PasswordLastChanged == user.CreatedDate;
    }
}

// 2. BAD: Detailed error messages in production
public class Startup
{
    public void Configure(IApplicationBuilder app, IWebHostEnvironment env)
    {
        if (env.IsDevelopment())
        {
            app.UseDeveloperExceptionPage(); // OK in dev
        }
        else
        {
            app.UseDeveloperExceptionPage(); // NEVER in production!
        }
    }
}

// GOOD: Generic errors in production
public class Startup
{
    public void Configure(IApplicationBuilder app, IWebHostEnvironment env)
    {
        if (env.IsDevelopment())
        {
            app.UseDeveloperExceptionPage();
        }
        else
        {
            app.UseExceptionHandler("/error");
            app.UseHsts();
        }
    }
}

// 3. BAD: Missing security headers
// No response headers configured

// GOOD: Security headers configured
public class SecurityHeadersMiddleware
{
    public async Task InvokeAsync(HttpContext context)
    {
        // Prevent clickjacking
        context.Response.Headers.Add("X-Frame-Options", "DENY");
        
        // Prevent MIME sniffing
        context.Response.Headers.Add("X-Content-Type-Options", "nosniff");
        
        // Enable XSS protection
        context.Response.Headers.Add("X-XSS-Protection", "1; mode=block");
        
        // Content Security Policy
        context.Response.Headers.Add("Content-Security-Policy", 
            "default-src 'self'; script-src 'self'; style-src 'self'");
        
        // Strict Transport Security
        context.Response.Headers.Add("Strict-Transport-Security", 
            "max-age=31536000; includeSubDomains");
        
        // Referrer Policy
        context.Response.Headers.Add("Referrer-Policy", "strict-origin-when-cross-origin");
        
        // Permissions Policy
        context.Response.Headers.Add("Permissions-Policy", 
            "geolocation=(), microphone=(), camera=()");
        
        await _next(context);
    }
}
```

### Secure Configuration Examples

```csharp
// Application Configuration
public class Startup
{
    public void ConfigureServices(IServiceCollection services)
    {
        services.AddControllers(options =>
        {
            // Require HTTPS
            options.Filters.Add(new RequireHttpsAttribute());
            
            // Set reasonable limits
            options.MaxModelBindingCollectionSize = 100;
        });
        
        // Configure CORS properly (not wide open)
        services.AddCors(options =>
        {
            options.AddPolicy("RestrictedPolicy", builder =>
            {
                builder.WithOrigins("https://myapp.com")
                       .WithMethods("GET", "POST")
                       .AllowCredentials();
            });
        });
        
        // Remove server header
        services.Configure<KestrelServerOptions>(options =>
        {
            options.AddServerHeader = false;
        });
        
        // Disable automatic model state validation
        services.Configure<ApiBehaviorOptions>(options =>
        {
            options.SuppressModelStateInvalidFilter = true;
        });
    }
    
    public void Configure(IApplicationBuilder app, IWebHostEnvironment env)
    {
        // Security headers
        app.UseSecurityHeaders();
        
        // HTTPS redirection
        app.UseHttpsRedirection();
        
        // HSTS
        if (!env.IsDevelopment())
        {
            app.UseHsts();
        }
        
        // Enable CORS with restricted policy
        app.UseCors("RestrictedPolicy");
        
        // Authentication before authorization
        app.UseAuthentication();
        app.UseAuthorization();
    }
}
```

### Database Configuration Security

```csharp
// appsettings.json - BAD
{
  "ConnectionStrings": {
    "DefaultConnection": "Server=localhost;Database=MyDb;User Id=sa;Password=Admin123;"
    // Using 'sa' account with excessive privileges!
  }
}

// appsettings.json - GOOD
{
  "ConnectionStrings": {
    "DefaultConnection": "Server=localhost;Database=MyDb;User Id=app_readonly;Password=<use-secrets-manager>;"
    // Dedicated account with minimal privileges
  }
}

// Program.cs - Use secrets manager
public class Program
{
    public static void Main(string[] args)
    {
        CreateHostBuilder(args).Build().Run();
    }

    public static IHostBuilder CreateHostBuilder(string[] args) =>
        Host.CreateDefaultBuilder(args)
            .ConfigureAppConfiguration((context, config) =>
            {
                if (!context.HostingEnvironment.IsDevelopment())
                {
                    // Use Azure Key Vault in production
                    var builtConfig = config.Build();
                    var keyVaultEndpoint = builtConfig["KeyVault:Endpoint"];
                    
                    if (!string.IsNullOrEmpty(keyVaultEndpoint))
                    {
                        config.AddAzureKeyVault(
                            new Uri(keyVaultEndpoint),
                            new DefaultAzureCredential());
                    }
                }
            })
            .ConfigureWebHostDefaults(webBuilder =>
            {
                webBuilder.UseStartup<Startup>();
            });
}
```

### Removing Unnecessary Features

```csharp
// Disable unnecessary HTTP methods
public class DisableHttpMethodsMiddleware
{
    private readonly HashSet<string> _allowedMethods = new()
    {
        "GET", "POST", "PUT", "DELETE"
    };
    
    public async Task InvokeAsync(HttpContext context)
    {
        if (!_allowedMethods.Contains(context.Request.Method))
        {
            context.Response.StatusCode = 405; // Method Not Allowed
            await context.Response.WriteAsync("Method not allowed");
            return;
        }
        
        await _next(context);
    }
}

// Remove sensitive files from deployment
// .csproj
<ItemGroup>
  <Content Remove=".git\**" />
  <Content Remove=".env" />
  <Content Remove="*.md" />
  <Content Remove="docs\**" />
</ItemGroup>

// Disable directory browsing
public class Startup
{
    public void Configure(IApplicationBuilder app)
    {
        // DON'T do this:
        // app.UseDirectoryBrowser();
        
        // DO limit static files
        app.UseStaticFiles(new StaticFileOptions
        {
            ServeUnknownFileTypes = false,
            FileProvider = new PhysicalFileProvider(
                Path.Combine(Directory.GetCurrentDirectory(), "wwwroot", "public")),
            RequestPath = "/public"
        });
    }
}
```

### Infrastructure as Code (Secure Defaults)

```yaml
# docker-compose.yml - GOOD security practices
version: '3.8'
services:
  webapp:
    image: myapp:latest
    restart: unless-stopped
    read_only: true  # Filesystem is read-only
    security_opt:
      - no-new-privileges:true  # Prevent privilege escalation
    cap_drop:
      - ALL  # Drop all capabilities
    cap_add:
      - NET_BIND_SERVICE  # Only add what's needed
    user: "1000:1000"  # Run as non-root user
    environment:
      - ASPNETCORE_ENVIRONMENT=Production
      - ASPNETCORE_URLS=https://+:443
    secrets:
      - db_password
    networks:
      - backend

secrets:
  db_password:
    external: true

networks:
  backend:
    internal: true  # No external access
```

### Configuration Checklist

```csharp
// Automated configuration verification
public class SecurityConfigurationValidator
{
    public List<string> ValidateConfiguration(IConfiguration config)
    {
        var errors = new List<string>();
        
        // Check HTTPS is enforced
        var urls = config["ASPNETCORE_URLS"];
        if (!urls?.Contains("https://") ?? true)
        {
            errors.Add("HTTPS not configured");
        }
        
        // Check for default secrets
        var dbPassword = config["ConnectionStrings:DefaultConnection"];
        if (dbPassword?.Contains("Password=admin") ?? false)
        {
            errors.Add("Default database password detected");
        }
        
        // Check environment
        var environment = config["ASPNETCORE_ENVIRONMENT"];
        if (environment == "Development" && IsProduction())
        {
            errors.Add("Development mode enabled in production");
        }
        
        // Check for exposed secrets
        var jwtSecret = config["Jwt:Secret"];
        if (jwtSecret == "your-secret-key" || jwtSecret?.Length < 32)
        {
            errors.Add("Weak or default JWT secret");
        }
        
        return errors;
    }
}
```
