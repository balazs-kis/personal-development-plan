## Introduction

Injection attacks occur when untrusted data is sent to an interpreter as part of a command or query. Attackers can trick the interpreter into executing unintended commands or accessing unauthorized data. While injection has dropped to #5 in the 2025 OWASP Top 10 (from #3 in 2021), it remains one of the most tested categories and can lead to complete system compromise.

## Key Concepts Summary

### How SQL Injection Works

```csharp
// VULNERABLE CODE
public class UserRepository
{
    public User GetUser(string username, string password)
    {
        // Building SQL with string concatenation - DANGEROUS!
        var sql = $"SELECT * FROM Users WHERE Username = '{username}' AND Password = '{password}'";
        
        using var connection = new SqlConnection(_connectionString);
        using var command = new SqlCommand(sql, connection);
        
        connection.Open();
        using var reader = command.ExecuteReader();
        
        if (reader.Read())
        {
            return new User
            {
                Id = reader.GetInt32(0),
                Username = reader.GetString(1)
            };
        }
        
        return null;
    }
}

// ATTACK SCENARIO
// Attacker inputs:
// username: admin' --
// password: (anything)

// Resulting SQL:
// SELECT * FROM Users WHERE Username = 'admin' --' AND Password = '...'
// The -- comments out the rest, bypassing password check!

// WORSE ATTACK
// username: admin'; DROP TABLE Users; --
// password: (anything)

// Resulting SQL:
// SELECT * FROM Users WHERE Username = 'admin'; DROP TABLE Users; --' AND Password = '...'
// This deletes the entire Users table!
```

### Preventing SQL Injection: Parameterized Queries

```csharp
// SECURE: Using parameterized queries
public class SecureUserRepository
{
    public async Task<User> GetUserAsync(string username, string password)
    {
        // Use parameters - SQL and data are separated
        var sql = "SELECT Id, Username FROM Users WHERE Username = @Username AND Password = @Password";
        
        await using var connection = new SqlConnection(_connectionString);
        await using var command = new SqlCommand(sql, connection);
        
        // Add parameters - values are treated as data, not SQL code
        command.Parameters.AddWithValue("@Username", username);
        command.Parameters.AddWithValue("@Password", password);
        
        await connection.OpenAsync();
        await using var reader = await command.ExecuteReaderAsync();
        
        if (await reader.ReadAsync())
        {
            return new User
            {
                Id = reader.GetInt32(0),
                Username = reader.GetString(1)
            };
        }
        
        return null;
    }
}

// With Entity Framework Core (even safer)
public class EFUserRepository
{
    private readonly AppDbContext _context;
    
    public async Task<User> GetUserAsync(string username, string password)
    {
        // EF Core uses parameterization by default
        return await _context.Users
            .FirstOrDefaultAsync(u => 
                u.Username == username && 
                u.Password == password);
    }
}

// With Dapper (parameterized by default)
public class DapperUserRepository
{
    public async Task<User> GetUserAsync(string username, string password)
    {
        await using var connection = new SqlConnection(_connectionString);
        
        // Dapper automatically parameterizes
        return await connection.QueryFirstOrDefaultAsync<User>(
            "SELECT Id, Username FROM Users WHERE Username = @Username AND Password = @Password",
            new { Username = username, Password = password });
    }
}
```

### Stored Procedures (Additional Layer)

```sql
-- Create stored procedure
CREATE PROCEDURE GetUser
    @Username NVARCHAR(100),
    @Password NVARCHAR(100)
AS
BEGIN
    SELECT Id, Username
    FROM Users
    WHERE Username = @Username AND PasswordHash = @Password;
END
```

### Command Injection

```csharp
// VULNERABLE: Command injection
public class FileProcessor
{
    public string ProcessFile(string filename)
    {
        // DANGEROUS: User input passed to shell
        var process = new Process
        {
            StartInfo = new ProcessStartInfo
            {
                FileName = "cmd.exe",
                Arguments = $"/c type {filename}",
                RedirectStandardOutput = true,
                UseShellExecute = false
            }
        };
        
        process.Start();
        return process.StandardOutput.ReadToEnd();
    }
}

// ATTACK:
// filename: "file.txt & del /s /q C:\*"
// This would delete all files!

// SECURE: Avoid shell commands, use .NET APIs
public class SecureFileProcessor
{
    private readonly string _allowedDirectory;
    
    public SecureFileProcessor(string allowedDirectory)
    {
        _allowedDirectory = allowedDirectory;
    }
    
    public async Task<string> ProcessFileAsync(string filename)
    {
        // 1. Validate filename (no path traversal)
        if (filename.Contains("..") || 
            filename.Contains("/") || 
            filename.Contains("\\"))
        {
            throw new ArgumentException("Invalid filename");
        }
        
        // 2. Use whitelist of allowed extensions
        var extension = Path.GetExtension(filename).ToLowerInvariant();
        var allowedExtensions = new[] { ".txt", ".csv", ".log" };
        
        if (!allowedExtensions.Contains(extension))
        {
            throw new ArgumentException("File type not allowed");
        }
        
        // 3. Construct safe path
        var safePath = Path.Combine(_allowedDirectory, filename);
        
        // 4. Verify path is within allowed directory
        var fullPath = Path.GetFullPath(safePath);
        if (!fullPath.StartsWith(_allowedDirectory))
        {
            throw new ArgumentException("Path traversal detected");
        }
        
        // 5. Use .NET file API (no shell execution)
        if (!File.Exists(fullPath))
        {
            throw new FileNotFoundException();
        }
        
        return await File.ReadAllTextAsync(fullPath);
    }
}
```

### LDAP Injection

```csharp
// VULNERABLE: LDAP injection
public class LdapAuthentication
{
    public bool Authenticate(string username, string password)
    {
        // Building LDAP filter with string concatenation
        var filter = $"(&(uid={username})(userPassword={password}))";
        
        // Attacker input: username = "*)(uid=*"
        // Resulting filter: (&(uid=*)(uid=*)(userPassword=...))
        // This bypasses authentication!
        
        return PerformLdapSearch(filter);
    }
}

// SECURE: Escape LDAP special characters
public class SecureLdapAuthentication
{
    public bool Authenticate(string username, string password)
    {
        // Escape special LDAP characters
        var safeUsername = EscapeLdapString(username);
        var safePassword = EscapeLdapString(password);
        
        var filter = $"(&(uid={safeUsername})(userPassword={safePassword}))";
        
        return PerformLdapSearch(filter);
    }
    
    private string EscapeLdapString(string input)
    {
        if (string.IsNullOrEmpty(input))
            return input;
        
        return input
            .Replace("\\", "\\5c")
            .Replace("*", "\\2a")
            .Replace("(", "\\28")
            .Replace(")", "\\29")
            .Replace("\0", "\\00");
    }
}
```

### Input Validation

```csharp
// Comprehensive input validation
public class InputValidator
{
    // Whitelist validation (preferred)
    public bool IsValidUsername(string username)
    {
        // Only allow alphanumeric and underscore
        return Regex.IsMatch(username, @"^[a-zA-Z0-9_]{3,20}$");
    }
    
    // Blacklist (less secure, use whitelist when possible)
    public string SanitizeInput(string input)
    {
        if (string.IsNullOrEmpty(input))
            return input;
        
        // Remove dangerous characters
        var dangerous = new[] { "'", "\"", ";", "--", "/*", "*/", "xp_", "sp_" };
        
        foreach (var pattern in dangerous)
        {
            input = input.Replace(pattern, "");
        }
        
        return input;
    }
    
    // Length validation
    public bool IsValidLength(string input, int maxLength = 100)
    {
        return input?.Length <= maxLength;
    }
    
    // Type validation
    public bool IsValidInteger(string input, out int value)
    {
        return int.TryParse(input, out value);
    }
}

// Usage in API controller
[ApiController]
[Route("api/[controller]")]
public class UsersController : ControllerBase
{
    [HttpGet("{userId}")]
    public async Task<IActionResult> GetUser(string userId)
    {
        // Validate input is actually an integer
        if (!int.TryParse(userId, out int id))
        {
            return BadRequest("Invalid user ID format");
        }
        
        // Validate range
        if (id <= 0)
        {
            return BadRequest("User ID must be positive");
        }
        
        // Now safe to use
        var user = await _userService.GetUserAsync(id);
        return Ok(user);
    }
}
```
