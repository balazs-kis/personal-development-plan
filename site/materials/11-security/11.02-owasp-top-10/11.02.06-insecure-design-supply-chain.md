## Introduction

This lesson covers two critical OWASP Top 10 categories: Insecure Design (#6) and Software Supply Chain Failures (#3). Insecure Design represents architectural and design flaws that cannot be fixed through perfect implementation alone. Software Supply Chain Failures addresses vulnerabilities from third-party dependencies, build tools, and the entire development pipeline—a growing threat with the highest exploit scores among OWASP categories.

## Key Concepts Summary

### Insecure Design vs. Insecure Implementation

```csharp
// INSECURE IMPLEMENTATION (can be fixed with code changes)
public class OrderService
{
    public async Task<Order> GetOrderAsync(int orderId)
    {
        // Missing authorization check - implementation flaw
        return await _repository.GetOrderAsync(orderId);
    }
}

// FIXED IMPLEMENTATION
public class OrderService
{
    public async Task<Order> GetOrderAsync(int orderId, int userId)
    {
        var order = await _repository.GetOrderAsync(orderId);
        
        // Added authorization - implementation fixed
        if (order.UserId != userId)
            throw new UnauthorizedAccessException();
        
        return order;
    }
}

// INSECURE DESIGN (architectural flaw)
// System allows unlimited password reset attempts without rate limiting
// Even with perfect code, the DESIGN is flawed
public class PasswordResetService
{
    public async Task<bool> ResetPasswordAsync(string email, string token, string newPassword)
    {
        // No rate limiting in the design
        // Attacker can brute force reset tokens
        var isValid = await _tokenService.ValidateTokenAsync(email, token);
        
        if (!isValid)
            return false;
        
        await _userService.UpdatePasswordAsync(email, newPassword);
        return true;
    }
}

// SECURE DESIGN (architectural solution)
// Design includes rate limiting, token expiration, and lockout
public class SecurePasswordResetService
{
    public async Task<ResetResult> ResetPasswordAsync(string email, string token, string newPassword)
    {
        // Check rate limit (design-level control)
        var attempts = await _rateLimiter.GetAttemptsAsync(email);
        if (attempts > 5)
        {
            await _alertService.NotifySecurityTeamAsync($"Password reset abuse: {email}");
            return ResetResult.RateLimited("Too many attempts. Account locked.");
        }
        
        // Validate token with expiration (design-level control)
        var validationResult = await _tokenService.ValidateTokenAsync(email, token);
        
        if (!validationResult.IsValid)
        {
            await _rateLimiter.IncrementAttemptsAsync(email);
            
            if (validationResult.IsExpired)
                return ResetResult.Failure("Token expired");
            
            return ResetResult.Failure("Invalid token");
        }
        
        // Update password
        await _userService.UpdatePasswordAsync(email, newPassword);
        await _rateLimiter.ClearAttemptsAsync(email);
        
        return ResetResult.Success();
    }
}
```

### Secure Design Principles

```csharp
// 1. Defense in Depth - Multiple security layers
public class SecurePaymentProcessor
{
    public async Task<PaymentResult> ProcessPaymentAsync(PaymentRequest request)
    {
        // Layer 1: Input validation
        if (!_validator.IsValid(request))
            return PaymentResult.Invalid();
        
        // Layer 2: Authentication & Authorization
        if (!await _authService.CanMakePayment(request.UserId, request.Amount))
            return PaymentResult.Unauthorized();
        
        // Layer 3: Fraud detection
        if (await _fraudService.IsSuspicious(request))
        {
            await _alertService.NotifyAsync("Suspicious payment", request);
            return PaymentResult.Blocked();
        }
        
        // Layer 4: Transaction limits
        if (await _limitService.ExceedsLimit(request))
            return PaymentResult.LimitExceeded();
        
        // Layer 5: Secure communication
        var result = await _paymentGateway.ProcessAsync(request); // Uses TLS
        
        // Layer 6: Audit logging
        await _auditLog.LogPaymentAsync(request, result);
        
        return result;
    }
}

// 2. Least Privilege - Minimal necessary access
public class DatabaseConfiguration
{
    public void ConfigureReadService()
    {
        // Read service gets read-only database user
        services.AddDbContext<ReadDbContext>(options =>
            options.UseSqlServer(_config["Database:ReadOnlyConnection"]));
    }
    
    public void ConfigureWriteService()
    {
        // Write service gets limited write permissions
        services.AddDbContext<WriteDbContext>(options =>
            options.UseSqlServer(_config["Database:WriteConnection"]));
    }
}

// 3. Separation of Duties
public class ApprovalWorkflow
{
    public async Task<bool> ApproveTransferAsync(int transferId, int approverId)
    {
        var transfer = await _repository.GetTransferAsync(transferId);
        
        // Requester cannot approve their own transfer
        if (transfer.RequesterId == approverId)
        {
            throw new InvalidOperationException("Cannot approve own request");
        }
        
        // Requires approval from different role
        if (!await _authService.HasRole(approverId, "Approver"))
        {
            throw new UnauthorizedAccessException("User not authorized to approve");
        }
        
        transfer.Status = TransferStatus.Approved;
        transfer.ApproverId = approverId;
        
        await _repository.UpdateAsync(transfer);
        return true;
    }
}
```

### Software Supply Chain Failures

```csharp
// Problem: Vulnerable dependencies
// Example: Using old version with known CVE

// packages.config or .csproj (BAD - outdated version)
<PackageReference Include="Newtonsoft.Json" Version="10.0.1" />
// Version 10.0.1 has CVE-2018-12356 (high severity)

// GOOD: Update to patched version
<PackageReference Include="Newtonsoft.Json" Version="13.0.3" />

// Problem: Transitive dependencies
// You depend on Package A
// Package A depends on Package B (has vulnerability)
// You're affected even though you don't directly use Package B!

// Solution: Use dependency scanning tools
```

### Dependency Scanning and Management

```xml
<!-- Enable NuGet audit in .csproj -->
<Project Sdk="Microsoft.NET.Sdk.Web">
  <PropertyGroup>
    <TargetFramework>net8.0</TargetFramework>
    <NuGetAudit>true</NuGetAudit>
    <NuGetAuditLevel>low</NuGetAuditLevel>
    <NuGetAuditMode>all</NuGetAuditMode>
  </PropertyGroup>
</Project>
```

```bash
# Use dotnet list package to check for vulnerabilities
dotnet list package --vulnerable

# Output shows:
# Project 'MyApp' has the following vulnerable packages
#    [net8.0]: 
#    Top-level Package      Requested   Resolved   Severity   Advisory URL
#    > Newtonsoft.Json      10.0.1      10.0.1     High       https://github.com/advisories/...

# Update vulnerable packages
dotnet add package Newtonsoft.Json --version 13.0.3

# Check for outdated packages
dotnet list package --outdated
```

### Software Bill of Materials (SBOM)

```bash
# Generate SBOM for your application
dotnet tool install --global CycloneDX

# Create SBOM in CycloneDX format
dotnet CycloneDX MyApp.csproj -o sbom.xml

# SBOM includes:
# - All direct dependencies
# - All transitive dependencies
# - Version numbers
# - License information
# - Known vulnerabilities
```

```xml
<!-- Example SBOM snippet -->
<bom xmlns="http://cyclonedx.org/schema/bom/1.4">
  <components>
    <component type="library">
      <name>Newtonsoft.Json</name>
      <version>13.0.3</version>
      <licenses>
        <license>
          <id>MIT</id>
        </license>
      </licenses>
      <purl>pkg:nuget/Newtonsoft.Json@13.0.3</purl>
    </component>
  </components>
</bom>
```

### CI/CD Pipeline Security

```yaml
# GitHub Actions workflow with security scanning
name: Build and Security Scan

on: [push, pull_request]

jobs:
  security-scan:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      
      # Restore dependencies
      - name: Restore dependencies
        run: dotnet restore
      
      # Check for vulnerable packages
      - name: Vulnerability scan
        run: dotnet list package --vulnerable --include-transitive
      
      # Fail build if high severity vulnerabilities found
      - name: Check vulnerabilities
        run: |
          if dotnet list package --vulnerable --include-transitive | grep -i "high\|critical"; then
            echo "High or Critical vulnerabilities found!"
            exit 1
          fi
      
      # Run SAST (Static Application Security Testing)
      - name: Run Security Code Scan
        run: dotnet tool run security-scan MyApp.sln
      
      # Generate SBOM
      - name: Generate SBOM
        run: |
          dotnet tool install --global CycloneDX
          dotnet CycloneDX MyApp.csproj -o sbom.json
      
      # Upload SBOM as artifact
      - name: Upload SBOM
        uses: actions/upload-artifact@v3
        with:
          name: sbom
          path: sbom.json
```

### Secure Package Sources

```xml
<!-- NuGet.config - Control package sources -->
<configuration>
  <packageSources>
    <clear />
    <!-- Only allow official NuGet.org -->
    <add key="nuget.org" value="https://api.nuget.org/v3/index.json" protocolVersion="3" />
    
    <!-- Private feed with authentication -->
    <add key="internal" value="https://company.pkgs.visualstudio.com/_packaging/feed/nuget/v3/index.json" />
  </packageSources>
  
  <packageSourceCredentials>
    <internal>
      <add key="Username" value="token" />
      <add key="ClearTextPassword" value="%AZURE_ARTIFACTS_TOKEN%" />
    </internal>
  </packageSourceCredentials>
</configuration>
```
