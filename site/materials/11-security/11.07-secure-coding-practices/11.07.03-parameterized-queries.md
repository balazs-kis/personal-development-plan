---

## Introduction

SQL injection remains one of the most dangerous and common vulnerabilities in web applications. It occurs when untrusted input is concatenated directly into SQL queries, allowing attackers to manipulate database operations. This lesson teaches you how to prevent SQL injection through parameterized queries and safe ORM usage.

**Key principle:** Never build SQL queries by concatenating strings. Always separate SQL code from data using parameters.

## Key Concepts Summary

### How SQL Injection Works

```csharp
// VULNERABLE CODE - NEVER DO THIS
public User GetUser(string username)
{
    var sql = "SELECT * FROM Users WHERE Username = '" + username + "'";
    // If username is: admin' OR '1'='1
    // Query becomes: SELECT * FROM Users WHERE Username = 'admin' OR '1'='1'
    // This returns ALL users!
    
    return ExecuteQuery(sql);
}

// Worse - allows data modification:
// If username is: '; DROP TABLE Users; --
// Query becomes: SELECT * FROM Users WHERE Username = ''; DROP TABLE Users; --'
// This deletes the entire Users table!
```

### Entity Framework LINQ (Safe by Default)

```csharp
// SAFE: Entity Framework automatically parameterizes LINQ queries
public class UserRepository
{
    private readonly AppDbContext _context;
    
    public User GetUserByUsername(string username)
    {
        // This is automatically parameterized - completely safe
        return _context.Users
            .FirstOrDefault(u => u.Username == username);
    }
    
    public List<Product> SearchProducts(string searchTerm, decimal minPrice)
    {
        // Complex queries are also safe with LINQ
        return _context.Products
            .Where(p => p.Name.Contains(searchTerm) && p.Price >= minPrice)
            .ToList();
    }
}
```

### Safe Raw SQL with FromSql

```csharp
// SAFE: Using FromSql with parameters
public List<User> GetUsersByRole(string role)
{
    return _context.Users
        .FromSql($"SELECT * FROM Users WHERE Role = {role}")
        .ToList();
    
    // EF Core automatically parameterizes interpolated values
    // Generated SQL: SELECT * FROM Users WHERE Role = @p0
}

// SAFE: Explicit parameter syntax
public List<User> GetActiveUsers(DateTime since)
{
    return _context.Users
        .FromSql($"SELECT * FROM Users WHERE LastLogin > {since} AND IsActive = {true}")
        .ToList();
}
```

### Dangerous Raw SQL (FromSqlRaw)

```csharp
// DANGEROUS: String concatenation with FromSqlRaw
public List<User> SearchUsers_UNSAFE(string searchTerm)
{
    // VULNERABLE TO SQL INJECTION!
    var sql = $"SELECT * FROM Users WHERE Username LIKE '%{searchTerm}%'";
    return _context.Users.FromSqlRaw(sql).ToList();
}

// SAFE: Using FromSqlRaw with explicit parameters
public List<User> SearchUsers_SAFE(string searchTerm)
{
    var sql = "SELECT * FROM Users WHERE Username LIKE '%' + @p0 + '%'";
    return _context.Users
        .FromSqlRaw(sql, searchTerm)
        .ToList();
}
```

### ADO.NET Parameterized Queries

```csharp
using System.Data.SqlClient;

public class UserRepository
{
    private readonly string _connectionString;
    
    public User GetUser(string username, string email)
    {
        // SAFE: Using SqlParameter
        using var connection = new SqlConnection(_connectionString);
        using var command = new SqlCommand(
            "SELECT * FROM Users WHERE Username = @username AND Email = @email",
            connection);
        
        // Add parameters - values are treated as data, not SQL code
        command.Parameters.AddWithValue("@username", username);
        command.Parameters.AddWithValue("@email", email);
        
        connection.Open();
        using var reader = command.ExecuteReader();
        
        if (reader.Read())
        {
            return new User
            {
                Id = reader.GetInt32(0),
                Username = reader.GetString(1),
                Email = reader.GetString(2)
            };
        }
        
        return null;
    }
}
```

### Using SqlParameter with Proper Types

```csharp
public void UpdateUser(int userId, string username, DateTime lastLogin)
{
    using var connection = new SqlConnection(_connectionString);
    using var command = new SqlCommand(
        "UPDATE Users SET Username = @username, LastLogin = @lastLogin WHERE Id = @id",
        connection);
    
    // Specify parameter types explicitly for better performance and security
    command.Parameters.Add("@id", SqlDbType.Int).Value = userId;
    command.Parameters.Add("@username", SqlDbType.NVarChar, 50).Value = username;
    command.Parameters.Add("@lastLogin", SqlDbType.DateTime).Value = lastLogin;
    
    connection.Open();
    command.ExecuteNonQuery();
}
```

### Stored Procedures (Also Safe)

```csharp
public List<User> GetUsersByDepartment(string department)
{
    using var connection = new SqlConnection(_connectionString);
    using var command = new SqlCommand("GetUsersByDepartment", connection);
    command.CommandType = CommandType.StoredProcedure;
    
    // Parameters in stored procedures are also safe
    command.Parameters.AddWithValue("@Department", department);
    
    connection.Open();
    using var reader = command.ExecuteReader();
    
    var users = new List<User>();
    while (reader.Read())
    {
        users.Add(MapUser(reader));
    }
    
    return users;
}
```

### Dynamic Queries (Be Careful)

```csharp
// If you MUST build dynamic queries, validate column names against allowlist
public List<Product> GetSortedProducts(string sortColumn, string sortDirection)
{
    // Validate against allowlist - parameters can't be used for column names
    var allowedColumns = new[] { "Name", "Price", "CreatedDate" };
    var allowedDirections = new[] { "ASC", "DESC" };
    
    if (!allowedColumns.Contains(sortColumn) || 
        !allowedDirections.Contains(sortDirection.ToUpper()))
    {
        throw new ArgumentException("Invalid sort parameters");
    }
    
    // Now safe to use in query (still use parameters for filter values)
    var sql = $"SELECT * FROM Products WHERE Price > @minPrice ORDER BY {sortColumn} {sortDirection}";
    
    return _context.Products
        .FromSqlRaw(sql, new SqlParameter("@minPrice", 0))
        .ToList();
}
```

### Entity Framework ExecuteSqlRaw

```csharp
// For non-query commands (INSERT, UPDATE, DELETE)
public int DeactivateOldUsers(DateTime cutoffDate)
{
    // SAFE: Using parameters with ExecuteSqlRaw
    return _context.Database.ExecuteSqlRaw(
        "UPDATE Users SET IsActive = 0 WHERE LastLogin < @p0",
        cutoffDate
    );
}

// UNSAFE: String concatenation
public int DeleteUser_UNSAFE(int userId)
{
    // VULNERABLE!
    return _context.Database.ExecuteSqlRaw(
        $"DELETE FROM Users WHERE Id = {userId}"
    );
}
```
