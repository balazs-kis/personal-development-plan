## Introduction

HashiCorp Vault is a popular open-source secret management solution that provides centralized secret storage, dynamic secrets, encryption as a service, and detailed audit logging. While Azure Key Vault is great for Azure-native applications, Vault works across any cloud or on-premises environment. This lesson covers how to integrate Vault with .NET applications and when to choose different secret stores.

**Key principle:** Use Vault Agent in production for automatic authentication and secret renewal. Don't manage Vault tokens directly in application code.

## Key Concepts Summary

### HashiCorp Vault Architecture

```
┌─────────────────────────────────────────┐
│           Vault Server                  │
│  ┌──────────┐  ┌──────────────┐       │
│  │ Secrets  │  │   Dynamic     │       │
│  │ Engine   │  │   Secrets     │       │
│  └──────────┘  └──────────────┘       │
│  ┌──────────┐  ┌──────────────┐       │
│  │   Auth   │  │    Audit      │       │
│  │ Methods  │  │    Logging    │       │
│  └──────────┘  └──────────────┘       │
└─────────────────────────────────────────┘
         ↑                    ↑
         │                    │
    ┌────┴────┐          ┌───┴────┐
    │   App   │          │  Vault │
    │         │          │  Agent │
    └─────────┘          └────────┘

Key Components:
- Secrets Engine: Stores and generates secrets
- Auth Methods: Authenticate clients (AppRole, Kubernetes, etc.)
- Dynamic Secrets: Generate credentials on-demand
- Audit Logging: Track all access to secrets
```

```bash

sudo apt update && sudo apt install vault

# Verify installation
vault --version

# Start Vault in dev mode (NOT for production!)
vault server -dev

# In another terminal, set environment variables
export VAULT_ADDR='http://127.0.0.1:8200'
export VAULT_TOKEN='root'  # Dev mode root token

# Store a secret
vault kv put secret/myapp/config username="admin" password="secret123"

# Retrieve a secret
vault kv get secret/myapp/config
```

### Using VaultSharp Library

```bash
# Install VaultSharp NuGet package
dotnet add package VaultSharp
```

```csharp
using VaultSharp;
using VaultSharp.V1.AuthMethods.Token;
using VaultSharp.V1.Commons;

public class VaultService
{
    private readonly IVaultClient _vaultClient;
    
    public VaultService(IConfiguration configuration)
    {
        var vaultUrl = configuration["Vault:Url"];
        var vaultToken = configuration["Vault:Token"];
        
        // Create auth method
        IAuthMethodInfo authMethod = new TokenAuthMethodInfo(vaultToken);
        
        // Create Vault settings
        var vaultClientSettings = new VaultClientSettings(vaultUrl, authMethod);
        
        // Initialize client
        _vaultClient = new VaultClient(vaultClientSettings);
    }
    
    public async Task<string> GetSecretAsync(string path, string key)
    {
        // Read secret from KV v2 engine
        Secret<SecretData> secret = await _vaultClient.V1.Secrets.KeyValue.V2
            .ReadSecretAsync(path: path, mountPoint: "secret");
        
        return secret.Data.Data[key].ToString();
    }
    
    public async Task<Dictionary<string, object>> GetAllSecretsAsync(string path)
    {
        Secret<SecretData> secret = await _vaultClient.V1.Secrets.KeyValue.V2
            .ReadSecretAsync(path: path, mountPoint: "secret");
        
        return secret.Data.Data;
    }
    
    public async Task WriteSecretAsync(string path, Dictionary<string, object> secrets)
    {
        await _vaultClient.V1.Secrets.KeyValue.V2.WriteSecretAsync(
            path: path,
            data: secrets,
            mountPoint: "secret");
    }
}

// Usage
var vaultService = new VaultService(configuration);
var dbPassword = await vaultService.GetSecretAsync("myapp/config", "password");
```

### AppRole Authentication (Production)

```csharp
// AppRole is recommended for applications in production
using VaultSharp.V1.AuthMethods.AppRole;

public class VaultService
{
    private readonly IVaultClient _vaultClient;

    {
        var vaultUrl = configuration["Vault:Url"];
        var roleId = configuration["Vault:RoleId"];
        var secretId = configuration["Vault:SecretId"];
        
        // Create AppRole auth method

        var vaultClientSettings = new VaultClientSettings(vaultUrl, authMethod);
        _vaultClient = new VaultClient(vaultClientSettings);
```

### Dynamic Database Credentials

```csharp
// Vault can generate database credentials on-demand
public class DynamicDatabaseService
{
    private readonly IVaultClient _vaultClient;
    
    public async Task<DatabaseCredentials> GetDatabaseCredentialsAsync()
    {
        // Vault generates temporary credentials
        Secret<UsernamePasswordCredentials> creds = 
            await _vaultClient.V1.Secrets.Database.GetCredentialsAsync(
                roleName: "myapp-role",
                mountPoint: "database");
        
        return new DatabaseCredentials
        {
            Username = creds.Data.Username,
            Password = creds.Data.Password,
            LeaseDuration = creds.LeaseDurationSeconds
        };
    }
}

public class DatabaseCredentials
{
    public string Username { get; set; }
    public string Password { get; set; }
    public int LeaseDuration { get; set; }
}

// Usage - credentials are automatically rotated
var creds = await dynamicDbService.GetDatabaseCredentialsAsync();
var connectionString = $"Server=db.example.com;Username={creds.Username};Password={creds.Password}";
```

### Using Vault Agent

```hcl
# vault-agent-config.hcl
pid_file = "./pidfile"

vault {
  address = "http://127.0.0.1:8200"
}

auto_auth {
  method {
    type = "approle"
    
    config = {
      role_id_file_path = "/path/to/role-id"
      secret_id_file_path = "/path/to/secret-id"
    }
  }
  
  sink {
    type = "file"
    config = {
      path = "/tmp/vault-token"
    }
  }
}

template {
  source      = "/path/to/template.ctmpl"
  destination = "/path/to/config.json"
  perms       = "0644"
}
```

```bash
# template.ctmpl - Consul Template format
{
  "database": {
    {{- with secret "secret/data/myapp/config" }}
    "host": "{{ .Data.data.db_host }}",
    "username": "{{ .Data.data.db_username }}",
    "password": "{{ .Data.data.db_password }}"
    {{- end }}
  },
  "api_keys": {
    {{- with secret "secret/data/myapp/api-keys" }}
    "stripe": "{{ .Data.data.stripe }}",
    "sendgrid": "{{ .Data.data.sendgrid }}"
    {{- end }}
  }
}
```

```bash
# Run Vault Agent
vault agent -config=vault-agent-config.hcl

# Agent will:
# 1. Authenticate with Vault
# 2. Fetch secrets
# 3. Render template to config.json
# 4. Watch for changes and re-render
```

### Comparing Secret Store Solutions

```csharp
/*
┌─────────────────────┬──────────────────┬─────────────────┬────────────────┐
│ Feature             │ Azure Key Vault  │ HashiCorp Vault │ AWS Secrets    │
│                     │                  │                 │ Manager        │
├─────────────────────┼──────────────────┼─────────────────┼────────────────┤
│ Cloud Native        │ Azure            │ Any/Hybrid      │ AWS            │
│ Dynamic Secrets     │ No               │ Yes             │ Limited        │
│ Encryption as Svc   │ Yes              │ Yes             │ Yes            │
│ Secret Versioning   │ Yes              │ Yes             │ Yes            │
│ Audit Logging       │ Yes              │ Yes             │ Yes            │
│ Managed Identity    │ Yes              │ No              │ Yes (IAM)      │
│ Open Source         │ No               │ Yes (OSS)       │ No             │
│ Self-Hosted         │ No               │ Yes             │ No             │
│ Multi-Cloud         │ Limited          │ Yes             │ Limited        │
│ Certificate Mgmt    │ Yes              │ Yes (PKI)       │ Yes            │
│ Learning Curve      │ Easy             │ Moderate        │ Easy           │
│ Cost                │ Pay per op       │ OSS/Enterprise  │ Pay per secret │
└─────────────────────┴──────────────────┴─────────────────┴────────────────┘

Choose Azure Key Vault when:
✅ Running primarily on Azure
✅ Want simple integration with Azure services
✅ Need managed certificates
✅ Prefer fully managed service

Choose HashiCorp Vault when:
✅ Multi-cloud or hybrid environment
✅ Need dynamic secrets generation
✅ Want encryption as a service
✅ Require self-hosted option
✅ Complex secret workflows

Choose AWS Secrets Manager when:
✅ Running primarily on AWS
✅ Want simple integration with AWS services
✅ Need automatic secret rotation for RDS
*/
```

### Best Practices for Vault Integration

```csharp
// ✅ DO: Use AppRole for application authentication
// Not tokens or user credentials

// ✅ DO: Use Vault Agent in production
// Let Agent handle authentication and token renewal

// ✅ DO: Use dynamic secrets when possible
var creds = await vault.GetDatabaseCredentialsAsync("myapp-role");

// ✅ DO: Implement lease renewal
public async Task RenewLeaseAsync(string leaseId)
{
    await _vaultClient.V1.System.RenewLeaseAsync(leaseId);
}

// ✅ DO: Handle Vault unavailability gracefully
try
{
    var secret = await vaultService.GetSecretAsync("path", "key");
}
catch (VaultApiException ex)
{
    _logger.LogError(ex, "Failed to retrieve secret from Vault");
    // Fall back to cached value or fail gracefully
}

// ❌ DON'T: Store Vault tokens in code or config
// Use AppRole or Kubernetes auth instead

// ❌ DON'T: Use root token in production
// Create specific policies and roles

// ❌ DON'T: Ignore lease expiration
// Renew leases or fetch new credentials

// ❌ DON'T: Bypass Vault Agent for token management
// Let Agent handle complexity
```

### Kubernetes Integration

```yaml
# Kubernetes deployment with Vault Agent sidecar
apiVersion: apps/v1
kind: Deployment
metadata:
  name: myapp
spec:
  template:
    metadata:
      annotations:
        vault.hashicorp.com/agent-inject: "true"
        vault.hashicorp.com/role: "myapp"
        vault.hashicorp.com/agent-inject-secret-config: "secret/data/myapp/config"
        vault.hashicorp.com/agent-inject-template-config: |
          {{- with secret "secret/data/myapp/config" -}}
          {
            "database": {
              "host": "{{ .Data.data.db_host }}",
              "password": "{{ .Data.data.db_password }}"
            }
          }
          {{- end }}
    spec:
      serviceAccountName: myapp
      containers:
      - name: myapp
        image: myapp:latest
        env:
        - name: CONFIG_FILE
          value: /vault/secrets/config
```

### Complete Example Application

```csharp
// Program.cs
var builder = WebApplication.CreateBuilder(args);

// Register Vault service
builder.Services.AddSingleton<VaultService>();

// Option 1: Direct integration
// builder.Services.AddSingleton<IVaultClient>(sp =>
// {
//     var config = sp.GetRequiredService<IConfiguration>();
//     return VaultClientFactory.CreateClient(config);
// });

// Option 2: Use config rendered by Vault Agent
if (File.Exists("/vault/secrets/config.json"))
{
    builder.Configuration.AddJsonFile(
        "/vault/secrets/config.json",
        optional: false,
        reloadOnChange: true);
}

builder.Services.AddControllers();

var app = builder.Build();

app.MapControllers();
app.Run();

// Controller
[ApiController]
[Route("api/[controller]")]
public class DatabaseController : ControllerBase
{
    private readonly VaultService _vaultService;
    
    public DatabaseController(VaultService vaultService)
    {
        _vaultService = vaultService;
    }
    
    [HttpGet("connection")]
    public async Task<IActionResult> GetConnectionString()
    {
        try
        {
            var secrets = await _vaultService.GetAllSecretsAsync("myapp/database");
            var host = secrets["host"];
            var username = secrets["username"];
            var password = secrets["password"];
            
            // Don't return actual password in response!
            return Ok(new
            {
                host = host,
                username = username,
                passwordSet = !string.IsNullOrEmpty(password?.ToString())
            });
        }
        catch (Exception ex)
        {
            return StatusCode(500, new { error = ex.Message });
        }
    }
}
```
