## Introduction

Asymmetric encryption uses a pair of mathematically related keys: a public key for encryption and a private key for decryption. This solves the key distribution problem of symmetric encryption but is much slower and can only encrypt small amounts of data. RSA is the most widely used asymmetric algorithm. This lesson teaches you when and how to use RSA effectively.

**Key principle:** Use RSA to encrypt keys, not data. For encrypting actual data, combine RSA with AES (hybrid encryption).

## Key Concepts Summary

### How Asymmetric Encryption Works

```
Alice wants to send Bob an encrypted message:
1. Bob generates a key pair (public key + private key)
2. Bob shares his PUBLIC key with Alice (can be sent insecurely)
3. Alice encrypts the message using Bob's PUBLIC key
4. Alice sends the encrypted message to Bob
5. Bob decrypts using his PRIVATE key (which only he has)

Key insight: Anyone can encrypt with the public key, but only 
the private key holder can decrypt.
```

### Generating RSA Key Pairs

```csharp
using System.Security.Cryptography;

// Generate a new RSA key pair
using var rsa = RSA.Create();
rsa.KeySize = 2048; // Minimum recommended: 2048 bits

// Export keys
string publicKey = rsa.ToXmlString(includePrivateParameters: false);
string privateKey = rsa.ToXmlString(includePrivateParameters: true);

// Alternative: Export in PEM format (more portable)
string publicKeyPem = rsa.ExportRSAPublicKeyPem();
string privateKeyPem = rsa.ExportRSAPrivateKeyPem();

Console.WriteLine($"Public Key:\n{publicKey}\n");
Console.WriteLine($"Private Key (keep secret!):\n{privateKey}");
```

### Basic RSA Encryption and Decryption

```csharp
public class RsaEncryptionService
{
    public static byte[] Encrypt(byte[] data, string publicKeyXml)
    {
        using var rsa = RSA.Create();
        rsa.FromXmlString(publicKeyXml);
        
        // Use OAEP padding (more secure than PKCS1)
        return rsa.Encrypt(data, RSAEncryptionPadding.OaepSHA256);
    }
    
    public static byte[] Decrypt(byte[] encryptedData, string privateKeyXml)
    {
        using var rsa = RSA.Create();
        rsa.FromXmlString(privateKeyXml);
        
        return rsa.Decrypt(encryptedData, RSAEncryptionPadding.OaepSHA256);
    }
}

// Usage
using var rsa = RSA.Create(2048);
string publicKey = rsa.ToXmlString(false);
string privateKey = rsa.ToXmlString(true);

string message = "Secret message";
byte[] messageBytes = Encoding.UTF8.GetBytes(message);

byte[] encrypted = RsaEncryptionService.Encrypt(messageBytes, publicKey);
byte[] decrypted = RsaEncryptionService.Decrypt(encrypted, privateKey);

string decryptedMessage = Encoding.UTF8.GetString(decrypted);
Console.WriteLine($"Decrypted: {decryptedMessage}");
```

### RSA Size Limitations

```csharp
// RSA can only encrypt data smaller than the key size minus padding
// For 2048-bit RSA with OAEP-SHA256 padding:
// Maximum data size ≈ (2048 / 8) - 2 - (2 * 32) = 190 bytes

public static int GetMaxDataSize(int keySize)
{
    // For OAEP-SHA256 padding
    return (keySize / 8) - 2 - (2 * 32);
}

// Example: 2048-bit key
int maxSize = GetMaxDataSize(2048); // ~190 bytes
Console.WriteLine($"Max data size for RSA-2048: {maxSize} bytes");

// ❌ BAD: Trying to encrypt large data
string largeData = new string('A', 500); // 500 bytes
// This will throw CryptographicException!
```

### Hybrid Encryption: RSA + AES

```csharp
public class HybridEncryption
{
    // Encrypt large data using AES, then encrypt AES key with RSA
    public static (byte[] encryptedData, byte[] encryptedKey, byte[] iv) 
        Encrypt(byte[] data, string rsaPublicKey)
    {
        // 1. Generate random AES key
        using var aes = Aes.Create();
        aes.GenerateKey();
        aes.GenerateIV();
        
        // 2. Encrypt data with AES
        byte[] encryptedData;
        using (var encryptor = aes.CreateEncryptor())
        using (var ms = new MemoryStream())
        using (var cs = new CryptoStream(ms, encryptor, CryptoStreamMode.Write))
        {
            cs.Write(data, 0, data.Length);
            cs.FlushFinalBlock();
            encryptedData = ms.ToArray();
        }
        
        // 3. Encrypt AES key with RSA
        using var rsa = RSA.Create();
        rsa.FromXmlString(rsaPublicKey);
        byte[] encryptedKey = rsa.Encrypt(aes.Key, RSAEncryptionPadding.OaepSHA256);
        
        return (encryptedData, encryptedKey, aes.IV);
    }
    
    // Decrypt: Use RSA to decrypt AES key, then decrypt data with AES
    public static byte[] Decrypt(
        byte[] encryptedData, 
        byte[] encryptedKey, 
        byte[] iv, 
        string rsaPrivateKey)
    {
        // 1. Decrypt AES key using RSA
        using var rsa = RSA.Create();
        rsa.FromXmlString(rsaPrivateKey);
        byte[] aesKey = rsa.Decrypt(encryptedKey, RSAEncryptionPadding.OaepSHA256);
        
        // 2. Decrypt data using AES
        using var aes = Aes.Create();
        aes.Key = aesKey;
        aes.IV = iv;
        
        using var decryptor = aes.CreateDecryptor();
        using var ms = new MemoryStream();
        using (var cs = new CryptoStream(ms, decryptor, CryptoStreamMode.Write))
        {
            cs.Write(encryptedData, 0, encryptedData.Length);
            cs.FlushFinalBlock();
        }
        
        return ms.ToArray();
    }
}

// Usage
using var rsa = RSA.Create(2048);
string publicKey = rsa.ToXmlString(false);
string privateKey = rsa.ToXmlString(true);

string largeMessage = "This can be megabytes of data...";
byte[] data = Encoding.UTF8.GetBytes(largeMessage);

// Encrypt
var (encryptedData, encryptedKey, iv) = HybridEncryption.Encrypt(data, publicKey);

// Decrypt
byte[] decrypted = HybridEncryption.Decrypt(encryptedData, encryptedKey, iv, privateKey);
string decryptedMessage = Encoding.UTF8.GetString(decrypted);
```

### Key Storage and Management

```csharp
public class KeyManagementService
{
    // Store keys securely (example using file system - use Key Vault in production)
    public static void SaveKeys(string publicKey, string privateKey, string keyName)
    {
        // In production, store private keys in:
        // - Azure Key Vault
        // - AWS KMS
        // - Hardware Security Module (HSM)
        // - Windows Certificate Store
        
        // For demo purposes only:
        File.WriteAllText($"{keyName}_public.xml", publicKey);
        
        // Encrypt private key before storing!
        File.WriteAllText($"{keyName}_private.xml", privateKey);
    }
    
    public static (string publicKey, string privateKey) LoadKeys(string keyName)
    {
        string publicKey = File.ReadAllText($"{keyName}_public.xml");
        string privateKey = File.ReadAllText($"{keyName}_private.xml");
        return (publicKey, privateKey);
    }
}
```

### Using PEM Format (Cross-Platform)

```csharp
// PEM format is more portable than XML (works with OpenSSL, other languages)
public class PemKeyManagement
{
    public static (string publicPem, string privatePem) GenerateKeyPair()
    {
        using var rsa = RSA.Create(2048);
        
        string publicPem = rsa.ExportRSAPublicKeyPem();
        string privatePem = rsa.ExportRSAPrivateKeyPem();
        
        return (publicPem, privatePem);
    }
    
    public static byte[] EncryptWithPem(byte[] data, string publicKeyPem)
    {
        using var rsa = RSA.Create();
        rsa.ImportFromPem(publicKeyPem);
        
        return rsa.Encrypt(data, RSAEncryptionPadding.OaepSHA256);
    }
    
    public static byte[] DecryptWithPem(byte[] encryptedData, string privateKeyPem)
    {
        using var rsa = RSA.Create();
        rsa.ImportFromPem(privateKeyPem);
        
        return rsa.Decrypt(encryptedData, RSAEncryptionPadding.OaepSHA256);
    }
}
```

### RSA Padding Modes

```csharp
// ✅ RECOMMENDED: OAEP with SHA-256 (most secure)
byte[] encrypted = rsa.Encrypt(data, RSAEncryptionPadding.OaepSHA256);

// ✅ ACCEPTABLE: OAEP with SHA-1 (for legacy compatibility)
byte[] encrypted = rsa.Encrypt(data, RSAEncryptionPadding.OaepSHA1);

// ⚠️ LEGACY ONLY: PKCS1 padding (older, less secure)
byte[] encrypted = rsa.Encrypt(data, RSAEncryptionPadding.Pkcs1);

// ❌ NEVER: No padding
// byte[] encrypted = rsa.Encrypt(data, null); // INSECURE!
```

### When to Use RSA vs AES

```csharp
// ✅ Use RSA for:
// - Encrypting symmetric keys (hybrid encryption)
// - Digital signatures (covered in next lesson)
// - Key exchange protocols
// - Small amounts of data (< 190 bytes for 2048-bit key)

// ✅ Use AES for:
// - Encrypting files and large data
// - Database encryption
// - Disk encryption
// - Any bulk data encryption

// ✅ Use BOTH (hybrid encryption) for:
// - Encrypting large data that needs to be sent to multiple recipients
// - Secure messaging systems
// - Email encryption (like PGP/GPG)
// - API payload encryption
```

### Real-World Example: Secure Message Exchange

```csharp
public class SecureMessageService
{
    private readonly string _recipientPublicKey;
    private readonly string _myPrivateKey;
    
    public SecureMessageService(string recipientPublicKey, string myPrivateKey)
    {
        _recipientPublicKey = recipientPublicKey;
        _myPrivateKey = myPrivateKey;
    }
    
    public EncryptedMessage SendMessage(string message)
    {
        byte[] messageBytes = Encoding.UTF8.GetBytes(message);
        
        // Use hybrid encryption
        var (encryptedData, encryptedKey, iv) = 
            HybridEncryption.Encrypt(messageBytes, _recipientPublicKey);
        
        return new EncryptedMessage
        {
            Data = Convert.ToBase64String(encryptedData),
            EncryptedKey = Convert.ToBase64String(encryptedKey),
            IV = Convert.ToBase64String(iv)
        };
    }
    
    public string ReceiveMessage(EncryptedMessage encrypted)
    {
        byte[] encryptedData = Convert.FromBase64String(encrypted.Data);
        byte[] encryptedKey = Convert.FromBase64String(encrypted.EncryptedKey);
        byte[] iv = Convert.FromBase64String(encrypted.IV);
        
        byte[] decrypted = HybridEncryption.Decrypt(
            encryptedData, encryptedKey, iv, _myPrivateKey);
        
        return Encoding.UTF8.GetString(decrypted);
    }
}

public class EncryptedMessage
{
    public string Data { get; set; }
    public string EncryptedKey { get; set; }
    public string IV { get; set; }
}
```

### Common Pitfalls

```csharp
// ❌ BAD: Using RSA to encrypt large data directly
var largeData = File.ReadAllBytes("large-file.pdf");
// var encrypted = rsa.Encrypt(largeData, ...); // Will fail!

// ✅ GOOD: Use hybrid encryption
var (encrypted, key, iv) = HybridEncryption.Encrypt(largeData, publicKey);

// ❌ BAD: Reusing the same key pair for encryption and signing
// Use separate key pairs for different purposes

// ❌ BAD: Not protecting the private key
File.WriteAllText("private.key", privateKey); // INSECURE!

// ✅ GOOD: Store in secure key vault
await keyVault.SetSecretAsync("rsa-private-key", privateKey);

// ❌ BAD: Using weak key sizes
using var rsa = RSA.Create(1024); // TOO WEAK!

// ✅ GOOD: Use at least 2048 bits
using var rsa = RSA.Create(2048);
```
