# Lesson 1: Testing Message Consumers with MassTransit

## Learning Objectives

By the end of this lesson, you will be able to:
- Understand the MassTransit Test Harness architecture
- Write unit tests for message consumers
- Test message publishing and consumption
- Verify request/response patterns
- Use consumer test doubles effectively

---

## Introduction

Testing message consumers is challenging because they depend on message brokers and asynchronous communication. The MassTransit Test Harness provides an in-memory transport that simplifies testing without requiring external dependencies.

### Setting Up the Test Harness

```csharp
using MassTransit;
using MassTransit.Testing;
using Microsoft.Extensions.DependencyInjection;
using Xunit;

public class OrderConsumerTests
{
    [Fact]
    public async Task Consume_ValidOrder_PublishesOrderCreated()
    {
        // Arrange
        await using var provider = new ServiceCollection()
            .AddMassTransitTestHarness(cfg =>
            {
                cfg.AddConsumer<CreateOrderConsumer>();
            })
            .BuildServiceProvider(true);

        var harness = await provider.StartTestHarness();

        try
        {
            var orderId = NewId.NextGuid();
            
            // Act
            await harness.Bus.Publish(new CreateOrder
            {
                OrderId = orderId,
                CustomerId = Guid.NewGuid(),
                Items = new[] { "Item1", "Item2" }
            });

            // Assert
            Assert.True(await harness.Consumed.Any<CreateOrder>());
            Assert.True(await harness.Published.Any<OrderCreated>(
                x => x.Context.Message.OrderId == orderId));
        }
        finally
        {
            await harness.Stop();
        }
    }
}
```

### Basic Consumer Example

```csharp
public record CreateOrder
{
    public Guid OrderId { get; init; }
    public Guid CustomerId { get; init; }
    public string[] Items { get; init; }
}

public record OrderCreated
{
    public Guid OrderId { get; init; }
    public DateTime CreatedAt { get; init; }
}

public class CreateOrderConsumer : IConsumer<CreateOrder>
{
    private readonly IOrderRepository _repository;

    public CreateOrderConsumer(IOrderRepository repository)
    {
        _repository = repository;
    }

    public async Task Consume(ConsumeContext<CreateOrder> context)
    {
        var order = new Order
        {
            Id = context.Message.OrderId,
            CustomerId = context.Message.CustomerId,
            Items = context.Message.Items,
            CreatedAt = DateTime.UtcNow
        };

        await _repository.SaveAsync(order);

        await context.Publish(new OrderCreated
        {
            OrderId = order.Id,
            CreatedAt = order.CreatedAt
        });
    }
}
```

### Testing Consumer with Dependencies

```csharp
public class OrderConsumerTests
{
    [Fact]
    public async Task Consume_ValidOrder_SavesToRepository()
    {
        // Arrange
        var repository = new InMemoryOrderRepository();
        
        await using var provider = new ServiceCollection()
            .AddMassTransitTestHarness(cfg =>
            {
                cfg.AddConsumer<CreateOrderConsumer>();
            })
            .AddSingleton<IOrderRepository>(repository)
            .BuildServiceProvider(true);

        var harness = await provider.StartTestHarness();

        try
        {
            var orderId = NewId.NextGuid();
            var customerId = Guid.NewGuid();
            
            // Act
            await harness.Bus.Publish(new CreateOrder
            {
                OrderId = orderId,
                CustomerId = customerId,
                Items = new[] { "Item1" }
            });

            // Wait for consumer to complete
            Assert.True(await harness.Consumed.Any<CreateOrder>());

            // Assert
            var savedOrder = await repository.GetByIdAsync(orderId);
            Assert.NotNull(savedOrder);
            Assert.Equal(customerId, savedOrder.CustomerId);
            Assert.Single(savedOrder.Items);
        }
        finally
        {
            await harness.Stop();
        }
    }
}
```

### Using Consumer Harness

```csharp
[Fact]
public async Task Consume_ValidOrder_ConsumerReceivesMessage()
{
    await using var provider = new ServiceCollection()
        .AddMassTransitTestHarness(cfg =>
        {
            cfg.AddConsumer<CreateOrderConsumer>();
        })
        .AddSingleton<IOrderRepository>(new InMemoryOrderRepository())
        .BuildServiceProvider(true);

    var harness = await provider.StartTestHarness();
    
    try
    {
        // Get consumer-specific harness
        var consumerHarness = harness.GetConsumerHarness<CreateOrderConsumer>();

        var orderId = NewId.NextGuid();
        
        await harness.Bus.Publish(new CreateOrder
        {
            OrderId = orderId,
            CustomerId = Guid.NewGuid(),
            Items = new[] { "Item1" }
        });

        // Assert using consumer harness
        Assert.True(await consumerHarness.Consumed.Any<CreateOrder>());
        
        // Verify specific message
        var consumed = consumerHarness.Consumed.Select<CreateOrder>().First();
        Assert.Equal(orderId, consumed.Context.Message.OrderId);
    }
    finally
    {
        await harness.Stop();
    }
}
```

### Testing Request/Response Patterns

```csharp
public record CheckOrderStatus
{
    public Guid OrderId { get; init; }
}

public record OrderStatus
{
    public Guid OrderId { get; init; }
    public string Status { get; init; }
}

public class CheckOrderStatusConsumer : IConsumer<CheckOrderStatus>
{
    private readonly IOrderRepository _repository;

    public CheckOrderStatusConsumer(IOrderRepository repository)
    {
        _repository = repository;
    }

    public async Task Consume(ConsumeContext<CheckOrderStatus> context)
    {
        var order = await _repository.GetByIdAsync(context.Message.OrderId);
        
        await context.RespondAsync(new OrderStatus
        {
            OrderId = order.Id,
            Status = order.Status
        });
    }
}

[Fact]
public async Task Request_ExistingOrder_ReturnsStatus()
{
    // Arrange
    var orderId = Guid.NewGuid();
    var repository = new InMemoryOrderRepository();
    await repository.SaveAsync(new Order 
    { 
        Id = orderId, 
        Status = "Shipped" 
    });

    await using var provider = new ServiceCollection()
        .AddMassTransitTestHarness(cfg =>
        {
            cfg.AddConsumer<CheckOrderStatusConsumer>();
        })
        .AddSingleton<IOrderRepository>(repository)
        .BuildServiceProvider(true);

    var harness = await provider.StartTestHarness();

    try
    {
        // Act
        var client = harness.GetRequestClient<CheckOrderStatus>();
        var response = await client.GetResponse<OrderStatus>(new CheckOrderStatus
        {
            OrderId = orderId
        });

        // Assert
        Assert.Equal(orderId, response.Message.OrderId);
        Assert.Equal("Shipped", response.Message.Status);
    }
    finally
    {
        await harness.Stop();
    }
}
```

### Testing with Multiple Consumers

```csharp
public class OrderCreatedNotificationConsumer : IConsumer<OrderCreated>
{
    private readonly INotificationService _notificationService;

    public OrderCreatedNotificationConsumer(INotificationService notificationService)
    {
        _notificationService = notificationService;
    }

    public async Task Consume(ConsumeContext<OrderCreated> context)
    {
        await _notificationService.SendAsync(
            $"Order {context.Message.OrderId} was created");
    }
}

[Fact]
public async Task OrderCreated_TriggersNotification()
{
    var notificationService = new FakeNotificationService();

    await using var provider = new ServiceCollection()
        .AddMassTransitTestHarness(cfg =>
        {
            cfg.AddConsumer<CreateOrderConsumer>();
            cfg.AddConsumer<OrderCreatedNotificationConsumer>();
        })
        .AddSingleton<IOrderRepository>(new InMemoryOrderRepository())
        .AddSingleton<INotificationService>(notificationService)
        .BuildServiceProvider(true);

    var harness = await provider.StartTestHarness();

    try
    {
        var orderId = NewId.NextGuid();
        
        // Act - Publish CreateOrder
        await harness.Bus.Publish(new CreateOrder
        {
            OrderId = orderId,
            CustomerId = Guid.NewGuid(),
            Items = new[] { "Item1" }
        });

        // Wait for both consumers
        Assert.True(await harness.Consumed.Any<CreateOrder>());
        Assert.True(await harness.Consumed.Any<OrderCreated>());

        // Assert notification was sent
        Assert.Single(notificationService.SentNotifications);
        Assert.Contains(orderId.ToString(), 
            notificationService.SentNotifications[0]);
    }
    finally
    {
        await harness.Stop();
    }
}
```

### Best Practices

**1. Use IAsyncLifetime for Setup/Teardown:**
```csharp
public class OrderConsumerTests : IAsyncLifetime
{
    private ServiceProvider _provider;
    private ITestHarness _harness;

    public async Task InitializeAsync()
    {
        _provider = new ServiceCollection()
            .AddMassTransitTestHarness(cfg =>
            {
                cfg.AddConsumer<CreateOrderConsumer>();
            })
            .AddSingleton<IOrderRepository>(new InMemoryOrderRepository())
            .BuildServiceProvider(true);

        _harness = await _provider.StartTestHarness();
    }

    public async Task DisposeAsync()
    {
        await _harness.Stop();
        await _provider.DisposeAsync();
    }

    [Fact]
    public async Task Test1()
    {
        // Test using _harness
    }

    [Fact]
    public async Task Test2()
    {
        // Each test gets clean harness
    }
}
```

**2. Use Test-Specific Message IDs:**
```csharp
// ✓ Good - Traceable IDs
var orderId = NewId.NextGuid();
await harness.Bus.Publish(new CreateOrder { OrderId = orderId });

// ✗ Bad - Hard to debug
await harness.Bus.Publish(new CreateOrder { OrderId = Guid.NewGuid() });
```

**3. Wait for Consumption Before Assertions:**
```csharp
// ✓ Good - Ensure message was consumed
await harness.Bus.Publish(message);
Assert.True(await harness.Consumed.Any<CreateOrder>());
// Now safe to check side effects

// ✗ Bad - Race condition
await harness.Bus.Publish(message);
var order = await repository.GetByIdAsync(orderId); // May not be saved yet!
```

**4. Use In-Memory Implementations:**
```csharp
// ✓ Good - Fast, isolated
.AddSingleton<IOrderRepository>(new InMemoryOrderRepository())

// ✗ Bad - Slow, shared state
.AddDbContext<OrderContext>(options => 
    options.UseSqlServer(connectionString))
```

**5. Test One Concern Per Test:**
```csharp
// ✓ Good - Tests message consumption
[Fact]
public async Task Consume_ValidOrder_ConsumesMessage() { }

// ✓ Good - Tests event publishing
[Fact]
public async Task Consume_ValidOrder_PublishesOrderCreated() { }

// ✗ Bad - Tests too much
[Fact]
public async Task Consume_ValidOrder_DoesEverything() { }
```
