By the end of this lesson, you will be able to:

Verifying that events are published correctly is crucial in event-driven architectures. This lesson covers techniques for testing event publication, from unit tests with mocks to integration tests with event collectors.

## Key Concepts Summary

### Event Publication Testing Levels

```
Unit Tests
├── Mock event bus
├── Verify publish called
└── Fast, isolated

Integration Tests
├── Real event bus (in-memory)
├── Event listeners/collectors
└── Verify end-to-end flow

End-to-End Tests
├── Real message broker
├── Full system integration
└── Slow, comprehensive
```

### Testing with Event Collectors

**Event Collector Pattern:**
```csharp
public class EventCollector
{
    private readonly Channel<object> _channel;
    private readonly ChannelReader<object> _reader;
    private readonly ChannelWriter<object> _writer;

    public EventCollector()
    {
        _channel = Channel.CreateUnbounded<object>();
        _reader = _channel.Reader;
        _writer = _channel.Writer;
    }

    public void Collect<T>(T @event) where T : class
    {
        _writer.TryWrite(@event);
    }

    public async Task<T> WaitForEvent<T>(
        TimeSpan? timeout = null,
        CancellationToken cancellationToken = default) 
        where T : class
    {
        timeout ??= TimeSpan.FromSeconds(5);
        
        using var cts = CancellationTokenSource
            .CreateLinkedTokenSource(cancellationToken);
        cts.CancelAfter(timeout.Value);

        await foreach (var @event in _reader.ReadAllAsync(cts.Token))
        {
            if (@event is T typedEvent)
            {
                return typedEvent;
            }
        }

        throw new TimeoutException(
            $"Event of type {typeof(T).Name} was not published within {timeout}");
    }

    public async Task<List<T>> WaitForEvents<T>(
        int count,
        TimeSpan? timeout = null,
        CancellationToken cancellationToken = default) 
        where T : class
    {
        timeout ??= TimeSpan.FromSeconds(5);
        var events = new List<T>();
        
        using var cts = CancellationTokenSource
            .CreateLinkedTokenSource(cancellationToken);
        cts.CancelAfter(timeout.Value);

        await foreach (var @event in _reader.ReadAllAsync(cts.Token))
        {
            if (@event is T typedEvent)
            {
                events.Add(typedEvent);
                if (events.Count >= count)
                {
                    return events;
                }
            }
        }

        throw new TimeoutException(
            $"Expected {count} events of type {typeof(T).Name}, got {events.Count}");
    }
}
```

### Event Bus Decorator for Testing

```csharp
public interface IEventBus
{
    Task PublishAsync<T>(T @event, CancellationToken cancellationToken = default)
        where T : class;
}

public class EventCollectorDecorator : IEventBus
{
    private readonly IEventBus _inner;
    private readonly EventCollector _collector;

    public EventCollectorDecorator(IEventBus inner, EventCollector collector)
    {
        _inner = inner;
        _collector = collector;
    }

    public async Task PublishAsync<T>(T @event, CancellationToken cancellationToken = default)
        where T : class
    {
        // Collect event for testing
        _collector.Collect(@event);
        
        // Publish to real bus
        await _inner.PublishAsync(@event, cancellationToken);
    }
}
```

### Integration Test Setup

```csharp
public class OrderServiceTests : IAsyncLifetime
{
    private WebApplicationFactory<Program> _factory;
    private HttpClient _client;
    private EventCollector _eventCollector;

    public async Task InitializeAsync()
    {
        _eventCollector = new EventCollector();

        _factory = new WebApplicationFactory<Program>()
            .WithWebHostBuilder(builder =>
            {
                builder.ConfigureServices(services =>
                {
                    // Replace event bus with decorated version
                    var descriptor = services.SingleOrDefault(
                        d => d.ServiceType == typeof(IEventBus));
                    if (descriptor != null)
                    {
                        services.Remove(descriptor);
                    }

                    services.AddScoped<IEventBus>(sp =>
                    {
                        var original = new InMemoryEventBus();
                        return new EventCollectorDecorator(original, _eventCollector);
                    });
                });
            });

        _client = _factory.CreateClient();
    }

    public async Task DisposeAsync()
    {
        await _factory.DisposeAsync();
    }

    [Fact]
    public async Task CreateOrder_PublishesOrderCreatedEvent()
    {
        // Arrange
        var request = new CreateOrderRequest
        {
            CustomerId = Guid.NewGuid(),
            Items = new[] { "Item1", "Item2" }
        };

        // Act
        var response = await _client.PostAsJsonAsync("/api/orders", request);

        // Assert
        response.EnsureSuccessStatusCode();
        
        var orderCreated = await _eventCollector.WaitForEvent<OrderCreatedEvent>();
        
        Assert.NotNull(orderCreated);
        Assert.Equal(request.CustomerId, orderCreated.CustomerId);
        Assert.Equal(2, orderCreated.ItemCount);
    }
}
```

### Unit Testing with Mocks

```csharp
public class OrderService
{
    private readonly IOrderRepository _repository;
    private readonly IEventBus _eventBus;

    public OrderService(IOrderRepository repository, IEventBus eventBus)
    {
        _repository = repository;
        _eventBus = eventBus;
    }

    public async Task CreateOrderAsync(CreateOrderCommand command)
    {
        var order = new Order
        {
            Id = Guid.NewGuid(),
            CustomerId = command.CustomerId,
            Items = command.Items,
            CreatedAt = DateTime.UtcNow
        };

        await _repository.SaveAsync(order);

        await _eventBus.PublishAsync(new OrderCreatedEvent
        {
            OrderId = order.Id,
            CustomerId = order.CustomerId,
            ItemCount = order.Items.Count,
            CreatedAt = order.CreatedAt
        });
    }
}

// Unit test with mock
[Fact]
public async Task CreateOrder_PublishesCorrectEvent()
{
    // Arrange
    var repository = new Mock<IOrderRepository>();
    var eventBus = new Mock<IEventBus>();
    var service = new OrderService(repository.Object, eventBus.Object);

    var command = new CreateOrderCommand
    {
        CustomerId = Guid.NewGuid(),
        Items = new List<string> { "Item1", "Item2" }
    };

    // Act
    await service.CreateOrderAsync(command);

    // Assert
    eventBus.Verify(x => x.PublishAsync(
        It.Is<OrderCreatedEvent>(e =>
            e.CustomerId == command.CustomerId &&
            e.ItemCount == 2),
        It.IsAny<CancellationToken>()),
        Times.Once);
}
```

### Testing Event Handlers

```csharp
public record OrderCreatedEvent
{
    public Guid OrderId { get; init; }
    public Guid CustomerId { get; init; }
    public int ItemCount { get; init; }
    public DateTime CreatedAt { get; init; }
}

public class SendOrderConfirmationHandler : IEventHandler<OrderCreatedEvent>
{
    private readonly IEmailService _emailService;
    private readonly ICustomerRepository _customerRepository;

    public SendOrderConfirmationHandler(
        IEmailService emailService,
        ICustomerRepository customerRepository)
    {
        _emailService = emailService;
        _customerRepository = customerRepository;
    }

    public async Task HandleAsync(
        OrderCreatedEvent @event,
        CancellationToken cancellationToken = default)
    {
        var customer = await _customerRepository
            .GetByIdAsync(@event.CustomerId, cancellationToken);

        await _emailService.SendAsync(new Email
        {
            To = customer.Email,
            Subject = $"Order {@event.OrderId} Confirmation",
            Body = $"Your order with {event.ItemCount} items has been created."
        }, cancellationToken);
    }
}

// Test event handler
[Fact]
public async Task HandleOrderCreated_SendsConfirmationEmail()
{
    // Arrange
    var customerId = Guid.NewGuid();
    var orderId = Guid.NewGuid();

    var emailService = new Mock<IEmailService>();
    var customerRepository = new Mock<ICustomerRepository>();
    
    customerRepository
        .Setup(x => x.GetByIdAsync(customerId, It.IsAny<CancellationToken>()))
        .ReturnsAsync(new Customer
        {
            Id = customerId,
            Email = "customer@example.com"
        });

    var handler = new SendOrderConfirmationHandler(
        emailService.Object,
        customerRepository.Object);

    var @event = new OrderCreatedEvent
    {
        OrderId = orderId,
        CustomerId = customerId,
        ItemCount = 3,
        CreatedAt = DateTime.UtcNow
    };

    // Act
    await handler.HandleAsync(@event);

    // Assert
    emailService.Verify(x => x.SendAsync(
        It.Is<Email>(email =>
            email.To == "customer@example.com" &&
            email.Subject.Contains(orderId.ToString()) &&
            email.Body.Contains("3 items")),
        It.IsAny<CancellationToken>()),
        Times.Once);
}
```

### Testing Event Publication in Aggregates

```csharp
public abstract class AggregateRoot
{
    private readonly List<object> _domainEvents = new();
    public IReadOnlyList<object> DomainEvents => _domainEvents.AsReadOnly();

    protected void AddDomainEvent(object domainEvent)
    {
        _domainEvents.Add(domainEvent);
    }

    public void ClearDomainEvents()
    {
        _domainEvents.Clear();
    }
}

public class Order : AggregateRoot
{
    public Guid Id { get; private set; }
    public string Status { get; private set; }

    public static Order Create(Guid customerId, List<string> items)
    {
        var order = new Order
        {
            Id = Guid.NewGuid(),
            Status = "Created"
        };

        order.AddDomainEvent(new OrderCreatedEvent
        {
            OrderId = order.Id,
            CustomerId = customerId,
            ItemCount = items.Count,
            CreatedAt = DateTime.UtcNow
        });

        return order;
    }

    public void Ship()
    {
        if (Status != "Created")
            throw new InvalidOperationException("Order cannot be shipped");

        Status = "Shipped";

        AddDomainEvent(new OrderShippedEvent
        {
            OrderId = Id,
            ShippedAt = DateTime.UtcNow
        });
    }
}

// Test aggregate events
[Fact]
public void CreateOrder_RaisesOrderCreatedEvent()
{
    // Arrange
    var customerId = Guid.NewGuid();
    var items = new List<string> { "Item1", "Item2" };

    // Act
    var order = Order.Create(customerId, items);

    // Assert
    Assert.Single(order.DomainEvents);
    var @event = Assert.IsType<OrderCreatedEvent>(order.DomainEvents[0]);
    Assert.Equal(order.Id, @event.OrderId);
    Assert.Equal(customerId, @event.CustomerId);
    Assert.Equal(2, @event.ItemCount);
}

[Fact]
public void ShipOrder_RaisesOrderShippedEvent()
{
    // Arrange
    var order = Order.Create(Guid.NewGuid(), new List<string> { "Item1" });
    order.ClearDomainEvents(); // Clear creation event

    // Act
    order.Ship();

    // Assert
    Assert.Single(order.DomainEvents);
    var @event = Assert.IsType<OrderShippedEvent>(order.DomainEvents[0]);
    Assert.Equal(order.Id, @event.OrderId);
    Assert.Equal("Shipped", order.Status);
}
```

### Testing Event Dispatcher

```csharp
public interface IEventDispatcher
{
    Task DispatchAsync(object @event, CancellationToken cancellationToken = default);
}

public class EventDispatcher : IEventDispatcher
{
    private readonly IServiceProvider _serviceProvider;

    public EventDispatcher(IServiceProvider serviceProvider)
    {
        _serviceProvider = serviceProvider;
    }

    public async Task DispatchAsync(
        object @event,
        CancellationToken cancellationToken = default)
    {
        var eventType = @event.GetType();
        var handlerType = typeof(IEventHandler<>).MakeGenericType(eventType);
        
        var handlers = _serviceProvider.GetServices(handlerType);

        foreach (var handler in handlers)
        {
            var handleMethod = handlerType.GetMethod("HandleAsync");
            await (Task)handleMethod.Invoke(handler, new[] { @event, cancellationToken });
        }
    }
}

// Test dispatcher
[Fact]
public async Task DispatchEvent_CallsAllHandlers()
{
    // Arrange
    var handler1 = new Mock<IEventHandler<OrderCreatedEvent>>();
    var handler2 = new Mock<IEventHandler<OrderCreatedEvent>>();

    var services = new ServiceCollection();
    services.AddSingleton(handler1.Object);
    services.AddSingleton(handler2.Object);
    var provider = services.BuildServiceProvider();

    var dispatcher = new EventDispatcher(provider);

    var @event = new OrderCreatedEvent
    {
        OrderId = Guid.NewGuid(),
        CustomerId = Guid.NewGuid(),
        ItemCount = 1,
        CreatedAt = DateTime.UtcNow
    };

    // Act
    await dispatcher.DispatchAsync(@event);

    // Assert
    handler1.Verify(x => x.HandleAsync(@event, It.IsAny<CancellationToken>()), 
        Times.Once);
    handler2.Verify(x => x.HandleAsync(@event, It.IsAny<CancellationToken>()), 
        Times.Once);
}
```

### Testing Event Ordering

```csharp
[Fact]
public async Task OrderWorkflow_PublishesEventsInCorrectOrder()
{
    // Arrange
    var eventCollector = new EventCollector();
    var eventBus = new EventCollectorDecorator(
        new InMemoryEventBus(), 
        eventCollector);
    
    var service = new OrderService(
        new InMemoryOrderRepository(),
        eventBus);

    // Act
    var orderId = await service.CreateAndShipOrderAsync(
        new CreateOrderCommand
        {
            CustomerId = Guid.NewGuid(),
            Items = new List<string> { "Item1" }
        });

    // Assert
    var events = await eventCollector.WaitForEvents<object>(2);
    
    Assert.IsType<OrderCreatedEvent>(events[0]);
    Assert.IsType<OrderShippedEvent>(events[1]);
    
    var created = (OrderCreatedEvent)events[0];
    var shipped = (OrderShippedEvent)events[1];
    
    Assert.Equal(created.OrderId, shipped.OrderId);
    Assert.True(shipped.ShippedAt > created.CreatedAt);
}
```

### Best Practices

**1. Use Event Collectors for Integration Tests:**
```csharp
// ✓ Good - Asynchronous event collection
var @event = await eventCollector.WaitForEvent<OrderCreatedEvent>();

// ✗ Bad - Polling with delays
await Task.Delay(1000);
var @event = publishedEvents.OfType<OrderCreatedEvent>().FirstOrDefault();
```

**2. Test Event Handlers in Isolation:**
```csharp
// ✓ Good - Test handler separately
[Fact]
public async Task Handler_ProcessesEvent() { }

// ✗ Bad - Test handler through event bus
[Fact]
public async Task EventBus_TriggerHandler() { }
```

**3. Verify Event Properties:**
```csharp
// ✓ Good - Verify all important properties
Assert.Equal(expectedId, @event.OrderId);
Assert.Equal(expectedCount, @event.ItemCount);
Assert.True(@event.CreatedAt <= DateTime.UtcNow);

// ✗ Bad - Only check event was published
Assert.NotNull(@event);
```

**4. Clear Events Between Tests:**
```csharp
// ✓ Good - Fresh state
public void Setup()
{
    _eventCollector = new EventCollector();
}

// ✗ Bad - Shared state
private static readonly EventCollector _eventCollector = new();
```
