# Lesson 5: Test Data Automation in CI/CD Pipelines

## Learning Objectives

By the end of this lesson, you will be able to:
- Automate test data provisioning in CI/CD pipelines
- Implement test data as code using version control
- Design efficient test data strategies for parallel test execution
- Manage test data lifecycle in continuous integration
- Optimize test data operations for fast feedback loops

---

## Introduction

Test data management becomes critical when tests run in CI/CD pipelines. Manual data setup doesn't scale. Data conflicts break parallel test runs. Slow provisioning delays feedback. This lesson teaches you how to automate test data management so your CI/CD pipelines run fast, reliably, and at scale.

**Key principle:** Test data should be treated as code—versioned, automated, and reproducible across all environments.

```csharp
public class TestDataProvisioner
```

### Parallel Test Execution with Data Isolation

```csharp
// Challenge: Running tests in parallel requires data isolation
// Solution 1: Separate databases per test class
public class IntegrationTestBase : IAsyncLifetime
{
    protected string DatabaseName { get; }
    protected string ConnectionString { get; }
    
    public IntegrationTestBase()
    {
        // Unique database per test class
        DatabaseName = $"testdb_{Guid.NewGuid():N}";
        ConnectionString = $"Host=localhost;Database={DatabaseName};Username=test;Password=test";
    }
    
    public async Task InitializeAsync()
    {
        // Create database
        await CreateDatabaseAsync();
        
        // Apply migrations
        await ApplyMigrationsAsync();
        
        // Seed test data
        await SeedTestDataAsync();
    }
    
    public async Task DisposeAsync()
    {
        // Clean up database
        await DropDatabaseAsync();
    }
}

// Solution 2: Schema-based isolation
public class SchemaIsolatedTestBase : IAsyncLifetime
{
    protected string SchemaName { get; }
    
    public SchemaIsolatedTestBase()
    {
        // Unique schema per test class
        SchemaName = $"test_{Guid.NewGuid():N}";
    }
    
    public async Task InitializeAsync()
    {
        // Create schema
        await _context.Database.ExecuteSqlRawAsync($"CREATE SCHEMA {SchemaName}");
        
        // Create tables in schema
        await CreateTablesInSchemaAsync();
        
        // Seed data
        await SeedTestDataAsync();
    }
    
    public async Task DisposeAsync()
    {
        // Drop schema
        await _context.Database.ExecuteSqlRawAsync($"DROP SCHEMA {SchemaName} CASCADE");
    }
}

// Solution 3: Test containers (best for CI/CD)
public class TestcontainersTestBase : IAsyncLifetime
{
    private PostgreSqlContainer _container = null!;
    
    public async Task InitializeAsync()
    {
        // Each test class gets its own container
        _container = new PostgreSqlBuilder()
            .WithImage("postgres:16")
            .Build();
        
        await _container.StartAsync();
        
        // Apply migrations and seed data
        await SetupDatabaseAsync();
    }
    
    public async Task DisposeAsync()
    {
        await _container.DisposeAsync();
    }
}
```

### Efficient Test Data Caching

```csharp
// Cache test data across test runs for performance
public class TestDataCache
{
    private static readonly string CacheDirectory = 
        Path.Combine(Path.GetTempPath(), "test-data-cache");
    
    public static async Task<bool> TryLoadFromCacheAsync(
        ApplicationDbContext context,
        string cacheKey)
    {
        var cachePath = Path.Combine(CacheDirectory, $"{cacheKey}.json");
        
        if (!File.Exists(cachePath))
            return false;
        
        var json = await File.ReadAllTextAsync(cachePath);
        var data = JsonSerializer.Deserialize<CachedTestData>(json);
        
        // Load cached data into database
        await context.Customers.AddRangeAsync(data.Customers);
        await context.Products.AddRangeAsync(data.Products);
        await context.Orders.AddRangeAsync(data.Orders);
        await context.SaveChangesAsync();
        
        return true;
    }
    
    public static async Task SaveToCacheAsync(
        ApplicationDbContext context,
        string cacheKey)
    {
        Directory.CreateDirectory(CacheDirectory);
        
        var data = new CachedTestData
        {
            Customers = await context.Customers.ToListAsync(),
            Products = await context.Products.ToListAsync(),
            Orders = await context.Orders.ToListAsync()
        };
        
        var json = JsonSerializer.Serialize(data);
        var cachePath = Path.Combine(CacheDirectory, $"{cacheKey}.json");
        await File.WriteAllTextAsync(cachePath, json);
    }
}

// Usage in tests
public class CustomerTests : IAsyncLifetime
{
    public async Task InitializeAsync()
    {
        // Try to load from cache first
        if (!await TestDataCache.TryLoadFromCacheAsync(_context, "customer-tests-v1"))
        {
            // Cache miss: generate and cache
            await GenerateTestDataAsync();
            await TestDataCache.SaveToCacheAsync(_context, "customer-tests-v1");
        }
    }
}
```

### Test Data Versioning

```csharp
// Version test data alongside code
public class VersionedTestData
{
    // Data version matches code version
    public const string DataVersion = "1.0.0";
    
    public static async Task EnsureCorrectVersionAsync(ApplicationDbContext context)
    {
        var metadataTable = await context.Set<TestDataMetadata>()
            .FirstOrDefaultAsync();
        
        if (metadataTable == null || metadataTable.Version != DataVersion)
        {
            // Regenerate test data for new version
            await RegenerateTestDataAsync(context);
            
            // Update version
            if (metadataTable == null)
            {
                await context.Set<TestDataMetadata>().AddAsync(
                    new TestDataMetadata { Version = DataVersion });
            }
            else
            {
                metadataTable.Version = DataVersion;
            }
            
            await context.SaveChangesAsync();
        }
    }
}

// In CI/CD, check data version
public class TestDataVersionChecker
{
    public static async Task<bool> NeedsRegenerationAsync(
        ApplicationDbContext context,
        string expectedVersion)
    {
        var currentVersion = await context.Set<TestDataMetadata>()
            .Select(m => m.Version)
            .FirstOrDefaultAsync();
        
        return currentVersion != expectedVersion;
    }
}
```

```csharp
// Execute SQL seed scripts in tests
public class SqlSeedDataLoader
{
    public static async Task ExecuteSeedScriptAsync(
        ApplicationDbContext context,
        string scriptPath)
    {
        var sql = await File.ReadAllTextAsync(scriptPath);
        
        // Split by GO statements (SQL Server) or semicolons
        var commands = sql.Split(new[] { "GO", ";" }, 
            StringSplitOptions.RemoveEmptyEntries);
        
        foreach (var command in commands)
        {
            if (!string.IsNullOrWhiteSpace(command))
            {
                await context.Database.ExecuteSqlRawAsync(command);
            }
        }
    }
}

// Usage
[Fact]
public async Task Test_WithSqlSeedData()
{
    await ResetDatabaseAsync();
    await SqlSeedDataLoader.ExecuteSeedScriptAsync(_context, "seed-data.sql");
    
    // Test with seeded data
    var productCount = await _context.Products.CountAsync();
    Assert.Equal(3, productCount);
}
```

### Centralized Test Data Repository

```csharp
// Centralized repository pattern for test data
public static class TestDataRepository
{
    // Pre-built test data that can be reused
    public static class Customers
    {
        public static Customer ActiveCustomer() => new Faker<Customer>()
            .RuleFor(c => c.FirstName, f => f.Name.FirstName())
            .RuleFor(c => c.LastName, f => f.Name.LastName())
            .RuleFor(c => c.Email, (f, c) => f.Internet.Email(c.FirstName, c.LastName))
            .RuleFor(c => c.IsActive, _ => true)
            .Generate();
        
        public static Customer InactiveCustomer() => new Faker<Customer>()
            .RuleFor(c => c.FirstName, f => f.Name.FirstName())
            .RuleFor(c => c.LastName, f => f.Name.LastName())
            .RuleFor(c => c.Email, (f, c) => f.Internet.Email(c.FirstName, c.LastName))
            .RuleFor(c => c.IsActive, _ => false)
            .Generate();
        
        public static List<Customer> Collection(int count) => 
            new Faker<Customer>()
                .RuleFor(c => c.FirstName, f => f.Name.FirstName())
                .RuleFor(c => c.LastName, f => f.Name.LastName())
                .RuleFor(c => c.Email, (f, c) => f.Internet.Email(c.FirstName, c.LastName))
                .Generate(count);
    }
    
    public static class Orders
    {
        public static Order PendingOrder(int customerId) => new Faker<Order>()
            .RuleFor(o => o.CustomerId, _ => customerId)
            .RuleFor(o => o.Status, _ => OrderStatus.Pending)
            .RuleFor(o => o.OrderDate, f => f.Date.Recent())
            .Generate();
        
        public static Order CompletedOrder(int customerId) => new Faker<Order>()
            .RuleFor(o => o.CustomerId, _ => customerId)
            .RuleFor(o => o.Status, _ => OrderStatus.Completed)
            .RuleFor(o => o.OrderDate, f => f.Date.Past())
            .RuleFor(o => o.CompletedAt, f => f.Date.Recent())
            .Generate();
    }
    
    public static class Products
    {
        public static Product InStockProduct() => new Faker<Product>()
            .RuleFor(p => p.Name, f => f.Commerce.ProductName())
            .RuleFor(p => p.Price, f => f.Finance.Amount(10, 1000))
            .RuleFor(p => p.Stock, f => f.Random.Int(10, 100))
            .Generate();
        
        public static Product OutOfStockProduct() => new Faker<Product>()
            .RuleFor(p => p.Name, f => f.Commerce.ProductName())
            .RuleFor(p => p.Price, f => f.Finance.Amount(10, 1000))
            .RuleFor(p => p.Stock, _ => 0)
            .Generate();
    }
}
```

### Monitoring Test Data Quality in CI/CD

```csharp
// Add test data health checks to CI/CD
public class TestDataHealthCheck
{
    public static async Task<HealthCheckResult> CheckAsync(
        ApplicationDbContext context)
    {
        var issues = new List<string>();
        
        // Check: Adequate data volume
        var customerCount = await context.Customers.CountAsync();
        if (customerCount < 10)
        {
            issues.Add($"Insufficient customers: {customerCount} (expected >= 10)");
        }
        
        // Check: Data relationships intact
        var ordersWithoutCustomers = await context.Orders
            .Where(o => o.Customer == null)
            .CountAsync();
        
        if (ordersWithoutCustomers > 0)
        {
            issues.Add($"Found {ordersWithoutCustomers} orders with invalid customer references");
        }
        
        // Check: No real PII (compliance)
        var suspiciousEmails = await context.Customers
            .Where(c => c.Email.Contains("gmail.com") || 
                       c.Email.Contains("yahoo.com"))
            .CountAsync();
        
        if (suspiciousEmails > 0)
        {
            issues.Add($"Found {suspiciousEmails} customers with potential real email addresses");
        }
        
        // Check: Data freshness
        var oldestOrder = await context.Orders
            .OrderBy(o => o.OrderDate)
            .Select(o => o.OrderDate)
            .FirstOrDefaultAsync();
        
        if (oldestOrder < DateTime.UtcNow.AddYears(-2))
        {
            issues.Add("Test data appears stale (orders older than 2 years)");
        }
        
        return issues.Any() 
            ? HealthCheckResult.Unhealthy(string.Join("; ", issues))
            : HealthCheckResult.Healthy("Test data is healthy");
    }
}

// CI/CD step
[Fact]
public async Task HealthCheck_TestData_IsHealthy()
{
    var result = await TestDataHealthCheck.CheckAsync(_context);
    
    Assert.True(result.Status == HealthStatus.Healthy, result.Description);
}
```

### Optimizing for Fast Feedback

```csharp
// Strategy: Lazy load test data only when needed
public class LazyTestDataProvider
{
    private readonly ApplicationDbContext _context;
    private bool _customersLoaded;
    private bool _productsLoaded;
    
    public async Task EnsureCustomersAsync()
    {
        if (_customersLoaded) return;
        
        await TestDataRepository.Customers
            .Collection(100)
            .ForEachAsync(async c => await _context.Customers.AddAsync(c));
        
        await _context.SaveChangesAsync();
        _customersLoaded = true;
    }
    
    public async Task EnsureProductsAsync()
    {
        if (_productsLoaded) return;
        
        var products = Enumerable.Range(1, 50)
            .Select(_ => TestDataRepository.Products.InStockProduct());
        
        await _context.Products.AddRangeAsync(products);
        await _context.SaveChangesAsync();
        _productsLoaded = true;
    }
}

// Tests only load what they need
[Fact]
public async Task GetCustomer_ReturnsCustomer()
{
    await _testData.EnsureCustomersAsync(); // Only customers needed
    
    // Test doesn't need products, so they're not loaded
    var service = new CustomerService(_context);
    var result = await service.GetAllCustomersAsync();
    
    Assert.NotEmpty(result);
}
```
