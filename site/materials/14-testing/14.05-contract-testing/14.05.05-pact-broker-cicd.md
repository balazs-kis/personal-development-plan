### Setting Up Pact Broker

**Option 1: Docker Compose (Recommended for Local/Test)**
```yaml
# docker-compose.yml
version: '3'
services:
  postgres:
    image: postgres:15
    healthcheck:
      test: psql postgres --command "select 1" -U postgres
    environment:
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: password
      POSTGRES_DB: pact_broker
    volumes:
      - postgres-data:/var/lib/postgresql/data

  pact-broker:
    image: pactfoundation/pact-broker:latest
    ports:
      - "9292:9292"
    depends_on:
      - postgres
    environment:
      PACT_BROKER_DATABASE_URL: "postgres://postgres:password@postgres/pact_broker"
      PACT_BROKER_BASIC_AUTH_USERNAME: pactbroker
      PACT_BROKER_BASIC_AUTH_PASSWORD: pactbroker
      PACT_BROKER_ALLOW_PUBLIC_READ: 'true'
      PACT_BROKER_BASE_URL: 'http://localhost:9292'

volumes:
  postgres-data:
```

**Start Pact Broker:**
```bash
docker-compose up -d
# Access at http://localhost:9292
# Username: pactbroker
# Password: pactbroker
```

### Publishing Pacts to Broker

**Using Pact CLI:**
```bash
# Install pact-broker CLI
npm install -g @pact-foundation/pact-node

# Or use Docker
docker pull pactfoundation/pact-cli

# Publish pacts
pact-broker publish ./pacts \
  --consumer-app-version $GIT_COMMIT \
  --branch $BRANCH_NAME \
  --broker-base-url https://your-pact-broker.com \
  --broker-token $PACT_BROKER_TOKEN
```

**In .NET Tests:**
```csharp
// Consumer test - automatically creates pact file
public class ProductApiConsumerTests : IDisposable
{
    private readonly IPactBuilderV4 _pactBuilder;

    public ProductApiConsumerTests()
    {
        var pact = Pact.V4("ProductConsumer", "ProductAPI", new PactConfig
        {
            PactDir = "../../../pacts/",  // Pacts written here
        });

        _pactBuilder = pact.WithHttpInteractions();
    }

    // Tests...

    public void Dispose()
    {
        // Pact file written on disposal
    }
}
```

**Publish Script:**
```bash
#!/bin/bash
# publish-pacts.sh

# Run consumer tests to generate pacts
dotnet test ./Consumer.Tests --filter Category=ContractTest

# Publish to Pact Broker
docker run --rm \
  -v ${PWD}/pacts:/pacts \
  pactfoundation/pact-cli:latest \
  publish /pacts \
  --consumer-app-version ${GIT_COMMIT} \
  --branch ${BRANCH_NAME} \
  --broker-base-url ${PACT_BROKER_URL} \
  --broker-token ${PACT_BROKER_TOKEN}
```

### Provider Verification with Broker

```csharp
[Fact]
public void EnsureProviderHonorsContractsFromBroker()
{
    var config = new PactVerifierConfig
    {
        Outputters = new[] { new XUnitOutput(_outputHelper) },
        LogLevel = PactLogLevel.Information
    };

    new PactVerifier(config)
        .WithHttpEndpoint(new Uri("http://localhost:9001"))
        .WithPactBrokerSource(new Uri("https://your-pact-broker.com"), options =>
        {
            options
                // Authentication
                .TokenAuthentication(_pactBrokerToken)
                
                // Which pacts to verify
                .ConsumerVersionSelectors(
                    new ConsumerVersionSelector { Branch = "main" },
                    new ConsumerVersionSelector { DeployedOrReleased = true }
                )
                
                // Provider version and publishing
                .ProviderVersion(_gitCommit)
                .ProviderBranch(_branchName)
                .PublishResults(publishResults: true);
        })
        .WithProviderStateUrl(new Uri("http://localhost:9001/provider-states"))
        .Verify();
}
```

### CI/CD Pipeline Integration

**Consumer Pipeline:**
```yaml
# .github/workflows/consumer-contract-tests.yml
name: Consumer Contract Tests

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

env:
  PACT_BROKER_URL: ${{ secrets.PACT_BROKER_URL }}
  PACT_BROKER_TOKEN: ${{ secrets.PACT_BROKER_TOKEN }}

jobs:
  test:
    runs-on: ubuntu-latest
    
    steps:
      - uses: actions/checkout@v4
      
      - name: Setup .NET
        uses: actions/setup-dotnet@v4
        with:
          dotnet-version: '8.0.x'
      
      - name: Run Contract Tests
        run: |
          dotnet test ./Consumer.Tests \
            --filter Category=ContractTest \
            --logger "trx;LogFileName=test-results.trx"
      
      - name: Publish Pacts to Broker
        if: success()
        run: |
          docker run --rm \
            -v ${PWD}/pacts:/pacts \
            pactfoundation/pact-cli:latest \
            publish /pacts \
            --consumer-app-version ${{ github.sha }} \
            --branch ${{ github.ref_name }} \
            --broker-base-url ${{ env.PACT_BROKER_URL }} \
            --broker-token ${{ env.PACT_BROKER_TOKEN }}
      
      - name: Check if safe to deploy
        if: github.ref == 'refs/heads/main'
        run: |
          docker run --rm \
            pactfoundation/pact-cli:latest \
            broker can-i-deploy \
            --pacticipant ProductConsumer \
            --version ${{ github.sha }} \
            --to-environment production \
            --retry-while-unknown 30 \
            --retry-interval 10 \
            --broker-base-url ${{ env.PACT_BROKER_URL }} \
            --broker-token ${{ env.PACT_BROKER_TOKEN }}
```

**Provider Pipeline:**
```yaml
# .github/workflows/provider-contract-tests.yml
name: Provider Contract Tests

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]
  # Triggered by Pact Broker webhook when new pact published
  repository_dispatch:
    types: [pact_changed]

env:
  PACT_BROKER_URL: ${{ secrets.PACT_BROKER_URL }}
  PACT_BROKER_TOKEN: ${{ secrets.PACT_BROKER_TOKEN }}

jobs:
  verify:
    runs-on: ubuntu-latest
    
    steps:
      - uses: actions/checkout@v4
      
      - name: Setup .NET
        uses: actions/setup-dotnet@v4
        with:
          dotnet-version: '8.0.x'
      
      - name: Run Provider Verification Tests
        env:
          GIT_COMMIT: ${{ github.sha }}
          BRANCH_NAME: ${{ github.ref_name }}
        run: |
          dotnet test ./Provider.Tests \
            --filter Category=ProviderVerification \
            --logger "trx;LogFileName=test-results.trx"
        # Verification results published automatically by test code
      
      - name: Check if safe to deploy
        if: github.ref == 'refs/heads/main'
        run: |
          docker run --rm \
            pactfoundation/pact-cli:latest \
            broker can-i-deploy \
            --pacticipant ProductAPI \
            --version ${{ github.sha }} \
            --to-environment production \
            --retry-while-unknown 30 \
            --retry-interval 10 \
            --broker-base-url ${{ env.PACT_BROKER_URL }} \
            --broker-token ${{ env.PACT_BROKER_TOKEN }}
      
      - name: Deploy
        if: success() && github.ref == 'refs/heads/main'
        run: |
          # Deploy application
          echo "Deploying to production..."
      
      - name: Record Deployment
        if: success() && github.ref == 'refs/heads/main'
        run: |
          docker run --rm \
            pactfoundation/pact-cli:latest \
            broker record-deployment \
            --pacticipant ProductAPI \
            --version ${{ github.sha }} \
            --environment production \
            --broker-base-url ${{ env.PACT_BROKER_URL }} \
            --broker-token ${{ env.PACT_BROKER_TOKEN }}
```

### Webhooks for Automatic Provider Verification

**Setting Up Webhook:**
```bash
# When consumer publishes new pact, trigger provider build
pact-broker create-webhook \
  'https://api.github.com/repos/your-org/provider-repo/dispatches' \
  --request=POST \
  --header 'Content-Type: application/json' \
  --header 'Accept: application/vnd.github+json' \
  --header 'Authorization: Bearer YOUR_GITHUB_TOKEN' \
  --data '{"event_type":"pact_changed","client_payload":{"pact_url":"${pactbroker.pactUrl}"}}' \
  --consumer ProductConsumer \
  --provider ProductAPI \
  --contract-content-changed \
  --broker-base-url https://your-pact-broker.com \
  --broker-token $PACT_BROKER_TOKEN
```

**Webhook Workflow:**
```
1. Consumer CI publishes new pact to Broker
   ↓
2. Pact Broker detects contract change
   ↓
3. Webhook triggers Provider CI build
   ↓
4. Provider verifies new contract
   ↓
5. Results published back to Broker
   ↓
6. Both teams can see verification status
```

### Best Practices for CI/CD Integration

**1. Parallel Test Execution:**
```yaml
# Run contract tests in parallel with unit tests
jobs:
  unit-tests:
    runs-on: ubuntu-latest
    steps:
      - run: dotnet test --filter Category!=ContractTest

  contract-tests:
    runs-on: ubuntu-latest
    steps:
      - run: dotnet test --filter Category=ContractTest
      - run: pact-broker publish ./pacts
```

**2. Fast Feedback:**
```yaml
# Fail fast if contract tests fail
- name: Contract Tests
  run: dotnet test --filter Category=ContractTest
  timeout-minutes: 5  # Don't wait forever

- name: Can I Deploy
  run: |
    pact-broker can-i-deploy \
      --retry-while-unknown 10 \  # Retry for 100 seconds
      --retry-interval 10
```

**3. Static Test Data:**
```csharp
// ✓ Good - Consistent data
.Given("product with ID 10 exists")
.WithJsonBody(new { id = 10, name = "Widget" })

// ✗ Bad - Random data creates unique contracts
.WithJsonBody(new { id = Random.Next(), name = Guid.NewGuid() })
```

**4. Separate Contract Tests:**
```xml
<!-- Tag contract tests -->
<ItemGroup>
  <AssemblyAttribute Include="Xunit.TraitAttribute">
    <_Parameter1>Category</_Parameter1>
    <_Parameter2>ContractTest</_Parameter2>
  </AssemblyAttribute>
</ItemGroup>
```

**5. Environment-Specific Verification:**
```csharp
// Verify different consumer versions per environment
var environmentName = Environment.GetEnvironmentVariable("ENVIRONMENT");

var selectors = environmentName switch
{
    "production" => new[] 
    { 
        new ConsumerVersionSelector { Environment = "production", DeployedOrReleased = true }
    },
    "staging" => new[]
    {
        new ConsumerVersionSelector { Environment = "staging", DeployedOrReleased = true },
        new ConsumerVersionSelector { Branch = "main" }
    },
    _ => new[]
    {
        new ConsumerVersionSelector { Branch = "main" }
    }
};
```

### Troubleshooting CI/CD Issues

**Common Issues:**

**1. Pacts Not Found:**
```
Error: No pacts found for ProductConsumer

Fix: Ensure pacts were published successfully
Check: Pact Broker UI for published pacts
```

**2. Verification Timeout:**
```
Error: Timeout waiting for provider verification

Fix: Increase retry interval and count
Check: Provider CI pipeline status
```

**3. Authentication Failures:**
```
Error: 401 Unauthorized

Fix: Verify PACT_BROKER_TOKEN is correct
Check: Token permissions in Pact Broker
```

**4. Version Conflicts:**
```
Error: Version abc123 already exists with different content

Fix: Ensure unique version per commit
Use: Git SHA instead of semantic version
```
