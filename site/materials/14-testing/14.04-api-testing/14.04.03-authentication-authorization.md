# Lesson 3: Testing Authentication and Authorization

## Learning Objectives

By the end of this lesson, you will be able to:
- Test protected API endpoints requiring authentication
- Configure fake authentication handlers for integration tests
- Generate and use JWT tokens in tests
- Test authorization policies and role-based access
- Verify different user scenarios (anonymous, authenticated, admin, etc.)
- Handle authentication state in WebApplicationFactory

---

## Introduction

Most real-world APIs require authentication and authorization. Testing protected endpoints presents unique challenges: you need to authenticate test users without coupling tests to production authentication systems. This lesson covers strategies for testing authentication and authorization in ASP.NET Core, including fake authentication handlers, JWT token generation, and testing various authorization scenarios.

### Approach 1: Fake Authentication Handler

Create a test authentication handler that accepts any token:

```csharp
using Microsoft.AspNetCore.Authentication;
using Microsoft.Extensions.Options;
using System.Security.Claims;
using System.Text.Encodings.Web;

public class TestAuthHandler : AuthenticationHandler<AuthenticationSchemeOptions>
{
    public TestAuthHandler(
        IOptionsMonitor<AuthenticationSchemeOptions> options,
        ILoggerFactory logger,
        UrlEncoder encoder,
        ISystemClock clock)
        : base(options, logger, encoder, clock)
    {
    }

    protected override Task<AuthenticateResult> HandleAuthenticateAsync()
    {
        // Check if Authorization header is present
        if (!Request.Headers.ContainsKey("Authorization"))
        {
            return Task.FromResult(AuthenticateResult.NoResult());
        }

        // Extract claims from header (e.g., "Bearer userId:role:claim")
        var authHeader = Request.Headers["Authorization"].ToString();
        var token = authHeader.Replace("Bearer ", "");
        var parts = token.Split(':');
        
        var claims = new List<Claim>
        {
            new Claim(ClaimTypes.NameIdentifier, parts[0]),
            new Claim(ClaimTypes.Name, parts[0])
        };
        
        if (parts.Length > 1)
        {
            claims.Add(new Claim(ClaimTypes.Role, parts[1]));
        }

        var identity = new ClaimsIdentity(claims, "Test");
        var principal = new ClaimsPrincipal(identity);
        var ticket = new AuthenticationTicket(principal, "Test");

        return Task.FromResult(AuthenticateResult.Success(ticket));
    }
}
```

Register the fake handler in your custom WebApplicationFactory:

```csharp
public class AuthenticatedWebApplicationFactory : WebApplicationFactory<Program>
{
    protected override void ConfigureWebHost(IWebHostBuilder builder)
    {
        builder.ConfigureTestServices(services =>
        {
            // Remove the production authentication
            services.AddAuthentication("Test")
                .AddScheme<AuthenticationSchemeOptions, TestAuthHandler>(
                    "Test", options => { });
        });
    }
}
```

Use in tests:

```csharp
[Fact]
public async Task GetProtectedResource_WithAuth_ReturnsOk()
```

### Approach 2: Reconfigure JWT Authentication (Preferred)

Override JWT bearer options to simplify validation:

```csharp
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.Extensions.Options;

public class TestWebApplicationFactory : WebApplicationFactory<Program>
{
    protected override void ConfigureWebHost(IWebHostBuilder builder)
    {
        builder.ConfigureTestServices(services =>
        {
            // Reconfigure JWT bearer options for testing
            services.PostConfigureAll<JwtBearerOptions>(options =>
            {
                // Use a test signing key
                var testKey = new SymmetricSecurityKey(
                    Encoding.UTF8.GetBytes("test-secret-key-min-32-chars!"));
                
                options.TokenValidationParameters = new TokenValidationParameters
                {
                    ValidateIssuerSigningKey = true,
                    IssuerSigningKey = testKey,
                    ValidateIssuer = true,
                    ValidIssuer = "test-issuer",
                    ValidateAudience = true,
                    ValidAudience = "test-audience",
                    ValidateLifetime = false, // Don't validate expiry in tests
                    ClockSkew = TimeSpan.Zero
                };
            });
        });
    }
}
```

Create a helper to generate test JWT tokens:

```csharp
using System.IdentityModel.Tokens.Jwt;
using Microsoft.IdentityModel.Tokens;

public class JwtTokenHelper
{
    private static readonly SymmetricSecurityKey Key = 
        new SymmetricSecurityKey(
            Encoding.UTF8.GetBytes("test-secret-key-min-32-chars!"));

    public static string GenerateToken(
        string userId, 
        string role = "User",
        params Claim[] additionalClaims)
    {
        var claims = new List<Claim>
        {
            new Claim(ClaimTypes.NameIdentifier, userId),
            new Claim(ClaimTypes.Name, userId),
            new Claim(ClaimTypes.Role, role)
        };
        
        claims.AddRange(additionalClaims);

        var credentials = new SigningCredentials(
            Key, 
            SecurityAlgorithms.HmacSha256);

        var token = new JwtSecurityToken(
            issuer: "test-issuer",
            audience: "test-audience",
            claims: claims,
            expires: DateTime.UtcNow.AddHours(1),
            signingCredentials: credentials);

        return new JwtSecurityTokenHandler().WriteToken(token);
    }
}
```

Use in tests:


### Testing User Context

Verify that authenticated user information is available:

```csharp
[Fact]
public async Task GetCurrentUser_ReturnsAuthenticatedUserInfo()
{
    var client = _factory.CreateClient();
    var token = JwtTokenHelper.GenerateToken("user123", "User");
    client.DefaultRequestHeaders.Authorization = 
        new AuthenticationHeaderValue("Bearer", token);
    
    var response = await client.GetAsync("/api/users/me");
    var user = await response.Content.ReadFromJsonAsync<UserDto>();
    
    response.StatusCode.Should().Be(HttpStatusCode.OK);
    user.Should().NotBeNull();
    user!.Id.Should().Be("user123");
}
```

### Best Practices

**1. Test Authorization Boundaries:**
```csharp
// Test all role combinations
[Theory]
[InlineData("User", "/api/admin", HttpStatusCode.Forbidden)]
[InlineData("Admin", "/api/admin", HttpStatusCode.OK)]
[InlineData("User", "/api/user", HttpStatusCode.OK)]
public async Task TestRoleAccess(string role, string endpoint, HttpStatusCode expected)
{
    var client = _factory.CreateClient();
    var token = JwtTokenHelper.GenerateToken("user", role);
    client.DefaultRequestHeaders.Authorization = 
        new AuthenticationHeaderValue("Bearer", token);
    
    var response = await client.GetAsync(endpoint);
    
    response.StatusCode.Should().Be(expected);
}
```

**2. Test Anonymous Access:**
```csharp
[Fact]
public async Task GetPublicEndpoint_WithoutAuth_ReturnsOk()
{
    var client = _factory.CreateClient();
    
    var response = await client.GetAsync("/api/public");
    
    response.StatusCode.Should().Be(HttpStatusCode.OK);
}
```

**3. Test Expired Tokens (if validating lifetime):**
```csharp
[Fact]
public async Task GetProtectedResource_WithExpiredToken_ReturnsUnauthorized()
{
    // Generate token that expired 1 hour ago
    var token = GenerateExpiredToken("user123");
    var client = _factory.CreateClient();
    client.DefaultRequestHeaders.Authorization = 
        new AuthenticationHeaderValue("Bearer", token);
    
    var response = await client.GetAsync("/api/protected");
    
    response.StatusCode.Should().Be(HttpStatusCode.Unauthorized);
}
```
