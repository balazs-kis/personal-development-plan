This lesson covers advanced API testing patterns that help you build comprehensive test suites for production-grade APIs. These patterns ensure your API behaves correctly under various conditions and maintains backward compatibility.

### Testing API Versioning

**URL-Based Versioning:**

```csharp
public class VersioningTests : IClassFixture<WebApplicationFactory<Program>>
{
    private readonly WebApplicationFactory<Program> _factory;

    public VersioningTests(WebApplicationFactory<Program> factory)
    {
        _factory = factory;
    }

    [Theory]
    [InlineData("/api/v1/products", "1.0")]
    [InlineData("/api/v2/products", "2.0")]
    public async Task GetProducts_DifferentVersions_ReturnsExpectedStructure(
        string url, 
        string version)
    {
        var client = _factory.CreateClient();

        var response = await client.GetAsync(url);
        
        response.EnsureSuccessStatusCode();
        
        // Verify each version returns its expected structure
        await Verify(response).UseParameters(version);
    }

    [Fact]
    public async Task GetProducts_V2_IncludesNewFields()
    {
        var client = _factory.CreateClient();

        var v1Response = await client.GetAsync("/api/v1/products");
        var v2Response = await client.GetAsync("/api/v2/products");
        
        var v1Content = await v1Response.Content.ReadFromJsonAsync<List<ProductV1>>();
        var v2Content = await v2Response.Content.ReadFromJsonAsync<List<ProductV2>>();

        // V2 should have additional fields
        v2Content![0].Should().Match<ProductV2>(p => 
            p.Category != null && p.Tags != null);
    }
}
```

### Test Data Builders and Factories

Create reusable test data builders:

```csharp
public class ProductBuilder
{
    private string _name = "Test Product";
    private decimal _price = 10.99m;
    private string _category = "Electronics";
    private bool _isAvailable = true;

    public ProductBuilder WithName(string name)
    {
        _name = name;
        return this;
    }

    public ProductBuilder WithPrice(decimal price)
    {
        _price = price;
        return this;
    }

    public ProductBuilder Unavailable()
    {
        _isAvailable = false;
        return this;
    }

    public Product Build()
    {
        return new Product
        {
            Name = _name,
            Price = _price,
            Category = _category,
            IsAvailable = _isAvailable
        };
    }

    public static ProductBuilder Create() => new();
}
```

Use in tests:

```csharp
[Fact]
public async Task CreateProduct_WithValidData_ReturnsCreated()
{
    var client = _factory.CreateClient();
    
    var product = ProductBuilder.Create()
        .WithName("Gaming Laptop")
        .WithPrice(1299.99m)
        .Build();

    var response = await client.PostAsJsonAsync("/api/products", product);
    
    response.StatusCode.Should().Be(HttpStatusCode.Created);
}
```

### Testing Rate Limiting

```csharp
[Fact]
public async Task RateLimiting_ExceedsLimit_ReturnsTooManyRequests()
{
    var client = _factory.CreateClient();
    
    // Make requests up to the rate limit
    for (int i = 0; i < 100; i++)
    {
        await client.GetAsync("/api/products");
    }
    
    // The 101st request should be rate limited
    var response = await client.GetAsync("/api/products");
    
    response.StatusCode.Should().Be(HttpStatusCode.TooManyRequests);
    response.Headers.Should().ContainKey("Retry-After");
}

[Fact]
public async Task RateLimiting_WithApiKey_HasHigherLimit()
{
    var client = _factory.CreateClient();
    client.DefaultRequestHeaders.Add("X-API-Key", "premium-key");
    
    var successfulRequests = 0;
    
    for (int i = 0; i < 1000; i++)
    {
        var response = await client.GetAsync("/api/products");
        if (response.IsSuccessStatusCode)
        {
            successfulRequests++;
        }
        else
        {
            break;
        }
    }
    
    successfulRequests.Should().BeGreaterThan(100); // Higher than free tier
}
```

### Testing CORS Configuration

```csharp
[Theory]
[InlineData("https://app.example.com", true)]
[InlineData("https://malicious.com", false)]
public async Task CORS_WithOrigin_ReturnsExpectedHeaders(string origin, bool shouldAllow)
{
    var client = _factory.CreateClient();
    
    var request = new HttpRequestMessage(HttpMethod.Options, "/api/products");
    request.Headers.Add("Origin", origin);
    request.Headers.Add("Access-Control-Request-Method", "GET");

    var response = await client.SendAsync(request);

    if (shouldAllow)
    {
        response.Headers.Should().ContainKey("Access-Control-Allow-Origin");
        response.Headers.GetValues("Access-Control-Allow-Origin")
            .Should().Contain(origin);
    }
    else
    {
        response.Headers.Should().NotContainKey("Access-Control-Allow-Origin");
    }
}

[Fact]
public async Task CORS_SupportsCredentials()
{
    var client = _factory.CreateClient();
    
    var request = new HttpRequestMessage(HttpMethod.Options, "/api/products");
    request.Headers.Add("Origin", "https://app.example.com");

    var response = await client.SendAsync(request);

    response.Headers.Should().ContainKey("Access-Control-Allow-Credentials");
    response.Headers.GetValues("Access-Control-Allow-Credentials")
        .Should().Contain("true");
}
```

### Testing Content Negotiation

```csharp
[Theory]
[InlineData("application/json")]
```

### Testing Pagination

```csharp
[Fact]
public async Task GetProducts_WithPagination_ReturnsPagedResults()
{
    var client = _factory.CreateClient();
    
    var response = await client.GetAsync("/api/products?page=1&pageSize=10");
    
    response.EnsureSuccessStatusCode();
    
    // Check pagination headers
    response.Headers.Should().ContainKey("X-Total-Count");
    response.Headers.Should().ContainKey("X-Page");
    response.Headers.Should().ContainKey("X-Page-Size");
    
    var products = await response.Content.ReadFromJsonAsync<List<Product>>();
    products.Should().HaveCount(10);
}

[Fact]
public async Task GetProducts_LastPage_ReturnsRemainingItems()
{
    var client = _factory.CreateClient();
    
    // First, get total count
    var firstResponse = await client.GetAsync("/api/products?page=1&pageSize=10");
    var totalCount = int.Parse(
        firstResponse.Headers.GetValues("X-Total-Count").First());
    
    // Calculate last page
    var lastPage = (int)Math.Ceiling(totalCount / 10.0);
    
    var response = await client.GetAsync($"/api/products?page={lastPage}&pageSize=10");
    var products = await response.Content.ReadFromJsonAsync<List<Product>>();
    
    var expectedCount = totalCount % 10 == 0 ? 10 : totalCount % 10;
    products.Should().HaveCount(expectedCount);
}
```

### Testing Filtering and Sorting

```csharp
[Fact]
public async Task GetProducts_WithFilter_ReturnsFilteredResults()
{
    var client = _factory.CreateClient();
    
    var response = await client.GetAsync("/api/products?category=Electronics");
    var products = await response.Content.ReadFromJsonAsync<List<Product>>();
    
    products.Should().OnlyContain(p => p.Category == "Electronics");
}

[Theory]
[InlineData("name", "asc")]
[InlineData("name", "desc")]
[InlineData("price", "asc")]
[InlineData("price", "desc")]
public async Task GetProducts_WithSorting_ReturnsSortedResults(
    string sortBy, 
    string order)
{
    var client = _factory.CreateClient();
    
    var response = await client.GetAsync(
        $"/api/products?sortBy={sortBy}&order={order}");
    var products = await response.Content.ReadFromJsonAsync<List<Product>>();
    
    products.Should().NotBeEmpty();
    
    if (sortBy == "name")
    {
        var names = products.Select(p => p.Name).ToList();
        var expected = order == "asc" 
            ? names.OrderBy(n => n).ToList()
            : names.OrderByDescending(n => n).ToList();
        names.Should().Equal(expected);
    }
}
```

### Advanced WebApplicationFactory Customization

**Multiple Database Contexts:**

```csharp
public class MultiDbWebApplicationFactory : WebApplicationFactory<Program>
{
    protected override void ConfigureWebHost(IWebHostBuilder builder)
    {
        builder.ConfigureTestServices(services =>
        {
            // Remove production databases
            services.RemoveAll<DbContextOptions<AppDbContext>>();
            services.RemoveAll<DbContextOptions<IdentityDbContext>>();
            
            // Add test databases
            services.AddDbContext<AppDbContext>(options =>
                options.UseInMemoryDatabase("TestAppDb"));
            
            services.AddDbContext<IdentityDbContext>(options =>
                options.UseInMemoryDatabase("TestIdentityDb"));
            
            // Seed data
            var sp = services.BuildServiceProvider();
            using var scope = sp.CreateScope();
            SeedTestData(scope);
        });
    }
    
    private void SeedTestData(IServiceScope scope)
    {
        var appDb = scope.ServiceProvider.GetRequiredService<AppDbContext>();
        var identityDb = scope.ServiceProvider
            .GetRequiredService<IdentityDbContext>();
        
        // Seed both databases
        appDb.Products.AddRange(CreateTestProducts());
        identityDb.Users.AddRange(CreateTestUsers());
        
        appDb.SaveChanges();
        identityDb.SaveChanges();
    }
}
```

**Mocking External Services:**

```csharp
public class MockedServicesWebApplicationFactory : WebApplicationFactory<Program>
{
    protected override void ConfigureWebHost(IWebHostBuilder builder)
    {
        builder.ConfigureTestServices(services =>
        {
            // Replace real external service with mock
            services.RemoveAll<IPaymentService>();
            services.AddSingleton<IPaymentService>(sp =>
            {
                var mock = new Mock<IPaymentService>();
                mock.Setup(x => x.ProcessPaymentAsync(It.IsAny<Payment>()))
                    .ReturnsAsync(new PaymentResult { Success = true });
                return mock.Object;
            });
        });
    }
}
```

### Testing Idempotency

```csharp
[Fact]
public async Task CreateOrder_CalledTwiceWithSameIdempotencyKey_CreatesOnce()
{
    var client = _factory.CreateClient();
    var idempotencyKey = Guid.NewGuid().ToString();
    var order = new { ProductId = 1, Quantity = 2 };
    
    // First request
    var request1 = new HttpRequestMessage(HttpMethod.Post, "/api/orders")
    {
        Content = JsonContent.Create(order)
    };
    request1.Headers.Add("Idempotency-Key", idempotencyKey);
    
    var response1 = await client.SendAsync(request1);
    var order1 = await response1.Content.ReadFromJsonAsync<Order>();
    
    // Second request with same key
    var request2 = new HttpRequestMessage(HttpMethod.Post, "/api/orders")
    {
        Content = JsonContent.Create(order)
    };
    request2.Headers.Add("Idempotency-Key", idempotencyKey);
    
    var response2 = await client.SendAsync(request2);
    var order2 = await response2.Content.ReadFromJsonAsync<Order>();
    
    // Should return the same order
    order1!.Id.Should().Be(order2!.Id);
    response2.StatusCode.Should().Be(HttpStatusCode.OK); // Not 201
}
```

### Testing Long-Running Operations

```csharp
[Fact]
public async Task ProcessBulkImport_ReturnsAccepted_AndJobId()
{
    var client = _factory.CreateClient();
    var importData = CreateLargeImportPayload();
    
    var response = await client.PostAsJsonAsync("/api/import", importData);
    
    response.StatusCode.Should().Be(HttpStatusCode.Accepted);
    
    var result = await response.Content.ReadFromJsonAsync<AsyncJobResult>();
    result!.JobId.Should().NotBeEmpty();
    result.StatusUrl.Should().NotBeNullOrEmpty();
}

[Fact]
public async Task GetJobStatus_CompletedJob_ReturnsResult()
{
    var client = _factory.CreateClient();
    
    // Start a job
    var startResponse = await client.PostAsJsonAsync("/api/import", new { });
    var job = await startResponse.Content.ReadFromJsonAsync<AsyncJobResult>();
    
    // Poll for completion
    JobStatus status = null!;
    for (int i = 0; i < 10; i++)
    {
        await Task.Delay(500);
        var statusResponse = await client.GetAsync(job!.StatusUrl);
        status = await statusResponse.Content.ReadFromJsonAsync<JobStatus>();
        
        if (status!.State == "Completed" || status.State == "Failed")
            break;
    }
    
    status.State.Should().Be("Completed");
}
```
