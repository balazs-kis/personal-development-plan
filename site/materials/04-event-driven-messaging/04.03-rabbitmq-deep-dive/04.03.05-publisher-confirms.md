### Publisher Confirms

**Basic Confirms:**
```csharp
public class PublisherWithConfirms
{
    public async Task PublishWithConfirmAsync(string message)
    {
        using var channel = _connection.CreateModel();
        
        // Enable publisher confirms
        channel.ConfirmSelect();
        
        var body = Encoding.UTF8.GetBytes(message);
        
        // Publish message
        channel.BasicPublish(
            exchange: "",
            routingKey: "orders",
            basicProperties: null,
            body: body
        );
        
        // Wait for confirm (blocking)
        if (channel.WaitForConfirms())
        {
            _logger.LogInformation("Message confirmed by broker");
        }
        else
        {
            _logger.LogError("Message not confirmed");
            throw new Exception("Publish failed");
        }
    }
}
```

**Async Confirms:**
```csharp
public class AsyncPublisherConfirms
{
    private readonly ConcurrentDictionary<ulong, string> _outstandingConfirms = new();
    
    public void SetupAsyncConfirms(IModel channel)
    {
        channel.ConfirmSelect();
        
        // Handle confirms
        channel.BasicAcks += (sender, ea) =>
        {
            CleanOutstandingConfirms(ea.DeliveryTag, ea.Multiple);
        };
        
        // Handle nacks (broker couldn't process)
        channel.BasicNacks += (sender, ea) =>
        {
            _outstandingConfirms.TryGetValue(ea.DeliveryTag, out var body);
            _logger.LogWarning($"Message nacked: {body}");
            
            // Retry logic here
            RetryPublish(body);
            
            CleanOutstandingConfirms(ea.DeliveryTag, ea.Multiple);
        };
    }
    
    public void PublishAsync(IModel channel, string message)
    {
        var deliveryTag = channel.NextPublishSeqNo;
        _outstandingConfirms.TryAdd(deliveryTag, message);
        
        channel.BasicPublish(
            exchange: "",
            routingKey: "orders",
            basicProperties: null,
            body: Encoding.UTF8.GetBytes(message)
        );
    }
    
    private void CleanOutstandingConfirms(ulong sequenceNumber, bool multiple)
    {
        if (multiple)
        {
            // Confirm all up to sequence number
            var confirmed = _outstandingConfirms
                .Where(k => k.Key <= sequenceNumber);
            foreach (var entry in confirmed)
            {
                _outstandingConfirms.TryRemove(entry.Key, out _);
            }
        }
        else
        {
            // Confirm single message
            _outstandingConfirms.TryRemove(sequenceNumber, out _);
        }
    }
}
```

### Complete Reliable Publishing

```csharp
public class ReliablePublisher
{
    private readonly IConnection _connection;
    private readonly IModel _channel;
    
    public ReliablePublisher()
    {
        // Connect to cluster
        var factory = new ConnectionFactory
        {
            Endpoints = new List<AmqpTcpEndpoint>
            {
                new AmqpTcpEndpoint("node1"),
                new AmqpTcpEndpoint("node2"),
                new AmqpTcpEndpoint("node3")
            },
            AutomaticRecoveryEnabled = true
        };
        
        _connection = factory.CreateConnection();
        _channel = _connection.CreateModel();
        
        // Enable confirms
        _channel.ConfirmSelect();
    }
    
    public async Task<bool> PublishReliablyAsync(Order order)
    {
        try
        {
            // Create persistent message
            var properties = _channel.CreateBasicProperties();
            properties.Persistent = true;
            properties.MessageId = Guid.NewGuid().ToString();
            
            var body = JsonSerializer.SerializeToUtf8Bytes(order);
            
            // Publish to quorum queue
            _channel.BasicPublish(
                exchange: "",
                routingKey: "orders",
                basicProperties: properties,
                body: body
            );
            
            // Wait for confirm
            return _channel.WaitForConfirms(TimeSpan.FromSeconds(5));
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to publish message");
            return false;
        }
    }
}
```

**Advanced Publisher with Retry Logic:**

```csharp
public class AdvancedReliablePublisher
{
    private readonly IConnection _connection;
    private readonly IModel _channel;
    private readonly ConcurrentDictionary<ulong, PendingMessage> _pending = new();
    
    public AdvancedReliablePublisher()
    {
        var factory = new ConnectionFactory
        {
            Endpoints = new List<AmqpTcpEndpoint>
            {
                new AmqpTcpEndpoint("node1"),
                new AmqpTcpEndpoint("node2"),
                new AmqpTcpEndpoint("node3")
            },
            AutomaticRecoveryEnabled = true,
            NetworkRecoveryInterval = TimeSpan.FromSeconds(10),
            TopologyRecoveryEnabled = true,
            RequestedHeartbeat = TimeSpan.FromSeconds(30)
        };
        
        _connection = factory.CreateConnection();
        _channel = _connection.CreateModel();
        _channel.ConfirmSelect();
        
        SetupConfirmHandlers();
    }
    
    private void SetupConfirmHandlers()
    {
        _channel.BasicAcks += (sender, ea) =>
        {
            CleanPending(ea.DeliveryTag, ea.Multiple, success: true);
        };
        
        _channel.BasicNacks += (sender, ea) =>
        {
            CleanPending(ea.DeliveryTag, ea.Multiple, success: false);
            
            // Retry nacked messages
            if (_pending.TryGetValue(ea.DeliveryTag, out var msg))
            {
                RetryPublish(msg);
            }
        };
        
        _channel.BasicReturn += (sender, ea) =>
        {
            // Message was returned (unroutable)
            _logger.LogWarning($"Message returned: {ea.RoutingKey}");
            // Handle unroutable messages
        };
    }
    
    public async Task<bool> PublishWithRetryAsync(
        string exchange,
        string routingKey,
        object message,
        int maxRetries = 3)
    {
        var attempt = 0;
        
        while (attempt < maxRetries)
        {
            try
            {
                var deliveryTag = _channel.NextPublishSeqNo;
                var props = _channel.CreateBasicProperties();
                props.Persistent = true;
                props.MessageId = Guid.NewGuid().ToString();
                props.Timestamp = new AmqpTimestamp(
                    DateTimeOffset.UtcNow.ToUnixTimeSeconds());
                
                var body = Encoding.UTF8.GetBytes(
                    JsonSerializer.Serialize(message));
                
                // Track pending
                _pending[deliveryTag] = new PendingMessage
                {
                    Exchange = exchange,
                    RoutingKey = routingKey,
                    Body = body,
                    Properties = props,
                    PublishedAt = DateTime.UtcNow
                };
                
                _channel.BasicPublish(exchange, routingKey, props, body);
                
                // Wait for confirm
                if (_channel.WaitForConfirms(TimeSpan.FromSeconds(5)))
                {
                    return true;
                }
                
                attempt++;
                _logger.LogWarning($"Publish attempt {attempt} failed, retrying...");
                
                // Exponential backoff
                await Task.Delay(TimeSpan.FromSeconds(Math.Pow(2, attempt)));
            }
            catch (Exception ex)
            {
                attempt++;
                _logger.LogError(ex, $"Publish attempt {attempt} threw exception");
                
                if (attempt >= maxRetries)
                    throw;
                
                await Task.Delay(TimeSpan.FromSeconds(Math.Pow(2, attempt)));
            }
        }
        
        return false;
    }
    
    private void CleanPending(ulong deliveryTag, bool multiple, bool success)
    {
        if (multiple)
        {
            var confirmed = _pending.Where(kvp => kvp.Key <= deliveryTag);
            foreach (var entry in confirmed)
            {
                _pending.TryRemove(entry.Key, out _);
                
                if (success)
                {
                    _metrics.IncrementCounter("messages_confirmed");
                }
                else
                {
                    _metrics.IncrementCounter("messages_nacked");
                }
            }
        }
        else
        {
            _pending.TryRemove(deliveryTag, out _);
        }
    }
    
    private void RetryPublish(PendingMessage msg)
    {
        _logger.LogWarning("Retrying nacked message");
        
        Task.Run(async () =>
        {
            await Task.Delay(1000);
            await PublishWithRetryAsync(
                msg.Exchange, 
                msg.RoutingKey, 
                Encoding.UTF8.GetString(msg.Body));
        });
    }
}

public class PendingMessage
{
    public string Exchange { get; set; }
    public string RoutingKey { get; set; }
    public byte[] Body { get; set; }
    public IBasicProperties Properties { get; set; }
    public DateTime PublishedAt { get; set; }
}
```

**Client-Side Load Balancing:**

```csharp
public class LoadBalancedConnection
{
    private readonly List<IConnection> _connections = new();
    private int _currentIndex = 0;
    
    public void SetupMultipleConnections()
    {
        var nodes = new[] { "node1", "node2", "node3" };
        
        foreach (var node in nodes)
        {
            var factory = new ConnectionFactory
            {
                HostName = node,
                Port = 5672,
                AutomaticRecoveryEnabled = true,
                NetworkRecoveryInterval = TimeSpan.FromSeconds(10)
            };
            
            try
            {
                var connection = factory.CreateConnection();
                _connections.Add(connection);
                _logger.LogInformation($"Connected to {node}");
            }
            catch (Exception ex)
            {
                _logger.LogWarning($"Failed to connect to {node}: {ex.Message}");
            }
        }
    }
    
    public IModel GetChannel()
    {
        // Round-robin across connections
        var connection = _connections[_currentIndex];
        _currentIndex = (_currentIndex + 1) % _connections.Count;
        
        return connection.CreateModel();
    }
}
```

```csharp
public class PartitionHandling
{
    public void ConfigurePartitionMode()
    {
        // Configure partition mode in rabbitmq.conf:
        // cluster_partition_handling = pause_minority
        
        // Options:
        // - pause_minority: Pause nodes in minority partition
        // - autoheal: Automatically heal by restarting minority
        // - ignore: Do nothing (data loss risk)
        
        // For quorum queues: partition tolerance built-in
    }
    
    public async Task DetectPartition()
    {
        var client = new HttpClient();
        // ... auth
        
        var response = await client.GetAsync(
            "http://localhost:15672/api/nodes");
        var nodes = await response.Content.ReadFromJsonAsync<List<NodeInfo>>();
        
        // Check for partitions
        foreach (var node in nodes)
        {
            if (node.Partitions != null && node.Partitions.Any())
            {
                _logger.LogError($"Partition detected on {node.Name}!");
                _logger.LogError($"  Partitioned from: " +
                    string.Join(", ", node.Partitions));
                
                await _alertService.AlertAsync(
                    $"🚨 Network partition detected in RabbitMQ cluster!");
            }
        }
    }
```

## Key Takeaways

- **Publisher confirms** ensure broker received message (completes reliability with acks)
- **Clustering** provides high availability (multiple nodes share load)
- **Quorum queues** replicate data across nodes (Raft consensus)
- **Automatic recovery** handles connection failures (reconnect + topology recovery)
- **Complete reliability** requires: confirms + acks + persistence + quorum
- **Monitor cluster health** (node status, memory, disk, queue depth)
- **Handle partitions** (pause_minority or autoheal mode)
- **Client-side HA** (list all nodes, enable auto-recovery)
