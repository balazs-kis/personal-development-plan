## Introduction

An **event store** is the heart of event sourcing—a specialized database optimized for storing and retrieving event streams. Unlike traditional databases that store current state, event stores maintain a complete, immutable history of all changes to aggregates as a sequence of events.

Think of an event store as a ledger or journal where every transaction is recorded but never modified. This append-only nature provides a complete audit trail, enables time travel, and forms the foundation for CQRS architectures.

## Key Concepts Summary

### What is an Event Store?

An event store is a database optimized for:
- **Append-only writes**: Events are never updated or deleted
- **Sequential reads**: Replay events in order for aggregate reconstruction
- **Stream isolation**: Each aggregate has its own event stream
- **Optimistic concurrency**: Prevent conflicting writes using version numbers

```
Traditional DB:
Order { Id: 123, Status: "Shipped", Total: 100 }
↓ (Update overwrites previous state)
Order { Id: 123, Status: "Delivered", Total: 100 }

Event Store:
Stream: Order-123
  Event 1: OrderCreated { Id: 123, Total: 100 }
  Event 2: OrderPaid { Id: 123, Amount: 100 }
  Event 3: OrderShipped { Id: 123, Carrier: "FedEx" }
  Event 4: OrderDelivered { Id: 123, DeliveredAt: "2024-01-15" }
↓ (All history preserved, can replay to any point in time)
```

**Database table structure:**

```sql
CREATE TABLE Events (
    Id UNIQUEIDENTIFIER PRIMARY KEY,
    StreamId NVARCHAR(255) NOT NULL,
    EventType NVARCHAR(255) NOT NULL,
    EventData NVARCHAR(MAX) NOT NULL,
    Metadata NVARCHAR(MAX),
    Version INT NOT NULL,
    Timestamp DATETIME2 NOT NULL,
    UserId NVARCHAR(255),
    CorrelationId NVARCHAR(255),
    
    -- Optimistic concurrency: prevent duplicate versions
    CONSTRAINT UQ_Stream_Version UNIQUE (StreamId, Version),
    
    -- Fast retrieval of stream
    INDEX IX_StreamId_Version (StreamId, Version)
);
```

### Implementing an Event Store

```csharp
public interface IEventStore
{
    Task AppendEventsAsync(string streamId, IEnumerable<IDomainEvent> events, int expectedVersion);
    Task<IEnumerable<IDomainEvent>> GetEventsAsync(string streamId);
    Task<IEnumerable<IDomainEvent>> GetEventsAsync(string streamId, int fromVersion);
}

public class SqlEventStore : IEventStore
{
    private readonly DbContext _db;
    
    public async Task AppendEventsAsync(
        string streamId,
        IEnumerable<IDomainEvent> events,
        int expectedVersion)
    {
        var storedEvents = new List<StoredEvent>();
        var version = expectedVersion;
        
        foreach (var evt in events)
        {
            version++;
            
            storedEvents.Add(new StoredEvent
            {
                Id = Guid.NewGuid(),
                StreamId = streamId,
                EventType = evt.GetType().Name,
                EventData = JsonSerializer.Serialize(evt),
                Version = version,
                Timestamp = DateTime.UtcNow,
                UserId = GetCurrentUserId(),
                CorrelationId = GetCorrelationId()
            });
        }
        
        try
        {
            _db.Events.AddRange(storedEvents);
            await _db.SaveChangesAsync();
        }
        catch (DbUpdateException ex) when (IsUniqueConstraintViolation(ex))
        {
            throw new ConcurrencyException(
                $"Stream {streamId} was modified by another process. " +
                $"Expected version {expectedVersion}, please retry.");
        }
    }
    
    public async Task<IEnumerable<IDomainEvent>> GetEventsAsync(string streamId)
    {
        var storedEvents = await _db.Events
            .Where(e => e.StreamId == streamId)
            .OrderBy(e => e.Version)
            .ToListAsync();
        
        return storedEvents.Select(DeserializeEvent);
    }
    
    public async Task<IEnumerable<IDomainEvent>> GetEventsAsync(string streamId, int fromVersion)
    {
        var storedEvents = await _db.Events
            .Where(e => e.StreamId == streamId && e.Version >= fromVersion)
            .OrderBy(e => e.Version)
            .ToListAsync();
        
        return storedEvents.Select(DeserializeEvent);
    }
    
    private IDomainEvent DeserializeEvent(StoredEvent stored)
    {
        var eventType = Type.GetType(stored.EventType);
        return (IDomainEvent)JsonSerializer.Deserialize(stored.EventData, eventType);
    }
}
```

### Optimistic Concurrency Control

```csharp
public class OrderAggregate
{
    private readonly List<IDomainEvent> _uncommittedEvents = new();
    
    public string Id { get; private set; }
    public int Version { get; private set; }  // Current version
    
    // Apply events to rebuild state
    public void Load(IEnumerable<IDomainEvent> events)
    {
        foreach (var evt in events)
        {
            Apply(evt);
            Version++;
        }
    }
    
    // Business method that generates events
    public void PlaceOrder(string customerId, List<OrderItem> items)
    {
        var evt = new OrderPlacedEvent
        {
            OrderId = Id,
            CustomerId = customerId,
            Items = items
        };
        
        Apply(evt);
        _uncommittedEvents.Add(evt);
    }
    
    public IEnumerable<IDomainEvent> GetUncommittedEvents() => _uncommittedEvents;
    public void MarkEventsAsCommitted() => _uncommittedEvents.Clear();
}

// Repository with optimistic concurrency
public class OrderRepository
{
    private readonly IEventStore _eventStore;
    
    public async Task<OrderAggregate> GetByIdAsync(string orderId)
    {
        var events = await _eventStore.GetEventsAsync($"Order-{orderId}");
        
        var order = new OrderAggregate { Id = orderId };
        order.Load(events);
        
        return order;
    }
    
    public async Task SaveAsync(OrderAggregate order)
    {
        var events = order.GetUncommittedEvents();
        
        // Save with optimistic concurrency check
        await _eventStore.AppendEventsAsync(
            $"Order-{order.Id}",
            events,
            order.Version  // Expected version
        );
        
        order.MarkEventsAsCommitted();
    }
}

// Usage with concurrent modification handling
try
{
    var order = await _repository.GetByIdAsync("123");
    order.Ship("FedEx", "1234567890");
    await _repository.SaveAsync(order);
}
catch (ConcurrencyException)
{
    // Another process modified this order
    // Reload and retry
    var order = await _repository.GetByIdAsync("123");
    order.Ship("FedEx", "1234567890");
    await _repository.SaveAsync(order);
}
```

### Storage Backend Options

**1. Relational Database (SQL Server, PostgreSQL)**
- ✅ Familiar technology
- ✅ ACID transactions
- ✅ Good for moderate scale
- ❌ May need partitioning at very large scale

**2. EventStoreDB (Purpose-built)**
- ✅ Optimized for event sourcing
- ✅ Built-in projections
- ✅ Stream subscriptions
- ❌ Additional infrastructure

**3. Azure Cosmos DB / DynamoDB**
- ✅ Highly scalable
- ✅ Global distribution
- ✅ Low latency
- ❌ Eventually consistent (careful with concurrency)

**4. Kafka (Event Streaming)**
- ✅ Extremely scalable
- ✅ Built-in retention
- ✅ Stream processing
- ❌ No queries, need separate read store

### Best Practices

1. **Immutable Events**: Never modify or delete events
2. **Unique Stream IDs**: One stream per aggregate instance
3. **Sequential Versions**: Use integer versions (1, 2, 3...)
4. **Small Events**: Keep event payloads small and focused
5. **Metadata**: Capture who, when, why for audit trails
6. **Idempotency**: Design events to be safe when replayed

### Advanced Event Store Patterns

**Pattern 1: Stream Categories**

```csharp
// Organize streams by category
public static class StreamIds
{
    public static string ForOrder(string orderId) => $"Order-{orderId}";
    public static string ForCustomer(string customerId) => $"Customer-{customerId}";
    public static string ForInventory(string productId) => $"Inventory-{productId}";
    
    public static string GetCategory(string streamId) => streamId.Split('-')[0];
    public static string GetId(string streamId) => streamId.Split('-')[1];
}

// Query all streams in a category
public async Task<IEnumerable<string>> GetOrderStreams()
{
    return await _db.Events
        .Where(e => e.StreamId.StartsWith("Order-"))
        .Select(e => e.StreamId)
        .Distinct()
        .ToListAsync();
}
```

**Pattern 2: Global Event Sequence**

```csharp
// Add global sequence number for ordered event processing
public class StoredEvent
{
    public Guid Id { get; set; }
    public long GlobalSequence { get; set; }  // Auto-incrementing
    public string StreamId { get; set; }
    public int StreamVersion { get; set; }
    public string EventType { get; set; }
    public string EventData { get; set; }
    public DateTime Timestamp { get; set; }
}

// Subscribe to all events in order
public async Task<IAsyncEnumerable<IDomainEvent>> SubscribeToAll(long fromSequence = 0)
{
    return _db.Events
        .Where(e => e.GlobalSequence > fromSequence)
        .OrderBy(e => e.GlobalSequence)
        .AsAsyncEnumerable()
        .Select(DeserializeEvent);
}
```

**Pattern 3: Event Subscriptions**

```csharp
public interface IEventSubscription
{
    Task OnEventAppended(string streamId, IDomainEvent evt, int version);
}

public class EventStoreWithSubscriptions : IEventStore
{
    private readonly List<IEventSubscription> _subscriptions = new();
    
    public void Subscribe(IEventSubscription subscription)
    {
        _subscriptions.Add(subscription);
    }
    
    public async Task AppendEventsAsync(string streamId, IEnumerable<IDomainEvent> events, int expectedVersion)
    {
        // Append to store
        var version = expectedVersion;
        var storedEvents = new List<StoredEvent>();
        
        foreach (var evt in events)
        {
            version++;
            storedEvents.Add(CreateStoredEvent(streamId, evt, version));
        }
        
        await _db.Events.AddRangeAsync(storedEvents);
        await _db.SaveChangesAsync();
        
        // Notify subscribers
        foreach (var stored in storedEvents)
        {
            var evt = DeserializeEvent(stored);
            foreach (var subscription in _subscriptions)
            {
                await subscription.OnEventAppended(streamId, evt, stored.Version);
            }
        }
    }
}

// Projection subscriber
public class ProjectionSubscription : IEventSubscription
{
    private readonly IProjection _projection;
    
    public async Task OnEventAppended(string streamId, IDomainEvent evt, int version)
    {
        if (_projection.CanHandle(evt))
        {
            await _projection.HandleAsync(evt);
        }
    }
}
```

**Caching Frequently Accessed Streams:**

```csharp
public class CachedEventStore : IEventStore
{
    private readonly IEventStore _inner;
    private readonly IMemoryCache _cache;
    private readonly int _cacheMinutes;
    
    public async Task<IEnumerable<IDomainEvent>> GetEventsAsync(string streamId)
    {
        var cacheKey = $"events:{streamId}";
        
        if (_cache.TryGetValue(cacheKey, out IEnumerable<IDomainEvent> cached))
        {
            return cached;
        }
        
        var events = await _inner.GetEventsAsync(streamId);
        
        _cache.Set(cacheKey, events, TimeSpan.FromMinutes(_cacheMinutes));
        
        return events;
    }
    
    public async Task AppendEventsAsync(string streamId, IEnumerable<IDomainEvent> events, int expectedVersion)
    {
        await _inner.AppendEventsAsync(streamId, events, expectedVersion);
        
        // Invalidate cache
        _cache.Remove($"events:{streamId}");
    }
}
```

### Event Store Migration Strategies

**Migrating from Traditional Database:**

```csharp
public class EventStoreMigration
{
    public async Task MigrateOrdersToEvents()
    {
        var orders = await _legacyDb.Orders.Include(o => o.Items).ToListAsync();
        
        foreach (var order in orders)
        {
            var events = new List<IDomainEvent>();
            
            // Create synthetic events from current state
            events.Add(new OrderPlacedEvent
            {
                OrderId = order.Id,
                CustomerId = order.CustomerId,
                Items = order.Items.Select(i => new OrderItem
                {
                    ProductId = i.ProductId,
                    Quantity = i.Quantity,
                    Price = i.Price
                }).ToList(),
                PlacedAt = order.CreatedAt
            });
            
            if (order.PaidAt.HasValue)
            {
                events.Add(new OrderPaidEvent
                {
                    OrderId = order.Id,
                    PaidAt = order.PaidAt.Value,
                    Amount = order.TotalAmount
                });
            }
            
            if (order.ShippedAt.HasValue)
            {
                events.Add(new OrderShippedEvent
                {
                    OrderId = order.Id,
                    ShippedAt = order.ShippedAt.Value,
                    Carrier = order.ShippingCarrier
                });
            }
            
            // Append synthetic event history
            await _eventStore.AppendEventsAsync($"Order-{order.Id}", events, 0);
        }
    }
}
```

### Monitoring and Diagnostics

```csharp
public class EventStoreMetrics
{
    private readonly Counter _eventsAppended;
    private readonly Counter _streamsCreated;
    private readonly Histogram _appendDuration;
    private readonly Gauge _totalEvents;
    
    public EventStoreMetrics(IMeterFactory meterFactory)
    {
        var meter = meterFactory.Create("EventStore");
        
        _eventsAppended = meter.CreateCounter<long>("events_appended_total");
        _streamsCreated = meter.CreateCounter<long>("streams_created_total");
        _appendDuration = meter.CreateHistogram<double>("append_duration_seconds");
        _totalEvents = meter.CreateGauge<long>("total_events");
    }
    
    public void RecordAppend(string streamId, int eventCount, TimeSpan duration)
    {
        var category = StreamIds.GetCategory(streamId);
        
        _eventsAppended.Add(eventCount, new("category", category));
        _appendDuration.Record(duration.TotalSeconds, new("category", category));
    }
    
    public async Task UpdateTotalEvents(IEventStore eventStore)
    {
        var count = await ((SqlEventStore)eventStore).GetTotalEventCount();
        _totalEvents.Record(count);
    }
}

// Diagnostic queries
public class EventStoreDiagnostics
{
    public async Task<EventStoreStats> GetStats()
    {
        return new EventStoreStats
        {
            TotalEvents = await _db.Events.CountAsync(),
            TotalStreams = await _db.Events.Select(e => e.StreamId).Distinct().CountAsync(),
            AverageEventsPerStream = await _db.Events
                .GroupBy(e => e.StreamId)
                .Select(g => g.Count())
                .AverageAsync(),
            LargestStream = await _db.Events
                .GroupBy(e => e.StreamId)
                .Select(g => new { StreamId = g.Key, Count = g.Count() })
                .OrderByDescending(x => x.Count)
                .FirstOrDefaultAsync()
        };
    }
    
    public async Task<IEnumerable<StreamHealth>> GetUnhealthyStreams(int threshold = 10000)
    {
        return await _db.Events
            .GroupBy(e => e.StreamId)
            .Select(g => new StreamHealth
            {
                StreamId = g.Key,
                EventCount = g.Count(),
                LastEventAt = g.Max(e => e.Timestamp)
            })
            .Where(s => s.EventCount > threshold)
            .ToListAsync();
    }
}
```

```csharp
public class EventStoreTests
{
    [Fact]
    public async Task AppendEvents_IncreasesVersion()
    {
        // Arrange
        var store = CreateEventStore();
        var event1 = new OrderPlacedEvent { OrderId = "123" };

        var store = CreateEventStore();

        // Act & Assert
        await Assert.ThrowsAsync<ConcurrencyException>(async () =>
            await store.AppendEventsAsync("Order-123", new[] { new OrderPaidEvent() }, 0)); // Wrong version!

        // Arrange
```
