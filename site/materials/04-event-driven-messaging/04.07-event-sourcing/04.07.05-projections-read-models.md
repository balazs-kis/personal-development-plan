## Introduction

**Projections** transform event streams into queryable read models. While event stores are optimized for writes and replaying history, they're terrible for queries like "find all unpaid orders" or "show customer purchase history." Projections solve this by building specialized read databases optimized for specific queries.

This is the "Q" (Query) side of CQRS—separate read models built from events, enabling complex queries without compromising the write model.

### Building a Read Model

```csharp
// Event
public class OrderPlacedEvent
{
    public string OrderId { get; set; }
    public string CustomerId { get; set; }
    public decimal TotalAmount { get; set; }
    public DateTime PlacedAt { get; set; }
}

// Read Model (optimized for queries)
public class OrderSummary
{
    public string OrderId { get; set; }
    public string CustomerId { get; set; }
    public string CustomerName { get; set; }  // Denormalized!
    public decimal TotalAmount { get; set; }
    public OrderStatus Status { get; set; }
    public DateTime PlacedAt { get; set; }
    public DateTime? ShippedAt { get; set; }
}

// Projection
public class OrderSummaryProjection
{
    private readonly DbContext _readDb;
    
    public async Task Handle(OrderPlacedEvent evt)
    {
        var customer = await _customerService.GetCustomerAsync(evt.CustomerId);
        
        _readDb.OrderSummaries.Add(new OrderSummary
        {
            OrderId = evt.OrderId,
            CustomerId = evt.CustomerId,
            CustomerName = customer.Name,  // Denormalize for query speed
            TotalAmount = evt.TotalAmount,
            Status = OrderStatus.Placed,
            PlacedAt = evt.PlacedAt
        });
        
        await _readDb.SaveChangesAsync();
    }
    
    public async Task Handle(OrderShippedEvent evt)
    {
        var summary = await _readDb.OrderSummaries.FindAsync(evt.OrderId);
        summary.Status = OrderStatus.Shipped;
        summary.ShippedAt = evt.ShippedAt;
        
        await _readDb.SaveChangesAsync();
    }
}
```

### Projection Manager

```csharp
public class ProjectionManager
{
    private readonly IEventStore _eventStore;
    private readonly IEnumerable<IProjection> _projections;
    
    public async Task RebuildAllProjections()
    {
        // Clear existing read models
        foreach (var projection in _projections)
        {
            await projection.ResetAsync();
        }
        
        // Replay all events
        var allStreams = await _eventStore.GetAllStreamsAsync();
        
        foreach (var streamId in allStreams)
        {
            var events = await _eventStore.GetEventsAsync(streamId);
            
            foreach (var evt in events)
            {
                foreach (var projection in _projections)
                {
                    if (projection.CanHandle(evt))
                    {
                        await projection.HandleAsync(evt);
                    }
                }
            }
        }
    }
    
    // Subscribe to new events
    public async Task StartListening()
    {
        await _eventStore.SubscribeToAll(async evt =>
        {
            foreach (var projection in _projections)
            {
                if (projection.CanHandle(evt))
                {
                    await projection.HandleAsync(evt);
                }
            }
        });
    }
}
```

### Eventual Consistency

```csharp
// Command side: Immediate
public async Task<string> PlaceOrder(PlaceOrderCommand cmd)
{
    var order = new OrderAggregate();
    order.PlaceOrder(cmd.CustomerId, cmd.Items);
    
    await _repository.SaveAsync(order);  // Events saved immediately
    
    return order.Id;
}

// Query side: Eventually consistent
public async Task<OrderSummary> GetOrder(string orderId)
{
    // May not see order immediately after placement
    // Projection needs time to process events
    
    var summary = await _readDb.OrderSummaries.FindAsync(orderId);
    
    if (summary == null)
    {
        // Option 1: Return 404 (eventual consistency)
        throw new NotFoundException();
        
        // Option 2: Fall back to event store
        var order = await _eventStoreRepo.GetByIdAsync(orderId);
        return MapToSummary(order);
    }
    
    return summary;
}
```

### Handling Projection Failures

```csharp
public class ResilientProjection : IProjection
{
    private readonly IProjection _inner;
    private readonly ILogger _logger;
    
    public async Task HandleAsync(IDomainEvent evt)
    {
        try
        {
            await _inner.HandleAsync(evt);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Projection failed for event {EventType}", evt.GetType().Name);
            
            // Store failed event for retry
            await _failedEventStore.SaveAsync(new FailedProjection
            {
                EventId = evt.Id,
                EventData = JsonSerializer.Serialize(evt),
                ProjectionName = _inner.GetType().Name,
                FailedAt = DateTime.UtcNow,
                Error = ex.Message
            });
        }
    }
}

// Retry failed projections
public class ProjectionRetryService : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            var failed = await _failedEventStore.GetPendingAsync();
            
            foreach (var failure in failed)
            {
                try
                {
                    var evt = DeserializeEvent(failure.EventData);
                    await _projection.HandleAsync(evt);
                    
                    await _failedEventStore.MarkResolvedAsync(failure.Id);
                }
                catch
                {
                    // Still failing, will retry later
                }
            }
            
            await Task.Delay(TimeSpan.FromMinutes(1), ct);
        }
    }
}
```

### Advanced Projection Patterns

**Pattern 1: Denormalized Read Models**

```csharp
// Projection that denormalizes data from multiple aggregates
public class OrderDetailsProjection : IProjection
{
    private readonly ReadDbContext _readDb;
    private readonly ICustomerService _customerService;
    private readonly IProductService _productService;
    
    public async Task HandleAsync(OrderPlacedEvent evt)
    {
        // Fetch related data
        var customer = await _customerService.GetAsync(evt.CustomerId);
        var products = await _productService.GetManyAsync(
            evt.Items.Select(i => i.ProductId));
        
        // Create denormalized view
        var details = new OrderDetails
        {
            OrderId = evt.OrderId,
            
            // Customer info (denormalized)
            CustomerId = customer.Id,
            CustomerName = customer.Name,
            CustomerEmail = customer.Email,
            
            // Items with product details (denormalized)
            Items = evt.Items.Select(item =>
            {
                var product = products.First(p => p.Id == item.ProductId);
                return new OrderItemDetails
                {
                    ProductId = item.ProductId,
                    ProductName = product.Name,
                    ProductImage = product.ImageUrl,
                    Quantity = item.Quantity,
                    UnitPrice = item.Price
                };
            }).ToList(),
            
            TotalAmount = evt.TotalAmount,
            PlacedAt = evt.PlacedAt
        };
        
        _readDb.OrderDetails.Add(details);
        await _readDb.SaveChangesAsync();
    }
}
```

**Pattern 2: Aggregated Statistics**

```csharp
public class CustomerStatsProjection : IProjection
{
    public async Task HandleAsync(OrderPlacedEvent evt)
    {
        var stats = await _db.CustomerStats.FindAsync(evt.CustomerId)
                    ?? new CustomerOrderStats { CustomerId = evt.CustomerId };
        
        stats.TotalOrders++;
        stats.TotalSpent += evt.TotalAmount;
        stats.AverageOrderValue = stats.TotalSpent / stats.TotalOrders;
        stats.LastOrderDate = evt.PlacedAt;
        
        if (stats.TotalOrders == 1)
            _db.CustomerStats.Add(stats);
        
        await _db.SaveChangesAsync();
    }
    
    public async Task HandleAsync(OrderCancelledEvent evt)
    {
        var stats = await _db.CustomerStats.FindAsync(evt.CustomerId);
        if (stats == null) return;
        
        stats.TotalOrders--;
        stats.TotalSpent -= evt.RefundAmount;
        stats.AverageOrderValue = stats.TotalOrders > 0 
            ? stats.TotalSpent / stats.TotalOrders 
            : 0;
        
        await _db.SaveChangesAsync();
    }
}
```

### Projection Coordination

```csharp
// Ensure projections process events in order
public class OrderedProjectionManager
{
    private readonly Dictionary<string, long> _positions = new();
    private readonly SemaphoreSlim _lock = new(1);
    
    public async Task ProcessEvent(IDomainEvent evt, long globalSequence)
    {
        await _lock.WaitAsync();
        
        try
        {
            foreach (var projection in _projections)
            {
                var projectionId = projection.GetType().Name;
                var lastProcessed = _positions.GetValueOrDefault(projectionId, 0);
                
                if (globalSequence != lastProcessed + 1)
                {
                    throw new InvalidOperationException(
                        $"Projection {projectionId} out of sync. " +
                        $"Expected sequence {lastProcessed + 1}, got {globalSequence}");
                }
                
                await projection.HandleAsync(evt);
                _positions[projectionId] = globalSequence;
                
                // Persist position
                await SavePosition(projectionId, globalSequence);
            }
        }
        finally
        {
            _lock.Release();
        }
    }
}
```

### Projection Monitoring

```csharp
public class ProjectionHealthMonitor : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            foreach (var projection in _projections)
            {
                var health = await CheckProjectionHealth(projection);
                
                if (!health.IsHealthy)
                {
                    _logger.LogWarning(
                        "Projection {ProjectionName} unhealthy: {Reason}",
                        projection.Name,
                        health.Reason);
                    
                    // Alert or auto-recover
                    if (health.Reason == "Stale")
                    {
                        await projection.CatchUpAsync();
                    }
                }
            }
            
            await Task.Delay(TimeSpan.FromMinutes(1), ct);
        }
    }
    
    private async Task<ProjectionHealth> CheckProjectionHealth(IProjection projection)
    {
        var lastProcessed = await GetLastProcessedSequence(projection);
        var latestEvent = await _eventStore.GetLatestGlobalSequence();
        
        var lag = latestEvent - lastProcessed;
        
        return new ProjectionHealth
        {
            IsHealthy = lag < 1000,  // Less than 1000 events behind
            Lag = lag,
            Reason = lag >= 1000 ? "Stale" : null
        };
    }
}
```

### Projection Rebuilding

```csharp
public class ProjectionRebuilder
{
    public async Task RebuildProjection(Type projectionType)
    {
        var projection = _projections.First(p => p.GetType() == projectionType);
        
        // 1. Stop live processing
        await projection.StopAsync();
        
        // 2. Clear existing read model
        await projection.ResetAsync();
        
        // 3. Replay all events
        var allEvents = await _eventStore.GetAllEventsAsync();
        
        var processedCount = 0;
        foreach (var evt in allEvents)
        {
            if (projection.CanHandle(evt))
            {
                await projection.HandleAsync(evt);
                processedCount++;
                
                if (processedCount % 1000 == 0)
                {
                    Console.WriteLine($"Rebuilt {processedCount} events...");
                }
            }
        }
        
        // 4. Resume live processing
        await projection.StartAsync();
        
        Console.WriteLine($"Projection {projectionType.Name} rebuilt: {processedCount} events");
    }
    
    // Blue-green rebuild (no downtime)
    public async Task RebuildProjectionBlueGreen(Type projectionType)
    {
        // Build into new database/tables
        var newProjection = CreateProjectionInstance(projectionType);
        await ConfigureGreenDatabase(newProjection);
        
        // Rebuild in background
        await RebuildProjection(newProjection.GetType());
        
        // Switch traffic to new projection
        await SwapProjections(projectionType, newProjection);
        
        // Clean up old projection
        await CleanupBlueProjection(projectionType);
    }
}
```

```csharp
        var projection = new OrderSummaryProjection(_readDb);

        {
            await projection.HandleAsync(evt);
        }
        
        // Assert
        var summary = await _readDb.OrderSummaries.FindAsync("123");
        Assert.NotNull(summary);
        Assert.Equal(OrderStatus.Shipped, summary.Status);
        Assert.Equal(100m, summary.TotalAmount);
    }
    
    [Fact]
    public async Task Projection_HandlesOutOfOrderEvents()
    {
        // Arrange: Ship before pay (should handle gracefully)
        var projection = new OrderSummaryProjection(_readDb);
        
        // Act
        await projection.HandleAsync(new OrderPlacedEvent { OrderId = "123" });
        await projection.HandleAsync(new OrderShippedEvent { OrderId = "123" });
        await projection.HandleAsync(new OrderPaidEvent { OrderId = "123" });
        
        // Assert: Projection should still be valid
        var summary = await _readDb.OrderSummaries.FindAsync("123");
        Assert.Equal(OrderStatus.Shipped, summary.Status);
    }
```
