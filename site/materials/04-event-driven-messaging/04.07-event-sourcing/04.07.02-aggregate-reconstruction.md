## Introduction

**Aggregate reconstruction** is the process of rebuilding an aggregate's current state by replaying all its events in sequence. This is the core mechanism of event sourcing—instead of loading state from a database row, you load and apply events one by one until you reach the current state.

Understanding reconstruction is essential for working with event-sourced systems, as it determines how aggregates are loaded, how business logic is applied, and how consistency is maintained.

## Key Concepts Summary

### Reconstruction Process

```csharp
// 1. Load events from store
var events = await _eventStore.GetEventsAsync("Order-123");

// 2. Create aggregate instance
var order = new OrderAggregate { Id = "123" };

// 3. Replay events to rebuild state
foreach (var evt in events)
{
    order.Apply(evt);
}

// 4. Aggregate now represents current state
Console.WriteLine($"Order status: {order.Status}");
Console.WriteLine($"Order total: {order.TotalAmount}");
```

### Implementing Apply Pattern

```csharp
public class OrderAggregate
{
    public string Id { get; private set; }
    public OrderStatus Status { get; private set; }
    public decimal TotalAmount { get; private set; }
    public List<OrderItem> Items { get; private set; } = new();
    public int Version { get; private set; }
    
    // Public method: business logic + event generation
    public void PlaceOrder(string customerId, List<OrderItem> items)
    {
        // Validate
        if (items == null || !items.Any())
            throw new InvalidOperationException("Order must have items");
        
        // Generate event
        var evt = new OrderPlacedEvent
        {
            OrderId = Id,
            CustomerId = customerId,
            Items = items,
            TotalAmount = items.Sum(i => i.Price * i.Quantity)
        };
        
        // Apply event (updates state)
        Apply(evt);
        
        // Store for persistence
        _uncommittedEvents.Add(evt);
    }
    
    // Internal: Apply event to state (used for both new events and replay)
    private void Apply(OrderPlacedEvent evt)
    {
        Id = evt.OrderId;
        Status = OrderStatus.Placed;
        Items = evt.Items.ToList();
        TotalAmount = evt.TotalAmount;
    }
    
    private void Apply(OrderPaidEvent evt)
    {
        Status = OrderStatus.Paid;
    }
    
    private void Apply(OrderShippedEvent evt)
    {
        Status = OrderStatus.Shipped;
    }
    
    // Generic apply dispatcher
    public void Apply(IDomainEvent evt)
    {
        var method = GetType().GetMethod("Apply",
            BindingFlags.NonPublic | BindingFlags.Instance,
            null,
            new[] { evt.GetType() },
            null);
        
        method?.Invoke(this, new object[] { evt });
        Version++;
    }
    
    // Reconstruct from events
    public void Load(IEnumerable<IDomainEvent> events)
    {
        foreach (var evt in events)
        {
            Apply(evt);
        }
    }
}
```

### Complete Repository Implementation

```csharp
public class EventSourcedRepository<T> where T : AggregateRoot, new()
{
    private readonly IEventStore _eventStore;
    
    public async Task<T> GetByIdAsync(string aggregateId)
    {
        // Load all events for this aggregate
        var streamId = $"{typeof(T).Name}-{aggregateId}";
        var events = await _eventStore.GetEventsAsync(streamId);
        
        if (!events.Any())
            return null;  // Aggregate doesn't exist
        
        // Create and reconstruct aggregate
        var aggregate = new T();
        aggregate.Id = aggregateId;
        aggregate.Load(events);
        
        return aggregate;
    }
    
    public async Task SaveAsync(T aggregate)
    {
        var events = aggregate.GetUncommittedEvents();
        if (!events.Any())
            return;  // No changes
        
        var streamId = $"{typeof(T).Name}-{aggregate.Id}";
        
        await _eventStore.AppendEventsAsync(
            streamId,
            events,
            aggregate.Version - events.Count()  // Expected version before new events
        );
        
        aggregate.MarkEventsAsCommitted();
    }
}

// Usage
var repository = new EventSourcedRepository<OrderAggregate>();

// Load existing order
var order = await repository.GetByIdAsync("123");

// Make changes
order.Ship("FedEx", "TRACK123");

// Save (appends ShippedEvent)
await repository.SaveAsync(order);
```

### Performance Optimization

```csharp
// Problem: Long event streams are slow to replay
var events = await _eventStore.GetEventsAsync("Order-123");  // 10,000 events!

foreach (var evt in events)  // Slow!
{
    order.Apply(evt);
}

// Solution 1: Load from snapshot (covered in Lesson 4)
var snapshot = await _snapshotStore.GetSnapshotAsync("Order-123");
var events = await _eventStore.GetEventsAsync("Order-123", snapshot.Version + 1);

order.RestoreFromSnapshot(snapshot);
foreach (var evt in events)  // Only replay events since snapshot
{
    order.Apply(evt);
}

// Solution 2: Cache frequently accessed aggregates
private readonly IMemoryCache _cache;

public async Task<T> GetByIdAsync(string id)
{
    if (_cache.TryGetValue(id, out T aggregate))
        return aggregate;
    
    aggregate = await LoadFromEventsAsync(id);
    _cache.Set(id, aggregate, TimeSpan.FromMinutes(5));
    
    return aggregate;
}
```

### Handling Invalid State

```csharp
public class OrderAggregate
{
    public void Cancel()
    {
        // Business rule: can't cancel shipped orders
        if (Status == OrderStatus.Shipped || Status == OrderStatus.Delivered)
        {
            throw new InvalidOperationException(
                "Cannot cancel order that has been shipped");
        }
        
        var evt = new OrderCancelledEvent { OrderId = Id };
        Apply(evt);
        _uncommittedEvents.Add(evt);
    }
    
    private void Apply(OrderCancelledEvent evt)
    {
        // Only update state, don't validate
        // (Events represent facts that already happened)
        Status = OrderStatus.Cancelled;
    }
}
```

### Time Travel and Temporal Queries

```csharp
public class OrderRepository
{
    // Get order state at specific point in time
    public async Task<OrderAggregate> GetAtPointInTimeAsync(
        string orderId,
        DateTime pointInTime)
    {
        var events = await _eventStore.GetEventsAsync($"Order-{orderId}");
        
        // Only replay events up to the specified time
        var historicalEvents = events.Where(e => e.Timestamp <= pointInTime);
        
        var order = new OrderAggregate { Id = orderId };
        order.Load(historicalEvents);
        
        return order;
    }
    
    // Get order state at specific version
    public async Task<OrderAggregate> GetAtVersionAsync(
        string orderId,
        int version)
    {
        var events = await _eventStore.GetEventsAsync($"Order-{orderId}");
        
        var historicalEvents = events.Take(version);
        
        var order = new OrderAggregate { Id = orderId };
        order.Load(historicalEvents);
        
        return order;
    }
}

// Usage: Audit query
var orderYesterday = await repository.GetAtPointInTimeAsync(
    "123",
    DateTime.UtcNow.AddDays(-1)
);

Console.WriteLine($"Order status yesterday: {orderYesterday.Status}");
```

### Advanced Apply Patterns

**Pattern 1: Reflective Dispatcher**

```csharp
public abstract class AggregateRoot
{
    private readonly Dictionary<Type, Action<IDomainEvent>> _handlers = new();
    
    protected AggregateRoot()
    {
        // Register handlers via reflection
        var handlerMethods = GetType()
            .GetMethods(BindingFlags.NonPublic | BindingFlags.Instance)
            .Where(m => m.Name == "Apply" && m.GetParameters().Length == 1)
            .ToList();
        
        foreach (var method in handlerMethods)
        {
            var eventType = method.GetParameters()[0].ParameterType;
            _handlers[eventType] = evt => method.Invoke(this, new object[] { evt });
        }
    }
    
    public void Apply(IDomainEvent evt)
    {
        if (_handlers.TryGetValue(evt.GetType(), out var handler))
        {
            handler(evt);
            Version++;
        }
    }
}
```

**Pattern 2: Event Sourced Entities (Not Just Aggregates)**

```csharp
public class OrderAggregate : AggregateRoot
{
    public string Id { get; private set; }
    public List<OrderLine> Lines { get; private set; } = new();
    
    private void Apply(OrderPlacedEvent evt)
    {
        Id = evt.OrderId;
    }
    
    private void Apply(OrderLineAddedEvent evt)
    {
        var line = new OrderLine(evt.LineId);
        line.Apply(new LineItemAddedEvent
        {
            ProductId = evt.ProductId,
            Quantity = evt.Quantity,
            Price = evt.Price
        });
        
        Lines.Add(line);
    }
}

// Child entity also event-sourced
public class OrderLine
{
    public string LineId { get; private set; }
    public string ProductId { get; private set; }
    public int Quantity { get; private set; }
    public decimal Price { get; private set; }
    
    public OrderLine(string lineId)
    {
        LineId = lineId;
    }
    
    public void Apply(LineItemAddedEvent evt)
    {
        ProductId = evt.ProductId;
        Quantity = evt.Quantity;
        Price = evt.Price;
    }
    
    public void Apply(LineQuantityChangedEvent evt)
    {
        Quantity = evt.NewQuantity;
    }
}
```

**Pattern 3: State Transition Validation**

```csharp
public class OrderAggregate : AggregateRoot
{
    public OrderStatus Status { get; private set; }
    
    public void Ship(string carrier, string trackingNumber)
    {
        // Validate state transitions
        if (Status != OrderStatus.Paid)
        {
            throw new InvalidOperationException(
                $"Cannot ship order in status {Status}. Order must be paid first.");
        }
        
        var evt = new OrderShippedEvent
        {
            OrderId = Id,
            Carrier = carrier,
            TrackingNumber = trackingNumber,
            ShippedAt = DateTime.UtcNow
        };
        
        Apply(evt);
        _uncommittedEvents.Add(evt);
    }
    
    private void Apply(OrderShippedEvent evt)
    {
        // Apply without validation (event already happened)
        Status = OrderStatus.Shipped;
    }
}
```

### Debugging Aggregate State

```csharp
public static class AggregateDebugger
{
    public static async Task<string> GenerateDebugReport<T>(
        string aggregateId,
        IEventStore eventStore) where T : AggregateRoot, new()
    {
        var streamId = $"{typeof(T).Name}-{aggregateId}";
        var events = await eventStore.GetEventsAsync(streamId);
        
        var sb = new StringBuilder();
        sb.AppendLine($"=== Aggregate Debug Report ===");
        sb.AppendLine($"Type: {typeof(T).Name}");
        sb.AppendLine($"ID: {aggregateId}");
        sb.AppendLine($"Total Events: {events.Count()}");
        sb.AppendLine();
        
        var aggregate = new T { Id = aggregateId };
        var version = 0;
        
        foreach (var evt in events)
        {
            version++;
            
            sb.AppendLine($"[{version}] {evt.GetType().Name} at {evt.Timestamp:yyyy-MM-dd HH:mm:ss}");
            
            // Snapshot state before applying event
            var stateBefore = SerializeState(aggregate);
            
            aggregate.Apply(evt);
            
            // Snapshot state after applying event
            var stateAfter = SerializeState(aggregate);
            
            // Show state changes
            var changes = CompareStates(stateBefore, stateAfter);
            foreach (var change in changes)
            {
                sb.AppendLine($"  {change.Property}: {change.OldValue} → {change.NewValue}");
            }
            
            sb.AppendLine();
        }
        
        sb.AppendLine($"=== Final State ===");
        sb.AppendLine(JsonSerializer.Serialize(aggregate, new JsonSerializerOptions { WriteIndented = true }));
        
        return sb.ToString();
    }
}

// Usage:
var report = await AggregateDebugger.GenerateDebugReport<OrderAggregate>(
    "123",
    _eventStore
);
File.WriteAllText("order-123-debug.txt", report);
```

### Performance Patterns

**Lazy Loading of Child Collections:**

```csharp
public class OrderAggregate : AggregateRoot
{
    private List<OrderLine> _lines;
    public IReadOnlyList<OrderLine> Lines => _lines ??= new List<OrderLine>();
    
    // Only materialize lines when accessed
    private void Apply(OrderLineAddedEvent evt)
    {
        (_lines ??= new List<OrderLine>()).Add(new OrderLine
        {
            LineId = evt.LineId,
            ProductId = evt.ProductId,
            Quantity = evt.Quantity,
            Price = evt.Price
        });
    }
}
```

**Parallel Event Application:**

```csharp
public class ParallelLoadingRepository<T> where T : AggregateRoot, new()
{
    public async Task<IEnumerable<T>> GetManyAsync(IEnumerable<string> aggregateIds)
    {
        // Load multiple aggregates in parallel
        var tasks = aggregateIds.Select(async id =>
        {
            var events = await _eventStore.GetEventsAsync($"{typeof(T).Name}-{id}");
            var aggregate = new T { Id = id };
            aggregate.Load(events);
            return aggregate;
        });
        
        return await Task.WhenAll(tasks);
    }
}
```

public class AggregateReconstructionTests

    [Fact]
    public void Aggregate_ReconstructsCorrectly_FromEvents()

        var order = new OrderAggregate { Id = "123" };

        // Assert
        Assert.Equal("123", order.Id);
        Assert.Equal(OrderStatus.Shipped, order.Status);
        Assert.Equal(100m, order.TotalAmount);

        Assert.Equal(3, order.Version);

    public void Aggregate_ThrowsError_WhenEventsOutOfOrder()

            new OrderPlacedEvent { OrderId = "123" },
            new OrderShippedEvent { OrderId = "123" },  // Can't ship before paying!
            new OrderPaidEvent { OrderId = "123" }
        };
        
        var order = new OrderAggregate { Id = "123" };
        
        // Act & Assert
        Assert.Throws<InvalidOperationException>(() => order.Load(events));
    }
    
    [Fact]
    public async Task Repository_LoadsAggregate_WithCorrectVersion()
    {
        // Arrange
        var eventStore = CreateEventStore();
        await eventStore.AppendEventsAsync("Order-123", new[]
        {
            new OrderPlacedEvent { OrderId = "123" },
            new OrderPaidEvent { OrderId = "123" }
        }, 0);
        
        var repository = new EventSourcedRepository<OrderAggregate>(eventStore);
        
        // Act
        var order = await repository.GetByIdAsync("123");
        
        // Assert
        Assert.Equal(2, order.Version);
    }
