# Lesson 3: Event Versioning and Upcasting

## Learning Objectives

By the end of this lesson, you will be able to:

- Perform event upcasting to migrate old events
- Design events for forward and backward compatibility
- Manage breaking changes in event schemas

---

## Introduction

In event-sourced systems, events are immutable and stored forever. As your system evolves, event schemas change—new fields are added, old fields are removed, types change. **Event versioning** and **upcasting** are techniques for handling this evolution without losing historical data or breaking replay.

Unlike traditional databases where you can migrate data with SQL scripts, event stores contain immutable history. You must design for evolution from day one.

## Key Concepts Summary

### Event Versioning Strategies

**1. Weak Schema (Add-Only)**
```csharp
// V1: Original event
public class OrderPlacedEvent_V1
{
    public string OrderId { get; set; }
    public decimal TotalAmount { get; set; }
}

// V2: Add new field (backward compatible)
public class OrderPlacedEvent
{
    public string OrderId { get; set; }
    public decimal TotalAmount { get; set; }
    public string Currency { get; set; } = "USD";  // Default value
}

// Old events deserialize fine, missing field gets default
```

**2. Multiple Event Versions**
```csharp
[EventType("OrderPlaced", Version = 1)]
public class OrderPlacedEvent_V1 : IDomainEvent
{
    public string OrderId { get; set; }
    public decimal TotalAmount { get; set; }
}

[EventType("OrderPlaced", Version = 2)]
public class OrderPlacedEvent_V2 : IDomainEvent
{
    public string OrderId { get; set; }
    public decimal TotalAmount { get; set; }
    public string Currency { get; set; }
    public Address ShippingAddress { get; set; }
}

// Aggregate handles both versions
private void Apply(OrderPlacedEvent_V1 evt) { /* ... */ }
private void Apply(OrderPlacedEvent_V2 evt) { /* ... */ }
```

**3. Upcasting (Migration on Read)**
```csharp
public interface IEventUpcaster
{
    bool CanUpcast(Type eventType, int version);
    IDomainEvent Upcast(IDomainEvent oldEvent);
}

public class OrderPlacedUpcaster : IEventUpcaster
{
    public bool CanUpcast(Type eventType, int version)
    {
        return eventType == typeof(OrderPlacedEvent_V1);
    }
    
    public IDomainEvent Upcast(IDomainEvent oldEvent)
    {
        var v1 = (OrderPlacedEvent_V1)oldEvent;
        
        return new OrderPlacedEvent_V2
        {
            OrderId = v1.OrderId,
            TotalAmount = v1.TotalAmount,
            Currency = "USD",  // Default for old events
            ShippingAddress = null  // Unknown in V1
        };
    }
}

// Event store with upcasting
public class UpcastingEventStore : IEventStore
{
    private readonly IEventStore _inner;
    private readonly IEnumerable<IEventUpcaster> _upcasters;
    
    public async Task<IEnumerable<IDomainEvent>> GetEventsAsync(string streamId)
    {
        var events = await _inner.GetEventsAsync(streamId);
        
        return events.Select(evt =>
        {
            // Try to upcast if needed
            var upcaster = _upcasters.FirstOrDefault(u => 
                u.CanUpcast(evt.GetType(), GetVersion(evt)));
            
            return upcaster != null ? upcaster.Upcast(evt) : evt;
        });
    }
}
```

### Breaking Changes

**Problem: Renamed Field**
```csharp
// V1
public class OrderPlacedEvent_V1
{
    public string CustomerId { get; set; }
}

// V2: Renamed field (BREAKING CHANGE)
public class OrderPlacedEvent_V2
{
    public string BuyerId { get; set; }  // Was CustomerId
}

// Solution: Upcaster
public class OrderPlacedV1ToV2Upcaster : IEventUpcaster
{
    public IDomainEvent Upcast(IDomainEvent oldEvent)
    {
        var v1 = (OrderPlacedEvent_V1)oldEvent;
        
        return new OrderPlacedEvent_V2
        {
            BuyerId = v1.CustomerId  // Map old name to new name
        };
    }
}
```

**Problem: Split Event**
```csharp
// V1: One event for multiple things
public class OrderProcessedEvent_V1
{
    public string OrderId { get; set; }
    public bool Paid { get; set; }
    public bool Shipped { get; set; }
}

// V2: Separate concerns (BREAKING CHANGE)
public class OrderPaidEvent { public string OrderId { get; set; } }
public class OrderShippedEvent { public string OrderId { get; set; } }

// Solution: Upcaster creates multiple events
public class OrderProcessedSplitUpcaster : IEventUpcaster
{
    public IEnumerable<IDomainEvent> Upcast(IDomainEvent oldEvent)
    {
        var v1 = (OrderProcessedEvent_V1)oldEvent;
        var events = new List<IDomainEvent>();
        
        if (v1.Paid)
            events.Add(new OrderPaidEvent { OrderId = v1.OrderId });
        
        if (v1.Shipped)
            events.Add(new OrderShippedEvent { OrderId = v1.OrderId });
        
        return events;
    }
}
```

### Upcasting Chain Pattern

```csharp
// V1 → V2 → V3 upcasting chain
public class ChainedUpcaster : IEventUpcaster
{
    private readonly List<IEventUpcaster> _upcasters;
    
    public ChainedUpcaster()
    {
        _upcasters = new List<IEventUpcaster>
        {
            new V1ToV2Upcaster(),
            new V2ToV3Upcaster()
        };
    }
    
    public IDomainEvent Upcast(IDomainEvent evt)
    {
        var current = evt;
        
        foreach (var upcaster in _upcasters)
        {
            if (upcaster.CanUpcast(current.GetType(), GetVersion(current)))
            {
                current = upcaster.Upcast(current);
            }
        }
        
        return current;
    }
}

// V1 to V2
public class V1ToV2Upcaster : IEventUpcaster
{
    public bool CanUpcast(Type eventType, int version) => 
        eventType == typeof(OrderPlacedEvent_V1);
    
    public IDomainEvent Upcast(IDomainEvent oldEvent)
    {
        var v1 = (OrderPlacedEvent_V1)oldEvent;
        return new OrderPlacedEvent_V2
        {
            OrderId = v1.OrderId,
            TotalAmount = v1.Total,
            Currency = "USD"
        };
    }
}

// V2 to V3
public class V2ToV3Upcaster : IEventUpcaster
{
    public bool CanUpcast(Type eventType, int version) =>
        eventType == typeof(OrderPlacedEvent_V2);
    
    public IDomainEvent Upcast(IDomainEvent oldEvent)
    {
        var v2 = (OrderPlacedEvent_V2)oldEvent;
        return new OrderPlacedEvent_V3
        {
            OrderId = v2.OrderId,
            Total = new Money(v2.TotalAmount, v2.Currency)
        };
    }
}
```

### Event Versioning Strategies Comparison

| Strategy | Pros | Cons | Use When |
|----------|------|------|----------|
| **Weak Schema** | Simple, minimal code | Can't handle breaking changes | Adding optional fields |
| **Multiple Versions** | Clear separation | More classes to maintain | Major structural changes |
| **Upcasting** | Centralized migration | Complex for chains | Gradual evolution |
| **Copy-Replace** | Preserves history | Duplicate code | Complete rewrites |

### Migration Scripts

```csharp
// One-time migration: Update all V1 events to V2 in database
public class EventMigrationService
{
    public async Task MigrateOrderPlacedEvents()
    {
        var v1Events = await _db.Events
            .Where(e => e.EventType == "OrderPlacedEvent_V1")
            .ToListAsync();
        
        foreach (var stored in v1Events)
        {
            var v1 = JsonSerializer.Deserialize<OrderPlacedEvent_V1>(stored.EventData);
            var v2 = new OrderPlacedEvent_V2
            {
                OrderId = v1.OrderId,
                TotalAmount = v1.Total,
                Currency = "USD"
            };
            
            // Update stored event
            stored.EventType = "OrderPlacedEvent_V2";
            stored.EventVersion = 2;
            stored.EventData = JsonSerializer.Serialize(v2);
        }
        
        await _db.SaveChangesAsync();
        
        Console.WriteLine($"Migrated {v1Events.Count} events from V1 to V2");
    }
}
```

### Versioning Best Practices

1. **Never modify existing event classes**
2. **Use semantic versioning** (V1, V2, V3)
3. **Document breaking changes** in code comments
4. **Test upcasters thoroughly** with real historical data
5. **Maintain upcasting chains** for all versions
6. **Consider migration costs** vs. runtime upcasting

### Monitoring Version Distribution

```csharp
public class EventVersionMonitoring
{
    private readonly Counter _eventsByVersion;
    
    public EventVersionMonitoring(IMeterFactory meterFactory)
    {
        var meter = meterFactory.Create("EventVersioning");
        _eventsByVersion = meter.CreateCounter<long>("events_by_version_total");
    }
    
    public void RecordEventVersion(string eventType, int version)
    {
        _eventsByVersion.Add(1, 
            new("event_type", eventType),
            new("version", version.ToString()));
    }
    
    public async Task<Dictionary<string, Dictionary<int, long>>> GetVersionDistribution()
    {
        var distribution = new Dictionary<string, Dictionary<int, long>>();
        
        var events = await _db.Events
            .GroupBy(e => new { e.EventType, e.EventVersion })
            .Select(g => new
            {
                g.Key.EventType,
                g.Key.EventVersion,
                Count = g.LongCount()
            })
            .ToListAsync();
        
        foreach (var item in events)
        {
            if (!distribution.ContainsKey(item.EventType))
                distribution[item.EventType] = new Dictionary<int, long>();
            
            distribution[item.EventType][item.EventVersion] = item.Count;
        }
        
        return distribution;
    }
    
    public async Task GenerateVersionReport()
    {
        var distribution = await GetVersionDistribution();
        
        Console.WriteLine("=== Event Version Distribution ===");
        foreach (var (eventType, versions) in distribution)
        {
            Console.WriteLine($"\n{eventType}:");
            foreach (var (version, count) in versions.OrderBy(v => v.Key))
            {
                var percentage = (count * 100.0) / versions.Values.Sum();
                Console.WriteLine($"  V{version}: {count:N0} events ({percentage:F1}%)");
            }
        }
    }
}
```

### Gradual Migration Strategy

```csharp
public class GradualMigrationService
{
    // Migrate events in batches to avoid locking
    public async Task MigrateEventsGradually(
        string eventType,
        int fromVersion,
        int toVersion,
        int batchSize = 1000)
    {
        var totalMigrated = 0;
        var hasMore = true;
        
        while (hasMore)
        {
            using var transaction = await _db.Database.BeginTransactionAsync();
            
            try
            {
                var events = await _db.Events
                    .Where(e => e.EventType == eventType && e.EventVersion == fromVersion)
                    .Take(batchSize)
                    .ToListAsync();
                
                if (!events.Any())
                {
                    hasMore = false;
                    break;
                }
                
                foreach (var evt in events)
                {
                    var oldEvent = DeserializeEvent(evt);
                    var newEvent = _upcaster.Upcast(oldEvent);
                    
                    evt.EventVersion = toVersion;
                    evt.EventData = JsonSerializer.Serialize(newEvent);
                }
                
                await _db.SaveChangesAsync();
                await transaction.CommitAsync();
                
                totalMigrated += events.Count;
                Console.WriteLine($"Migrated {totalMigrated} events...");
                
                // Throttle to avoid overloading database
                await Task.Delay(100);
            }
            catch (Exception ex)
            {
                await transaction.RollbackAsync();
                Console.WriteLine($"Migration batch failed: {ex.Message}");
                throw;
            }
        }
        
        Console.WriteLine($"Migration complete: {totalMigrated} events migrated from V{fromVersion} to V{toVersion}");
    }
}
```
