## Introduction

**Message filtering and routing** enables selective message delivery based on message content, headers, or metadata. Instead of broadcasting to all subscribers or load-balancing to any consumer, routing delivers messages only to interested parties based on routing rules.

This pattern is essential for building flexible, scalable systems where different consumers need different subsets of messages. Think of it as a smart post office that reads the address and routes letters accordingly, rather than delivering all mail to everyone.

## Key Concepts Summary

### Routing Strategies

**1. Direct Routing (Exact Match)**
- Route based on exact routing key match
- Simple and efficient
- Example: Route "high-priority" messages to priority queue

**2. Topic-Based Routing (Pattern Match)**
- Route based on pattern matching
- Flexible with wildcards
- Example: `order.*.created` matches `order.US.created`, `order.EU.created`

**3. Content-Based Routing (Message Inspection)**
- Route based on message content/properties
- Most flexible but requires content inspection
- Example: Route orders >$1000 to special processing

### RabbitMQ Topic Exchange

```csharp
// Producer with topic routing
public class EventPublisher
{
    private readonly IModel _channel;
    
    public EventPublisher(IConnection connection)
    {
        _channel = connection.CreateModel();
        
        // Declare topic exchange
        _channel.ExchangeDeclare(
            exchange: "events",
            type: ExchangeType.Topic,
            durable: true
        );
    }
    
    public void PublishOrderEvent(Order order, string region, string eventType)
    {
        // Routing key pattern: <entity>.<region>.<eventType>
        var routingKey = $"order.{region}.{eventType}";
        
        var message = JsonSerializer.Serialize(order);
        var body = Encoding.UTF8.GetBytes(message);
        
        _channel.BasicPublish(
            exchange: "events",
            routingKey: routingKey,  // e.g., "order.us.created"
            basicProperties: null,
            body: body
        );
        
        Console.WriteLine($"Published with routing key: {routingKey}");
    }
}

// Consumer 1: All US orders
public class USOrderConsumer
{
    public USOrderConsumer(IConnection connection)
    {
        var channel = connection.CreateModel();
        
        channel.ExchangeDeclare("events", ExchangeType.Topic, durable: true);
        
        var queueName = channel.QueueDeclare().QueueName;
        
        // Bind with pattern: order.us.*
        // Matches: order.us.created, order.us.updated, order.us.cancelled
        channel.QueueBind(
            queue: queueName,
            exchange: "events",
            routingKey: "order.us.*"
        );
        
        var consumer = new EventingBasicConsumer(channel);
        consumer.Received += (model, ea) =>
        {
            Console.WriteLine($"[US] Received: {ea.RoutingKey}");
            // Process US order
        };
        
        channel.BasicConsume(queueName, autoAck: true, consumer: consumer);
    }
}

public class OrderCreatedConsumer

// Consumer 3: Multiple patterns
public class MultiPatternConsumer
{
    public MultiPatternConsumer(IConnection connection)
    {
        var channel = connection.CreateModel();
        
        channel.ExchangeDeclare("events", ExchangeType.Topic, durable: true);
        
        var queueName = channel.QueueDeclare().QueueName;
        
        // Multiple bindings to same queue
        channel.QueueBind(queueName, "events", "order.eu.*");
        channel.QueueBind(queueName, "events", "order.asia.*");

    }

```

**Wildcard patterns:**
- `*` (star): Matches exactly one word
  - `order.*.created` matches `order.us.created` but NOT `order.us.priority.created`
- `#` (hash): Matches zero or more words
  - `order.#` matches `order.us.created`, `order.us.priority.created`, `order.cancelled`

### Kafka Header-Based Filtering

```csharp
// Producer with headers
public class OrderProducer
{
    private readonly IProducer<string, Order> _producer;
    
    public async Task PublishOrder(Order order, string region, decimal amount)
    {
        var headers = new Headers
        {
            { "region", Encoding.UTF8.GetBytes(region) },
            { "amount", BitConverter.GetBytes(amount) },
            { "priority", Encoding.UTF8.GetBytes(amount > 1000 ? "high" : "normal") }
        };
        
        await _producer.ProduceAsync("orders", new Message<string, Order>
        {
            Key = order.Id,
            Value = order,
            Headers = headers
        });
    }
}

// Consumer with filtering
public class HighValueOrderConsumer
{
    private readonly IConsumer<string, Order> _consumer;
    
    public void Start(CancellationToken ct)
    {
        _consumer.Subscribe("orders");
        
        while (!ct.IsCancellationRequested)
        {
            var result = _consumer.Consume(ct);
            
            // Filter based on headers
            if (result.Message.Headers.TryGetLastBytes("priority", out var priorityBytes))
            {
                var priority = Encoding.UTF8.GetString(priorityBytes);
                
                if (priority == "high")
                {
                    // Process high-priority order
                    Console.WriteLine($"Processing high-value order: {result.Message.Value.Id}");
                    ProcessHighValueOrder(result.Message.Value);
                }
            }
            
            _consumer.Commit(result);
        }
    }
}

// Consumer with complex filtering
public class RegionalConsumer
{
    public void Start(string interestedRegion, CancellationToken ct)
    {
        _consumer.Subscribe("orders");
        
        while (!ct.IsCancellationRequested)
        {
            var result = _consumer.Consume(ct);
            
            // Multi-criteria filtering
            if (result.Message.Headers.TryGetLastBytes("region", out var regionBytes))
            {
                var region = Encoding.UTF8.GetString(regionBytes);
                
                if (region == interestedRegion)
                {
                    if (result.Message.Headers.TryGetLastBytes("amount", out var amountBytes))
                    {
                        var amount = BitConverter.ToDecimal(amountBytes);
                        
                        if (amount > 500)
                        {
                            // Process: correct region AND high amount
                            ProcessOrder(result.Message.Value);
                        }
                    }
                }
            }
            
            _consumer.Commit(result);
        }
    }
}
```

**Note:** Kafka filtering happens on consumer side (all messages still delivered), whereas RabbitMQ filtering happens on broker side (only matching messages delivered).

### Content-Based Routing

```csharp
// RabbitMQ with headers exchange
public class ContentBasedPublisher
{
    private readonly IModel _channel;
    
    public ContentBasedPublisher(IConnection connection)
    {
        _channel = connection.CreateModel();
        
        // Headers exchange: routes based on message headers
        _channel.ExchangeDeclare(
            exchange: "orders-router",
            type: ExchangeType.Headers,
            durable: true
        );
    }
    
    public void PublishOrder(Order order)
    {
        var properties = _channel.CreateBasicProperties();
        properties.Headers = new Dictionary<string, object>
        {
            { "region", order.Region },
            { "amount", order.TotalAmount },
            { "customer-tier", order.CustomerTier }
        };
        
        var body = Encoding.UTF8.GetBytes(JsonSerializer.Serialize(order));
        
        _channel.BasicPublish(
            exchange: "orders-router",
            routingKey: "",  // Ignored for headers exchange
            basicProperties: properties,
            body: body
        );
    }
}

// Consumer with header matching
public class VIPConsumer
{
    public VIPConsumer(IConnection connection)
    {
        var channel = connection.CreateModel();
        
        channel.ExchangeDeclare("orders-router", ExchangeType.Headers, durable: true);
        
        var queueName = channel.QueueDeclare().QueueName;
        
        // Match headers: customer-tier = "vip" AND amount > 1000
        var arguments = new Dictionary<string, object>
        {
            { "x-match", "all" },  // ALL headers must match
            { "customer-tier", "vip" },
            { "amount-min", 1000 }
        };
        
        channel.QueueBind(
            queue: queueName,
            exchange: "orders-router",
            routingKey: "",
            arguments: arguments
        );
    }
}
```

### Routing Performance Considerations

**RabbitMQ routing complexity:**
- **Direct exchange:** O(1) lookup - fastest, use for exact matching
- **Fanout exchange:** O(n) where n = bound queues - still very fast
- **Topic exchange:** O(n) pattern matching - slower but flexible
- **Headers exchange:** O(n) with full header inspection - slowest

**Performance test results (approximate):**
```
Direct:  500,000 msgs/sec
Fanout:  400,000 msgs/sec
Topic:   200,000 msgs/sec
Headers: 100,000 msgs/sec
```

**Kafka filtering trade-offs:**
- **Broker-side:** No filtering - all messages sent to all consumers in different groups
- **Consumer-side:** Application filters after receiving
- **Network overhead:** Consumers receive and discard unwanted messages
- **When to use separate topics:** Very different message types, high-volume selective consumption

```csharp
// Example: Cost of consumer-side filtering
public class FilteringConsumer
{
    private long _received = 0;
    private long _processed = 0;
    
    public void Start(string interestedRegion, CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            var result = _consumer.Consume(ct);
            _received++;
            
            if (GetRegion(result.Message) == interestedRegion)
            {
                ProcessMessage(result.Message.Value);
                _processed++;
            }
            
            _consumer.Commit(result);
            
            // Log filtering efficiency
            if (_received % 1000 == 0)
            {
                var efficiency = (_processed / (double)_received) * 100;
                Console.WriteLine($"Filtering efficiency: {efficiency:F1}%");
                
                if (efficiency < 10)
                {
                    Console.WriteLine("⚠️ Consider using separate topics");
                }
            }
        }
    }
}
```

**Pattern 4: Priority-Based Routing**

```csharp
// RabbitMQ with priority queues
public class PriorityRouter
{
    private readonly IModel _channel;
    
    public PriorityRouter(IConnection connection)
    {
        _channel = connection.CreateModel();
        
        _channel.ExchangeDeclare("orders", ExchangeType.Direct, durable: true);
        
        // High priority queue
        var highPriorityArgs = new Dictionary<string, object>
        {
            { "x-max-priority", 10 }
        };
        _channel.QueueDeclare("orders-high", true, false, false, highPriorityArgs);
        _channel.QueueBind("orders-high", "orders", "high");
        
        // Normal priority queue
        _channel.QueueDeclare("orders-normal", true, false, false);
        _channel.QueueBind("orders-normal", "orders", "normal");
    }
    
    public void PublishOrder(Order order)
    {
        var routingKey = order.TotalAmount > 1000 ? "high" : "normal";
        var priority = order.TotalAmount > 1000 ? (byte)9 : (byte)5;
        
        var properties = _channel.CreateBasicProperties();
        properties.Priority = priority;
        properties.Persistent = true;
        
        var body = Encoding.UTF8.GetBytes(JsonSerializer.Serialize(order));
        
        _channel.BasicPublish("orders", routingKey, properties, body);
        Console.WriteLine($"Published order {order.Id} with priority {priority}");
    }
}
```

### Kafka Multi-Criteria Filtering Optimization

```csharp
public class OptimizedKafkaFilter
{
    // Pre-compile filter predicates for better performance
    private readonly Func<ConsumeResult<string, Order>, bool> _filterPredicate;
    
    public OptimizedKafkaFilter(FilterCriteria criteria)
    {
        // Build composite predicate
        var predicates = new List<Func<ConsumeResult<string, Order>, bool>>();
        
        if (criteria.Regions != null)
        {
            predicates.Add(result => 
                GetHeaderValue(result, "region") is string region &&
                criteria.Regions.Contains(region));
        }
        
        if (criteria.MinAmount.HasValue)
        {
            predicates.Add(result =>
                GetHeaderValue(result, "amount") is decimal amount &&
                amount >= criteria.MinAmount.Value);
        }
        
        if (criteria.CustomerTiers != null)
        {
            predicates.Add(result =>
                GetHeaderValue(result, "customer-tier") is string tier &&
                criteria.CustomerTiers.Contains(tier));
        }
        
        // Combine all predicates with AND logic
        _filterPredicate = result => predicates.All(p => p(result));
    }
    
    public void Start(CancellationToken ct)
    {
        var stats = new FilterStats();
        
        while (!ct.IsCancellationRequested)
        {
            var result = _consumer.Consume(ct);
            stats.TotalMessages++;
            
            if (_filterPredicate(result))
            {
                ProcessMessage(result.Message.Value);
                stats.ProcessedMessages++;
            }
            else
            {
                stats.FilteredMessages++;
            }
            
            _consumer.Commit(result);
            
            // Periodic stats logging
            if (stats.TotalMessages % 10000 == 0)
            {
                LogFilteringStats(stats);
            }
        }
    }
    
    private object GetHeaderValue(ConsumeResult<string, Order> result, string key)
    {
        if (result.Message.Headers.TryGetLastBytes(key, out var bytes))
        {
            // Try to deserialize based on expected type
            var str = Encoding.UTF8.GetString(bytes);
            
            if (decimal.TryParse(str, out var decimalValue))
                return decimalValue;
                
            return str;
        }
        return null;
    }
}
```

### Routing Debugging and Troubleshooting

```csharp
public class RoutingDebugger
{
    // Test routing logic without actual broker
    public void TestRouting(Order order)
    {
        Console.WriteLine($"\n=== Routing Analysis for Order {order.Id} ===");
        Console.WriteLine($"Amount: ${order.TotalAmount}");
        Console.WriteLine($"Region: {order.Region}");
        Console.WriteLine($"Customer Tier: {order.CustomerTier}");
        
        // Test against all routing patterns
        var patterns = new[]
        {
            ("order.*.created", "All order creation events"),
            ("order.us.*", "All US orders"),
            ("order.*.cancelled", "All cancellations"),
            ($"order.{order.Region}.*", $"All {order.Region} orders"),
            ("order.#", "All order events")
        };
        
        var actualKey = $"order.{order.Region}.{order.Status}";
        Console.WriteLine($"\nRouting Key: {actualKey}");
        Console.WriteLine("\nMatching Patterns:");
        
        foreach (var (pattern, description) in patterns)
        {
            if (MatchesPattern(actualKey, pattern))
            {
                Console.WriteLine($"  ✓ {pattern} - {description}");
            }
        }
    }
    
    private bool MatchesPattern(string routingKey, string pattern)
    {
        // Simple pattern matching (actual implementation more complex)
        var keyParts = routingKey.Split('.');
        var patternParts = pattern.Split('.');
        
        if (pattern.Contains('#'))
        {
            return true; // Simplification: # matches everything
        }
        
        if (keyParts.Length != patternParts.Length)
            return false;
            
        for (int i = 0; i < patternParts.Length; i++)
        {
            if (patternParts[i] != "*" && patternParts[i] != keyParts[i])
                return false;
        }
        
        return true;
    }
}
```
