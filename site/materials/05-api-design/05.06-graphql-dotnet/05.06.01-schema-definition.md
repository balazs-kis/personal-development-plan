# Lesson 1: Hot Chocolate Schema Definition

## Learning Objectives

By the end of this lesson, you will be able to:
- Set up Hot Chocolate in an ASP.NET Core application
- Define GraphQL schemas using annotations and code-first approach
- Create object types, queries, and mutations
- Use Hot Chocolate's type system features
- Configure Hot Chocolate services

---

## Introduction

Hot Chocolate is a modern, feature-rich GraphQL server for .NET. It offers multiple approaches to defining schemas: annotation-based (code-first), pure code-first, and schema-first. This lesson focuses on the annotation-based approach, which is the most common and provides excellent developer experience with C# types.

### Mutations

```csharp
public class Mutation
{
    public Book AddBook(
        AddBookInput input,
        [Service] IBookRepository repository)
    {
        var book = new Book
        {
            Title = input.Title,
            Author = input.Author,
            Year = input.Year
        };
        
        return repository.Add(book);
    }
    
    public Book UpdateBook(
        int id,
        UpdateBookInput input,
        [Service] IBookRepository repository)
    {
        var book = repository.GetById(id)
            ?? throw new GraphQLException("Book not found");
        
        book.Title = input.Title ?? book.Title;
        book.Author = input.Author ?? book.Author;
        
        return repository.Update(book);
    }
    
    public bool DeleteBook(
        int id,
        [Service] IBookRepository repository)
    {
        return repository.Delete(id);
    }
}

// Input types
public record AddBookInput(string Title, string Author, int Year);

public record UpdateBookInput(string? Title, string? Author);
```

### Error Handling

```csharp
// Custom error
public class BookNotFoundError : Exception
{
    public BookNotFoundError(int id)
        : base($"Book with ID {id} not found")
    {
        BookId = id;
    }
    
    public int BookId { get; }
}

// Error filter
public class BookErrorFilter : IErrorFilter
{
    public IError OnError(IError error)
    {
        if (error.Exception is BookNotFoundError notFound)
        {
            return error.WithMessage(notFound.Message)
                .SetCode("BOOK_NOT_FOUND")
                .SetExtension("bookId", notFound.BookId);
        }
        
        return error;
    }
}

// Register error filter
builder.Services
    .AddGraphQLServer()
    .AddQueryType<Query>()
    .AddErrorFilter<BookErrorFilter>();
```

### Pure Code-First Approach

```csharp
// Define types using fluent API
public class BookType : ObjectType<Book>
{
    protected override void Configure(IObjectTypeDescriptor<Book> descriptor)
    {
        descriptor.Description("Represents a book in the library");
        
        descriptor
            .Field(b => b.Id)
            .Description("The unique identifier");
        
        descriptor
            .Field(b => b.Title)
            .Description("The book's title")
            .Type<NonNullType<StringType>>();
        
        descriptor
            .Field(b => b.Author)
            .Description("The book's author");
        
        descriptor
            .Field(b => b.PublishedDate)
            .Type<DateTimeType>();
        
        // Computed field
        descriptor
            .Field("ageInYears")
            .Type<IntType>()
            .Resolve(ctx =>
            {
                var book = ctx.Parent<Book>();
                return DateTime.Now.Year - book.PublishedDate.Year;
            });
        
        // Ignore field
        descriptor.Ignore(b => b.InternalNotes);
    }
}

// Register type
builder.Services
    .AddGraphQLServer()
    .AddQueryType<Query>()
    .AddType<BookType>();
```

// Custom middleware
public class LoggingMiddleware
{
    private readonly FieldDelegate _next;
    
    public LoggingMiddleware(FieldDelegate next)
    {
        _next = next;
    }
    
    public async Task InvokeAsync(IMiddlewareContext context)
    {
        var start = DateTime.UtcNow;
        
        await _next(context);
        
        var duration = DateTime.UtcNow - start;

            $"Field: {context.Selection.Field.Name}, Duration: {duration.TotalMilliseconds}ms");
    }
}

[ExtendObjectType(typeof(Book))]
public class BookExtensions
{

### Global Object Identification

// Implement Node interface
[Node]
[ExtendObjectType(typeof(Book), IgnoreProperties = new[] { nameof(Book.Id) })]
public class BookNode
{
    // Encode ID

    public int Id([Parent] Book book) => book.Id;

### Validation

```csharp
// Input validation with FluentValidation
public class AddBookInputValidator : AbstractValidator<AddBookInput>
{
    public AddBookInputValidator()
    {
        RuleFor(x => x.Title)
            .NotEmpty()
            .MinimumLength(3)
            .MaximumLength(200);
        
        RuleFor(x => x.Author)
            .NotEmpty()
            .MaximumLength(100);
        
        RuleFor(x => x.PublishedDate)
            .LessThanOrEqualTo(DateTime.Now)
            .WithMessage("Publication date cannot be in the future");
    }
}

// Use in mutation
public class Mutation
{
    public async Task<AddBookPayload> AddBookAsync(
        AddBookInput input,
        [Service] IValidator<AddBookInput> validator,
        [Service] IBookRepository repository)
    {
        var validationResult = await validator.ValidateAsync(input);
        
        if (!validationResult.IsValid)
        {
            return new AddBookPayload
            {
                Errors = validationResult.Errors
                    .Select(e => new UserError(e.ErrorMessage, e.PropertyName))
                    .ToList()
            };
        }
        
        var book = new Book
        {
            Title = input.Title,
            Author = input.Author,
            PublishedDate = input.PublishedDate
        };
        
        var created = await repository.AddAsync(book);
        
        return new AddBookPayload { Book = created };
    }
}

// Register validator
builder.Services.AddValidatorsFromAssemblyContaining<AddBookInputValidator>();
```
