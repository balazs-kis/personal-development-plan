## Learning Objectives

By the end of this lesson, you will be able to:
- Implement GraphQL subscriptions in Hot Chocolate

## Introduction

GraphQL subscriptions enable real-time, event-driven communication between client and server. Hot Chocolate provides first-class support for subscriptions using WebSockets, allowing you to build reactive applications.

### Basic Subscription

```csharp
// Program.cs
builder.Services
    .AddGraphQLServer()
    .AddQueryType<Query>()
    .AddMutationType<Mutation>()
    .AddSubscriptionType<Subscription>()
    .AddInMemorySubscriptions();

app.UseWebSockets();
app.MapGraphQL();

// Subscription.cs
public class Subscription
{
    [Subscribe]
    [Topic]
    public Book OnBookAdded([EventMessage] Book book)
    {
        return book;
    }
}

// Mutation.cs
public class Mutation
{
    public async Task<Book> AddBookAsync(
        AddBookInput input,
        [Service] IBookRepository repository,
        [Service] ITopicEventSender eventSender)
    {
        var book = new Book
        {
            Title = input.Title,
            Author = input.Author
        };
        
        var created = await repository.AddAsync(book);
        
        // Publish event
        await eventSender.SendAsync(nameof(Subscription.OnBookAdded), created);
        
        return created;
    }
}

// Client subscription:
// subscription {
//   onBookAdded {
//     title
//     author
//   }
// }
```

### Filtered Subscriptions

```csharp
public class Subscription
{
    [Subscribe]
    public async ValueTask<ISourceStream<Book>> OnBookAddedAsync(
        string? author,
        [Service] ITopicEventReceiver eventReceiver)
    {
        var stream = await eventReceiver.SubscribeAsync<Book>("BookAdded");
        
        if (author is not null)
        {
            return stream.Where(book => book.Author == author);
        }
        
        return stream;
    }
}
```

### Advanced Subscription Patterns

```csharp
// Multiple topics
public class Subscription
{
    [Subscribe]
    [Topic("BookAdded", "BookUpdated", "BookDeleted")]
    public BookEvent OnBookEvent([EventMessage] BookEvent bookEvent)
    {
        return bookEvent;
    }
}

public record BookEvent(BookEventType Type, Book Book);

public enum BookEventType
{
    Added,
    Updated,
    Deleted
}

// Publish different event types
public class Mutation
{
    public async Task<Book> AddBookAsync(
        AddBookInput input,
        [Service] IBookRepository repository,
        [Service] ITopicEventSender eventSender)
    {
        var book = await repository.AddAsync(input);
        
        await eventSender.SendAsync(
            "BookAdded",
            new BookEvent(BookEventType.Added, book));
        
        return book;
    }
    
    public async Task<Book> UpdateBookAsync(
        int id,
        UpdateBookInput input,
        [Service] IBookRepository repository,
        [Service] ITopicEventSender eventSender)
    {
        var book = await repository.UpdateAsync(id, input);
        
        await eventSender.SendAsync(
            "BookUpdated",
            new BookEvent(BookEventType.Updated, book));
        
        return book;
    }
}
```

### Subscription Lifecycle

```csharp
// Handle subscription lifecycle
public class SubscriptionLifetimeService : IDisposable
{
    private readonly ConcurrentDictionary<string, int> _subscriberCount = new();
    
    public void OnSubscribed(string topic)
    {
        _subscriberCount.AddOrUpdate(topic, 1, (_, count) => count + 1);
        Console.WriteLine($"Subscriber added to {topic}. Total: {_subscriberCount[topic]}");
    }
    
    public void OnUnsubscribed(string topic)
    {
        if (_subscriberCount.TryGetValue(topic, out var count))
        {
            var newCount = count - 1;
            if (newCount <= 0)
            {
                _subscriberCount.TryRemove(topic, out _);
            }
            else
            {
                _subscriberCount[topic] = newCount;
            }
            
            Console.WriteLine($"Subscriber removed from {topic}. Remaining: {newCount}");
        }
    }
    
    public void Dispose()
    {
        _subscriberCount.Clear();
    }
}

// Register service
builder.Services.AddSingleton<SubscriptionLifetimeService>();
```

### Batched Subscription Events

```csharp
public class Subscription
{
    [Subscribe]
    public async IAsyncEnumerable<Book> OnBooksAddedBatchAsync(
        [Service] ITopicEventReceiver eventReceiver,
        [EnumeratorCancellation] CancellationToken cancellationToken)
    {
        var stream = await eventReceiver.SubscribeAsync<Book>("BookAdded");
        
        // Buffer events for 1 second or 10 items
        await foreach (var batch in stream.Buffer(TimeSpan.FromSeconds(1), 10)
            .WithCancellation(cancellationToken))
        {
            foreach (var book in batch)
            {
                yield return book;
            }
        }
    }
}
```

### Subscription Error Handling

```csharp
public class Subscription
{
    [Subscribe]
    public async ValueTask<ISourceStream<Book>> OnBookAddedAsync(
        [Service] ITopicEventReceiver eventReceiver,
        [Service] ILogger<Subscription> logger)
    {
        try
        {
            var stream = await eventReceiver.SubscribeAsync<Book>("BookAdded");
            
            return stream.Select(book =>
            {
                try
                {
                    // Validate book
                    if (string.IsNullOrEmpty(book.Title))
                    {
                        logger.LogWarning("Received book with empty title");
                        throw new GraphQLException("Invalid book data");
                    }
                    
                    return book;
                }
                catch (Exception ex)
                {
                    logger.LogError(ex, "Error processing book event");
                    throw;
                }
            });
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Error subscribing to book events");
            throw new GraphQLException("Failed to subscribe to book events");
        }
    }
}
```

### Real-Time Dashboard Example

```csharp
// Dashboard subscription aggregating multiple streams
public class Subscription
{
    [Subscribe]
    public async IAsyncEnumerable<DashboardUpdate> OnDashboardUpdateAsync(
        [Service] ITopicEventReceiver eventReceiver,
        [EnumeratorCancellation] CancellationToken cancellationToken)
    {
        // Subscribe to multiple topics
        var bookStream = await eventReceiver.SubscribeAsync<Book>("BookAdded");
        var userStream = await eventReceiver.SubscribeAsync<User>("UserRegistered");
        var orderStream = await eventReceiver.SubscribeAsync<Order>("OrderPlaced");
        
        // Merge streams
        await foreach (var update in MergeStreams(bookStream, userStream, orderStream)
            .WithCancellation(cancellationToken))
        {
            yield return update;
        }
    }
    
    private async IAsyncEnumerable<DashboardUpdate> MergeStreams(
        ISourceStream<Book> books,
        ISourceStream<User> users,
        ISourceStream<Order> orders)
    {
        var bookTask = books.ReadEventsAsync().GetAsyncEnumerator();
        var userTask = users.ReadEventsAsync().GetAsyncEnumerator();
        var orderTask = orders.ReadEventsAsync().GetAsyncEnumerator();
        
        while (true)
        {
            var tasks = new[]
            {
                bookTask.MoveNextAsync().AsTask(),
                userTask.MoveNextAsync().AsTask(),
                orderTask.MoveNextAsync().AsTask()
            };
            
            var completed = await Task.WhenAny(tasks);
            
            if (completed == tasks[0] && bookTask.Current != null)
            {
                yield return new DashboardUpdate
                {
                    Type = "BookAdded",
                    Data = bookTask.Current
                };
            }
            else if (completed == tasks[1] && userTask.Current != null)
            {
                yield return new DashboardUpdate
                {
                    Type = "UserRegistered",
                    Data = userTask.Current
                };
            }
            else if (completed == tasks[2] && orderTask.Current != null)
            {
                yield return new DashboardUpdate
                {
                    Type = "OrderPlaced",
                    Data = orderTask.Current
                };
            }
        }
    }
}

public class DashboardUpdate
{
    public string Type { get; set; } = string.Empty;
    public object Data { get; set; } = null!;
}
```

### Subscription Middleware

```csharp
// Custom subscription middleware
public class SubscriptionLoggingMiddleware
{
    private readonly SubscribeDelegate _next;
    
    public SubscriptionLoggingMiddleware(SubscribeDelegate next)
    {
        _next = next;
    }
    
    public async ValueTask<ISourceStream> InvokeAsync(IMiddlewareContext context)
    {
        var logger = context.Services.GetRequiredService<ILogger<SubscriptionLoggingMiddleware>>();
        
        logger.LogInformation(
            "Subscription started: {SubscriptionName}",
            context.Selection.Field.Name);
        
        var stream = await _next(context);
        
        return new LoggingSourceStream(stream, logger);
    }
}

public class LoggingSourceStream : ISourceStream
{
    private readonly ISourceStream _inner;
    private readonly ILogger _logger;
    
    public LoggingSourceStream(ISourceStream inner, ILogger logger)
    {
        _inner = inner;
        _logger = logger;
    }
    
    public async IAsyncEnumerable<object> ReadEventsAsync()
    {
        await foreach (var item in _inner.ReadEventsAsync())
        {
            _logger.LogDebug("Subscription event: {EventType}", item?.GetType().Name);
            yield return item;
        }
    }
}

// Register middleware
builder.Services
    .AddGraphQLServer()
    .AddSubscriptionType<Subscription>()
    .UseSubscription<SubscriptionLoggingMiddleware>();
```

### Monitoring Subscriptions

```csharp
// Track active subscriptions
public class SubscriptionMonitor
{
    private readonly ConcurrentDictionary<string, List<string>> _activeSubscriptions = new();
    
    public void RecordSubscription(string topic, string connectionId)
    {
        _activeSubscriptions.AddOrUpdate(
            topic,
            _ => new List<string> { connectionId },
            (_, list) =>
            {
                list.Add(connectionId);
                return list;
            });
    }
    
    public void RemoveSubscription(string topic, string connectionId)
    {
        if (_activeSubscriptions.TryGetValue(topic, out var list))
        {
            list.Remove(connectionId);
            if (list.Count == 0)
            {
                _activeSubscriptions.TryRemove(topic, out _);
            }
        }
    }
    
    public Dictionary<string, int> GetSubscriptionCounts()
    {
        return _activeSubscriptions.ToDictionary(
            kvp => kvp.Key,
            kvp => kvp.Value.Count);
    }
}
```

### Complete Real-World Example

```csharp
// Complete subscription system
public class ChatSubscription
{
    [Subscribe]
    [Authorize]
    public async ValueTask<ISourceStream<ChatMessage>> OnMessageAsync(
        string roomId,
        ClaimsPrincipal user,
        [Service] ITopicEventReceiver eventReceiver,
        [Service] IChatService chatService,
        [Service] ILogger<ChatSubscription> logger)
    {
        var userId = user.FindFirst(ClaimTypes.NameIdentifier)?.Value;
        
        if (userId == null)
        {
            throw new GraphQLException("User not authenticated");
        }
        
        // Verify user has access to room
        if (!await chatService.UserHasAccessToRoomAsync(userId, roomId))
        {
            throw new GraphQLException("Access denied to chat room");
        }
        
        logger.LogInformation(
            "User {UserId} subscribed to room {RoomId}",
            userId, roomId);
        
        var stream = await eventReceiver.SubscribeAsync<ChatMessage>($"room:{roomId}");
        
        // Filter out messages from blocked users
        var blockedUsers = await chatService.GetBlockedUsersAsync(userId);
        
        return stream.Where(msg => !blockedUsers.Contains(msg.SenderId));
    }
}

public class ChatMutation
{
    [Authorize]
    public async Task<ChatMessage> SendMessageAsync(
        string roomId,
        string content,
        ClaimsPrincipal user,
        [Service] IChatService chatService,
        [Service] ITopicEventSender eventSender,
        [Service] ILogger<ChatMutation> logger)
    {
        var userId = user.FindFirst(ClaimTypes.NameIdentifier)?.Value;
        
        if (userId == null)
        {
            throw new GraphQLException("User not authenticated");
        }
        
        // Verify access
        if (!await chatService.UserHasAccessToRoomAsync(userId, roomId))
        {
            throw new GraphQLException("Access denied to chat room");
        }
        
        // Save message
        var message = await chatService.SaveMessageAsync(new ChatMessage
        {
            RoomId = roomId,
            SenderId = userId,
            Content = content,
            Timestamp = DateTime.UtcNow
        });
        
        // Publish to subscribers
        await eventSender.SendAsync($"room:{roomId}", message);
        
        logger.LogInformation(
            "User {UserId} sent message in room {RoomId}",
            userId, roomId);
        
        return message;
    }
}

public class ChatMessage
{
    public string Id { get; set; } = Guid.NewGuid().ToString();
    public string RoomId { get; set; } = string.Empty;
    public string SenderId { get; set; } = string.Empty;
    public string Content { get; set; } = string.Empty;
    public DateTime Timestamp { get; set; }
}
```
