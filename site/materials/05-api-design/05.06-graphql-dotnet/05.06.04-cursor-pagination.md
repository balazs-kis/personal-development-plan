## Introduction

Cursor-based pagination (also known as Relay-style pagination) provides a robust way to paginate through large datasets. Unlike offset-based pagination, cursor-based pagination is more efficient and handles concurrent modifications better.

### Implementing Pagination from Scratch

```csharp
// Manual pagination implementation
public class Query
{
    public async Task<BookConnection> GetBooksAsync(
        [Service] BookDbContext context,
        int? first = null,
        string? after = null,
        int? last = null,
        string? before = null)
    {
        var query = context.Books.AsQueryable();
        
        // Decode cursors
        var afterId = after != null ? DecodeCursor(after) : 0;
        var beforeId = before != null ? DecodeCursor(before) : int.MaxValue;
        
        // Apply cursor filters
        query = query.Where(b => b.Id > afterId && b.Id < beforeId);
        
        // Determine direction and limit
        if (first.HasValue)
        {
            query = query.OrderBy(b => b.Id).Take(first.Value + 1);
        }
        else if (last.HasValue)
        {
            query = query.OrderByDescending(b => b.Id).Take(last.Value + 1);
        }
        
        var books = await query.ToListAsync();
        
        // Check if there are more pages
        var hasNextPage = first.HasValue && books.Count > first.Value;
        var hasPreviousPage = last.HasValue && books.Count > last.Value;
        
        if (hasNextPage) books.RemoveAt(books.Count - 1);
        if (hasPreviousPage) books.RemoveAt(0);
        
        if (last.HasValue) books.Reverse();
        
        // Build connection
        var edges = books.Select(b => new BookEdge
        {
            Node = b,
            Cursor = EncodeCursor(b.Id)
        }).ToList();
        
        return new BookConnection
        {
            Edges = edges,
            PageInfo = new PageInfo
            {
                HasNextPage = hasNextPage,
                HasPreviousPage = hasPreviousPage,
                StartCursor = edges.FirstOrDefault()?.Cursor,
                EndCursor = edges.LastOrDefault()?.Cursor
            },
            TotalCount = await context.Books.CountAsync()
        };
    }
    
    private string EncodeCursor(int id)
    {
        return Convert.ToBase64String(Encoding.UTF8.GetBytes(id.ToString()));
    }
    
    private int DecodeCursor(string cursor)
    {
        var bytes = Convert.FromBase64String(cursor);
        return int.Parse(Encoding.UTF8.GetString(bytes));
    }
}
```

### Pagination with DataLoader

```csharp
// Paginated DataLoader
public class PaginatedBooksDataLoader : BatchDataLoader<PaginationInput, Connection<Book>>
{
    private readonly IDbContextFactory<BookDbContext> _contextFactory;
    
    public PaginatedBooksDataLoader(
        IDbContextFactory<BookDbContext> contextFactory,
        IBatchScheduler batchScheduler)
        : base(batchScheduler)
    {
        _contextFactory = contextFactory;
    }
    
    protected override async Task<IReadOnlyDictionary<PaginationInput, Connection<Book>>> 
        LoadBatchAsync(
            IReadOnlyList<PaginationInput> keys,
            CancellationToken cancellationToken)
    {
        await using var context = await _contextFactory.CreateDbContextAsync(cancellationToken);
        
        var results = new Dictionary<PaginationInput, Connection<Book>>();
        
        foreach (var input in keys)
        {
            var query = context.Books.AsQueryable();
            
            // Apply pagination
            var books = await query
                .Skip(input.Skip)
                .Take(input.Take)
                .ToListAsync(cancellationToken);
            
            var connection = new Connection<Book>
            {
                Edges = books.Select(b => new Edge<Book>
                {
                    Node = b,
                    Cursor = Convert.ToBase64String(Encoding.UTF8.GetBytes(b.Id.ToString()))
                }).ToList(),
                PageInfo = new ConnectionPageInfo
                {
                    HasNextPage = books.Count == input.Take,
                    HasPreviousPage = input.Skip > 0
                }
            };
            
            results[input] = connection;
        }
        
        return results;
    }
}

public record PaginationInput(int Skip, int Take);
```

### Keyset Pagination (High Performance)

```csharp
// Keyset pagination for large datasets
public class Query
{
    public async Task<BookConnection> GetBooksByKeysetAsync(
        [Service] BookDbContext context,
        int? first = null,
        DateTime? afterDate = null,
        int? afterId = null)
    {
        var query = context.Books.AsQueryable();
        
        // Apply keyset filter
        if (afterDate.HasValue && afterId.HasValue)
        {
            query = query.Where(b =>
                b.PublishedDate > afterDate.Value ||
                (b.PublishedDate == afterDate.Value && b.Id > afterId.Value));
        }
        
        query = query
            .OrderBy(b => b.PublishedDate)
            .ThenBy(b => b.Id);
        
        var pageSize = Math.Min(first ?? 20, 100);
        var books = await query.Take(pageSize + 1).ToListAsync();
        
        var hasNextPage = books.Count > pageSize;
        if (hasNextPage) books.RemoveAt(books.Count - 1);
        
        var edges = books.Select(b => new BookEdge
        {
            Node = b,
            Cursor = $"{b.PublishedDate:O}:{b.Id}"
        }).ToList();
        
        return new BookConnection
        {
            Edges = edges,
            PageInfo = new PageInfo
            {
                HasNextPage = hasNextPage,
                EndCursor = edges.LastOrDefault()?.Cursor
            }
        };
    }
}
```

### Connection Extensions

```csharp
// Add custom fields to connection
[ExtendObjectType("BookConnection")]
public class BookConnectionExtensions
{
    public async Task<decimal> GetAveragePriceAsync(
        [Parent] Connection<Book> connection)
    {
        var books = connection.Edges.Select(e => e.Node);
        return books.Any() ? books.Average(b => b.Price) : 0;
    }
    
    public int GetTotalAuthorsAsync(
        [Parent] Connection<Book> connection)
    {
        return connection.Edges
            .Select(e => e.Node.AuthorId)
            .Distinct()
            .Count();
    }
```

### Custom Cursor Format

```csharp
// Custom cursor serializer
public class CustomCursorSerializer : ICursorSerializer
{
    public string Serialize(int position)
    {
        // Custom format: base64(timestamp:position)
        var data = $"{DateTime.UtcNow.Ticks}:{position}";
        return Convert.ToBase64String(Encoding.UTF8.GetBytes(data));
    }
    
    public int Deserialize(string cursor)
    {
        var decoded = Encoding.UTF8.GetString(Convert.FromBase64String(cursor));
        var parts = decoded.Split(':');
        return int.Parse(parts[1]);
    }
}

// Register custom serializer
```

### Pagination Performance Patterns

```csharp
// Efficient pagination with count caching
public class Query
{
    [UsePaging]
    public async Task<Connection<Book>> GetBooksWithCachedCountAsync(
        [Service] BookDbContext context,
        [Service] IMemoryCache cache,
        PagingArguments args)
    {
        var cacheKey = "books:totalCount";
        
        if (!cache.TryGetValue(cacheKey, out int totalCount))
        {
            totalCount = await context.Books.CountAsync();
            cache.Set(cacheKey, totalCount, TimeSpan.FromMinutes(5));
        }
        
        var query = context.Books.OrderBy(b => b.Id);
        var items = await query
            .Skip(args.Skip ?? 0)
            .Take(args.Take ?? 20)
            .ToListAsync();
        
        // Build connection manually with cached count
        return new Connection<Book>(
            items,
            new ConnectionPageInfo(
                hasNextPage: (args.Skip ?? 0) + items.Count < totalCount,
                hasPreviousPage: (args.Skip ?? 0) > 0),
            totalCount);
    }
}
```

### Seek Pagination (Alternative to Cursor)

```csharp
// Seek pagination for time-series data
public class Query
{
    public async Task<BookSeekResult> GetBooksSeekAsync(
        [Service] BookDbContext context,
        DateTime? afterTimestamp = null,
        int? afterId = null,
        int take = 20)
    {
        var query = context.Books.AsQueryable();
        
        if (afterTimestamp.HasValue && afterId.HasValue)
        {
            query = query.Where(b =>
                b.CreatedAt > afterTimestamp.Value ||
                (b.CreatedAt == afterTimestamp.Value && b.Id > afterId.Value));
        }
        
        var books = await query
            .OrderBy(b => b.CreatedAt)
            .ThenBy(b => b.Id)
            .Take(take + 1)
            .ToListAsync();
        
        var hasMore = books.Count > take;
        if (hasMore) books.RemoveAt(books.Count - 1);
        
        return new BookSeekResult
        {
            Books = books,
            HasMore = hasMore,
            NextTimestamp = books.LastOrDefault()?.CreatedAt,
            NextId = books.LastOrDefault()?.Id
        };
    }
}

public class BookSeekResult
{
    public List<Book> Books { get; set; } = new();
    public bool HasMore { get; set; }
    public DateTime? NextTimestamp { get; set; }
    public int? NextId { get; set; }
}
```

### Stable Cursor Implementation

```csharp
// Ensure stable cursors even with data changes
public class StableCursorPagination
{
    public async Task<Connection<Book>> GetBooksStableAsync(
        [Service] BookDbContext context,
        int? first,
        string? after)
    {
        // Decode cursor to get stable position (e.g., timestamp + ID)
        var (afterTimestamp, afterId) = after != null
            ? DecodeCursor(after)
            : (DateTime.MinValue, 0);
        
        var books = await context.Books
            .Where(b =>
                b.CreatedAt > afterTimestamp ||
                (b.CreatedAt == afterTimestamp && b.Id > afterId))
            .OrderBy(b => b.CreatedAt)
            .ThenBy(b => b.Id)
            .Take((first ?? 20) + 1)
            .ToListAsync();
        
        var hasNextPage = books.Count > (first ?? 20);
        if (hasNextPage) books.RemoveAt(books.Count - 1);
        
        var edges = books.Select(b => new Edge<Book>
        {
            Node = b,
            Cursor = EncodeCursor(b.CreatedAt, b.Id)
        }).ToList();
        
        return new Connection<Book>
        {
            Edges = edges,
            PageInfo = new ConnectionPageInfo
            {
                HasNextPage = hasNextPage,
                EndCursor = edges.LastOrDefault()?.Cursor
            }
        };
    }
    
    private string EncodeCursor(DateTime timestamp, int id)
    {
        var data = $"{timestamp.Ticks}:{id}";
        return Convert.ToBase64String(Encoding.UTF8.GetBytes(data));
    }
    
    private (DateTime, int) DecodeCursor(string cursor)
    {
        var decoded = Encoding.UTF8.GetString(Convert.FromBase64String(cursor));
        var parts = decoded.Split(':');
        return (new DateTime(long.Parse(parts[0])), int.Parse(parts[1]));
    }
}
```
