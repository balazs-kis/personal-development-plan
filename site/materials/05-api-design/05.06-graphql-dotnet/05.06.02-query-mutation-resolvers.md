## Introduction

Resolvers are the heart of your GraphQL API—they fetch the data for each field. Hot Chocolate provides powerful features for implementing efficient resolvers, including DataLoader for batching, dependency injection, and parent resolvers for handling relationships.

### DataLoader Implementation

```csharp
// BatchDataLoader for 1:1 relationships
public class BookDataLoader : BatchDataLoader<int, Book>
{
    private readonly IBookRepository _repository;
    
    public BookDataLoader(
        IBookRepository repository,
        IBatchScheduler batchScheduler,
        DataLoaderOptions? options = null)
        : base(batchScheduler, options)
    {
        _repository = repository;
    }
    
    protected override async Task<IReadOnlyDictionary<int, Book>> LoadBatchAsync(
        IReadOnlyList<int> keys,
        CancellationToken cancellationToken)
    {
        var books = await _repository.GetBooksByIdsAsync(keys, cancellationToken);
        return books.ToDictionary(b => b.Id);
    }
}

// GroupDataLoader for 1:N relationships
public class BooksByAuthorDataLoader : GroupedDataLoader<int, Book>
{
    private readonly IBookRepository _repository;
    
    public BooksByAuthorDataLoader(
        IBookRepository repository,
        IBatchScheduler batchScheduler,
        DataLoaderOptions? options = null)
        : base(batchScheduler, options)
    {
        _repository = repository;
    }
    
    protected override async Task<ILookup<int, Book>> LoadGroupedBatchAsync(
        IReadOnlyList<int> keys,
        CancellationToken cancellationToken)
    {
        var books = await _repository.GetBooksByAuthorIdsAsync(keys, cancellationToken);
        return books.ToLookup(b => b.AuthorId);
    }
}

// Register DataLoaders
builder.Services
    .AddGraphQLServer()
    .AddDataLoader<BookDataLoader>()
    .AddDataLoader<BooksByAuthorDataLoader>();
```

### Mutations with Validation

```csharp
public class Mutation
{
    public async Task<AddBookPayload> AddBookAsync(
        AddBookInput input,
        [Service] IBookRepository repository,
        [Service] IValidator<AddBookInput> validator)
    {
        var validationResult = await validator.ValidateAsync(input);
        if (!validationResult.IsValid)
        {
            return new AddBookPayload(
                errors: validationResult.Errors
                    .Select(e => new UserError(e.ErrorMessage, "VALIDATION_ERROR"))
                    .ToList());
        }
        
        var book = new Book
        {
            Title = input.Title,
            Author= input.AuthorId,
            PublishedDate = input.PublishedDate
        };
        
        var created = await repository.AddAsync(book);
        return new AddBookPayload(created);
    }
}

// Payload type
public class AddBookPayload
{
    public AddBookPayload(Book? book = null, IReadOnlyList<UserError>? errors = null)
    {
        Book = book;
        Errors = errors ?? Array.Empty<UserError>();
    }
    
    public Book? Book { get; }
    public IReadOnlyList<UserError> Errors { get; }
}
```

### Advanced DataLoader Patterns

```csharp
// CacheDataLoader for expensive computations
public class BookStatisticsDataLoader : CacheDataLoader<int, BookStatistics>
{
    private readonly IBookRepository _repository;
    
    public BookStatisticsDataLoader(
        IBookRepository repository,
        IBatchScheduler batchScheduler,
        DataLoaderOptions? options = null)
        : base(batchScheduler, options)
    {
        _repository = repository;
    }
    
    protected override async Task<BookStatistics> LoadSingleAsync(
        int key,
        CancellationToken cancellationToken)
    {
        var book = await _repository.GetByIdAsync(key, cancellationToken);
        
        // Expensive calculation
        return new BookStatistics
        {
            ViewCount = await _repository.GetViewCountAsync(key),
            AverageRating = await _repository.GetAverageRatingAsync(key),
            ReviewCount = await _repository.GetReviewCountAsync(key)
        };
    }
}

// DataLoader with custom cache key
public class BookByIsbnDataLoader : BatchDataLoader<string, Book>
{
    private readonly IBookRepository _repository;
    
    public BookByIsbnDataLoader(
        IBookRepository repository,
        IBatchScheduler batchScheduler)
        : base(batchScheduler, new DataLoaderOptions
        {
            // Custom cache key function
            CacheKeyComparer = StringComparer.OrdinalIgnoreCase
        })
    {
        _repository = repository;
    }
    
    protected override async Task<IReadOnlyDictionary<string, Book>> LoadBatchAsync(
        IReadOnlyList<string> isbns,
        CancellationToken cancellationToken)
    {
        var books = await _repository.GetByIsbnsAsync(isbns, cancellationToken);
        return books.ToDictionary(b => b.ISBN, StringComparer.OrdinalIgnoreCase);
    }
}
```

### Error Handling Strategies

```csharp
// Custom exception types
public class BookNotFoundException : Exception
{
    public BookNotFoundException(int id)
        : base($"Book with ID {id} not found")
    {
        BookId = id;
    }
    
    public int BookId { get; }
}

public class InsufficientPermissionsException : Exception
{
    public InsufficientPermissionsException(string action)
        : base($"Insufficient permissions for: {action}")
    {
        Action = action;
    }
    
    public string Action { get; }
}

// Global error filter
public class GlobalErrorFilter : IErrorFilter
{
    private readonly ILogger<GlobalErrorFilter> _logger;
    
    public GlobalErrorFilter(ILogger<GlobalErrorFilter> logger)
    {
        _logger = logger;
    }
    
    public IError OnError(IError error)
    {
        // Log all errors
        _logger.LogError(error.Exception, "GraphQL error occurred");
        
        return error.Exception switch
        {
            BookNotFoundException ex => error
                .WithMessage(ex.Message)
                .SetCode("BOOK_NOT_FOUND")
                .SetExtension("bookId", ex.BookId)
                .RemoveException(),
            
            InsufficientPermissionsException ex => error
                .WithMessage("You don't have permission to perform this action")
                .SetCode("FORBIDDEN")
                .SetExtension("action", ex.Action)
                .RemoveException(),
            
            ValidationException ex => error
                .WithMessage("Validation failed")
                .SetCode("VALIDATION_ERROR")
                .SetExtension("errors", ex.Errors)
                .RemoveException(),
            
            _ => error
                .WithMessage("An unexpected error occurred")
                .SetCode("INTERNAL_SERVER_ERROR")
                .RemoveException()
        };
    }
}

// Register error filter
builder.Services
    .AddGraphQLServer()
    .AddErrorFilter<GlobalErrorFilter>();
```

### Mutation Patterns

```csharp
// Mutation with optimistic concurrency
public class Mutation
{
    public async Task<UpdateBookPayload> UpdateBookAsync(
        UpdateBookInput input,
        [Service] IBookRepository repository)
    {
        try
        {
            var book = await repository.GetByIdAsync(input.Id);
            
            if (book == null)
            {
                return new UpdateBookPayload
                {
                    Errors = new[]
                    {
                        new UserError("Book not found", "BOOK_NOT_FOUND")
                    }
                };
            }
            
            // Check version for optimistic concurrency
            if (book.Version != input.Version)
            {
                return new UpdateBookPayload
                {
                    Errors = new[]
                    {
                        new UserError(
                            "Book has been modified by another user",
                            "CONFLICT")
                    }
                };
            }
            
            book.Title = input.Title;
            book.Author = input.Author;
            book.Version++;
            
            var updated = await repository.UpdateAsync(book);
            
            return new UpdateBookPayload { Book = updated };
        }
        catch (DbUpdateConcurrencyException)
        {
            return new UpdateBookPayload
            {
                Errors = new[]
                {
                    new UserError(
                        "Book has been modified by another user",
                        "CONFLICT")
                }
            };
        }
    }
}

// Batch mutations
public class Mutation
{
    public async Task<BatchUpdateBooksPayload> UpdateBooksAsync(
        List<UpdateBookInput> inputs,
        [Service] IBookRepository repository)
    {
        var results = new List<BookResult>();
        
        foreach (var input in inputs)
        {
            try
            {
                var book = await repository.UpdateAsync(input.Id, input);
                results.Add(new BookResult { Book = book });
            }
            catch (Exception ex)
            {
                results.Add(new BookResult
                {
                    Error = new UserError(ex.Message, "UPDATE_FAILED")
                });
            }
        }
        
        return new BatchUpdateBooksPayload { Results = results };
    }
}

public class BookResult
{
    public Book? Book { get; set; }
    public UserError? Error { get; set; }
}
```

### Transaction Management

```csharp
public class Mutation
{
    [UseDbContext(typeof(BookDbContext))]
    public async Task<CreateBookWithReviewsPayload> CreateBookWithReviewsAsync(
        CreateBookWithReviewsInput input,
        [ScopedService] BookDbContext context)
    {
        // Start transaction
        await using var transaction = await context.Database.BeginTransactionAsync();
        
        try
        {
            // Create book
            var book = new Book
            {
                Title = input.Title,
                Author = input.Author
            };
            
            context.Books.Add(book);
            await context.SaveChangesAsync();
            
            // Create reviews
            foreach (var reviewInput in input.Reviews)
            {
                var review = new Review
                {
                    BookId = book.Id,
                    Content = reviewInput.Content,
                    Rating = reviewInput.Rating
                };
                
                context.Reviews.Add(review);
            }
            
            await context.SaveChangesAsync();
            await transaction.CommitAsync();
            
            return new CreateBookWithReviewsPayload { Book = book };
        }
        catch (Exception)
        {
            await transaction.RollbackAsync();
            throw;
        }
    }
}
```

### Resolver Composition

```csharp
// Reusable resolver logic
public class AuthorizedResolverBase
{
    protected async Task<T> ExecuteAuthorizedAsync<T>(
        ClaimsPrincipal user,
        string requiredRole,
        Func<Task<T>> action)
    {
        if (!user.IsInRole(requiredRole))
        {
            throw new GraphQLException(
                new Error("Insufficient permissions", "FORBIDDEN"));
        }
        
        return await action();
    }
}

public class BookResolvers : AuthorizedResolverBase
{
    public async Task<Book> DeleteBookAsync(
        int id,
        ClaimsPrincipal user,
        [Service] IBookRepository repository)
    {
        return await ExecuteAuthorizedAsync(
            user,
            "Admin",
            async () =>
            {
                var book = await repository.GetByIdAsync(id);
                await repository.DeleteAsync(id);
                return book;
            });
    }
}
```

### Field Middleware

```csharp
// Caching middleware
public class CacheFieldMiddleware
{
    private readonly FieldDelegate _next;
    private readonly IMemoryCache _cache;
    
    public CacheFieldMiddleware(FieldDelegate next, IMemoryCache cache)
    {
        _next = next;
        _cache = cache;
    }
    
    public async Task InvokeAsync(IMiddlewareContext context)
    {
        var cacheKey = $"{context.Selection.Field.Name}:{string.Join(":", context.ArgumentLiterals)}";
        
        if (_cache.TryGetValue(cacheKey, out object? cachedValue))
        {
            context.Result = cachedValue;
            return;
        }
        
        await _next(context);
        
        if (context.Result is not null)
        {
            _cache.Set(cacheKey, context.Result, TimeSpan.FromMinutes(5));
        }
    }
}

// Apply to field
public class Query
{
    [UseMiddleware(typeof(CacheFieldMiddleware))]
    public async Task<List<Book>> GetBooksAsync([Service] IBookRepository repository)
    {
        return await repository.GetAllAsync();
    }
}
```
