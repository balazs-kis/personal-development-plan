# Lesson 3: Filtering and Sorting

## Learning Objectives

By the end of this lesson, you will be able to:
- Add filtering capabilities to your GraphQL API
- Implement sorting functionality

- Configure and customize filters

## Introduction

Hot Chocolate provides built-in support for filtering and sorting through the `HotChocolate.Data` package. These features allow clients to filter and sort data without writing custom resolver logic, making your API more flexible and powerful.

### Custom Filter Operations

```csharp
// Define custom filter operation
public class StringContainsAnyFilterOperationHandler
    : FilterOperationHandler<StringOperationFilterInput, string>
{
    public override bool CanHandle(
        ITypeCompletionContext context,
        IFilterInputTypeDefinition typeDefinition,
        IFilterFieldDefinition fieldDefinition)
    {
        return fieldDefinition.Member is PropertyInfo p &&
               p.PropertyType == typeof(string);
    }
    
    public override Expression HandleOperation(
        QueryableFilterContext context,
        IFilterField field,
        StringOperationFilterInput filterInput,
        object? parsedValue)
    {
        if (parsedValue is string[] values)
        {
            var property = Expression.Property(
                context.GetInstance(),
                field.Member as PropertyInfo);
            
            // Build OR expression for each value
            Expression? expression = null;
            foreach (var value in values)
            {
                var containsCall = Expression.Call(
                    property,
                    typeof(string).GetMethod(nameof(string.Contains), new[] { typeof(string) })!,
                    Expression.Constant(value));
                
                expression = expression == null
                    ? containsCall
                    : Expression.OrElse(expression, containsCall);
            }
            
            return expression!;
        }
        
        throw new InvalidOperationException();
    }
}

// Register custom operation
builder.Services
    .AddGraphQLServer()
    .AddFiltering(x => x
        .AddOperationHandler<StringContainsAnyFilterOperationHandler>());
```

### Projection with Computed Fields

public enum BookSortField

public class BookSortType : SortInputType<Book>

        // Custom sort logic
        descriptor
            .Field("popularity")

            });
    }
}

### Filtering with Custom Logic

```csharp
// Custom filter operation
public class IsPopularFilterOperationHandler
    : FilterOperationHandler<BooleanOperationFilterInput, bool>
{
    public override Expression HandleOperation(
        QueryableFilterContext context,
        IFilterField field,
        BooleanOperationFilterInput filterInput,
        object? parsedValue)
    {
        var book = Expression.Parameter(typeof(Book), "book");
        
        // Define "popular" as rating > 4.5 AND viewCount > 1000
        var ratingCheck = Expression.GreaterThan(
            Expression.Property(book, nameof(Book.Rating)),
            Expression.Constant(4.5));
        
        var viewCountCheck = Expression.GreaterThan(
            Expression.Property(book, nameof(Book.ViewCount)),
            Expression.Constant(1000));
        
        var combined = Expression.AndAlso(ratingCheck, viewCountCheck);
        
        return Expression.Lambda<Func<Book, bool>>(combined, book);
    }
}
```
