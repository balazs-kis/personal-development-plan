## Introduction

The N+1 problem is a common performance issue in GraphQL where fetching a list of items and their related data results in N+1 database queries (1 query for the list + N queries for each item's relations). DataLoader solves this by batching and caching data fetches, significantly improving performance.

## Key Concepts Summary

### The N+1 Problem

```graphql
# Query
{
  users {           # 1 query to get users
    id
    name
    posts {         # N queries (one per user!)
      title
    }
  }
}

# Without DataLoader:
# SELECT * FROM users;  (1 query)
# SELECT * FROM posts WHERE author_id = 1;  (query 1)
# SELECT * FROM posts WHERE author_id = 2;  (query 2)
# SELECT * FROM posts WHERE author_id = 3;  (query 3)
# ... N more queries!
```

### DataLoader Solution

```typescript
import DataLoader from 'dataloader';

// Create DataLoader
const postLoader = new DataLoader(async (userIds) => {
  // Batch load all posts for all users in one query
  const posts = await db.posts.findByAuthorIds(userIds);
  
  // Group posts by author ID
  const postsByUserId = groupBy(posts, 'authorId');
  
  // Return in same order as input userIds
  return userIds.map(id => postsByUserId[id] || []);
});

// Use in resolver
const resolvers = {
  User: {
    posts: (user, args, { postLoader }) => {
      return postLoader.load(user.id);  // Batches automatically
    }
  }
};

// Result:
// SELECT * FROM users;  (1 query)
// SELECT * FROM posts WHERE author_id IN (1, 2, 3, ...);  (1 batched query!)
```

### Complete Example

```typescript
// loaders.ts
import DataLoader from 'dataloader';

export const createLoaders = (db) => ({
  userLoader: new DataLoader(async (ids) => {
    const users = await db.users.findByIds(ids);
    const userMap = new Map(users.map(u => [u.id, u]));
    return ids.map(id => userMap.get(id));
  }),
  
  postsByAuthorLoader: new DataLoader(async (authorIds) => {
    const posts = await db.posts.findByAuthorIds(authorIds);
    const postsByAuthor = groupBy(posts, 'authorId');
    return authorIds.map(id => postsByAuthor[id] || []);
  })
});

// resolvers.ts
const resolvers = {
  Query: {
    users: (_, __, { db }) => db.users.findAll()
  },
  
  User: {
    posts: (user, _, { loaders }) => {
      return loaders.postsByAuthorLoader.load(user.id);
    }
  },
  
  Post: {
    author: (post, _, { loaders }) => {
      return loaders.userLoader.load(post.authorId);
    }
  }
};
```

### DataLoader Configuration

```typescript
// Basic DataLoader

const userLoader = new DataLoader(
  async (ids) => {
    const users = await db.users.findByIds(ids);
    return ids.map(id => users.find(u => u.id === id));

  {
    // Disable caching

    // Custom cache implementation

  }
);
```


**One-to-Many:**

```typescript
// User has many Posts
const postsByAuthorLoader = new DataLoader(async (authorIds) => {
  const posts = await db.posts.findByAuthorIds(authorIds);
  
  // Group by authorId
  const grouped = {};
  posts.forEach(post => {
    if (!grouped[post.authorId]) {
      grouped[post.authorId] = [];
    }
    grouped[post.authorId].push(post);
  });
  
  // Return in order, empty array if no posts
  return authorIds.map(id => grouped[id] || []);
});

const resolvers = {
  User: {
    posts: (user, _, { loaders }) => {
      return loaders.postsByAuthorLoader.load(user.id);
    }
  }
};
```

**Many-to-Many:**

```typescript
// Post has many Tags (through junction table)
const tagsByPostLoader = new DataLoader(async (postIds) => {
  // Query junction table + tags
  const postTags = await db.query(`
    SELECT pt.post_id, t.*
    FROM post_tags pt
    JOIN tags t ON pt.tag_id = t.id
    WHERE pt.post_id IN (${postIds.join(',')})
  `);
  
  // Group by post_id
  const grouped = {};
  postTags.forEach(({ post_id, ...tag }) => {
    if (!grouped[post_id]) {
      grouped[post_id] = [];
    }
    grouped[post_id].push(tag);
  });
  
  return postIds.map(id => grouped[id] || []);
});

const resolvers = {
  Post: {
    tags: (post, _, { loaders }) => {
      return loaders.tagsByPostLoader.load(post.id);
    }
  }
};
```

### Advanced DataLoader Patterns

**Composite Keys:**

```typescript
// Load by composite key (userId + status)
const orderLoader = new DataLoader(async (keys) => {
  // keys = [{ userId: '1', status: 'PENDING' }, { userId: '2', status: 'SHIPPED' }]
  
  const results = await Promise.all(
    keys.map(({ userId, status }) => 
      db.orders.findByUserAndStatus(userId, status)
    )
  );
  
  return results;
}, {
  cacheKeyFn: (key) => `${key.userId}:${key.status}`
});

// Usage
const orders = await orderLoader.load({
  userId: user.id,
  status: 'PENDING'
});
```

**Priming the Cache:**

```typescript
const userLoader = new DataLoader(async (ids) => {
  const users = await db.users.findByIds(ids);
  return ids.map(id => users.find(u => u.id === id));
});

// Prime cache when creating user
const resolvers = {
  Mutation: {
    createUser: async (parent, { input }, { db, loaders }) => {
      const user = await db.users.create(input);
      
      // Prime the loader cache
      loaders.userLoader.prime(user.id, user);
      
      return user;
    }
  }
};
```

**Clearing Cache:**

```typescript
const resolvers = {
  Mutation: {
    updateUser: async (parent, { id, input }, { db, loaders }) => {
      const user = await db.users.update(id, input);
      
      // Clear cache for this user
      loaders.userLoader.clear(id);
      
      // Or clear all cache
      loaders.userLoader.clearAll();
      
      return user;
    }
  }
};
```

**Counting with DataLoader:**

```typescript
// Count posts by author
```

### DataLoader with Filtering

```typescript
// Load posts by author with filters
const createPostsByAuthorLoader = (filters) => {
  return new DataLoader(async (authorIds) => {
    const posts = await db.posts.findByAuthorIds(authorIds, filters);
    const grouped = groupBy(posts, 'authorId');
    return authorIds.map(id => grouped[id] || []);
  }, {
    cacheKeyFn: (id) => `${id}:${JSON.stringify(filters)}`
  });
};

const resolvers = {
  User: {
    posts: (user, { status, first }, { db }) => {
      // Create loader with filters
      const loader = createPostsByAuthorLoader({ status, first });
      return loader.load(user.id);
    }
  }
};
```

### Error Handling

```typescript
const userLoader = new DataLoader(async (ids) => {
  try {
    const users = await db.users.findByIds(ids);
    const userMap = new Map(users.map(u => [u.id, u]));
    
    return ids.map(id => {
      const user = userMap.get(id);
      
      // Return error for missing users
      if (!user) {
        return new Error(`User ${id} not found`);
      }
      
      return user;
    });
  } catch (error) {
    // Return error for each id
    return ids.map(() => error);
  }
});

// In resolver
const resolvers = {
  Post: {
    author: async (post, _, { loaders }) => {
      try {
        return await loaders.userLoader.load(post.authorId);
      } catch (error) {
        console.error('Failed to load author:', error);
        return null;  // Or throw error
      }
    }
  }
};
```

### Performance Monitoring

```typescript
const createInstrumentedLoader = (name, batchLoadFn) => {
  return new DataLoader(async (keys) => {
    const start = Date.now();
    console.log(`[${name}] Batching ${keys.length} keys`);
    
    try {
      const results = await batchLoadFn(keys);
      const duration = Date.now() - start;
      
      console.log(`[${name}] Completed in ${duration}ms`);
      
      return results;
    } catch (error) {
      const duration = Date.now() - start;
      console.error(`[${name}] Failed after ${duration}ms:`, error);
      throw error;
    }
  });
};

// Usage
const userLoader = createInstrumentedLoader(
  'userLoader',
  async (ids) => {
    const users = await db.users.findByIds(ids);
    const userMap = new Map(users.map(u => [u.id, u]));
    return ids.map(id => userMap.get(id));
  }
);
```

### DataLoader Best Practices

```typescript
// ✅ Good: Create loaders per request
const server = new ApolloServer({
  typeDefs,
  resolvers,
  context: ({ req }) => ({
    user: req.user,
    loaders: createLoaders(db)  // Fresh loaders per request
  })
});

// ❌ Bad: Reusing loaders across requests
const loaders = createLoaders(db);  // Don't do this!
const server = new ApolloServer({
  context: ({ req }) => ({
    loaders  // Shared cache across users!
  })
});

// ✅ Good: Return results in same order as keys
const userLoader = new DataLoader(async (ids) => {
  const users = await db.users.findByIds(ids);
  const userMap = new Map(users.map(u => [u.id, u]));
  return ids.map(id => userMap.get(id));  // Same order!
});

// ❌ Bad: Wrong order
const badLoader = new DataLoader(async (ids) => {
  const users = await db.users.findByIds(ids);
  return users;  // Wrong order!
});

// ✅ Good: Handle missing values
return ids.map(id => userMap.get(id) || null);

// ❌ Bad: Undefined for missing
return ids.map(id => userMap.get(id));  // undefined breaks GraphQL
```

### Real-World Example

```typescript
// Complete loader setup
export const createLoaders = (db) => {
  return {
    // Users
    userLoader: new DataLoader(async (ids) => {
      const users = await db.users.findByIds(ids);
      const map = new Map(users.map(u => [u.id, u]));
      return ids.map(id => map.get(id) || null);
    }),
    
    // Posts by author
    postsByAuthorLoader: new DataLoader(async (authorIds) => {
      const posts = await db.posts.findByAuthorIds(authorIds);
      const grouped = groupBy(posts, 'authorId');
      return authorIds.map(id => grouped[id] || []);
    }),
    
    // Comments by post
    commentsByPostLoader: new DataLoader(async (postIds) => {
      const comments = await db.comments.findByPostIds(postIds);
      const grouped = groupBy(comments, 'postId');
      return postIds.map(id => grouped[id] || []);
    }),
    
    // Like count
    likeCountLoader: new DataLoader(async (postIds) => {
      const counts = await db.likes.countByPostIds(postIds);
      const map = new Map(counts.map(c => [c.postId, c.count]));
      return postIds.map(id => map.get(id) || 0);
    }),
    
    // Has liked (for current user)
    hasLikedLoader: new DataLoader(async (postIds) => {
      if (!this.userId) return postIds.map(() => false);
      
      const likes = await db.likes.findByUserAndPosts(
        this.userId,
        postIds
      );
      const likedSet = new Set(likes.map(l => l.postId));
      return postIds.map(id => likedSet.has(id));
    })
  };
};

// Usage in resolvers
const resolvers = {
  Query: {
    feed: async (parent, { first }, { db }) => {
      return db.posts.findRecent(first);
    }
  },
  
  Post: {
    author: (post, _, { loaders }) => {
      return loaders.userLoader.load(post.authorId);
    },
    
    comments: (post, _, { loaders }) => {
      return loaders.commentsByPostLoader.load(post.id);
    },
    
    likeCount: (post, _, { loaders }) => {
      return loaders.likeCountLoader.load(post.id);
    },
    
    liked: (post, _, { loaders }) => {
      return loaders.hasLikedLoader.load(post.id);
    }
  },
  
  Comment: {
    author: (comment, _, { loaders }) => {
      return loaders.userLoader.load(comment.authorId);
    }
  }
};
```
