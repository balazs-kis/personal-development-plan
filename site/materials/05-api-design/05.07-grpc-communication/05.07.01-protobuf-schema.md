# Lesson 1: Protocol Buffers Schema Design

## Learning Objectives

By the end of this lesson, you will be able to:
- Understand Protocol Buffers syntax and data types
- Design efficient and evolvable .proto schemas
- Use scalar types, enums, and nested messages
- Apply best practices for field numbering and naming
- Handle schema evolution and backward compatibility

---

## Introduction

Protocol Buffers (protobuf) is Google's language-neutral, platform-neutral mechanism for serializing structured data. It's the foundation of gRPC, providing strong typing, efficient binary serialization, and excellent support for schema evolution.

## Key Concepts Summary

### Basic Proto3 Syntax

```protobuf
syntax = "proto3";

package ecommerce.v1;

option csharp_namespace = "ECommerce.V1";

// Simple message
message Product {
  int64 id = 1;
  string name = 2;
  string description = 3;
  double price = 4;
  bool in_stock = 5;
}

// Message with enum
message Order {
  enum Status {
    STATUS_UNSPECIFIED = 0;
    PENDING = 1;
    PROCESSING = 2;
    SHIPPED = 3;
    DELIVERED = 4;
    CANCELLED = 5;
  }
  
  int64 id = 1;
  int64 customer_id = 2;
  Status status = 3;
  repeated OrderItem items = 4;
  google.protobuf.Timestamp created_at = 5;
}

// Nested message
message OrderItem {
  int64 product_id = 1;
  int32 quantity = 2;
  double unit_price = 3;
}
```

### Scalar Types

```protobuf
message ScalarTypes {
  // Integers
  int32 age = 1;           // Variable-length encoding
  int64 id = 2;            // Variable-length encoding
  uint32 count = 3;        // Unsigned
  uint64 timestamp = 4;    // Unsigned
  sint32 temperature = 5;  // Signed (efficient for negative)
  sint64 offset = 6;       // Signed (efficient for negative)
  fixed32 ip_address = 7;  // Fixed 4 bytes
  fixed64 hash = 8;        // Fixed 8 bytes
  sfixed32 x = 9;          // Signed fixed
  sfixed64 y = 10;         // Signed fixed
  
  // Floating point
  float ratio = 11;        // 4 bytes
  double precision = 12;   // 8 bytes
  
  // Boolean
  bool is_active = 13;
  
  // String and bytes
  string name = 14;        // UTF-8 or ASCII
  bytes data = 15;         // Arbitrary bytes
}
```

### Enums

```protobuf
// Enum (first value must be 0)

  PRODUCT_CATEGORY_UNSPECIFIED = 0;  // Required
  ELECTRONICS = 1;
  CLOTHING = 2;
  BOOKS = 3;
  HOME = 4;

// Using enum in message

  int64 id = 1;
  string name = 2;
```

### Field Numbering Best Practices

```protobuf
message Product {
  // 1-15: Single byte encoding (use for frequently used fields)
  int64 id = 1;
  string name = 2;
  double price = 3;
  bool in_stock = 4;
  
  // 16-2047: Two byte encoding
  string long_description = 16;
  repeated string tags = 17;
  map<string, string> attributes = 18;
  
  // Reserved for future use or deprecated fields
  reserved 10, 11, 12;
  reserved "old_field_name", "deprecated_field";
}
```

### Schema Evolution

```protobuf
// Version 1
message Product {
  int64 id = 1;
  string name = 2;
  double price = 3;
}

// Version 2 (backward compatible)
message Product {
  int64 id = 1;
  string name = 2;
  double price = 3;
  string description = 4;  // Added field
  repeated string tags = 5;  // Added field
  
  // Old field deprecated
  string old_category = 6 [deprecated = true];
}

// Version 3 (using reserved)
message Product {
  int64 id = 1;
  string name = 2;
  double price = 3;
  string description = 4;
  repeated string tags = 5;
  
  reserved 6;  // Can't reuse field number
  reserved "old_category";  // Can't reuse name
  
  ProductCategory category = 7;  // New field with different type
}
```

### Advanced Schema Patterns

**Schema Evolution Rules:**

```protobuf
// Safe changes (backward compatible)
message Product {
  int64 id = 1;
  string name = 2;
  double price = 3;
  
  // Safe: Add optional fields
  string description = 4;
  repeated string tags = 5;
  
  // Safe: Deprecate fields
  string old_field = 6 [deprecated = true];
}

// Unsafe changes (breaking)
// ❌ Don't change field numbers
// ❌ Don't change field types
// ❌ Don't change singular/repeated
// ❌ Don't delete fields without reserving

// Safe evolution with reserved
message Product {
  int64 id = 1;
  string name = 2;
  double price = 3;
  
  reserved 4, 5;  // Reserved numbers
  reserved "old_field", "deprecated";  // Reserved names
  
  string new_field = 6;
}
```

**Versioning Strategies:**

```protobuf
// Package-based versioning
package products.v1;
// vs
package products.v2;

// Field-level versioning
message Product {
  int64 id = 1;
  string name = 2;
  
  oneof price {
    double price_v1 = 3 [deprecated = true];
    Price price_v2 = 4;
  }
}

message Price {
  double amount = 1;
  string currency = 2;
}
```

**Performance Optimization:**

```protobuf
message ProductMetrics {
  // Use smallest appropriate type
  int32 view_count = 1;      // Up to ~2B
  uint32 like_count = 2;     // Only positive
  sint32 temperature = 3;    // Often negative (efficient encoding)
  
  // Fixed-size for large values
  fixed64 unique_id = 4;     // Always 8 bytes
  
  // Packed repeated primitives (smaller encoding)
  repeated int32 related_ids = 5 [packed = true];
  repeated double prices = 6 [packed = true];
}
```

**Documentation Best Practices:**

```protobuf
// Service-level documentation
// ProductCatalog provides operations for managing product inventory.
//
// This service supports:
// - Product CRUD operations
// - Search and filtering
// - Bulk updates
// - Inventory tracking

// Message-level documentation
// Product represents an item in the catalog.
//
// Products contain all information necessary for display, pricing,
// and inventory management. Each product has a unique SKU and may
// have multiple variants (sizes, colors, etc.).
message Product {
  // Field-level documentation
  // Unique identifier assigned by the system.
  // Read-only after creation.
  int64 id = 1;
  
  // Display name shown to customers.
  // Must be 3-200 characters.
  // Required field.
  string name = 2;
  
  // Stock Keeping Unit for inventory tracking.
  // Format: XXX-YYYYY-ZZ
  // Must be unique across all products.
  string sku = 3;
}

// Enum documentation
// ProductStatus represents the lifecycle state of a product.
enum ProductStatus {
  // Default value, should not be used.
  PRODUCT_STATUS_UNSPECIFIED = 0;
  
  // Product is being prepared, not visible to customers.
  DRAFT = 1;
  
  // Product is live and available for purchase.
  ACTIVE = 2;
  
  // Product is temporarily unavailable.
  INACTIVE = 3;
  
  // Product is permanently removed from catalog.
  DISCONTINUED = 4;
}
```

**Real-World Complex Schema:**

```protobuf
syntax = "proto3";

import "google/protobuf/timestamp.proto";
import "google/protobuf/wrappers.proto";
import "google/protobuf/duration.proto";

option csharp_namespace = "ECommerce.Catalog.V1";

// Complete product with all features
message Product {
  // Identity (1-5)
  int64 id = 1;
  string name = 2;
  string sku = 3;
  string slug = 4;
  string barcode = 5;
  
  // Categorization (6-10)
  ProductCategory category = 6;
  int64 brand_id = 7;
  repeated string tags = 8;
  repeated int64 collection_ids = 9;
  reserved 10;
  
  // Pricing (11-15)
  Price base_price = 11;
  Price sale_price = 12;
  google.protobuf.Timestamp sale_start = 13;
  google.protobuf.Timestamp sale_end = 14;
  TaxInfo tax_info = 15;
  
  // Content (16-25)
  string short_description = 16;
  string long_description = 17;
  repeated MediaAsset images = 18;
  repeated MediaAsset videos = 19;
  repeated Document documents = 20;
  map<string, string> specifications = 21;
  repeated string features = 22;
  SeoMetadata seo = 23;
  map<string, LocalizedContent> localizations = 24;
  reserved 25;
  
  // Inventory (26-30)
  InventoryInfo inventory = 26;
  ShippingInfo shipping = 27;
  bool allow_backorder = 28;
  google.protobuf.Duration estimated_delivery = 29;
  reserved 30;
  
  // Variants (31-35)
  repeated ProductVariant variants = 31;
  repeated VariantOption variant_options = 32;
  reserved 33, 34, 35;
  
  // Social proof (36-40)
  double average_rating = 36;
  int32 review_count = 37;
  int32 purchase_count = 38;
  int32 view_count = 39;
  reserved 40;
  
  // Status & flags (41-45)
  ProductStatus status = 41;
  bool is_featured = 42;
  bool is_new = 43;
  bool is_bestseller = 44;
  google.protobuf.BoolValue is_limited_edition = 45;
  
  // Relationships (46-50)
  repeated int64 related_product_ids = 46;
  repeated int64 cross_sell_product_ids = 47;
  repeated int64 up_sell_product_ids = 48;
  int64 parent_product_id = 49;
  reserved 50;
  
  // Metadata (51-55)
  google.protobuf.Timestamp created_at = 51;
  google.protobuf.Timestamp updated_at = 52;
  google.protobuf.Timestamp published_at = 53;
  string created_by = 54;
  string updated_by = 55;
}

message Price {
  int64 amount = 1;  // In smallest currency unit (cents)
  string currency = 2;  // ISO 4217 code
  string formatted = 3;  // Display string: "$19.99"
}

message TaxInfo {
  bool taxable = 1;
  string tax_code = 2;
  double tax_rate = 3;
}

message MediaAsset {
  string id = 1;
  string url = 2;
  string thumbnail_url = 3;
  string alt_text = 4;
  int32 sort_order = 5;
  int32 width = 6;
  int32 height = 7;
  int64 file_size = 8;
}

message Document {
  string id = 1;
  string name = 2;
  string url = 3;
  string mime_type = 4;
  int64 file_size = 5;
}

message SeoMetadata {
  string meta_title = 1;
  string meta_description = 2;
  repeated string meta_keywords = 3;
  string canonical_url = 4;
  string og_image = 5;
  map<string, string> structured_data = 6;
}

message LocalizedContent {
  string name = 1;
  string short_description = 2;
  string long_description = 3;
  SeoMetadata seo = 4;
}

message InventoryInfo {
  int32 available_quantity = 1;
  int32 reserved_quantity = 2;
  int32 total_quantity = 3;
  bool track_inventory = 4;
  int32 low_stock_threshold = 5;
  InventoryLocation location = 6;
}

message InventoryLocation {
  string warehouse_id = 1;
  string aisle = 2;
  string shelf = 3;
  string bin = 4;
}

message ShippingInfo {
  Weight weight = 1;
  Dimensions dimensions = 2;
  bool requires_shipping = 3;
  bool is_fragile = 4;
  repeated string allowed_shipping_methods = 5;

  double value = 1;

  double height = 3;

message ProductVariant {
```

## Practice Exercise

Design a Protocol Buffers schema for a library management system:

**Requirements:**
1. **Messages:**
   - Book (id, title, ISBN, authors, publication date, categories, available copies)
   - Author (id, name, biography, birth date)
   - Member (id, name, email, membership type, join date)
   - Loan (id, book_id, member_id, due date, return date, status)

2. **Features:**
   - Use appropriate scalar types
   - Include enums for membership type and loan status
   - Use repeated fields for book authors and categories
   - Use timestamps for dates
   - Include proper field numbering
   - Add comments for documentation
