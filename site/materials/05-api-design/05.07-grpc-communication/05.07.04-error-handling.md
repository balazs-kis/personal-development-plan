### Status Codes

```csharp
// Common status codes
public override async Task<GetProductResponse> GetProduct(
    GetProductRequest request,
    ServerCallContext context)
{
    // NOT_FOUND
    if (product == null)
    {
        throw new RpcException(new Status(
            StatusCode.NotFound,
            "Product not found"));
    }
    
    // INVALID_ARGUMENT
    if (request.Id <= 0)
    {
        throw new RpcException(new Status(
            StatusCode.InvalidArgument,
            "Product ID must be positive"));
    }
    
    // UNAUTHENTICATED
    if (!context.GetHttpContext().User.Identity.IsAuthenticated)
    {
        throw new RpcException(new Status(
            StatusCode.Unauthenticated,
            "User not authenticated"));
    }
    
    // PERMISSION_DENIED
    if (!await _authService.CanAccessProduct(userId, productId))
    {
        throw new RpcException(new Status(
            StatusCode.PermissionDenied,
            "Access denied"));
    }
}
```

### Complete Status Code Guide

```csharp
public class ProductService : Products.V1.ProductService.ProductServiceBase
{
    public override async Task<Product> GetProduct(
        GetProductRequest request,
        ServerCallContext context)
    {
        // OK (0) - Success (implicit, don't throw)
        // Just return the response
        
        // CANCELLED (1) - Operation cancelled by client
        if (context.CancellationToken.IsCancellationRequested)
        {
            throw new RpcException(new Status(
                StatusCode.Cancelled,
                "Operation cancelled by client"));
        }
        
        // UNKNOWN (2) - Unknown error
        // Let unhandled exceptions become UNKNOWN
        
        // INVALID_ARGUMENT (3) - Invalid request parameters
        if (request.Id <= 0)
        {
            throw new RpcException(new Status(
                StatusCode.InvalidArgument,
                "Product ID must be positive"));
        }
        
        // DEADLINE_EXCEEDED (4) - Operation timeout
        if (DateTime.UtcNow > context.Deadline)
        {
            throw new RpcException(new Status(
                StatusCode.DeadlineExceeded,
                "Request deadline exceeded"));
        }
        
        // NOT_FOUND (5) - Resource not found
        var product = await _repository.GetByIdAsync(request.Id);
        if (product == null)
        {
            throw new RpcException(new Status(
                StatusCode.NotFound,
                $"Product {request.Id} not found"));
        }
        
        // ALREADY_EXISTS (6) - Resource already exists
        if (await _repository.ExistsBySkuAsync(request.Sku))
        {
            throw new RpcException(new Status(
                StatusCode.AlreadyExists,
                $"Product with SKU {request.Sku} already exists"));
        }
        
        // PERMISSION_DENIED (7) - Insufficient permissions
        if (!await _authService.CanAccessProduct(userId, productId))
        {
            throw new RpcException(new Status(
                StatusCode.PermissionDenied,
                "You don't have permission to access this product"));
        }
        
        // RESOURCE_EXHAUSTED (8) - Rate limit or quota exceeded
        if (!await _rateLimiter.AllowRequest(userId))
        {
            throw new RpcException(new Status(
                StatusCode.ResourceExhausted,
                "Rate limit exceeded. Try again later."));
        }
        
        // FAILED_PRECONDITION (9) - Precondition not met
        if (product.Status != ProductStatus.Draft)
        {
            throw new RpcException(new Status(
                StatusCode.FailedPrecondition,
                "Product must be in Draft status to be edited"));
        }
        
        // ABORTED (10) - Concurrency conflict
        if (product.Version != request.ExpectedVersion)
        {
            throw new RpcException(new Status(
                StatusCode.Aborted,
                "Product was modified by another user. Please refresh and try again."));
        }
        
        // OUT_OF_RANGE (11) - Value out of range
        if (request.Quantity > product.MaxOrderQuantity)
        {
            throw new RpcException(new Status(
                StatusCode.OutOfRange,
                $"Quantity must be between 1 and {product.MaxOrderQuantity}"));
        }
        
        // UNIMPLEMENTED (12) - Not implemented
        throw new RpcException(new Status(
            StatusCode.Unimplemented,
            "This feature is not yet implemented"));
        
        // INTERNAL (13) - Internal server error
        try
        {
            // Some operation
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Internal error processing request");
            throw new RpcException(new Status(
                StatusCode.Internal,
                "An internal error occurred"));
        }
        
        // UNAVAILABLE (14) - Service unavailable
        if (!await _healthCheck.IsHealthyAsync())
        {
            throw new RpcException(new Status(
                StatusCode.Unavailable,
                "Service is temporarily unavailable"));
        }
        
        // DATA_LOSS (15) - Data corruption
        if (!await _validator.ValidateIntegrityAsync(product))
        {
            throw new RpcException(new Status(
                StatusCode.DataLoss,
                "Data integrity check failed"));
        }
        
        // UNAUTHENTICATED (16) - Missing or invalid authentication
        if (!context.GetHttpContext().User.Identity.IsAuthenticated)
        {
            throw new RpcException(new Status(
                StatusCode.Unauthenticated,
                "Authentication required"));
        }
        
        return MapToProto(product);
    }
}
```

### Structured Error Details

```protobuf
// Define error details in proto
syntax = "proto3";

package errors.v1;

message ErrorDetail {
  string code = 1;
  string message = 2;
  map<string, string> metadata = 3;
}

message ValidationError {
  message FieldViolation {
    string field = 1;
    string description = 2;
  }
  
  repeated FieldViolation violations = 1;
}

message RetryInfo {
  google.protobuf.Duration retry_delay = 1;
}
```

```csharp
using Google.Protobuf;
using Google.Protobuf.WellKnownTypes;

// Server: Send structured error details
public override async Task<Product> CreateProduct(
    CreateProductRequest request,
    ServerCallContext context)
{
    var validationErrors = ValidateRequest(request);
    
    if (validationErrors.Any())
    {
        var errorDetail = new ValidationError();
        errorDetail.Violations.AddRange(validationErrors.Select(e => 
            new ValidationError.Types.FieldViolation
            {
                Field = e.Field,
                Description = e.Message
            }));
        
        var trailers = new Metadata();
        
        // Serialize error details to trailer
        var errorBytes = errorDetail.ToByteArray();
        trailers.Add("error-details-bin", errorBytes);
        
        throw new RpcException(
            new Status(StatusCode.InvalidArgument, "Validation failed"),
            trailers);
    }
    
    // Process request...
}

// Client: Parse structured error details
try
{
    var product = await client.CreateProductAsync(request);
}
catch (RpcException ex)
{
    if (ex.StatusCode == StatusCode.InvalidArgument)
    {
        var errorBytes = ex.Trailers.GetValueBytes("error-details-bin");
        if (errorBytes != null)
        {
            var validationError = ValidationError.Parser.ParseFrom(errorBytes);
            
            Console.WriteLine("Validation errors:");
            foreach (var violation in validationError.Violations)
            {
                Console.WriteLine($"  {violation.Field}: {violation.Description}");
            }
        }
    }
}
```

### Retry Policies

```csharp
using Polly;
using Polly.Extensions.Http;

// Configure retry policy
var retryPolicy = HttpPolicyExtensions
    .HandleTransientHttpError()
    .Or<RpcException>(ex => 
        ex.StatusCode == StatusCode.Unavailable ||
        ex.StatusCode == StatusCode.ResourceExhausted ||
        ex.StatusCode == StatusCode.DeadlineExceeded)
    .WaitAndRetryAsync(
        retryCount: 3,
        sleepDurationProvider: retryAttempt => 
            TimeSpan.FromSeconds(Math.Pow(2, retryAttempt)),
        onRetry: (outcome, timespan, retryAttempt, context) =>
        {
            Console.WriteLine(
                $"Retry {retryAttempt} after {timespan.TotalSeconds}s delay");
        });

// Apply to gRPC client
var channel = GrpcChannel.ForAddress("https://localhost:5001");

services.AddGrpcClient<ProductService.ProductServiceClient>(o =>
{
    o.Address = new Uri("https://localhost:5001");
})
.AddPolicyHandler(retryPolicy);

// Manual retry with exponential backoff
async Task<Product> GetProductWithRetry(int productId, int maxRetries = 3)
{
    var retryCount = 0;
    var delay = TimeSpan.FromMilliseconds(100);
    
    while (true)
    {
        try
        {
            return await client.GetProductAsync(new GetProductRequest 
            { 
                Id = productId 
            });
        }
        catch (RpcException ex) when (
            (ex.StatusCode == StatusCode.Unavailable ||
             ex.StatusCode == StatusCode.DeadlineExceeded) &&
            retryCount < maxRetries)
        {
            retryCount++;
            
            // Check for retry info in trailers
            var retryDelay = ParseRetryDelay(ex.Trailers) ?? delay;
            
            Console.WriteLine(
                $"Retry {retryCount}/{maxRetries} after {retryDelay.TotalMilliseconds}ms");
            
            await Task.Delay(retryDelay);
            
            // Exponential backoff
            delay = TimeSpan.FromMilliseconds(delay.TotalMilliseconds * 2);
        }
    }
}

TimeSpan? ParseRetryDelay(Metadata trailers)
{
    var retryAfter = trailers.GetValue("retry-after");
    if (int.TryParse(retryAfter, out var seconds))
    {
        return TimeSpan.FromSeconds(seconds);
    }
    return null;
}
```

### Deadlines and Timeouts

```csharp
// Server: Check deadline
public override async Task<Product> GetProduct(
    GetProductRequest request,
    ServerCallContext context)
{
    // Check if we have enough time
    var timeRemaining = context.Deadline - DateTime.UtcNow;
    if (timeRemaining < TimeSpan.FromSeconds(1))
    {
        throw new RpcException(new Status(
            StatusCode.DeadlineExceeded,
            "Insufficient time to complete request"));
    }
    
    // Propagate deadline to downstream calls
    using var cts = CancellationTokenSource.CreateLinkedTokenSource(
        context.CancellationToken);
    cts.CancelAfter(timeRemaining - TimeSpan.FromMilliseconds(500));
    
    var product = await _repository.GetByIdAsync(
        request.Id, 
        cts.Token);
    
    return MapToProto(product);
}

// Client: Set deadline
var deadline = DateTime.UtcNow.AddSeconds(5);

try
{
    var response = await client.GetProductAsync(
        new GetProductRequest { Id = 1 },
        deadline: deadline);
}
catch (RpcException ex) when (ex.StatusCode == StatusCode.DeadlineExceeded)
{
    Console.WriteLine("Request timed out");
}

// Client: Set timeout
var callOptions = new CallOptions()
    .WithDeadline(DateTime.UtcNow.AddSeconds(30))
    .WithCancellationToken(cancellationToken);

var response = await client.GetProductAsync(request, callOptions);
```

### Global Error Handling

```csharp
// Exception interceptor
public class ExceptionInterceptor : Interceptor
{
    private readonly ILogger<ExceptionInterceptor> _logger;
    
    public ExceptionInterceptor(ILogger<ExceptionInterceptor> logger)
    {
        _logger = logger;
    }
    
    public override async Task<TResponse> UnaryServerHandler<TRequest, TResponse>(
        TRequest request,
        ServerCallContext context,
        UnaryServerMethod<TRequest, TResponse> continuation)
    {
        try
        {
            return await continuation(request, context);
        }
        catch (RpcException)
        {
            // Already an RpcException, rethrow
            throw;
        }
        catch (ValidationException ex)
        {
            _logger.LogWarning(ex, "Validation error");
            
            throw new RpcException(new Status(
                StatusCode.InvalidArgument,
                ex.Message));
        }
        catch (UnauthorizedAccessException ex)
        {
            _logger.LogWarning(ex, "Unauthorized access");
            
            throw new RpcException(new Status(
                StatusCode.PermissionDenied,
                "Access denied"));
        }
        catch (DbUpdateConcurrencyException ex)
        {
            _logger.LogWarning(ex, "Concurrency conflict");
            
            throw new RpcException(new Status(
                StatusCode.Aborted,
                "The resource was modified by another user"));
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Unhandled exception in {Method}", context.Method);
            
            throw new RpcException(new Status(
                StatusCode.Internal,
                "An internal error occurred"));
        }
    }
}

// Register interceptor
builder.Services.AddGrpc(options =>
{
    options.Interceptors.Add<ExceptionInterceptor>();
});
```

### Cancellation Handling

```csharp
// Server: Handle cancellation gracefully
public override async Task<ProcessResponse> ProcessData(
    ProcessRequest request,
    ServerCallContext context)
{
    try
    {
        // Register cancellation callback
        context.CancellationToken.Register(() =>
        {
            _logger.LogInformation("Request cancelled by client");
        });
        
        // Long-running operation with cancellation support
        var result = await _processor.ProcessAsync(
            request.Data,
            context.CancellationToken);
        
        return new ProcessResponse { Result = result };
    }
    catch (OperationCanceledException)
    {
        // Cleanup resources
        await CleanupAsync();
        
        throw new RpcException(new Status(
            StatusCode.Cancelled,
            "Operation was cancelled"));
    }
}

// Client: Cancel request
var cts = new CancellationTokenSource();

var call = client.ProcessDataAsync(
    new ProcessRequest { Data = data },
    cancellationToken: cts.Token);

// Cancel after 10 seconds
cts.CancelAfter(TimeSpan.FromSeconds(10));

try
{
    var response = await call;
}
catch (RpcException ex) when (ex.StatusCode == StatusCode.Cancelled)
{
    Console.WriteLine("Request was cancelled");
}
```

### Error Handling in Streaming

```csharp
// Server: Error handling in server streaming
public override async Task StreamProducts(
    StreamRequest request,
    IServerStreamWriter<Product> responseStream,
    ServerCallContext context)
{
    try
    {
        await foreach (var product in GetProductsAsync())
        {
            try
            {
                await responseStream.WriteAsync(product);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to write product {Id}", product.Id);
                
                // Can't send error in stream, throw to end stream
                throw new RpcException(new Status(
                    StatusCode.Internal,
                    $"Failed to stream product {product.Id}"));
            }
        }
    }
    catch (OperationCanceledException)
    {
        _logger.LogInformation("Stream cancelled");
        throw new RpcException(new Status(
            StatusCode.Cancelled,
            "Stream cancelled by client"));
    }
}

// Client: Handle streaming errors
try
{
    var call = client.StreamProducts(new StreamRequest());
    
    await foreach (var product in call.ResponseStream.ReadAllAsync())
    {
        ProcessProduct(product);
    }
}
catch (RpcException ex)
{
    Console.WriteLine($"Stream error: {ex.Status.StatusCode} - {ex.Status.Detail}");
    
    if (ex.StatusCode == StatusCode.Internal)
    {
        // Retry entire stream
        await RetryStreamAsync();
    }
}
```
