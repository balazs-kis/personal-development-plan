## Introduction

## Key Concepts Summary

### Basic Service Definition

```protobuf
syntax = "proto3";

package products.v1;

option csharp_namespace = "Products.V1";

service ProductService {
  // Unary RPC
  rpc GetProduct(GetProductRequest) returns (GetProductResponse);
  
  // Server streaming
  rpc ListProducts(ListProductsRequest) returns (stream Product);
  
  // Client streaming
  rpc CreateProducts(stream CreateProductRequest) returns (CreateProductsResponse);
  
  // Bidirectional streaming
  rpc SyncProducts(stream ProductUpdate) returns (stream ProductUpdate);
}

message GetProductRequest {
  int64 id = 1;
}

message GetProductResponse {
  Product product = 1;
}

message Product {
  int64 id = 1;
  string name = 2;
  double price = 3;
}
```

### Request/Response Patterns

```protobuf
// Pattern 1: Simple request/response
rpc GetProduct(GetProductRequest) returns (GetProductResponse);

message GetProductRequest {
  int64 id = 1;
}

message GetProductResponse {
  Product product = 1;
}

// Pattern 2: List with pagination
rpc ListProducts(ListProductsRequest) returns (ListProductsResponse);

message ListProductsRequest {
  int32 page_size = 1;
  string page_token = 2;
  string filter = 3;
}

message ListProductsResponse {
  repeated Product products = 1;
  string next_page_token = 2;
  int32 total_count = 3;
}

// Pattern 3: Empty responses
import "google/protobuf/empty.proto";

rpc DeleteProduct(DeleteProductRequest) returns (google.protobuf.Empty);

message DeleteProductRequest {
  int64 id = 1;
}
```

### ASP.NET Core Implementation

```csharp
// Install packages:
// dotnet add package Grpc.AspNetCore
// dotnet add package Grpc.Tools

// Program.cs
var builder = WebApplication.CreateBuilder(args);
builder.Services.AddGrpc();

var app = builder.Build();
app.MapGrpcService<ProductService>();
app.Run();

// ProductService.cs
using Grpc.Core;
using Products.V1;

public class ProductService : Products.V1.ProductService.ProductServiceBase
{
    private readonly IProductRepository _repository;
    
    public ProductService(IProductRepository repository)
    {
        _repository = repository;
    }
    
    public override async Task<GetProductResponse> GetProduct(
        GetProductRequest request,
        ServerCallContext context)
    {
        var product = await _repository.GetByIdAsync(request.Id);
        
        if (product == null)
        {
            throw new RpcException(new Status(
                StatusCode.NotFound,
                $"Product with ID {request.Id} not found"));
        }
        
        return new GetProductResponse
        {
            Product = new Product
            {
                Id = product.Id,
                Name = product.Name,
                Price = product.Price
            }
        };
    }
}
```

### Service Design Patterns

**CRUD Operations:**

```protobuf
syntax = "proto3";

package products.v1;

import "google/protobuf/empty.proto";
import "google/protobuf/timestamp.proto";
import "google/protobuf/field_mask.proto";

service ProductService {
  // Create
  rpc CreateProduct(CreateProductRequest) returns (Product);
  rpc CreateProducts(CreateProductsRequest) returns (CreateProductsResponse);
  
  // Read
  rpc GetProduct(GetProductRequest) returns (Product);
  rpc ListProducts(ListProductsRequest) returns (ListProductsResponse);
  rpc SearchProducts(SearchProductsRequest) returns (SearchProductsResponse);
  
  // Update
  rpc UpdateProduct(UpdateProductRequest) returns (Product);
  rpc UpdateProducts(UpdateProductsRequest) returns (UpdateProductsResponse);
  
  // Delete
  rpc DeleteProduct(DeleteProductRequest) returns (google.protobuf.Empty);
  rpc DeleteProducts(DeleteProductsRequest) returns (DeleteProductsResponse);
}

// Create requests
message CreateProductRequest {
  string name = 1;
  double price = 2;
  string description = 3;
  repeated string tags = 4;
}

message CreateProductsRequest {
  repeated CreateProductRequest products = 1;
}

message CreateProductsResponse {
  repeated Product products = 1;
  int32 created_count = 2;
  repeated string errors = 3;
}

// Read requests
message GetProductRequest {
  int64 id = 1;
}

message ListProductsRequest {
  int32 page_size = 1;
  string page_token = 2;
  string filter = 3;
  string order_by = 4;
}

message ListProductsResponse {
  repeated Product products = 1;
  string next_page_token = 2;
  int32 total_count = 3;
}

message SearchProductsRequest {
  string query = 1;
  int32 page_size = 2;
  string page_token = 3;
  repeated string categories = 4;
  PriceRange price_range = 5;
}

message SearchProductsResponse {
  repeated Product products = 1;
  string next_page_token = 2;
  int32 total_count = 3;
  SearchMetadata metadata = 4;
}

message PriceRange {
  double min = 1;
  double max = 2;
}

message SearchMetadata {
  int32 took_ms = 1;
  map<string, int32> facets = 2;
}

// Update requests
message UpdateProductRequest {
  int64 id = 1;
  string name = 2;
  double price = 3;
  string description = 4;
  google.protobuf.FieldMask update_mask = 5;  // Partial updates
}

message UpdateProductsRequest {
  repeated UpdateProductRequest products = 1;
}

message UpdateProductsResponse {
  repeated Product products = 1;
  int32 updated_count = 2;
  repeated string errors = 3;
}

// Delete requests
message DeleteProductRequest {
  int64 id = 1;
}

message DeleteProductsRequest {
  repeated int64 ids = 1;
}

message DeleteProductsResponse {
  int32 deleted_count = 2;
  repeated string errors = 3;
}

message Product {
  int64 id = 1;
  string name = 2;
  double price = 3;
  string description = 4;
  repeated string tags = 5;
  google.protobuf.Timestamp created_at = 6;
  google.protobuf.Timestamp updated_at = 7;
}
```

**Implementing CRUD in ASP.NET Core:**

```csharp
using Grpc.Core;
using Microsoft.EntityFrameworkCore;
using Products.V1;
using Google.Protobuf.WellKnownTypes;

public class ProductService : Products.V1.ProductService.ProductServiceBase
{
    private readonly ApplicationDbContext _context;
    private readonly ILogger<ProductService> _logger;
    
    public ProductService(
        ApplicationDbContext context,
        ILogger<ProductService> logger)
    {
        _context = context;
        _logger = logger;
    }
    
    // Create
    public override async Task<Product> CreateProduct(
        CreateProductRequest request,
        ServerCallContext context)
    {
        var product = new ProductEntity
        {
            Name = request.Name,
            Price = request.Price,
            Description = request.Description,
            Tags = request.Tags.ToList(),
            CreatedAt = DateTime.UtcNow,
            UpdatedAt = DateTime.UtcNow
        };
        
        _context.Products.Add(product);
        await _context.SaveChangesAsync(context.CancellationToken);
        
        _logger.LogInformation(
            "Created product {ProductId}: {ProductName}",
            product.Id,
            product.Name);
        
        return MapToProto(product);
    }
    
    // Read
    public override async Task<Product> GetProduct(
        GetProductRequest request,
        ServerCallContext context)
    {
        var product = await _context.Products
            .FirstOrDefaultAsync(
                p => p.Id == request.Id,
                context.CancellationToken);
        
        if (product == null)
        {
            throw new RpcException(new Status(
                StatusCode.NotFound,
                $"Product with ID {request.Id} not found"));
        }
        
        return MapToProto(product);
    }
    
    public override async Task<ListProductsResponse> ListProducts(
        ListProductsRequest request,
        ServerCallContext context)
    {
        var pageSize = request.PageSize > 0 ? request.PageSize : 50;
        pageSize = Math.Min(pageSize, 100); // Max page size
        
        var query = _context.Products.AsQueryable();
        
        // Apply filter
        if (!string.IsNullOrEmpty(request.Filter))
        {
            query = ApplyFilter(query, request.Filter);
        }
        
        // Apply ordering
        if (!string.IsNullOrEmpty(request.OrderBy))
        {
            query = ApplyOrdering(query, request.OrderBy);
        }
        
        // Get total count
        var totalCount = await query.CountAsync(context.CancellationToken);
        
        // Apply pagination
        var skip = ParsePageToken(request.PageToken);
        var products = await query
            .Skip(skip)
            .Take(pageSize)
            .ToListAsync(context.CancellationToken);
        
        var response = new ListProductsResponse
        {
            TotalCount = totalCount
        };
        
        response.Products.AddRange(products.Select(MapToProto));
        
        // Generate next page token
        if (skip + pageSize < totalCount)
        {
            response.NextPageToken = GeneratePageToken(skip + pageSize);
        }
        
        return response;
    }
    
    // Update
    public override async Task<Product> UpdateProduct(
        UpdateProductRequest request,
        ServerCallContext context)
    {
        var product = await _context.Products
            .FirstOrDefaultAsync(
                p => p.Id == request.Id,
                context.CancellationToken);
        
        if (product == null)
        {
            throw new RpcException(new Status(
                StatusCode.NotFound,
                $"Product with ID {request.Id} not found"));
        }
        
        // Apply partial update using field mask
        if (request.UpdateMask != null && request.UpdateMask.Paths.Any())
        {
            foreach (var path in request.UpdateMask.Paths)
            {
                switch (path)
                {
                    case "name":
                        product.Name = request.Name;
                        break;
                    case "price":
                        product.Price = request.Price;
                        break;
                    case "description":
                        product.Description = request.Description;
                        break;
                }
            }
        }
        else
        {
            // Full update
            product.Name = request.Name;
            product.Price = request.Price;
            product.Description = request.Description;
        }
        
        product.UpdatedAt = DateTime.UtcNow;
        
        await _context.SaveChangesAsync(context.CancellationToken);
        
        _logger.LogInformation(
            "Updated product {ProductId}",
            product.Id);
        
        return MapToProto(product);
    }
    
    // Delete
    public override async Task<Empty> DeleteProduct(
        DeleteProductRequest request,
        ServerCallContext context)
    {
        var product = await _context.Products
            .FirstOrDefaultAsync(
                p => p.Id == request.Id,
                context.CancellationToken);
        
        if (product == null)
        {
            throw new RpcException(new Status(
                StatusCode.NotFound,
                $"Product with ID {request.Id} not found"));
        }
        
        _context.Products.Remove(product);
        await _context.SaveChangesAsync(context.CancellationToken);
        
        _logger.LogInformation(
            "Deleted product {ProductId}",
            request.Id);
        
        return new Empty();
    }
    
    // Helper methods
    private static Product MapToProto(ProductEntity entity)
    {
        return new Product
        {
            Id = entity.Id,
            Name = entity.Name,
            Price = entity.Price,
            Description = entity.Description,
            Tags = { entity.Tags },
            CreatedAt = Timestamp.FromDateTime(entity.CreatedAt),
            UpdatedAt = Timestamp.FromDateTime(entity.UpdatedAt)
        };
    }
    
    private static IQueryable<ProductEntity> ApplyFilter(
        IQueryable<ProductEntity> query,
        string filter)
    {
        // Simple filter implementation
        // Production: use expression parser or OData-style filters
        if (filter.StartsWith("name="))
        {
            var name = filter.Substring(5);
            query = query.Where(p => p.Name.Contains(name));
        }
        
        return query;
    }
    
    private static IQueryable<ProductEntity> ApplyOrdering(
        IQueryable<ProductEntity> query,
        string orderBy)
    {
        return orderBy.ToLower() switch
        {
            "name" => query.OrderBy(p => p.Name),
            "name desc" => query.OrderByDescending(p => p.Name),
            "price" => query.OrderBy(p => p.Price),
            "price desc" => query.OrderByDescending(p => p.Price),
            "created" => query.OrderBy(p => p.CreatedAt),
            "created desc" => query.OrderByDescending(p => p.CreatedAt),
            _ => query.OrderBy(p => p.Id)
        };
    }
    
    private static int ParsePageToken(string token)
    {
        if (string.IsNullOrEmpty(token))
            return 0;
        
        if (int.TryParse(token, out var offset))
            return offset;
        
        return 0;
    }
    
    private static string GeneratePageToken(int offset)
    {
        return offset.ToString();
    }
}
```

### Client Implementation

**Basic Client Usage:**

```csharp
using Grpc.Net.Client;
using Products.V1;

// Create channel
var channel = GrpcChannel.ForAddress("https://localhost:5001");
var client = new ProductService.ProductServiceClient(channel);

// Create product
var createRequest = new CreateProductRequest
{
    Name = "Laptop",
    Price = 999.99,
    Description = "High-performance laptop"
};
createRequest.Tags.Add("electronics");
createRequest.Tags.Add("computers");

var product = await client.CreateProductAsync(createRequest);
Console.WriteLine($"Created product: {product.Id}");

// Get product
var getRequest = new GetProductRequest { Id = product.Id };
var retrieved = await client.GetProductAsync(getRequest);
Console.WriteLine($"Product: {retrieved.Name} - ${retrieved.Price}");

// List products
var listRequest = new ListProductsRequest
{
    PageSize = 20,
    OrderBy = "name"
};

var listResponse = await client.ListProductsAsync(listRequest);
foreach (var p in listResponse.Products)
{
    Console.WriteLine($"{p.Id}: {p.Name}");
}

// Update product
var updateRequest = new UpdateProductRequest
{
    Id = product.Id,
    Name = "Gaming Laptop",
    Price = 1299.99,
    Description = "High-performance gaming laptop"
};
// Partial update using field mask
updateRequest.UpdateMask = new Google.Protobuf.WellKnownTypes.FieldMask();
updateRequest.UpdateMask.Paths.Add("name");
updateRequest.UpdateMask.Paths.Add("price");

var updated = await client.UpdateProductAsync(updateRequest);
Console.WriteLine($"Updated: {updated.Name} - ${updated.Price}");

// Delete product
var deleteRequest = new DeleteProductRequest { Id = product.Id };
await client.DeleteProductAsync(deleteRequest);
Console.WriteLine("Product deleted");
```

### Validation and Error Handling

```csharp
using FluentValidation;
using Grpc.Core;

// Define validator
public class CreateProductRequestValidator : AbstractValidator<CreateProductRequest>
{
    public CreateProductRequestValidator()
    {
        RuleFor(x => x.Name)
            .NotEmpty()
            .MinimumLength(3)
            .MaximumLength(200);
        
        RuleFor(x => x.Price)
            .GreaterThan(0)
            .LessThanOrEqualTo(1000000);
        
        RuleFor(x => x.Description)
            .MaximumLength(2000);
    }
}

// Use validator in service
public class ProductService : Products.V1.ProductService.ProductServiceBase
{
    private readonly IValidator<CreateProductRequest> _validator;
    
    public ProductService(IValidator<CreateProductRequest> validator)
    {
        _validator = validator;
    }
    
    public override async Task<Product> CreateProduct(
        CreateProductRequest request,
        ServerCallContext context)
    {
        // Validate request
        var validationResult = await _validator.ValidateAsync(request);
        if (!validationResult.IsValid)
        {
            var errors = string.Join("; ", 
                validationResult.Errors.Select(e => e.ErrorMessage));
            
            throw new RpcException(new Status(
                StatusCode.InvalidArgument,
                errors));
        }
        
        // Process request...
    }
}

// Register validator
builder.Services.AddValidatorsFromAssemblyContaining<CreateProductRequestValidator>();
```

### Service Composition

```protobuf

package ecommerce.v1;
```

### Service Versioning

```protobuf
// v1/products.proto
syntax = "proto3";
package products.v1;
option csharp_namespace = "Products.V1";

service ProductService {
  rpc GetProduct(GetProductRequest) returns (Product);
}

message Product {
  int64 id = 1;
  string name = 2;
  double price = 3;
}

// v2/products.proto
syntax = "proto3";
package products.v2;
option csharp_namespace = "Products.V2";

service ProductService {
  rpc GetProduct(GetProductRequest) returns (Product);
  rpc GetProductDetails(GetProductRequest) returns (ProductDetails);
}

message Product {
  int64 id = 1;
  string name = 2;
  Price price = 3;  // Changed to complex type
  string description = 4;  // New field
}

message Price {
  double amount = 1;
  string currency = 2;
}
```

```csharp
// Host both versions
app.MapGrpcService<Products.V1.ProductService>();
app.MapGrpcService<Products.V2.ProductService>();

// Client selects version by namespace
var v1Client = new Products.V1.ProductService.ProductServiceClient(channel);
var v2Client = new Products.V2.ProductService.ProductServiceClient(channel);
```
