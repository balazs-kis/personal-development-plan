### Secure API Key Generation

```csharp
public class ApiKeyGenerator
{
    public static string GenerateApiKey()
    {
        var key = new byte[32];
        using (var generator = RandomNumberGenerator.Create())
        {
            generator.GetBytes(key);
        }
        return Convert.ToBase64String(key);
    }
    
    public static string GenerateApiKeyWithPrefix(string prefix = "sk")
    {
        var key = GenerateApiKey();
        return $"{prefix}_{key}";
    }
}
```

### Storing API Keys in Azure Key Vault

```csharp
// Program.cs
using Azure.Identity;
using Azure.Security.KeyVault.Secrets;

var builder = WebApplication.CreateBuilder(args);

// Add Azure Key Vault
var keyVaultEndpoint = new Uri(builder.Configuration["KeyVault:Endpoint"]!);
builder.Configuration.AddAzureKeyVault(keyVaultEndpoint, new DefaultAzureCredential());

// API Key service
builder.Services.AddSingleton<IApiKeyService, ApiKeyService>();

public interface IApiKeyService
{
    Task<bool> ValidateApiKeyAsync(string apiKey);
    Task<string> CreateApiKeyAsync(string clientId);
    Task RevokeApiKeyAsync(string apiKey);
}

public class ApiKeyService : IApiKeyService
{
    private readonly SecretClient _secretClient;
    private readonly ILogger<ApiKeyService> _logger;
    
    public ApiKeyService(IConfiguration configuration, ILogger<ApiKeyService> logger)
    {
        var keyVaultUrl = configuration["KeyVault:Endpoint"];
        _secretClient = new SecretClient(new Uri(keyVaultUrl!), new DefaultAzureCredential());
        _logger = logger;
    }
    
    public async Task<string> CreateApiKeyAsync(string clientId)
    {
        var apiKey = ApiKeyGenerator.GenerateApiKeyWithPrefix("sk");
        var secretName = $"ApiKey-{clientId}";
        
        await _secretClient.SetSecretAsync(secretName, apiKey);
        _logger.LogInformation("Created API key for client {ClientId}", clientId);
        
        return apiKey;
    }
    
    public async Task<bool> ValidateApiKeyAsync(string apiKey)
    {
        try
        {
            var secrets = _secretClient.GetPropertiesOfSecretsAsync();
            await foreach (var secret in secrets)
            {
                var secretValue = await _secretClient.GetSecretAsync(secret.Name);
                if (secretValue.Value.Value == apiKey && secret.Enabled == true)
                {
                    return true;
                }
            }
            return false;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error validating API key");
            return false;
        }
    }
    
    public async Task RevokeApiKeyAsync(string apiKey)
    {
        var secrets = _secretClient.GetPropertiesOfSecretsAsync();
        await foreach (var secret in secrets)
        {
            var secretValue = await _secretClient.GetSecretAsync(secret.Name);
            if (secretValue.Value.Value == apiKey)
            {
                await _secretClient.UpdateSecretPropertiesAsync(
                    new SecretProperties(secret.Name) { Enabled = false });
                _logger.LogInformation("Revoked API key: {SecretName}", secret.Name);
                return;
            }
        }
    }
}
```

### Database-Backed API Keys

```csharp
public class ApiKey
{
    public int Id { get; set; }
    public string Key { get; set; } = string.Empty;
    public string ClientId { get; set; } = string.Empty;
    public DateTime CreatedAt { get; set; }
    public DateTime? ExpiresAt { get; set; }
    public DateTime? RevokedAt { get; set; }
    public bool IsActive => RevokedAt == null && (ExpiresAt == null || ExpiresAt > DateTime.UtcNow);
    public string? Description { get; set; }
    public int RequestCount { get; set; }
    public DateTime? LastUsedAt { get; set; }
}

public class ApiKeyRepository : IApiKeyRepository
{
    private readonly ApplicationDbContext _context;
    
    public async Task<ApiKey?> GetByKeyAsync(string key)
    {
        return await _context.ApiKeys
            .FirstOrDefaultAsync(k => k.Key == key && k.IsActive);
    }
    
    public async Task<ApiKey> CreateAsync(string clientId, string description, int daysValid = 365)
    {
        var apiKey = new ApiKey
        {
            Key = ApiKeyGenerator.GenerateApiKeyWithPrefix("sk"),
            ClientId = clientId,
            Description = description,
            CreatedAt = DateTime.UtcNow,
            ExpiresAt = DateTime.UtcNow.AddDays(daysValid)
        };
        
        _context.ApiKeys.Add(apiKey);
        await _context.SaveChangesAsync();
        
        return apiKey;
    }
    
    public async Task RevokeAsync(string key)
    {
        var apiKey = await _context.ApiKeys.FirstOrDefaultAsync(k => k.Key == key);
        if (apiKey != null)
        {
            apiKey.RevokedAt = DateTime.UtcNow;
            await _context.SaveChangesAsync();
        }
    }
    
    public async Task TrackUsageAsync(string key)
    {
        var apiKey = await _context.ApiKeys.FirstOrDefaultAsync(k => k.Key == key);
        if (apiKey != null)
        {
            apiKey.RequestCount++;
            apiKey.LastUsedAt = DateTime.UtcNow;
            await _context.SaveChangesAsync();
        }
    }
}
```

### API Key Hashing

```csharp
using System.Security.Cryptography;

public class ApiKeyHasher
{
    public static (string apiKey, string hashedKey) GenerateAndHashKey()
    {
        // Generate plain key
        var key = new byte[32];
        using (var generator = RandomNumberGenerator.Create())
        {
            generator.GetBytes(key);
        }
        var apiKey = $"sk_{Convert.ToBase64String(key)}";
        
        // Hash the key for storage
        var hashedKey = HashKey(apiKey);
        
        return (apiKey, hashedKey);
    }
    
    public static string HashKey(string apiKey)
    {
        using var sha256 = SHA256.Create();
        var keyBytes = Encoding.UTF8.GetBytes(apiKey);
        var hashBytes = sha256.ComputeHash(keyBytes);
        return Convert.ToBase64String(hashBytes);
    }
    
    public static bool VerifyKey(string providedKey, string storedHash)
    {
        var providedHash = HashKey(providedKey);
        return providedHash == storedHash;
    }
}

// Updated repository to use hashing
public class HashedApiKeyRepository : IApiKeyRepository
{
    private readonly ApplicationDbContext _context;
    
    public async Task<ApiKey> CreateAsync(string clientId, string description)
    {
        var (plainKey, hashedKey) = ApiKeyHasher.GenerateAndHashKey();
        
        var apiKey = new ApiKey
        {
            KeyHash = hashedKey, // Store hash, not plain key
            ClientId = clientId,
            Description = description,
            CreatedAt = DateTime.UtcNow,
            ExpiresAt = DateTime.UtcNow.AddYears(1)
        };
        
        _context.ApiKeys.Add(apiKey);
        await _context.SaveChangesAsync();
        
        // Return plain key only once
        apiKey.Key = plainKey; // Temporary, not persisted
        return apiKey;
    }
    
    public async Task<ApiKey?> ValidateKeyAsync(string providedKey)
    {
        var hashedKey = ApiKeyHasher.HashKey(providedKey);
        
        return await _context.ApiKeys
            .FirstOrDefaultAsync(k => k.KeyHash == hashedKey && k.IsActive);
    }
}
```

### API Key Scopes and Permissions

```csharp
public class ApiKey
{
    public int Id { get; set; }
    public string KeyHash { get; set; } = string.Empty;
    public string ClientId { get; set; } = string.Empty;
    public string? Description { get; set; }
    
    // Permissions
    public List<string> Scopes { get; set; } = new();
    public string Tier { get; set; } = "free"; // free, standard, premium
    
    // Rate limiting
    public int RateLimitPerMinute { get; set; } = 60;
    public int RateLimitPerHour { get; set; } = 1000;
    
    // Lifecycle
    public DateTime CreatedAt { get; set; }
    public DateTime? ExpiresAt { get; set; }
    public DateTime? RevokedAt { get; set; }
    public DateTime? LastUsedAt { get; set; }
    public int RequestCount { get; set; }
    
    // IP restrictions
    public List<string> AllowedIpAddresses { get; set; } = new();
    
    public bool IsActive => 
        RevokedAt == null && 
        (ExpiresAt == null || ExpiresAt > DateTime.UtcNow);
}

// Authorization based on scopes
public class ApiKeyScopeRequirement : IAuthorizationRequirement
{
    public string RequiredScope { get; }
    
    public ApiKeyScopeRequirement(string requiredScope)
    {
        RequiredScope = requiredScope;
    }
}

public class ApiKeyScopeHandler : AuthorizationHandler<ApiKeyScopeRequirement>
{
    private readonly IApiKeyRepository _repository;
    
    protected override async Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        ApiKeyScopeRequirement requirement)
    {
        var apiKeyId = context.User.FindFirst("ApiKeyId")?.Value;
        
        if (string.IsNullOrEmpty(apiKeyId))
        {
            return;
        }
        
        var apiKey = await _repository.GetByIdAsync(int.Parse(apiKeyId));
        
        if (apiKey != null && apiKey.Scopes.Contains(requirement.RequiredScope))
        {
            context.Succeed(requirement);
        }
    }
}

// Register
builder.Services.AddAuthorization(options =>
{
    options.AddPolicy("Products.Read", policy =>
        policy.Requirements.Add(new ApiKeyScopeRequirement("products:read")));
    
    options.AddPolicy("Products.Write", policy =>
        policy.Requirements.Add(new ApiKeyScopeRequirement("products:write")));
});
```

### IP Whitelisting for API Keys

```csharp
public class ApiKeyIpRestrictionMiddleware
{
    private readonly RequestDelegate _next;
    
    public ApiKeyIpRestrictionMiddleware(RequestDelegate next)
    {
        _next = next;
    }
    
    public async Task InvokeAsync(
        HttpContext context,
        IApiKeyRepository apiKeyRepository)
    {
        // Only check if authenticated with API key
        if (context.User.Identity?.AuthenticationType != "ApiKey")
        {
            await _next(context);
            return;
        }
        
        var apiKeyId = context.User.FindFirst("ApiKeyId")?.Value;
        if (string.IsNullOrEmpty(apiKeyId))
        {
            await _next(context);
            return;
        }
        
        var apiKey = await apiKeyRepository.GetByIdAsync(int.Parse(apiKeyId));
        
        if (apiKey == null || !apiKey.AllowedIpAddresses.Any())
        {
            await _next(context);
            return;
        }
        
        var clientIp = context.Connection.RemoteIpAddress?.ToString();
        
        if (!apiKey.AllowedIpAddresses.Contains(clientIp))
        {
            context.Response.StatusCode = StatusCodes.Status403Forbidden;
            await context.Response.WriteAsJsonAsync(new
            {
                Error = "IP address not allowed",
                ClientIp = clientIp
            });
            return;
        }
        
        await _next(context);
    }
}
```

### Automated API Key Rotation

```csharp
public class ApiKeyRotationService : BackgroundService
{
    private readonly IServiceProvider _serviceProvider;
    private readonly ILogger<ApiKeyRotationService> _logger;
    private readonly TimeSpan _checkInterval = TimeSpan.FromHours(6);
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                await RotateExpiredKeysAsync(stoppingToken);
                await NotifyUpcomingExpirationsAsync(stoppingToken);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error during key rotation check");
            }
            
            await Task.Delay(_checkInterval, stoppingToken);
        }
    }
    
    private async Task RotateExpiredKeysAsync(CancellationToken cancellationToken)
    {
        using var scope = _serviceProvider.CreateScope();
        var repository = scope.ServiceProvider.GetRequiredService<IApiKeyRepository>();
        var notificationService = scope.ServiceProvider
            .GetRequiredService<INotificationService>();
        
        // Find keys expiring in next 30 days
        var expiringKeys = await repository.GetKeysExpiringInDaysAsync(30);
        
        foreach (var key in expiringKeys)
        {
            if (key.AutoRotate)
            {
                var newKey = await repository.CreateAsync(
                    key.ClientId,
                    $"Auto-rotated from {key.Id}");
                
                // Notify client of new key
                await notificationService.SendKeyRotationNotificationAsync(
                    key.ClientId,
                    newKey.Key!,
                    key.ExpiresAt!.Value);
                
                _logger.LogInformation(
                    "Rotated API key for client {ClientId}",
                    key.ClientId);
            }
        }
    }
    
    private async Task NotifyUpcomingExpirationsAsync(CancellationToken cancellationToken)
    {
        using var scope = _serviceProvider.CreateScope();
        var repository = scope.ServiceProvider.GetRequiredService<IApiKeyRepository>();
        var notificationService = scope.ServiceProvider
            .GetRequiredService<INotificationService>();
        
        // Notify 7 days before expiration
        var keys = await repository.GetKeysExpiringInDaysAsync(7);
        
        foreach (var key in keys.Where(k => !k.AutoRotate))
        {
            await notificationService.SendExpirationWarningAsync(
                key.ClientId,
                key.ExpiresAt!.Value);
        }
    }
}

// Register
builder.Services.AddHostedService<ApiKeyRotationService>();
```

### API Key Management Controller

```csharp
[ApiController]
[Route("api/[controller]")]
[Authorize]
public class ApiKeysController : ControllerBase
{
    private readonly IApiKeyRepository _repository;
    private readonly ILogger<ApiKeysController> _logger;
    
    [HttpGet]
    public async Task<ActionResult<List<ApiKeyDto>>> GetMyKeys()
    {
        var userId = User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
        var keys = await _repository.GetByClientIdAsync(userId);
        
        return Ok(keys.Select(k => new ApiKeyDto
        {
            Id = k.Id,
            Description = k.Description,
            CreatedAt = k.CreatedAt,
            ExpiresAt = k.ExpiresAt,
            LastUsedAt = k.LastUsedAt,
            RequestCount = k.RequestCount,
            IsActive = k.IsActive,
            Scopes = k.Scopes,
            Tier = k.Tier
            // Never return the actual key or hash
        }));
    }
    
    [HttpPost]
    public async Task<ActionResult<CreateApiKeyResponse>> CreateKey(
        [FromBody] CreateApiKeyRequest request)
    {
        var userId = User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
        
        // Validate user can create more keys
        var existingKeys = await _repository.GetByClientIdAsync(userId);
        if (existingKeys.Count >= 10)
        {
            return BadRequest("Maximum number of API keys reached");
        }
        
        var (plainKey, hashedKey) = ApiKeyHasher.GenerateAndHashKey();
        
        var apiKey = new ApiKey
        {
            KeyHash = hashedKey,
            ClientId = userId,
            Description = request.Description,
            Scopes = request.Scopes,
            Tier = request.Tier ?? "free",
            CreatedAt = DateTime.UtcNow,
            ExpiresAt = request.ExpiresInDays.HasValue
                ? DateTime.UtcNow.AddDays(request.ExpiresInDays.Value)
                : null,
            AllowedIpAddresses = request.AllowedIpAddresses ?? new List<string>()
        };
        
        await _repository.CreateAsync(apiKey);
        
        _logger.LogInformation(
            "Created API key {KeyId} for user {UserId}",
            apiKey.Id,
            userId);
        
        // Return plain key only once
        return Ok(new CreateApiKeyResponse
        {
            ApiKey = plainKey, // Show only once!
            KeyId = apiKey.Id,
            Message = "Save this key securely. It will not be shown again."
        });
    }
    
    [HttpDelete("{keyId}")]
    public async Task<ActionResult> RevokeKey(int keyId)
    {
        var userId = User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
        var apiKey = await _repository.GetByIdAsync(keyId);
        
        if (apiKey == null)
        {
            return NotFound();
        }
        
        if (apiKey.ClientId != userId)
        {
            return Forbid();
        }
        
        apiKey.RevokedAt = DateTime.UtcNow;
        await _repository.UpdateAsync(apiKey);
        
        _logger.LogInformation(
            "Revoked API key {KeyId} for user {UserId}",
            keyId,
            userId);
        
        return NoContent();
    }
    
    [HttpPut("{keyId}/scopes")]
    public async Task<ActionResult> UpdateScopes(
        int keyId,
        [FromBody] UpdateScopesRequest request)
    {
        var userId = User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
        var apiKey = await _repository.GetByIdAsync(keyId);
        
        if (apiKey == null)
        {
            return NotFound();
        }
        
        if (apiKey.ClientId != userId)
        {
            return Forbid();
        }
        
        apiKey.Scopes = request.Scopes;
        await _repository.UpdateAsync(apiKey);
        
        return NoContent();
    }
    
    [HttpGet("{keyId}/usage")]
    public async Task<ActionResult<ApiKeyUsageDto>> GetKeyUsage(int keyId)
    {
        var userId = User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
        var apiKey = await _repository.GetByIdAsync(keyId);
        
        if (apiKey == null)
        {
            return NotFound();
        }
        
        if (apiKey.ClientId != userId)
        {
            return Forbid();
        }
        
        var usage = await _repository.GetUsageStatsAsync(keyId);
        
        return Ok(new ApiKeyUsageDto
        {
            TotalRequests = apiKey.RequestCount,
            LastUsedAt = apiKey.LastUsedAt,
            RequestsByDay = usage.RequestsByDay,
            RequestsByEndpoint = usage.RequestsByEndpoint,
            ErrorRate = usage.ErrorRate
        });
    }
}
```

### Monitoring and Alerts

```csharp
public class ApiKeyMonitoringService
{
    private readonly IApiKeyRepository _repository;
    private readonly ILogger<ApiKeyMonitoringService> _logger;
    private readonly IMetricsCollector _metrics;
    
    public async Task TrackRequestAsync(string apiKey, HttpContext context)
    {
        var key = await _repository.GetByKeyAsync(apiKey);
        if (key == null) return;
        
        // Track usage
        await _repository.TrackUsageAsync(apiKey);
        
        // Collect metrics
        _metrics.IncrementCounter("api_key_requests", new Dictionary<string, string>
        {
            { "client_id", key.ClientId },
            { "tier", key.Tier },
            { "endpoint", context.Request.Path }
        });
        
        // Check for suspicious activity
        await DetectAnomaliesAsync(key, context);
    }
    
    private async Task DetectAnomaliesAsync(ApiKey key, HttpContext context)
    {
        // Check for unusual request rate
        var recentRequests = await _repository.GetRecentRequestCountAsync(
            key.Id,
            TimeSpan.FromMinutes(5));
        
        if (recentRequests > key.RateLimitPerMinute * 5)
        {
            _logger.LogWarning(
                "Unusual request rate detected for API key {KeyId}: {Count} requests in 5 minutes",
                key.Id,
                recentRequests);
            
            // Could trigger auto-suspension here
        }
        
        // Check for new IP address
        var clientIp = context.Connection.RemoteIpAddress?.ToString();
        var isNewIp = await _repository.IsNewIpAddressAsync(key.Id, clientIp);
        
        if (isNewIp)
        {
            _logger.LogInformation(
                "New IP address {IpAddress} detected for API key {KeyId}",
                clientIp,
                key.Id);
        }
    }
}
```
