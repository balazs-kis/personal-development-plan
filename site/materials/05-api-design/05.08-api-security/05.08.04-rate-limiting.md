## Learning Objectives

By the end of this lesson, you will be able to:
- Implement rate limiting in ASP.NET Core

- Configure different limits for different endpoints

- Monitor and adjust rate limits

## Introduction

Rate limiting protects your API from abuse, ensures fair usage among clients, and maintains service quality. ASP.NET Core 7+ includes built-in rate limiting middleware that's easy to configure and customize.

## Key Concepts Summary

### Basic Rate Limiting

```csharp
using System.Threading.RateLimiting;

var builder = WebApplication.CreateBuilder(args);

// Add rate limiting
builder.Services.AddRateLimiter(options =>
{
    // Fixed window limiter
    options.AddFixedWindowLimiter("fixed", options =>
    {
        options.PermitLimit = 100;
        options.Window = TimeSpan.FromMinutes(1);
        options.QueueProcessingOrder = QueueProcessingOrder.OldestFirst;
        options.QueueLimit = 5;
    });
    
    // Sliding window limiter
    options.AddSlidingWindowLimiter("sliding", options =>
    {
        options.PermitLimit = 100;
        options.Window = TimeSpan.FromMinutes(1);
        options.SegmentsPerWindow = 6;
        options.QueueProcessingOrder = QueueProcessingOrder.OldestFirst;
        options.QueueLimit = 5;
    });
    
    // Token bucket limiter
    options.AddTokenBucketLimiter("token", options =>
    {
        options.TokenLimit = 100;
        options.ReplenishmentPeriod = TimeSpan.FromMinutes(1);
        options.TokensPerPeriod = 20;
        options.QueueProcessingOrder = QueueProcessingOrder.OldestFirst;
        options.QueueLimit = 5;
    });
    
    // Concurrency limiter
    options.AddConcurrencyLimiter("concurrency", options =>
    {
        options.PermitLimit = 10;
        options.QueueProcessingOrder = QueueProcessingOrder.OldestFirst;
        options.QueueLimit = 5;
    });
    
    options.OnRejected = async (context, token) =>
    {
        context.HttpContext.Response.StatusCode = StatusCodes.Status429TooManyRequests;
        await context.HttpContext.Response.WriteAsync(
            "Too many requests. Please try again later.", token);
    };
});

var app = builder.Build();

app.UseRateLimiter();

// Apply rate limiting to endpoint
app.MapGet("/api/products", () => "Products")
    .RequireRateLimiting("fixed");

app.Run();
```

### Custom Rate Limit Response

```csharp
builder.Services.AddRateLimiter(options =>
{
    options.OnRejected = async (context, cancellationToken) =>
    {
        context.HttpContext.Response.StatusCode = StatusCodes.Status429TooManyRequests;
        context.HttpContext.Response.Headers.Add("Retry-After", "60");
        
        if (context.Lease.TryGetMetadata(MetadataName.RetryAfter, out var retryAfter))
        {
            context.HttpContext.Response.Headers.Add("Retry-After", 
                retryAfter.TotalSeconds.ToString());
        }
        
        await context.HttpContext.Response.WriteAsJsonAsync(new
        {
            Error = "Too many requests",
            Message = "Rate limit exceeded. Please try again later.",
            RetryAfter = retryAfter?.TotalSeconds
        }, cancellationToken);
    };
});
```

### Distributed Rate Limiting with Redis

```csharp
// Custom distributed rate limiter
public class RedisRateLimiter : RateLimiter
{
    private readonly IConnectionMultiplexer _redis;
    private readonly string _key;
    private readonly int _permitLimit;
    private readonly TimeSpan _window;
    
    public override async ValueTask<RateLimitLease> AcquireAsync(
        int permitCount = 1,
        CancellationToken cancellationToken = default)
    {
        var db = _redis.GetDatabase();
        var current = await db.StringIncrementAsync(_key);
        
        if (current == 1)
        {
            await db.KeyExpireAsync(_key, _window);
        }
        
        if (current <= _permitLimit)
        {
            return new RedisRateLimitLease(true);
        }
        
        return new RedisRateLimitLease(false);
    }
}
```

### Advanced Rate Limiting Patterns

**Tiered Rate Limiting:**

```csharp
public class TieredRateLimiterPolicy
{
    public string Name { get; set; } = string.Empty;
    public Dictionary<string, RateLimitConfig> Tiers { get; set; } = new();
}

public class RateLimitConfig
{
    public int PermitLimit { get; set; }
    public TimeSpan Window { get; set; }
    public int BurstLimit { get; set; }
}

builder.Services.AddRateLimiter(options =>
{
    options.AddPolicy("tiered", context =>
    {
        var tier = context.User.FindFirst("tier")?.Value ?? "free";
        
        var config = tier switch
        {
            "enterprise" => new RateLimitConfig
            {
                PermitLimit = 10000,
                Window = TimeSpan.FromMinutes(1),
                BurstLimit = 100
            },
            "professional" => new RateLimitConfig
            {
                PermitLimit = 1000,
                Window = TimeSpan.FromMinutes(1),
                BurstLimit = 50
            },
            "standard" => new RateLimitConfig
            {
                PermitLimit = 100,
                Window = TimeSpan.FromMinutes(1),
                BurstLimit = 20
            },
            _ => new RateLimitConfig
            {
                PermitLimit = 10,
                Window = TimeSpan.FromMinutes(1),
                BurstLimit = 5
            }
        };
        
        var partitionKey = context.User.Identity?.Name ?? 
                          context.Connection.RemoteIpAddress?.ToString() ?? 
                          "anonymous";
        
        return RateLimitPartition.GetTokenBucketLimiter(partitionKey, _ =>
            new TokenBucketRateLimiterOptions
            {
                TokenLimit = config.PermitLimit,
                ReplenishmentPeriod = config.Window,
                TokensPerPeriod = config.PermitLimit / 10,
                QueueLimit = config.BurstLimit
            });
    });
});
```

**Adaptive Rate Limiting:**

```csharp
public class AdaptiveRateLimiterService
{
    private readonly IMetricsCollector _metrics;
    private readonly ILogger<AdaptiveRateLimiterService> _logger;
    private int _currentLimit = 100;
    
    public async Task AdjustLimitsAsync()
    {
        var serverLoad = await _metrics.GetServerLoadAsync();
        var errorRate = await _metrics.GetErrorRateAsync();
        var responseTime = await _metrics.GetAverageResponseTimeAsync();
        
        if (serverLoad > 0.9 || errorRate > 0.05 || responseTime > 1000)
        {
            // Reduce limits when under stress
            _currentLimit = Math.Max(10, _currentLimit - 10);
            _logger.LogWarning(
                "Reducing rate limit to {Limit} due to high load",
                _currentLimit);
        }
        else if (serverLoad < 0.5 && errorRate < 0.01 && responseTime < 200)
        {
            // Increase limits when server is healthy
            _currentLimit = Math.Min(1000, _currentLimit + 10);
            _logger.LogInformation(
                "Increasing rate limit to {Limit}",
                _currentLimit);
        }
    }
    
    public int GetCurrentLimit() => _currentLimit;
}

// Use adaptive limits
builder.Services.AddRateLimiter(options =>
{
    options.AddPolicy("adaptive", context =>
    {
        var adaptiveService = context.RequestServices
            .GetRequiredService<AdaptiveRateLimiterService>();
        
        var limit = adaptiveService.GetCurrentLimit();
        
        return RateLimitPartition.GetFixedWindowLimiter("adaptive", _ =>
            new FixedWindowRateLimiterOptions
            {
                PermitLimit = limit,
                Window = TimeSpan.FromMinutes(1)
            });
    });
});
```

**Rate Limit Headers:**

```csharp
public class RateLimitHeadersMiddleware
{
    private readonly RequestDelegate _next;
    
    public async Task InvokeAsync(HttpContext context)
    {
        await _next(context);
        
        // Add standard rate limit headers
        var lease = context.GetEndpoint()?.Metadata
            .GetMetadata<EnableRateLimitingAttribute>();
        
        if (lease != null)
        {
            context.Response.Headers.Add("X-RateLimit-Limit", "100");
            context.Response.Headers.Add("X-RateLimit-Remaining", "95");
            context.Response.Headers.Add("X-RateLimit-Reset", 
                DateTimeOffset.UtcNow.AddMinutes(1).ToUnixTimeSeconds().ToString());
        }
    }
}

// Enhanced response on limit exceeded
builder.Services.AddRateLimiter(options =>
{
    options.OnRejected = async (context, cancellationToken) =>
    {
        context.HttpContext.Response.StatusCode = StatusCodes.Status429TooManyRequests;
        
        var retryAfter = 60;
        if (context.Lease.TryGetMetadata(MetadataName.RetryAfter, out var retryAfterMetadata))
        {
            retryAfter = (int)retryAfterMetadata.TotalSeconds;
        }
        
        context.HttpContext.Response.Headers.Add("Retry-After", retryAfter.ToString());
        context.HttpContext.Response.Headers.Add("X-RateLimit-Limit", "100");
        context.HttpContext.Response.Headers.Add("X-RateLimit-Remaining", "0");
        context.HttpContext.Response.Headers.Add("X-RateLimit-Reset",
            DateTimeOffset.UtcNow.AddSeconds(retryAfter).ToUnixTimeSeconds().ToString());
        
        await context.HttpContext.Response.WriteAsJsonAsync(new
        {
            Error = "Rate limit exceeded",
            Message = $"Too many requests. Please retry after {retryAfter} seconds.",
            RetryAfter = retryAfter,
            Limit = 100,
            Remaining = 0
        }, cancellationToken);
    };
});
```

**Custom Rate Limiter with Cost-Based Limiting:**

```csharp
public class CostBasedRateLimiter
{
    private readonly Dictionary<string, int> _costs = new()
    {
        { "/api/search", 1 },
        { "/api/reports/generate", 10 },
        { "/api/export", 20 },
        { "/api/batch-process", 50 }
    };
    
    private readonly ConcurrentDictionary<string, (int tokens, DateTime resetTime)> 
        _userTokens = new();
    
    public bool TryConsume(string userId, string endpoint, out int remainingTokens)
    {
        var cost = _costs.GetValueOrDefault(endpoint, 1);
        var maxTokens = 100;
        var refillRate = TimeSpan.FromMinutes(1);
        
        var (tokens, resetTime) = _userTokens.GetOrAdd(userId, 
            _ => (maxTokens, DateTime.UtcNow.Add(refillRate)));
        
        // Refill tokens if time has passed
        if (DateTime.UtcNow >= resetTime)
        {
            tokens = maxTokens;
            resetTime = DateTime.UtcNow.Add(refillRate);
        }
        
        if (tokens >= cost)
        {
            tokens -= cost;
            _userTokens[userId] = (tokens, resetTime);
            remainingTokens = tokens;
            return true;
        }
        
        remainingTokens = tokens;
        return false;
    }
}

public class CostBasedRateLimitMiddleware
{
    private readonly RequestDelegate _next;
    private readonly CostBasedRateLimiter _limiter;
    
    public async Task InvokeAsync(HttpContext context)
    {
        var userId = context.User.FindFirst(ClaimTypes.NameIdentifier)?.Value ?? "anonymous";
        var endpoint = context.Request.Path.ToString();
        
        if (!_limiter.TryConsume(userId, endpoint, out var remaining))
        {
            context.Response.StatusCode = StatusCodes.Status429TooManyRequests;
            context.Response.Headers.Add("X-RateLimit-Remaining", remaining.ToString());
            
            await context.Response.WriteAsJsonAsync(new
            {
                Error = "Rate limit exceeded",
                Message = "Insufficient tokens for this operation",
                Remaining = remaining
            });
            return;
        }
        
        context.Response.Headers.Add("X-RateLimit-Remaining", remaining.ToString());
        await _next(context);
    }
}
```

**Rate Limit Monitoring and Analytics:**

```csharp
public class RateLimitMonitoringService : BackgroundService
{
    private readonly IServiceProvider _serviceProvider;
    private readonly ILogger<RateLimitMonitoringService> _logger;
    
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                using var scope = _serviceProvider.CreateScope();
                var metrics = scope.ServiceProvider
                    .GetRequiredService<IRateLimitMetrics>();
                
                var stats = await metrics.GetStatisticsAsync();
                
                _logger.LogInformation(
                    "Rate Limit Stats - Total: {Total}, Rejected: {Rejected}, Rate: {Rate}%",
                    stats.TotalRequests,
                    stats.RejectedRequests,
                    stats.RejectionRate * 100);
                
                // Alert if rejection rate is too high
                if (stats.RejectionRate > 0.1)
                {
                    _logger.LogWarning(
                        "High rate limit rejection rate: {Rate}%",
                        stats.RejectionRate * 100);
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error collecting rate limit metrics");
            }
            
            await Task.Delay(TimeSpan.FromMinutes(5), stoppingToken);
        }
    }
}

public interface IRateLimitMetrics
{
    Task<RateLimitStatistics> GetStatisticsAsync();
    Task RecordRequestAsync(string userId, bool accepted);
}

public class RateLimitStatistics
{
    public long TotalRequests { get; set; }
    public long RejectedRequests { get; set; }
    public double RejectionRate => TotalRequests > 0 
        ? (double)RejectedRequests / TotalRequests 
        : 0;
    public Dictionary<string, long> RejectionsByUser { get; set; } = new();
    public Dictionary<string, long> RejectionsByEndpoint { get; set; } = new();
}
```
