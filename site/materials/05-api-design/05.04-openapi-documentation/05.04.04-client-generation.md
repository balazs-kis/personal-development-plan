# Lesson 4: Client Code Generation and SDKs

## Learning Objectives

By the end of this lesson, you will be able to:
- Generate client SDKs from OpenAPI specifications
- Use NSwag for .NET client generation
- Use AutoRest and Kiota for client generation
- Generate TypeScript and other language clients
- Integrate generated clients into applications

---

## Introduction

OpenAPI specifications enable automatic generation of type-safe client SDKs. Instead of manually writing HTTP requests, you can generate client libraries that provide strongly-typed methods, handle serialization, and include intellisense support.

### NSwag with Dependency Injection

```csharp
// Generated client with DI support
public partial class ProductsClient
{
    private readonly HttpClient _httpClient;
    private readonly ILogger<ProductsClient> _logger;
    
    public ProductsClient(HttpClient httpClient, ILogger<ProductsClient> logger)
    {
        _httpClient = httpClient;
        _logger = logger;
    }
}

// Registration in Program.cs
builder.Services.AddHttpClient<IProductsClient, ProductsClient>(client =>
{
    client.BaseAddress = new Uri("https://api.example.com");
    client.DefaultRequestHeaders.Add("User-Agent", "MyApp/1.0");
    client.Timeout = TimeSpan.FromSeconds(30);
});

// Configure with authentication
builder.Services.AddHttpClient<IProductsClient, ProductsClient>(client =>
{
    client.BaseAddress = new Uri("https://api.example.com");
})
.AddHttpMessageHandler<AuthenticationDelegatingHandler>();

public class AuthenticationDelegatingHandler : DelegatingHandler
{
    private readonly ITokenProvider _tokenProvider;
    
    public AuthenticationDelegatingHandler(ITokenProvider tokenProvider)
    {
        _tokenProvider = tokenProvider;
    }
    
    protected override async Task<HttpResponseMessage> SendAsync(
        HttpRequestMessage request, 
        CancellationToken cancellationToken)
    {
        var token = await _tokenProvider.GetTokenAsync(cancellationToken);
        request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
        
        return await base.SendAsync(request, cancellationToken);
    }
}

// Usage in services
public class ProductService
{
    private readonly IProductsClient _productsClient;
    
    public ProductService(IProductsClient productsClient)
    {
        _productsClient = productsClient;
    }
    
    public async Task<IEnumerable<Product>> GetProductsAsync()
    {
        var response = await _productsClient.GetProductsAsync(1, 20);
        return response.Data;
    }
}
```

### Error Handling in Generated Clients

```csharp
// NSwag generates typed exception classes
public class ProductService
{
    private readonly IProductsClient _client;
    private readonly ILogger<ProductService> _logger;
    
    public ProductService(IProductsClient client, ILogger<ProductService> logger)
    {
        _client = client;
        _logger = logger;
    }
    
    public async Task<Product?> GetProductAsync(int id)
    {
        try
        {
            return await _client.GetProductAsync(id);
        }
        catch (ApiException<Error> ex) when (ex.StatusCode == 404)
        {
            _logger.LogWarning("Product {ProductId} not found", id);
            return null;
        }
        catch (ApiException<Error> ex) when (ex.StatusCode == 400)
        {
            _logger.LogError("Bad request for product {ProductId}: {Message}", 
                id, ex.Result.Message);
            throw new ValidationException(ex.Result.Message);
        }
        catch (ApiException ex) when (ex.StatusCode >= 500)
        {
            _logger.LogError(ex, "Server error getting product {ProductId}", id);
            throw new ServiceUnavailableException("Product service unavailable", ex);
        }
    }
}
```

### Retry Policies with Polly

```csharp
// Install Polly
// dotnet add package Microsoft.Extensions.Http.Polly

builder.Services.AddHttpClient<IProductsClient, ProductsClient>(client =>
{
    client.BaseAddress = new Uri("https://api.example.com");
})
.AddTransientHttpErrorPolicy(policyBuilder =>
    policyBuilder.WaitAndRetryAsync(3, retryAttempt => 
        TimeSpan.FromSeconds(Math.Pow(2, retryAttempt))))
.AddTransientHttpErrorPolicy(policyBuilder =>
    policyBuilder.CircuitBreakerAsync(5, TimeSpan.FromSeconds(30)));

// Custom retry logic
builder.Services.AddHttpClient<IProductsClient, ProductsClient>(client =>
{
    client.BaseAddress = new Uri("https://api.example.com");
})
.AddPolicyHandler((services, request) =>
{
    var logger = services.GetRequiredService<ILogger<ProductsClient>>();
    
    return Policy<HttpResponseMessage>
        .Handle<HttpRequestException>()
        .OrResult(response => (int)response.StatusCode >= 500)
        .WaitAndRetryAsync(
            retryCount: 3,
            sleepDurationProvider: retryAttempt => 
                TimeSpan.FromSeconds(Math.Pow(2, retryAttempt)),
            onRetry: (outcome, timespan, retryCount, context) =>
            {
                logger.LogWarning(
                    "Retry {RetryCount} after {Delay}ms due to {Reason}",
                    retryCount, timespan.TotalMilliseconds, outcome.Exception?.Message);
            });
});
```

### React Integration

```typescript
// hooks/useProducts.ts
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { ApiClient, CreateProductRequest } from '../api-client';

const apiClient = new ApiClient(process.env.REACT_APP_API_URL);

export function useProducts(page: number = 1, pageSize: number = 20) {
  return useQuery({
    queryKey: ['products', page, pageSize],
    queryFn: () => apiClient.getProducts(page, pageSize)
  });
}

export function useProduct(id: number) {
  return useQuery({
    queryKey: ['product', id],
    queryFn: () => apiClient.getProduct(id),
    enabled: id > 0
  });
}

export function useCreateProduct() {
  const queryClient = useQueryClient();
  
  return useMutation({
    mutationFn: (request: CreateProductRequest) => 
      apiClient.createProduct(request),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['products'] });
    }
  });
}

// Component usage
import { useProducts, useCreateProduct } from './hooks/useProducts';

function ProductList() {
  const { data: products, isLoading, error } = useProducts(1, 20);
  const createProduct = useCreateProduct();
  
  if (isLoading) return <div>Loading...</div>;
  if (error) return <div>Error: {error.message}</div>;
  
  return (
    <div>
      {products?.map(product => (
        <div key={product.id}>{product.name}</div>
      ))}
    </div>
  );
}
```

### Client Caching Strategy

```csharp
// Caching decorator for generated client
public class CachedProductsClient : IProductsClient
{
    private readonly IProductsClient _innerClient;
    private readonly IMemoryCache _cache;
    private readonly ILogger<CachedProductsClient> _logger;
    
    public CachedProductsClient(
        IProductsClient innerClient,
        IMemoryCache cache,
        ILogger<CachedProductsClient> logger)
    {
        _innerClient = innerClient;
        _cache = cache;
        _logger = logger;
    }
    
    public async Task<Product> GetProductAsync(int id, CancellationToken ct = default)
    {
        var cacheKey = $"product:{id}";
        
        if (_cache.TryGetValue(cacheKey, out Product cachedProduct))
        {
            _logger.LogDebug("Cache hit for product {ProductId}", id);
            return cachedProduct;
        }
        
        var product = await _innerClient.GetProductAsync(id, ct);
        
        var cacheOptions = new MemoryCacheEntryOptions()
            .SetAbsoluteExpiration(TimeSpan.FromMinutes(5))
            .SetSlidingExpiration(TimeSpan.FromMinutes(1));
            
        _cache.Set(cacheKey, product, cacheOptions);
        
        return product;
    }
    
    public async Task<Product> CreateProductAsync(
        CreateProductRequest request, 
        CancellationToken ct = default)
    {
        var product = await _innerClient.CreateProductAsync(request, ct);
        
        // Invalidate list cache
        _cache.Remove("products:list");
        
        return product;
    }
}

// Register with decoration
builder.Services.AddHttpClient<ProductsClient>(/* config */);
builder.Services.AddSingleton<IProductsClient>(sp =>
{
    var client = sp.GetRequiredService<ProductsClient>();
    var cache = sp.GetRequiredService<IMemoryCache>();
    var logger = sp.GetRequiredService<ILogger<CachedProductsClient>>();
    return new CachedProductsClient(client, cache, logger);
});
```

```csharp

  }

// Program.cs
var useMock = builder.Configuration.GetValue<bool>("ApiClient:UseMock");
var baseUrl = builder.Configuration.GetValue<string>("ApiClient:BaseUrl");

builder.Services.AddHttpClient<IProductsClient, ProductsClient>(client =>
```
