## Introduction

**Snapshots** are point-in-time captures of aggregate state that eliminate the need to replay all events. Instead of loading 10,000 events, you load the latest snapshot (event 9,500) and replay only the last 500 events. Snapshots are a performance optimization—a cache of reconstructed state.

Understanding snapshots is crucial for building event-sourced systems that scale, as replay performance degrades linearly with event count.

## Key Concepts Summary

### When to Use Snapshots

```csharp
// Without snapshot: Load all events
var events = await _eventStore.GetEventsAsync("Order-123");  // 5,000 events
foreach (var evt in events)  // Slow!
{
    order.Apply(evt);
}

// With snapshot: Load snapshot + recent events
var snapshot = await _snapshotStore.GetAsync("Order-123");  // At event 4,900
var events = await _eventStore.GetEventsAsync("Order-123", fromVersion: 4901);
order.RestoreFromSnapshot(snapshot);
foreach (var evt in events)  // Only 100 events!
{
    order.Apply(evt);
}
```

**Use snapshots when:**
- Aggregates have > 100 events
- Reconstruction time > 50ms
- Frequent reads of same aggregate
- Performance is critical

### Implementation

```csharp
public interface ISnapshotStore
{
    Task SaveSnapshotAsync<T>(string aggregateId, T state, int version);
    Task<Snapshot<T>> GetSnapshotAsync<T>(string aggregateId);
}

public class SqlSnapshotStore : ISnapshotStore
{
    private readonly DbContext _db;
    
    public async Task SaveSnapshotAsync<T>(string aggregateId, T state, int version)
    {
        var snapshot = new SnapshotRecord
        {
            AggregateId = aggregateId,
            AggregateType = typeof(T).Name,
            Version = version,
            SnapshotData = JsonSerializer.Serialize(state),
            CreatedAt = DateTime.UtcNow
        };
        
        // Upsert: replace existing snapshot
        var existing = await _db.Snapshots.FindAsync(aggregateId);
        if (existing != null)
        {
            existing.Version = version;
            existing.SnapshotData = snapshot.SnapshotData;
            existing.CreatedAt = snapshot.CreatedAt;
        }
        else
        {
            _db.Snapshots.Add(snapshot);
        }
        
        await _db.SaveChangesAsync();
    }
    
    public async Task<Snapshot<T>> GetSnapshotAsync<T>(string aggregateId)
    {
        var record = await _db.Snapshots.FindAsync(aggregateId);
        if (record == null)
            return null;
        
        return new Snapshot<T>
        {
            AggregateId = aggregateId,
            Version = record.Version,
            State = JsonSerializer.Deserialize<T>(record.SnapshotData),
            CreatedAt = record.CreatedAt
        };
    }
}
```

### Repository with Snapshots

```csharp
public class SnapshotRepository<T> where T : AggregateRoot, new()
{
    private readonly IEventStore _eventStore;
    private readonly ISnapshotStore _snapshotStore;
    private readonly int _snapshotFrequency;
    
    public SnapshotRepository(
        IEventStore eventStore,
        ISnapshotStore snapshotStore,
        int snapshotFrequency = 100)
    {
        _eventStore = eventStore;
        _snapshotStore = snapshotStore;
        _snapshotFrequency = snapshotFrequency;
    }
    
    public async Task<T> GetByIdAsync(string aggregateId)
    {
        var streamId = $"{typeof(T).Name}-{aggregateId}";
        
        // Try to load snapshot
        var snapshot = await _snapshotStore.GetSnapshotAsync<T>(aggregateId);
        
        IEnumerable<IDomainEvent> events;
        var aggregate = new T { Id = aggregateId };
        
        if (snapshot != null)
        {
            // Restore from snapshot
            aggregate.RestoreFromSnapshot(snapshot.State);
            aggregate.Version = snapshot.Version;
            
            // Load events since snapshot
            events = await _eventStore.GetEventsAsync(streamId, snapshot.Version + 1);
        }
        else
        {
            // No snapshot, load all events
            events = await _eventStore.GetEventsAsync(streamId);
        }
        
        // Replay remaining events
        aggregate.Load(events);
        
        return aggregate;
    }
    
    public async Task SaveAsync(T aggregate)
    {
        var events = aggregate.GetUncommittedEvents();
        if (!events.Any())
            return;
        
        var streamId = $"{typeof(T).Name}-{aggregate.Id}";
        
        // Save events
        await _eventStore.AppendEventsAsync(streamId, events, aggregate.Version - events.Count());
        
        aggregate.MarkEventsAsCommitted();
        
        // Take snapshot if threshold reached
        if (aggregate.Version % _snapshotFrequency == 0)
        {
            await _snapshotStore.SaveSnapshotAsync(
                aggregate.Id,
                aggregate.ToSnapshot(),
                aggregate.Version
            );
        }
    }
}

// Aggregate with snapshot support
public abstract class AggregateRoot
{
    public string Id { get; set; }
    public int Version { get; set; }
    
    public abstract object ToSnapshot();
    public abstract void RestoreFromSnapshot(object snapshot);
}

public class OrderAggregate : AggregateRoot
{
    public OrderStatus Status { get; private set; }
    public decimal TotalAmount { get; private set; }
    public List<OrderItem> Items { get; private set; } = new();
    
    public override object ToSnapshot()
    {
        return new OrderSnapshot
        {
            OrderId = Id,
            Status = Status,
            TotalAmount = TotalAmount,
            Items = Items.ToList()
        };
    }
    
    public override void RestoreFromSnapshot(object snapshot)
    {
        var snap = (OrderSnapshot)snapshot;
        Id = snap.OrderId;
        Status = snap.Status;
        TotalAmount = snap.TotalAmount;
        Items = snap.Items.ToList();
    }
}
```

### Snapshot Strategies

**1. Fixed Frequency (Every N Events)**
```csharp
if (aggregate.Version % 100 == 0)  // Every 100 events
{
    await TakeSnapshot(aggregate);
}
```

**2. Time-Based (Every N Hours)**
```csharp
var lastSnapshot = await GetLastSnapshotTime(aggregateId);
if (DateTime.UtcNow - lastSnapshot > TimeSpan.FromHours(6))
{
    await TakeSnapshot(aggregate);
}
```

**3. On-Demand (After Load)**
```csharp
var aggregate = await LoadAggregate(id);
var eventsReplayed = aggregate.Version - snapshot.Version;

if (eventsReplayed > 50)  // If we replayed many events
{
    await TakeSnapshot(aggregate);  // Take new snapshot for next time
}
```

### Cleanup and Archival

```csharp
public class SnapshotMaintenanceService
{
    // Keep only the latest snapshot per aggregate
    public async Task CleanupOldSnapshots()
    {
        // If storing multiple snapshots per aggregate
        var oldSnapshots = await _db.Snapshots
            .GroupBy(s => s.AggregateId)
            .SelectMany(g => g.OrderByDescending(s => s.Version).Skip(1))
            .ToListAsync();
        
        _db.Snapshots.RemoveRange(oldSnapshots);
        await _db.SaveChangesAsync();
    }
    
    // Archive old events after snapshot
    public async Task ArchiveOldEvents(string aggregateId)
    {
        var snapshot = await _snapshotStore.GetAsync<OrderAggregate>(aggregateId);
        if (snapshot == null)
            return;
        
        // Move events before snapshot to cold storage
        var oldEvents = await _eventStore.GetEventsAsync(
            $"Order-{aggregateId}",
            toVersion: snapshot.Version
        );
        
        await _archiveStore.ArchiveEvents(oldEvents);
        await _eventStore.DeleteEvents($"Order-{aggregateId}", toVersion: snapshot.Version);
    }
}
```

### Snapshot Invalidation Strategies

```csharp
// Strategy 1: Version-based invalidation
public class VersionedSnapshotStore : ISnapshotStore
{
    private const int CurrentSnapshotVersion = 2;
    
    public async Task<Snapshot<T>> GetSnapshotAsync<T>(string aggregateId)
    {
        var snapshot = await _inner.GetSnapshotAsync<T>(aggregateId);
        
        if (snapshot != null && snapshot.SnapshotVersion < CurrentSnapshotVersion)
        {
            // Snapshot schema changed, invalidate
            return null;
        }
        
        return snapshot;
    }
}

// Strategy 2: TTL-based invalidation
public class TtlSnapshotStore : ISnapshotStore
{
    private readonly TimeSpan _ttl = TimeSpan.FromHours(24);
    
    public async Task<Snapshot<T>> GetSnapshotAsync<T>(string aggregateId)
    {
        var snapshot = await _inner.GetSnapshotAsync<T>(aggregateId);
        
        if (snapshot != null && DateTime.UtcNow - snapshot.CreatedAt > _ttl)
        {
            // Snapshot too old, invalidate
            return null;
        }
        
        return snapshot;
    }
}
```

### Snapshot Compression

```csharp
public class CompressedSnapshotStore : ISnapshotStore
{
    public async Task SaveSnapshotAsync<T>(string aggregateId, T state, int version)
    {
        var json = JsonSerializer.Serialize(state);
        var bytes = Encoding.UTF8.GetBytes(json);
        
        // Compress with GZip
        using var compressedStream = new MemoryStream();
        using (var gzip = new GZipStream(compressedStream, CompressionMode.Compress))
        {
            await gzip.WriteAsync(bytes, 0, bytes.Length);
        }
        
        var compressed = compressedStream.ToArray();
        
        await _db.Snapshots.AddAsync(new SnapshotRecord
        {
            AggregateId = aggregateId,
            Version = version,
            Data = compressed,
            IsCompressed = true,
            CreatedAt = DateTime.UtcNow
        });
        
        await _db.SaveChangesAsync();
        
        Console.WriteLine($"Snapshot compressed: {bytes.Length} → {compressed.Length} bytes " +
                         $"({100 - (compressed.Length * 100 / bytes.Length)}% reduction)");
    }
    
    public async Task<Snapshot<T>> GetSnapshotAsync<T>(string aggregateId)
    {
        var record = await _db.Snapshots.FindAsync(aggregateId);
        if (record == null) return null;
        
        byte[] data = record.Data;
        
        if (record.IsCompressed)
        {
            // Decompress
            using var compressedStream = new MemoryStream(data);
            using var gzip = new GZipStream(compressedStream, CompressionMode.Decompress);
            using var decompressedStream = new MemoryStream();
            
            await gzip.CopyToAsync(decompressedStream);
            data = decompressedStream.ToArray();
        }
        
        var json = Encoding.UTF8.GetString(data);
        var state = JsonSerializer.Deserialize<T>(json);
        
        return new Snapshot<T>
        {
            AggregateId = aggregateId,
            Version = record.Version,
            State = state,
            CreatedAt = record.CreatedAt
        };
    }
}
```

```csharp
public class SnapshotUpcaster
{
    public Snapshot<OrderAggregate> UpcastOrderSnapshot(VersionedSnapshot<object> snapshot)
    {
        return snapshot.SnapshotVersion switch
        {
            1 => UpcastV1ToV2((OrderSnapshotV1)snapshot.State),
            2 => new Snapshot<OrderAggregate>
```

### Monitoring Snapshot Effectiveness

```csharp
public class SnapshotMetrics
{
    private readonly Histogram _loadWithSnapshot;
    private readonly Histogram _loadWithoutSnapshot;
    private readonly Gauge _snapshotHitRate;
    
    public void RecordLoad(string aggregateId, bool snapshotUsed, TimeSpan duration, int eventsReplayed)
    {
        if (snapshotUsed)
        {
            _loadWithSnapshot.Record(duration.TotalMilliseconds,
                new("aggregate_type", GetAggregateType(aggregateId)),
                new("events_replayed", eventsReplayed.ToString()));
        }
        else
        {
            _loadWithoutSnapshot.Record(duration.TotalMilliseconds,
                new("aggregate_type", GetAggregateType(aggregateId)));
        }
    }
    
    public async Task UpdateSnapshotHitRate()
    {
        var total = _loadsWithSnapshot + _loadsWithoutSnapshot;
        var rate = total > 0 ? (_loadsWithSnapshot / (double)total) * 100 : 0;
        
        _snapshotHitRate.Record(rate);
        
        if (rate < 50)
        {
            Console.WriteLine($"⚠️ Low snapshot hit rate: {rate:F1}%");
        }
    }
}
```

### Advanced Snapshot Patterns

**Pattern: Snapshot with Deltas**

```csharp
// Instead of full snapshots, store deltas
public class DeltaSnapshot
{
    public string AggregateId { get; set; }
    public int BaseVersion { get; set; }
    public object BaseState { get; set; }
    public List<EventDelta> Deltas { get; set; }  // Recent events
}

// Reduces snapshot size but requires some event replay
public async Task<T> LoadWithDelta<T>(string aggregateId) where T : AggregateRoot, new()
{
    var deltaSnapshot = await _deltaStore.GetAsync(aggregateId);
    
    var aggregate = new T { Id = aggregateId };
    aggregate.RestoreFromSnapshot(deltaSnapshot.BaseState);
    aggregate.Version = deltaSnapshot.BaseVersion;
    
    // Replay cached deltas
    foreach (var delta in deltaSnapshot.Deltas)
    {
        aggregate.Apply(delta.Event);
    }
    
    // Fetch any events after deltas
    var newEvents = await _eventStore.GetEventsAsync(
        aggregateId, 
        fromVersion: deltaSnapshot.BaseVersion + deltaSnapshot.Deltas.Count + 1);
    
    aggregate.Load(newEvents);
    
    return aggregate;
}
```
