---

## Introduction

Query parameter versioning places the version in the query string (e.g., `/api/products?api-version=1.0`), while media type versioning uses custom media types in the Accept header (e.g., `Accept: application/vnd.company.product-v2+json`). These approaches offer different trade-offs in terms of discoverability, caching, and REST compliance.

## Key Concepts Summary

### Query Parameter Versioning

```csharp
// Program.cs - Configure query parameter versioning
using Asp.Versioning;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddApiVersioning(options =>
{
    options.DefaultApiVersion = new ApiVersion(1, 0);
    options.AssumeDefaultVersionWhenUnspecified = true;
    options.ReportApiVersions = true;
    
    // Read version from query string
    options.ApiVersionReader = new QueryStringApiVersionReader("api-version");
    
    // Can also use alternative parameter name
    // options.ApiVersionReader = new QueryStringApiVersionReader("v");
});

builder.Services.AddControllers();
var app = builder.Build();
app.MapControllers();
app.Run();
```

### Controller with Query Versioning

```csharp
[ApiController]
[Route("api/products")]
[ApiVersion("1.0")]
[ApiVersion("2.0")]
public class ProductsController : ControllerBase
{
    // GET /api/products?api-version=1.0
    [MapToApiVersion("1.0")]
    [HttpGet]
    public ActionResult<IEnumerable<ProductV1Dto>> GetProductsV1()
    {
        var products = _productService.GetAllProducts();
        return Ok(products.Select(MapToV1Dto));
    }
    
    // GET /api/products?api-version=2.0
    [MapToApiVersion("2.0")]
    [HttpGet]
    public ActionResult<ProductsV2Response> GetProductsV2(
        [FromQuery] string? category = null,
        [FromQuery] int page = 1,
        [FromQuery] int pageSize = 20)
    {
        // Note: api-version and other query params coexist
        var products = _productService.GetProducts(category, page, pageSize);
        var total = _productService.GetProductCount(category);
        
        return Ok(new ProductsV2Response
        {
            Data = products.Select(MapToV2Dto).ToList(),
            Page = page,
            PageSize = pageSize,
            TotalCount = total
        });
    }
    
    // GET /api/products/123?api-version=1.0
    // GET /api/products/123?api-version=2.0
    [HttpGet("{id}")]
    public ActionResult GetProduct(int id, ApiVersion version)
    {
        var product = _productService.GetProductById(id);
        
        if (product == null)
            return NotFound();
        
        return version.MajorVersion switch
        {
            1 => Ok(MapToV1Dto(product)),
            2 => Ok(MapToV2Dto(product)),
            _ => BadRequest($"Unsupported version: {version}")
        };
    }
}

// Client usage:
// curl "http://localhost:5000/api/products?api-version=1.0"
// curl "http://localhost:5000/api/products?api-version=2.0&category=electronics&page=2"
```

### Custom Media Type Versioning

```csharp
// Define custom media types
public static class CustomMediaTypes
{
    public const string ProductV1 = "application/vnd.mycompany.product.v1+json";
    public const string ProductV2 = "application/vnd.mycompany.product.v2+json";
    public const string ProductListV1 = "application/vnd.mycompany.product-list.v1+json";
    public const string ProductListV2 = "application/vnd.mycompany.product-list.v2+json";
}

// Program.cs
builder.Services.AddApiVersioning(options =>
{
    options.DefaultApiVersion = new ApiVersion(1, 0);
    options.AssumeDefaultVersionWhenUnspecified = true;
    options.ReportApiVersions = true;
    
    // Read version from Accept header media type
    options.ApiVersionReader = new MediaTypeApiVersionReader();
});

builder.Services.AddControllers(options =>
{
    // Add custom media type formatters
    options.FormatterMappings.SetMediaTypeMappingForFormat(
        "v1",
        CustomMediaTypes.ProductV1);
    options.FormatterMappings.SetMediaTypeMappingForFormat(
        "v2",
        CustomMediaTypes.ProductV2);
});

// Controller with media type versioning
[ApiController]
[Route("api/products")]
[ApiVersion("1.0")]
[ApiVersion("2.0")]
public class ProductsController : ControllerBase
{
    // Responds to: Accept: application/vnd.mycompany.product-list.v1+json
    [MapToApiVersion("1.0")]
    [HttpGet]
    [Produces(CustomMediaTypes.ProductListV1)]
    public ActionResult<IEnumerable<ProductV1Dto>> GetProductsV1()
    {
        var products = _productService.GetAllProducts();
        return Ok(products.Select(MapToV1Dto));
    }
    
    // Responds to: Accept: application/vnd.mycompany.product-list.v2+json
    [MapToApiVersion("2.0")]
    [HttpGet]
    [Produces(CustomMediaTypes.ProductListV2)]
    public ActionResult<ProductsV2Response> GetProductsV2(
        [FromQuery] string? category = null)
    {
        var products = _productService.GetProducts(category);
        return Ok(new ProductsV2Response { Data = products.Select(MapToV2Dto).ToList() });
    }
    
    // Single resource with versioned media types
    [HttpGet("{id}")]
    public ActionResult GetProduct(int id, ApiVersion version)
    {
        var product = _productService.GetProductById(id);
        
        if (product == null)
            return NotFound();
        
        if (version.MajorVersion == 1)
        {
            Response.ContentType = CustomMediaTypes.ProductV1;
            return Ok(MapToV1Dto(product));
        }
        else
        {
            Response.ContentType = CustomMediaTypes.ProductV2;
            return Ok(MapToV2Dto(product));
        }
    }
}

// Client usage:
// curl -H "Accept: application/vnd.mycompany.product-list.v1+json" http://localhost:5000/api/products
// curl -H "Accept: application/vnd.mycompany.product-list.v2+json" http://localhost:5000/api/products
```

### Date-Based Versioning

```csharp
// Some APIs use date-based versions (e.g., Azure)
// Format: YYYY-MM-DD

[ApiController]
[Route("api/products")]
[ApiVersion("2024-01-15")]
[ApiVersion("2024-06-01")]
public class ProductsController : ControllerBase
{
    // GET /api/products?api-version=2024-01-15
    [MapToApiVersion("2024-01-15")]
    [HttpGet]
    public ActionResult<IEnumerable<ProductDto>> GetProducts_2024_01()
    {
        // January 2024 version
    }
    
    // GET /api/products?api-version=2024-06-01
    [MapToApiVersion("2024-06-01")]
    [HttpGet]
    public ActionResult<IEnumerable<ProductDto>> GetProducts_2024_06()
    {
        // June 2024 version with breaking changes
    }
}

// Date-based versioning pros:
// ✅ Clear when version was released
// ✅ Easy to track timeline
// ✅ Works well for frequently updated APIs

// Date-based versioning cons:
// ❌ No semantic meaning (is it breaking?)
// ❌ Hard to know compatibility
// ❌ Clients must track dates
```

```
Choose URL Path when:
✅ Building public APIs
✅ Discoverability is important
✅ Non-technical users will consume API
✅ You want simple, obvious versioning

Choose Header when:
✅ Building internal/enterprise APIs
✅ URL stability is important
✅ Sophisticated API clients
✅ REST purity matters

Choose Query Parameter when:
✅ Need discoverable + clean URLs
✅ Building analytics/reporting APIs
✅ Version alongside other query params
✅ Want flexible, optional versioning

Choose Media Type when:
✅ Building hypermedia APIs
✅ Content negotiation is core feature
✅ Following strict REST principles
✅ Different representations of same resource
```

### Combining Strategies

```csharp
// Support multiple versioning strategies simultaneously
builder.Services.AddApiVersioning(options =>
{
    options.DefaultApiVersion = new ApiVersion(1, 0);
    options.AssumeDefaultVersionWhenUnspecified = true;
    options.ReportApiVersions = true;
    
    // Accept version from multiple sources
    options.ApiVersionReader = ApiVersionReader.Combine(
        new UrlSegmentApiVersionReader(),           // /api/v1/products
        new HeaderApiVersionReader("X-API-Version"), // X-API-Version: 1.0
        new QueryStringApiVersionReader("v"),        // ?v=1.0
        new MediaTypeApiVersionReader()              // Accept header
    );
});

// Now all of these work:
// GET /api/v1/products
// GET /api/products with header X-API-Version: 1.0
// GET /api/products?v=1.0
// GET /api/products with Accept: application/json;v=1.0

// First found takes precedence:
// 1. URL segment
// 2. Header
// 3. Query string
// 4. Media type
```

### Testing Different Strategies

```csharp
public class VersioningStrategiesTests
{
    [Theory]
    [InlineData("/api/v1/products", null, null)]  // URL versioning
    [InlineData("/api/products", "1.0", null)]     // Header versioning
    [InlineData("/api/products?v=1.0", null, null)] // Query versioning
    public async Task GetProducts_DifferentVersioningStrategies_Work(
        string url,
        string? headerVersion,
        string? acceptType)
    {
        // Arrange
        var request = new HttpRequestMessage(HttpMethod.Get, url);
        
        if (headerVersion != null)
            request.Headers.Add("X-API-Version", headerVersion);
        
        if (acceptType != null)
            request.Headers.Add("Accept", acceptType);
        
        // Act
        var response = await _client.SendAsync(request);
        
        // Assert
        response.EnsureSuccessStatusCode();
        var content = await response.Content.ReadAsStringAsync();
        Assert.NotEmpty(content);
    }
}
```

```

- Docs: https://learn.microsoft.com/en-us/azure/azure-resource-manager/management/api-versions

AWS APIs:
- Uses custom media types and headers
- Service-specific versioning
- Example: x-amz-target: DynamoDB_20120810.GetItem

- Docs: https://docs.aws.amazon.com/general/latest/gr/sigv4_elements.html

- Example: https://api.twitter.com/2/tweets

Salesforce API:
- Uses URL path: /services/data/v52.0/sobjects
- Major+minor version
- Example: https://yourinstance.salesforce.com/services/data/v52.0/sobjects/Account
- Docs: https://developer.salesforce.com/docs/atlas.en-us.api_rest.meta/api_rest/

Google APIs:
- Mixed approach: URL path for major, query for minor
- Example: https://www.googleapis.com/calendar/v3/calendars
- Some use discovery documents
- Docs: https://developers.google.com/discovery

Netflix API (internal):
- Uses custom media types
- Example: Accept: application/vnd.netflix.v1+json
- Content negotiation based

Twilio API:
- Uses URL path: /2010-04-01/Accounts
- Date-based in URL
- Example: https://api.twilio.com/2010-04-01/Accounts/AC.../Messages.json

Key Takeaways:
- No single "correct" approach
- URL path most common for public APIs (Twitter, Salesforce)
- Query parameters common for enterprise (Azure)
```

### Migration Between Versioning Strategies

```csharp
// Scenario: Migrating from URL versioning to header versioning

// Phase 1: Support both URL and header
builder.Services.AddApiVersioning(options =>
{
    options.DefaultApiVersion = new ApiVersion(2, 0);
    options.AssumeDefaultVersionWhenUnspecified = true;
    
    // Accept version from both URL and header
    options.ApiVersionReader = ApiVersionReader.Combine(
        new UrlSegmentApiVersionReader(),           // /api/v1/products (legacy)
        new HeaderApiVersionReader("X-API-Version") // X-API-Version: 2.0 (new)
    );
});

// Controllers work with both
[ApiController]
[Route("api/v{version:apiVersion}/products")]  // URL versioning
[Route("api/products")]                          // Header versioning
[ApiVersion("1.0")]
[ApiVersion("2.0")]
public class ProductsController : ControllerBase
{
    // Works with both:
    // GET /api/v1/products
    // GET /api/products with header X-API-Version: 1.0
}

// Phase 2: Deprecate URL versioning
// Mark URL routes as deprecated, encourage header usage

// Phase 3: Remove URL versioning
// Keep only header versioning
[ApiController]
[Route("api/products")]
[ApiVersion("2.0")]
[ApiVersion("3.0")]
public class ProductsController : ControllerBase { }
```

### Performance Considerations

```csharp
// Caching with different versioning strategies

// URL versioning: Each version has unique URL - simple caching
[ResponseCache(Duration = 300, Location = ResponseCacheLocation.Any)]
[HttpGet("/api/v1/products/{id}")]
public ActionResult GetProductV1(int id) { }

[ResponseCache(Duration = 300, Location = ResponseCacheLocation.Any)]
[HttpGet("/api/v2/products/{id}")]
public ActionResult GetProductV2(int id) { }

// Header/Query versioning: Need Vary header
[HttpGet("/api/products/{id}")]
public ActionResult GetProduct(int id, ApiVersion version)
{
    // Add Vary header for proper caching
    Response.Headers.Add("Vary", "X-API-Version");
    Response.Headers.Add("Cache-Control", "public, max-age=300");
    
    return version.MajorVersion switch
    {
        1 => Ok(MapToV1Dto(product)),
        2 => Ok(MapToV2Dto(product)),
        _ => BadRequest()
    };
}

// Query parameter versioning: Vary on query string
[HttpGet("/api/products/{id}")]
public ActionResult GetProduct(int id, [FromQuery(Name = "api-version")] string version)
{
    // Query parameters automatically part of cache key
    Response.Headers.Add("Cache-Control", "public, max-age=300");
    
    return version switch
    {
        "1.0" => Ok(MapToV1Dto(product)),
        "2.0" => Ok(MapToV2Dto(product)),
        _ => BadRequest()
    };
}

// Performance comparison:
// URL versioning: Best for CDN caching (unique URLs)
// Header versioning: Requires Vary header support
// Query versioning: Good caching, slightly more complex URLs
// Media type: Most complex caching setup
```

### Error Handling Across Strategies

```csharp
// Consistent error handling regardless of versioning strategy

public class ApiVersionExceptionFilter : IExceptionFilter
{
    public void OnException(ExceptionContext context)
    {
        if (context.Exception is ApiVersionException versionEx)
        {
            var error = new
            {
                error = "UnsupportedApiVersion",
                message = versionEx.Message,
                requestedVersion = versionEx.RequestedVersion?.ToString(),
                supportedVersions = new[] { "1.0", "2.0", "3.0" },
                documentation = "https://api.example.com/docs/versioning"
            };
            
            context.Result = new ObjectResult(error)
            {
                StatusCode = 400
            };
            
            context.ExceptionHandled = true;
        }
    }
}

// Register filter
builder.Services.AddControllers(options =>
{
    options.Filters.Add<ApiVersionExceptionFilter>();
});

// Works for all versioning strategies:
// GET /api/v99/products → 400 with clear error
// GET /api/products?api-version=99.0 → 400 with clear error
// GET /api/products with header X-API-Version: 99.0 → 400 with clear error
```

## Self-Check Questions

1. **What are the advantages of query parameter versioning?**
   - Discoverable (in URL but not in path)
   - Browser-testable (easy to add query param)
   - Works alongside other query parameters
   - Optional versioning (can have default)
   - Good for analytics APIs

2. **How does media type versioning differ from custom header versioning?**
   - Media type: Version in Accept header content negotiation (application/json;v=1.0)
   - Custom header: Dedicated version header (X-API-Version: 1.0)
   - Media type more RESTful, custom header more explicit
   - Media type tied to content format, header independent

3. **When would you use date-based versioning?**
   - APIs that change frequently
   - When timeline matters more than semantic meaning
   - Enterprise APIs (Azure, Stripe style)
   - When you want clear release dates
   - When breaking changes happen on schedule

4. **What happens when multiple versioning strategies are combined?**
   - First reader finds version (precedence order)
   - Typically: URL → Header → Query → Media Type
   - Provides flexibility for different clients
   - Can support migration between strategies
   - More complex to test and document

5. **Which strategy offers the best caching support?**
   - URL path: Best (unique URLs, simple caching)
   - Query parameter: Good (URLs still unique)
   - Header: Complex (requires Vary header)
   - Media type: Most complex (content negotiation)

6. **How do real-world APIs (GitHub, Stripe, Azure) version their APIs?**
   - GitHub: Header with dates (X-GitHub-Api-Version: 2022-11-28)
   - Stripe: Header with dates (Stripe-Version: 2020-08-27)
   - Azure: Query parameter with dates (?api-version=2021-04-01)
   - Twitter: URL path (/2/tweets)
   - AWS: Custom media types and service-specific
   - No single standard - choose based on needs
