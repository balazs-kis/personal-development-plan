---

## Introduction

URL path versioning is the most common and visible versioning strategy. The version number appears directly in the URL path (e.g., `/api/v1/products`, `/api/v2/products`). This approach is intuitive, easily discoverable, and works well with HTTP caching, but requires careful routing configuration and can lead to code duplication.

## Key Concepts Summary

### URL Path Versioning Structure

```
Path Versioning Patterns:

✅ Good patterns:
/api/v1/products
/api/v2/products
/api/v1.5/products          (minor versions if needed)
/api/2024-01-15/products    (date-based versioning)

❌ Bad patterns:
/api/products/v1            (version at end)
/api/v1.0.0/products        (too granular)
/api/version1/products      (verbose)
/api/1/products             (no 'v' prefix)
```

### Basic Implementation in ASP.NET Core

```csharp
// Install NuGet package:
// Install-Package Asp.Versioning.Mvc

using Asp.Versioning;

// Program.cs - Configure API versioning
var builder = WebApplication.CreateBuilder(args);

builder.Services.AddApiVersioning(options =>
{
    // Default version when none is specified
    options.DefaultApiVersion = new ApiVersion(1, 0);
    
    // Assume default version when client doesn't specify
    options.AssumeDefaultVersionWhenUnspecified = true;
    
    // Report supported/deprecated versions in response headers
    options.ReportApiVersions = true;
    
    // Read version from URL path
    options.ApiVersionReader = new UrlSegmentApiVersionReader();
})
.AddApiExplorer(options =>
{
    // Format version as 'v'major[.minor]
    options.GroupNameFormat = "'v'VVV";
    
    // Substitute version in route template
    options.SubstituteApiVersionInUrl = true;
});

builder.Services.AddControllers();

var app = builder.Build();
app.MapControllers();
app.Run();
```

### Version 1 Controller

```csharp
using Asp.Versioning;
using Microsoft.AspNetCore.Mvc;

[ApiController]
[Route("api/v{version:apiVersion}/products")]
[ApiVersion("1.0")]
public class ProductsV1Controller : ControllerBase
{
    private readonly IProductService _productService;
    
    public ProductsV1Controller(IProductService productService)
    {
        _productService = productService;
    }
    
    // GET /api/v1/products
    [HttpGet]
    public ActionResult<IEnumerable<ProductV1Dto>> GetProducts()
    {
        var products = _productService.GetAllProducts();
        
        // V1 returns simple product model
        return Ok(products.Select(p => new ProductV1Dto
        {
            Id = p.Id,
            Name = p.Name,
            Price = p.Price
        }));
    }
    
    // GET /api/v1/products/{id}
    [HttpGet("{id}")]
    public ActionResult<ProductV1Dto> GetProduct(int id)
    {
        var product = _productService.GetProductById(id);
        
        if (product == null)
            return NotFound();
        
        return Ok(new ProductV1Dto
        {
            Id = product.Id,
            Name = product.Name,
            Price = product.Price
        });
    }
    
    // POST /api/v1/products
    [HttpPost]
    public ActionResult<ProductV1Dto> CreateProduct([FromBody] CreateProductV1Request request)
    {
        var product = _productService.CreateProduct(request);
        
        return CreatedAtAction(
            nameof(GetProduct),
            new { id = product.Id, version = "1.0" },
            new ProductV1Dto
            {
                Id = product.Id,
                Name = product.Name,
                Price = product.Price
            });
    }
}

// V1 DTOs
public class ProductV1Dto
{
    public int Id { get; set; }
    public string Name { get; set; }
    public decimal Price { get; set; }
    // Note: No category, tags, or other fields that come in V2
}

public class CreateProductV1Request
{
    public string Name { get; set; }
    public decimal Price { get; set; }
}
```

### Version 2 Controller (Breaking Changes)

```csharp
[ApiController]
[Route("api/v{version:apiVersion}/products")]
[ApiVersion("2.0")]
public class ProductsV2Controller : ControllerBase
{
    private readonly IProductService _productService;
    
    public ProductsV2Controller(IProductService productService)
    {
        _productService = productService;
    }
    
    // GET /api/v2/products
    [HttpGet]
    public ActionResult<IEnumerable<ProductV2Dto>> GetProducts(
        [FromQuery] string? category = null,
        [FromQuery] int page = 1,
        [FromQuery] int pageSize = 20)
    {
        // V2 adds pagination and filtering
        var products = _productService.GetProducts(category, page, pageSize);
        
        return Ok(products.Select(p => new ProductV2Dto
        {
            Id = p.Id,
            Name = p.Name,
            Price = p.Price,
            Category = p.Category,      // New in V2
            Tags = p.Tags,               // New in V2
            CreatedAt = p.CreatedAt      // New in V2
        }));
    }
    
    // GET /api/v2/products/{id}
    [HttpGet("{id}")]
    public ActionResult<ProductV2Dto> GetProduct(int id)
    {
        var product = _productService.GetProductById(id);
        
        if (product == null)
            return NotFound();
        
        return Ok(new ProductV2Dto
        {
            Id = product.Id,
            Name = product.Name,
            Price = p.Price,
            Category = product.Category,
            Tags = product.Tags,
            CreatedAt = product.CreatedAt
        });
    }
    
    // POST /api/v2/products
    [HttpPost]
    public ActionResult<ProductV2Dto> CreateProduct([FromBody] CreateProductV2Request request)
    {
        // V2 requires category (breaking change from V1)
        if (string.IsNullOrEmpty(request.Category))
            return BadRequest("Category is required in API v2");
        
        var product = _productService.CreateProduct(request);
        
        return CreatedAtAction(
            nameof(GetProduct),
            new { id = product.Id, version = "2.0" },
            MapToV2Dto(product));
    }
}

// V2 DTOs
public class ProductV2Dto
{
    public int Id { get; set; }
    public string Name { get; set; }
    public decimal Price { get; set; }
    public string Category { get; set; }     // New in V2
    public List<string> Tags { get; set; }   // New in V2
    public DateTime CreatedAt { get; set; }  // New in V2
}

public class CreateProductV2Request
{
    public string Name { get; set; }
    public decimal Price { get; set; }
    public string Category { get; set; }     // Required in V2
    public List<string>? Tags { get; set; }
}
```

### Supporting Multiple Versions on Same Controller

```csharp
// Alternative: Single controller supporting multiple versions
[ApiController]
[Route("api/v{version:apiVersion}/products")]
[ApiVersion("1.0")]
[ApiVersion("2.0")]
public class ProductsController : ControllerBase
{
    // This method only in V1
    [MapToApiVersion("1.0")]
    [HttpGet]
    public ActionResult<IEnumerable<ProductV1Dto>> GetProductsV1()
    {
        // V1 implementation
    }
    
    // This method only in V2
    [MapToApiVersion("2.0")]
    [HttpGet]
    public ActionResult<IEnumerable<ProductV2Dto>> GetProductsV2(
        [FromQuery] string? category = null)
    {
        // V2 implementation with new parameters
    }
    
    // This method in both versions (same behavior)
    [HttpGet("{id}")]
    public ActionResult GetProduct(int id, ApiVersion version)
    {
        var product = _productService.GetProductById(id);
        
        if (product == null)
            return NotFound();
        
        // Return appropriate DTO based on version
        if (version.MajorVersion == 1)
            return Ok(MapToV1Dto(product));
        else
            return Ok(MapToV2Dto(product));
    }
}
```

### URL Generation with Versioning

```csharp
public class ProductsController : ControllerBase
{
    private readonly IUrlHelper _urlHelper;
    
    [HttpGet("{id}")]
    public ActionResult<ProductDto> GetProduct(int id, ApiVersion version)
    {
        var product = _productService.GetProductById(id);
        
        if (product == null)
            return NotFound();
        
        var dto = MapToDto(product);
        
        // Generate versioned URLs for related resources
        dto.Links = new
        {
            Self = _urlHelper.Action("GetProduct", new { id, version = version.ToString() }),
            Orders = _urlHelper.Action("GetProductOrders", new { id, version = version.ToString() }),
            Reviews = _urlHelper.Action("GetProductReviews", new { id, version = version.ToString() })
        };
        
        return Ok(dto);
    }
}

// Response includes versioned URLs:
{
  "id": 123,
  "name": "Product",
  "price": 29.99,
  "links": {
    "self": "/api/v2/products/123",
    "orders": "/api/v2/products/123/orders",
    "reviews": "/api/v2/products/123/reviews"
  }
}
```

### Advantages and Disadvantages

```
✅ Advantages of URL Path Versioning:

1. Discoverability
   - Version visible in URL
   - Easy to share and bookmark
   - Obvious to anyone looking at logs or URLs
   - Great for documentation

2. Browser-friendly
   - Can test directly in browser
   - No special headers needed
   - Works with simple HTTP clients
   - Easy to debug

3. Caching
   - Each version has unique URL
   - Standard HTTP caching works perfectly
   - CDNs cache easily
   - No Vary header needed

4. API Gateway routing
   - Easy to route by path
   - Simple load balancing
   - Version-specific infrastructure
   - Traffic analysis straightforward

5. Developer experience
   - Intuitive and obvious
   - Matches common patterns
   - Easy to document
   - Less error-prone

❌ Disadvantages:

1. URL proliferation
   - /api/v1/products, /api/v2/products
   - Duplicate route definitions
   - More configuration

2. Less "RESTful"
   - Resource has multiple URLs
   - Violates single URI principle
   - Version in URL not pure REST

3. Code duplication potential
   - May need separate controllers
   - Can lead to copy-paste code
   - Requires discipline to share code

4. Breaking URL contracts
   - URL changes break clients
   - Bookmarks may break
   - Can't change URL pattern easily

When to use URL path versioning:
✅ Public APIs for external developers
✅ Simple, discoverable versioning needed
✅ Browser-based testing important
✅ Standard caching required
✅ Most common choice (industry standard)
```

### Shared Code Between Versions

```csharp
// Avoid code duplication by sharing business logic
public interface IProductService
{
    Task<IEnumerable<Product>> GetAllProducts();
    Task<IEnumerable<Product>> GetProducts(string? category, int page, int pageSize);
    Task<Product?> GetProductById(int id);
    Task<Product> CreateProduct(CreateProductCommand command);
}

// Separate mappers for each version
public class ProductV1Mapper
{
    public static ProductV1Dto MapToV1Dto(Product product)
    {
        return new ProductV1Dto
        {
            Id = product.Id,
            Name = product.Name,
            Price = product.Price
        };
    }
}

public class ProductV2Mapper
{
    public static ProductV2Dto MapToV2Dto(Product product)
    {
        return new ProductV2Dto
        {
            Id = product.Id,
            Name = product.Name,
            Price = product.Price,
            Category = product.Category,
            Tags = product.Tags,
            CreatedAt = product.CreatedAt
        };
    }
}

// Controllers use shared service, different mappers
[ApiVersion("1.0")]
public class ProductsV1Controller : ControllerBase
{
    private readonly IProductService _service;
    
    [HttpGet]
    public async Task<ActionResult<IEnumerable<ProductV1Dto>>> GetProducts()
    {
        var products = await _service.GetAllProducts();
        return Ok(products.Select(ProductV1Mapper.MapToV1Dto));
    }
}

[ApiVersion("2.0")]
public class ProductsV2Controller : ControllerBase
{
    private readonly IProductService _service;  // Same service!
    
    [HttpGet]
    public async Task<ActionResult<IEnumerable<ProductV2Dto>>> GetProducts(
        [FromQuery] string? category = null,
        [FromQuery] int page = 1,
        [FromQuery] int pageSize = 20)
    {
        var products = await _service.GetProducts(category, page, pageSize);
        return Ok(products.Select(ProductV2Mapper.MapToV2Dto));
    }
}
```

## Self-Check Questions

1. **What are the advantages of URL path versioning?**
   - Most discoverable (version in URL)
   - Browser-friendly testing
   - Simple HTTP caching
   - Easy API Gateway routing
   - Industry standard approach

2. **When should you increment the major version vs minor version?**
   - Major (v2.0): Breaking changes (removed fields, changed types, new required fields)
   - Minor (v1.1): Backward-compatible additions (new optional fields, new endpoints)
   - Patch (v1.0.1): Bug fixes without contract changes

3. **How do you handle routes that exist in multiple versions?**
   - Use separate controllers (ProductsV1Controller, ProductsV2Controller)
   - Or use single controller with [MapToApiVersion("1.0")] attributes
   - Share business logic, separate DTOs/mappers

4. **What happens when a client requests an unsupported version?**
   - Returns 400 Bad Request
   - Error message indicates unsupported version
   - Response headers show supported versions
   - Can configure default version fallback

5. **How do you generate URLs that include the version?**
   - Use Url.Action with version parameter
   - Include version in route values
   - Example: `Url.Action("GetProduct", new { id = 123, version = "2.0" })`

6. **Should you use separate controllers or one controller with version mapping?**
   - Separate controllers: Clearer separation, easier to maintain, better for major versions
   - Single controller with mapping: Less code duplication, better for minor versions
   - Choose based on how different the versions are
