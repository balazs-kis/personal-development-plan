## Introduction

Resolvers are functions that fetch the data for each field in your GraphQL schema. The GraphQL execution engine calls resolvers for each field in a query, passing along the parent object, arguments, context, and field information. Understanding resolvers is key to implementing efficient GraphQL APIs.

## Key Concepts Summary

### Basic Resolver Structure

```typescript
// Resolver signature
type Resolver = (
  parent: any,
  args: any,
  context: any,
  info: GraphQLResolveInfo
) => any

// Example resolvers
const resolvers = {
  Query: {
    user: async (parent, { id }, context, info) => {
      return context.db.users.findById(id);
    },
    
    users: async (parent, { first, after }, context) => {
      return context.db.users.findMany({
        limit: first,
        cursor: after
      });
    }
  },
  
  User: {
    posts: async (user, { first }, context) => {
      return context.db.posts.findByAuthorId(user.id, { limit: first });
    },
    
    email: (user, args, context) => {
      // Check authorization
      if (context.user?.id !== user.id && !context.user?.isAdmin) {
        return null;
      }
      return user.email;
    }
  },
  
  Mutation: {
    createPost: async (parent, { input }, context) => {
      if (!context.user) {
        throw new Error('Unauthorized');
      }
      
      return context.db.posts.create({
        ...input,
        authorId: context.user.id
      });
    }
  }
};
```

### Execution Flow

```
Query:
{
  user(id: "123") {
    name
    posts {
      title
    }
  }
}

Execution:
1. Query.user(id: "123") → returns User object
2. User.name(user) → returns "John"
3. User.posts(user) → returns [Post, Post]
4. Post.title(post1) → returns "Title 1"
5. Post.title(post2) → returns "Title 2"
```

### Resolver Context

```typescript
// Creating context
const server = new ApolloServer({
  typeDefs,
  resolvers,
  context: async ({ req }) => {
    // Extract auth token
    const token = req.headers.authorization?.replace('Bearer ', '');
    
    // Verify and decode token
    const user = token ? await verifyToken(token) : null;
    
    // Return context object
    return {
      user,
      db: database,
      loaders: createDataLoaders(database),
      req
    };
  }
});

// Using context in resolvers
const resolvers = {
  Query: {
    me: (parent, args, context) => {
      // Access user from context
      if (!context.user) {
        throw new Error('Not authenticated');
      }
      return context.user;
    },
    
    posts: async (parent, { first }, context) => {
      // Access db from context
      return context.db.posts.findMany({ limit: first });
    }
  }
};
```

### Field-Level Resolvers

```typescript
const resolvers = {
  // Root resolvers
  Query: {
    user: (parent, { id }, { db }) => db.users.findById(id)
  },
  
  // Field resolvers for User type
  User: {
    // Explicit resolver (transforms data)
    email: (user, args, context) => {
      // Hide email from non-owners
      if (context.user?.id !== user.id && !context.user?.isAdmin) {
        return null;
      }
      return user.email;
    },
    
    // Resolver with async data fetching
    posts: async (user, { first, after }, { db }) => {
      return db.posts.findByAuthorId(user.id, {
        limit: first,
        cursor: after
      });
    },
    
    // Computed field
    fullName: (user) => {
      return `${user.firstName} ${user.lastName}`;
    },
    
    // Count field (aggregate)
    postCount: async (user, args, { db }) => {
      return db.posts.countByAuthorId(user.id);
    },
    
    // Default resolver (returns user.name)
    // name: (user) => user.name
  }
};
```

### Async Resolvers

```typescript
const resolvers = {
  Query: {
    // Promise-based resolver
    user: (parent, { id }, { db }) => {
      return db.users.findById(id);  // Returns Promise<User>
    },
    
    // Async/await resolver
    users: async (parent, { first }, { db }) => {
      const users = await db.users.findMany({ limit: first });
      return users;
    },
    
    // Multiple async operations
    dashboard: async (parent, args, { db, user }) => {
      if (!user) throw new Error('Not authenticated');
      
      const [posts, followers, notifications] = await Promise.all([
        db.posts.findByAuthorId(user.id),
        db.follows.getFollowers(user.id),
        db.notifications.findByUserId(user.id, { unread: true })
      ]);
      
      return { posts, followers, notifications };
    }
  }
};
```

### Resolver Arguments

```typescript
// Four resolver arguments
type Resolver = (
  parent: any,       // Parent object (from parent resolver)
  args: any,         // Field arguments
  context: any,      // Shared context
  info: GraphQLResolveInfo  // Query info (advanced)
) => any

// Example using all arguments
const resolvers = {
  Query: {
    posts: async (parent, args, context, info) => {
      // args: field arguments
      const { first, after, filter } = args;
      
      // context: shared data
      const { db, user } = context;
      
      // info: query information (rarely used)
      const requestedFields = getFieldNames(info);
      
      // Optimize based on requested fields
      if (requestedFields.includes('author')) {
        return db.posts.findManyWithAuthor(first, after, filter);
      }
      
      return db.posts.findMany(first, after, filter);
    }
  },
  
  Post: {
    author: async (post, args, context) => {
      // parent: Post object from Query.posts
      return context.loaders.userLoader.load(post.authorId);
    }
  }
};
```

### Error Handling in Resolvers

```typescript
import { GraphQLError } from 'graphql';

const resolvers = {
  Query: {
    user: async (parent, { id }, { db }) => {
      const user = await db.users.findById(id);
      
      if (!user) {
        // Throw GraphQL error
        throw new GraphQLError('User not found', {
          extensions: {
            code: 'NOT_FOUND',
            userId: id
          }
        });
      }
      
      return user;
    }
  },
  
  Mutation: {
    createPost: async (parent, { input }, { user, db }) => {
      // Authentication error
      if (!user) {
        throw new GraphQLError('Not authenticated', {
          extensions: { code: 'UNAUTHENTICATED' }
        });
      }
      
      // Validation error
      if (!input.title || input.title.length < 5) {
        throw new GraphQLError('Title must be at least 5 characters', {
          extensions: {
            code: 'BAD_USER_INPUT',
            field: 'title'
          }
        });
      }
      
      try {
        return await db.posts.create({
          ...input,
          authorId: user.id
        });
      } catch (error) {
        // Database error
        throw new GraphQLError('Failed to create post', {
          extensions: {
            code: 'INTERNAL_SERVER_ERROR',
            originalError: error.message
          }
        });
      }
    }
  }
};
```

### Resolver Chains

```typescript
// Schema
type Query {
  post(id: ID!): Post
}

type Post {
  id: ID!
  title: String!
  author: User!
}

type User {
  id: ID!
  name: String!
  email: String!
}

// Resolvers with chaining
const resolvers = {
  Query: {
    post: async (parent, { id }, { db }) => {
      // Step 1: Fetch post
      const post = await db.posts.findById(id);
      console.log('Query.post:', post);
      return post;  // { id: "1", title: "...", authorId: "123" }
    }
  },
  
  Post: {
    // Step 2: post object passed as parent
    author: async (post, args, { loaders }) => {
      console.log('Post.author parent:', post);
      const author = await loaders.userLoader.load(post.authorId);
      return author;  // { id: "123", name: "...", email: "..." }
    }
  },
  
  User: {
    // Step 3: author object passed as parent
    email: (user, args, { currentUser }) => {
      console.log('User.email parent:', user);
      // Authorization logic
      if (currentUser?.id !== user.id) {
        return null;
      }
      return user.email;
    }
  }
};

// Query execution flow:
// 1. Query.post(id: "1") → { id: "1", title: "...", authorId: "123" }
// 2. Post.author(post) → { id: "123", name: "John", email: "john@..." }
// 3. User.email(user) → "john@example.com" (or null if unauthorized)
```

### Default Resolvers

```typescript
// GraphQL provides default resolvers for simple fields
type User {
  id: ID!
  name: String!
  email: String!
}

// These resolvers are automatic (you don't need to write them):
const defaultResolvers = {
  User: {
    id: (user) => user.id,
    name: (user) => user.name,
    email: (user) => user.email
  }
};

// Only override when you need custom logic:
const customResolvers = {
  User: {
    // Default works: id, name
    
    // Override for custom logic
    email: (user, args, { currentUser }) => {
      if (currentUser?.id !== user.id) {
        return null;  // Hide email
      }
      return user.email;
    }
  }
};
```

### Resolver Best Practices

```typescript
// ✅ Good: Thin resolvers, logic in services
const resolvers = {
  Query: {
    user: async (parent, { id }, { services }) => {
      return services.userService.getById(id);
    }
  },
  
  Mutation: {
    createPost: async (parent, { input }, { services, user }) => {
      return services.postService.create(input, user);
    }
  }
};

// ❌ Bad: Fat resolvers with business logic
const badResolvers = {
  Mutation: {
    createPost: async (parent, { input }, { db, user }) => {
      // Don't put all logic in resolver
      if (!user) throw new Error('Unauthorized');
      if (!input.title) throw new Error('Title required');
      const sanitized = sanitizeHtml(input.content);
      const slug = generateSlug(input.title);
      const post = await db.posts.create({...});
      await db.search.index(post);
      await emailService.notify(post);
      return post;
    }
  }
};

// ✅ Better: Extract to service layer
class PostService {
  async create(input, user) {
    this.validateUser(user);
    this.validateInput(input);
    
    const post = await this.db.posts.create({
      ...input,
      content: this.sanitizeHtml(input.content),
      slug: this.generateSlug(input.title),
      authorId: user.id
    });
    
    await this.searchService.index(post);
    await this.notificationService.notifyFollowers(post);
    
    return post;
  }
}
```

### Parallel Execution

```typescript
// GraphQL executes sibling fields in parallel
const resolvers = {
  Query: {
    dashboard: async (parent, args, { services }) => {
      // These run sequentially (waterfall)
      const user = await services.userService.getMe();
      const posts = await services.postService.getByAuthor(user.id);
      const followers = await services.userService.getFollowers(user.id);
      
      return { user, posts, followers };
    }
  },
  
  Dashboard: {
    // These run in parallel!
    user: (dashboard) => dashboard.user,
    posts: (dashboard) => dashboard.posts,
    followers: (dashboard) => dashboard.followers
  }
};

// Better: Return early, let field resolvers handle fetching
const betterResolvers = {
  Query: {
    dashboard: (parent, args, { user }) => {
      // Just return user ID
      return { userId: user.id };
    }
  },
  
  Dashboard: {
    // These all run in parallel automatically
    user: ({ userId }, args, { services }) => {
      return services.userService.getById(userId);
    },
    
    posts: ({ userId }, args, { services }) => {
      return services.postService.getByAuthor(userId);
    },
    
    followers: ({ userId }, args, { services }) => {
      return services.userService.getFollowers(userId);
    }
  }
};
```

### Subscription Resolvers

```typescript
import { PubSub } from 'graphql-subscriptions';

const pubsub = new PubSub();

const resolvers = {
  Mutation: {
    createPost: async (parent, { input }, { user, db }) => {
      const post = await db.posts.create({
        ...input,
        authorId: user.id
      });
      
      // Publish event
      pubsub.publish('POST_CREATED', { postCreated: post });
      
      return post;
    }
  },
  
  Subscription: {
    postCreated: {
      subscribe: () => pubsub.asyncIterator(['POST_CREATED'])
    },
    
    // Subscription with filter
    commentAdded: {
      subscribe: withFilter(
        () => pubsub.asyncIterator(['COMMENT_ADDED']),
        (payload, variables) => {
          // Only send if comment is on the specified post
          return payload.commentAdded.postId === variables.postId;
        }
      )
    }
  }
};
```
