---

## Introduction

GraphQL's Schema Definition Language (SDL) is a human-readable syntax for defining your API's type system. The schema serves as a contract between client and server, describing exactly what data can be queried and what operations are available. Unlike REST where the shape of responses is implicit, GraphQL makes the API structure explicit and type-safe.

## Key Concepts Summary

### Basic Schema Structure

```graphql
# Schema definition - entry points for queries and mutations
schema {
  query: Query
  mutation: Mutation
  subscription: Subscription
}

# Query type - all read operations
type Query {
  user(id: ID!): User
  users(first: Int, after: String): UserConnection!
  posts(authorId: ID): [Post!]!
}

# Mutation type - all write operations
type Mutation {
  createUser(input: CreateUserInput!): User!
  updateUser(id: ID!, input: UpdateUserInput!): User!
  deleteUser(id: ID!): Boolean!
}

# Subscription type - real-time updates
type Subscription {
  postCreated: Post!
  userUpdated(id: ID!): User!
}
```

### Object Types

```graphql
# Basic object type
type User {
  id: ID!              # Non-null ID
  email: String!       # Non-null String
  name: String         # Nullable String
  age: Int
  isActive: Boolean!
  role: UserRole!      # Enum type
  posts: [Post!]!      # Non-null list of non-null Posts
  createdAt: DateTime! # Custom scalar
}

# Object with relationships
type Post {
  id: ID!
  title: String!
  content: String!
  published: Boolean!
  author: User!        # Reference to User type
  comments: [Comment!]!
  tags: [String!]!
  viewCount: Int!
  createdAt: DateTime!
  updatedAt: DateTime!
}

# Nested object
type Comment {
  id: ID!
  content: String!
  author: User!
  post: Post!
  createdAt: DateTime!
}
```

### Scalars

```graphql
# Built-in scalars
type Example {
  intField: Int         # Signed 32-bit integer
  floatField: Float     # Signed double-precision floating-point
  stringField: String   # UTF-8 character sequence
  booleanField: Boolean # true or false
  idField: ID           # Unique identifier (serialized as String)
}

# Custom scalars (defined elsewhere in implementation)
scalar DateTime
scalar Email
scalar URL
scalar JSON
scalar Decimal

# Usage of custom scalars
type User {
  id: ID!
  email: Email!
  website: URL
  createdAt: DateTime!
  metadata: JSON
  balance: Decimal!
}
```

### Enums

```graphql
# Enum type
enum UserRole {
  ADMIN
  MODERATOR
  USER
  GUEST
}

enum PostStatus {
  DRAFT
  PUBLISHED
  ARCHIVED
}

enum SortOrder {
  ASC
  DESC
}

# Usage in types
type User {
  id: ID!
  role: UserRole!
  status: AccountStatus!
}

# Usage in arguments
type Query {
  users(
    role: UserRole
    sortBy: UserSortField
    sortOrder: SortOrder
  ): [User!]!
}
```

### Interfaces

```graphql
# Interface definition
interface Node {
  id: ID!
  createdAt: DateTime!
  updatedAt: DateTime!
}

# Types implementing interface
type User implements Node {
  id: ID!
  createdAt: DateTime!
  updatedAt: DateTime!
  email: String!
  name: String!
}

type Post implements Node {
  id: ID!
  createdAt: DateTime!
  updatedAt: DateTime!
  title: String!
  content: String!
  author: User!
}

# Query returning interface
type Query {
  node(id: ID!): Node
  nodes(ids: [ID!]!): [Node!]!
}

# Multiple interfaces
interface Timestamps {
  createdAt: DateTime!
  updatedAt: DateTime!
}

interface Authored {
  author: User!
}

type BlogPost implements Node & Timestamps & Authored {
  id: ID!
  createdAt: DateTime!
  updatedAt: DateTime!
  author: User!
  title: String!
  content: String!
}
```

### Unions

```graphql
# Union type
union SearchResult = User | Post | Comment

# Query returning union
type Query {
  search(query: String!): [SearchResult!]!
}

# Another example
union MediaContent = Image | Video | Document

type Image {
  url: String!
  width: Int!
  height: Int!
  alt: String
}

type Video {
  url: String!
  duration: Int!
  thumbnail: String!
}

type Document {
  url: String!
  filename: String!
  fileSize: Int!
}

type Post {
  id: ID!
  title: String!
  media: [MediaContent!]!
}
```

### Input Types

```graphql
# Input type for mutations
input CreateUserInput {
  email: String!
  name: String!
  password: String!
  role: UserRole
}

input UpdateUserInput {
  email: String
  name: String
  role: UserRole
}

# Nested input
input CreatePostInput {
  title: String!
  content: String!
  published: Boolean
  tags: [String!]
  metadata: PostMetadataInput
}

input PostMetadataInput {
  category: String!
  featured: Boolean
  allowComments: Boolean
}

# Usage in mutations
type Mutation {
  createUser(input: CreateUserInput!): User!
  updateUser(id: ID!, input: UpdateUserInput!): User!
  createPost(input: CreatePostInput!): Post!
}

# Input for pagination and filtering
input UserFilterInput {
  role: UserRole
  isActive: Boolean
  searchTerm: String
}

input PaginationInput {
  first: Int
  after: String
  last: Int
  before: String
}

type Query {
  users(
    filter: UserFilterInput
    pagination: PaginationInput
  ): UserConnection!
}
```

### Lists and Non-Null

```graphql
type Examples {
  # Nullable list, nullable items
  field1: [String]
  # Non-null list, nullable items
  field2: [String]!
  # Nullable list, non-null items
  field3: [String!]
  # Non-null list, non-null items
  field4: [String!]!
}

# Practical examples
type User {
  # User might not have posts (null), or have empty array
  posts: [Post!]
  # User always has array of tags (can be empty)
  tags: [String!]!
  # User's emails are all valid (no nulls in array)
  emails: [Email!]!
}
```

### Field Arguments

```graphql
type Query {
  # Single argument
  user(id: ID!): User
  
  # Multiple arguments
  users(
    first: Int = 10
    after: String
    role: UserRole
    isActive: Boolean = true
  ): UserConnection!
  
  # Complex arguments
  posts(
    filter: PostFilterInput
    sort: PostSortInput
    pagination: PaginationInput
  ): PostConnection!
}

# Arguments with defaults
type User {
  posts(
    first: Int = 10
    includeD rafts: Boolean = false
    sortBy: PostSortField = CREATED_AT
    sortOrder: SortOrder = DESC
  ): [Post!]!
}
```

### Directives

```graphql
# Built-in directives
type Query {
  user(id: ID!): User
  deprecatedField: String @deprecated(reason: "Use `user` instead")
}

type User {
  id: ID!
  name: String!
  email: String @deprecated(reason: "Use `emailAddress`")
  emailAddress: String!
}

# Conditional inclusion
query GetUser($includeEmail: Boolean!) {
  user(id: "123") {
    id
    name
    email @include(if: $includeEmail)
  }
}

query GetUser($skipEmail: Boolean!) {
  user(id: "123") {
    id
    name
    email @skip(if: $skipEmail)
  }
}

# Custom directives (implementation-specific)
type User {
  email: String! @auth(requires: ADMIN)
  posts: [Post!]! @cacheControl(maxAge: 60)
}
```

### Schema Documentation

```graphql
"""
Main query type for read operations
"""
type Query {
  """
  Get currently authenticated user
  """
  me: User
  
  """
  Fetch a user by ID
  """
  user(
    "User ID"
    id: ID!
  ): User
  
  """
  List users with pagination and filtering
  """
  users(
    "Number of items to fetch"
    first: Int
    "Cursor for pagination"
    after: String
    "Filter by role"
    role: UserRole
  ): UserConnection!
}

# Using triple-quote comments for documentation
type User {
  "Unique identifier"
  id: ID!
  "User's email address"
  email: Email!
  "Display name"
  name: String!
}
```

### Schema Design Patterns

**Pagination (Relay-style Connections):**

```graphql
type UserConnection {
  edges: [UserEdge!]!
  pageInfo: PageInfo!
  totalCount: Int!
}

type UserEdge {
  node: User!
  cursor: String!
}

type PageInfo {
  hasNextPage: Boolean!
  hasPreviousPage: Boolean!
  startCursor: String
  endCursor: String
}

type Query {
  users(
    first: Int
    after: String
    last: Int
    before: String
  ): UserConnection!
}
```

**Error Handling:**

```graphql
# Result type pattern
type CreateUserResult {
  user: User
  errors: [UserError!]!
}

type UserError {
  field: String
  message: String!
  code: ErrorCode!
}

enum ErrorCode {
  VALIDATION_ERROR
  NOT_FOUND
  UNAUTHORIZED
  FORBIDDEN
}

type Mutation {
  createUser(input: CreateUserInput!): CreateUserResult!
}

# Union pattern
union CreateUserResult = User | ValidationError | NotFoundError

type ValidationError {
  fields: [FieldError!]!
}

type FieldError {
  field: String!
  message: String!
}

type NotFoundError {
  message: String!
  resourceType: String!
  resourceId: ID!
}
```

**Filtering and Sorting:**

```graphql
input PostFilterInput {
  status: PostStatus
  authorId: ID
  tags: [String!]
  searchTerm: String
  createdAfter: DateTime
  createdBefore: DateTime
}

enum PostSortField {
  CREATED_AT
  UPDATED_AT
  TITLE
  VIEW_COUNT
}

input PostSortInput {
  field: PostSortField!
  order: SortOrder!
}

enum SortOrder {
  ASC
  DESC
}

type Query {
  posts(
    filter: PostFilterInput
    sort: PostSortInput
    first: Int
    after: String
  ): PostConnection!
}
```

**Polymorphic Types:**

```graphql
# Using interfaces for common fields
interface Notification {
  id: ID!
  recipient: User!
  message: String!
  read: Boolean!
  createdAt: DateTime!
}

type CommentNotification implements Notification {
  id: ID!
  recipient: User!
  message: String!
  read: Boolean!
  createdAt: DateTime!
  comment: Comment!
  post: Post!
}

type LikeNotification implements Notification {
  id: ID!
  recipient: User!
  message: String!
  read: Boolean!
  createdAt: DateTime!
  post: Post!
  likedBy: User!
}

type FollowNotification implements Notification {
  id: ID!
  recipient: User!
  message: String!
  read: Boolean!
  createdAt: DateTime!
  follower: User!
}

type Query {
  notifications(unreadOnly: Boolean): [Notification!]!
}
```

**Batch Operations:**

```graphql
type Mutation {
  # Single operation
  createPost(input: CreatePostInput!): Post!
  
  # Batch operation
  createPosts(inputs: [CreatePostInput!]!): [PostResult!]!
  
  # Batch delete
  deletePosts(ids: [ID!]!): BatchDeleteResult!
}

type PostResult {
  post: Post
  error: Error
}

type BatchDeleteResult {
  deletedIds: [ID!]!
  errors: [Error!]!
  successCount: Int!
  failureCount: Int!
}
```

### Schema Validation Rules

```graphql
# Name conventions
type User {  # PascalCase for types
  firstName: String!  # camelCase for fields
}

enum UserRole {
  ADMIN  # SCREAMING_SNAKE_CASE for enum values
  MODERATOR
  USER
}

# Best practices
type User {
  posts: [Post!]!  # ✅ Plural for lists
  isActive: Boolean!  # ✅ Clear boolean
  email: String!  # ✅ Simple, descriptive
}
```

### Complete Example Schema

```graphql
schema {
  query: Query
  mutation: Mutation
  subscription: Subscription
}

# Custom scalars
scalar DateTime
scalar Email
scalar URL
scalar JSON

# Enums
enum UserRole {
  ADMIN
  MODERATOR
  USER
  GUEST
}

enum PostStatus {
  DRAFT
  PUBLISHED
  ARCHIVED
}

enum SortOrder {
  ASC
  DESC
}

# Interfaces
interface Node {
  id: ID!
}

interface Timestamps {
  createdAt: DateTime!
  updatedAt: DateTime!
}

# Types
type User implements Node & Timestamps {
  id: ID!
  email: Email!
  name: String!
  bio: String
  avatar: URL
  role: UserRole!
  isActive: Boolean!
  posts(
    first: Int
    after: String
    status: PostStatus
  ): PostConnection!
  followers: UserConnection!
  following: UserConnection!
  createdAt: DateTime!
  updatedAt: DateTime!
}

type Post implements Node & Timestamps {
  id: ID!
  title: String!
  content: String!
  excerpt: String
  status: PostStatus!
  published: Boolean!
  author: User!
  comments: [Comment!]!
  likes: [User!]!
  tags: [String!]!
  viewCount: Int!
  likeCount: Int!
  commentCount: Int!
  createdAt: DateTime!
  updatedAt: DateTime!
  publishedAt: DateTime
}

type Comment implements Node & Timestamps {
  id: ID!
  content: String!
  author: User!
  post: Post!
  replies: [Comment!]!
  parent: Comment
  createdAt: DateTime!
  updatedAt: DateTime!
}

# Connections for pagination
type UserConnection {
  edges: [UserEdge!]!
  pageInfo: PageInfo!
  totalCount: Int!
}

type UserEdge {
  node: User!
  cursor: String!
}

type PostConnection {
  edges: [PostEdge!]!
  pageInfo: PageInfo!
  totalCount: Int!
}

type PostEdge {
  node: Post!
  cursor: String!
}

type PageInfo {
  hasNextPage: Boolean!
  hasPreviousPage: Boolean!
  startCursor: String
  endCursor: String
}

# Input types
input CreateUserInput {
  email: Email!
  name: String!
  password: String!
  bio: String
  role: UserRole = USER
}

input UpdateUserInput {
  email: Email
  name: String
  bio: String
  avatar: URL
}

input CreatePostInput {
  title: String!
  content: String!
  excerpt: String
  status: PostStatus = DRAFT
  tags: [String!]
}

input UpdatePostInput {
  title: String
  content: String
  excerpt: String
  status: PostStatus
  tags: [String!]
}

input PostFilterInput {
  status: PostStatus
  authorId: ID
  tags: [String!]
  searchTerm: String
}

# Queries
type Query {
  # User queries
  me: User
  user(id: ID!): User
  users(
    first: Int
    after: String
    role: UserRole
  ): UserConnection!
  
  # Post queries
  post(id: ID!): Post
  posts(
    filter: PostFilterInput
    first: Int
    after: String
  ): PostConnection!
  
  # Search
  search(query: String!): [SearchResult!]!
}

# Search result union
union SearchResult = User | Post | Comment

# Mutations
type Mutation {
  # User mutations
  createUser(input: CreateUserInput!): User!
  updateUser(id: ID!, input: UpdateUserInput!): User!
  deleteUser(id: ID!): Boolean!
  
  # Post mutations
  createPost(input: CreatePostInput!): Post!
  updatePost(id: ID!, input: UpdatePostInput!): Post!
  deletePost(id: ID!): Boolean!
  publishPost(id: ID!): Post!
  
  # Interaction mutations
  likePost(postId: ID!): Post!
  unlikePost(postId: ID!): Post!
  followUser(userId: ID!): User!
  unfollowUser(userId: ID!): User!
}

# Subscriptions
type Subscription {
  postCreated: Post!
  postPublished: Post!
  commentAdded(postId: ID!): Comment!
  userFollowed(userId: ID!): User!
}
```

## Practice Exercise

Design a GraphQL schema for a simple e-commerce system:

**Requirements:**
1. **Types:**
   - Product (id, name, description, price, category, inStock)
   - Category (id, name, products)
   - Order (id, customer, items, total, status)
   - OrderItem (product, quantity, price)
   - Customer (id, name, email, orders)

2. **Enums:**
   - OrderStatus (PENDING, PROCESSING, SHIPPED, DELIVERED, CANCELLED)
   - ProductCategory

3. **Queries:**
   - Get product by ID
   - List products (with optional category filter)
   - Get customer by ID
   - Get order by ID

4. **Mutations:**
   - Create product
   - Update product stock
   - Create order
   - Update order status

5. **Features:**
   - Use interfaces for common fields (id, timestamps)
   - Use input types for mutations
   - Include pagination for product lists
   - Add appropriate non-null constraints
