# Lesson 2: Queries, Mutations, and Subscriptions

## Learning Objectives

By the end of this lesson, you will be able to:
- Write GraphQL queries to fetch data
- Use variables, fragments, and aliases
- Create mutations to modify data
- Understand GraphQL subscriptions for real-time updates
- Use operation names and directives

---

## Introduction

GraphQL provides three types of operations: queries for reading data, mutations for writing data, and subscriptions for real-time updates. Unlike REST where you might need multiple endpoints to get related data, GraphQL lets you request exactly what you need in a single request, reducing over-fetching and under-fetching.

## Key Concepts Summary

### Basic Queries

```graphql
# Simple query
{
  user(id: "123") {
    id
    name
    email
  }
}

# Response
{
  "data": {
    "user": {
      "id": "123",
      "name": "John Doe",
      "email": "john@example.com"
    }
  }
}

# Query with nested fields
{
  user(id: "123") {
    id
    name
    posts {
      id
      title
      comments {
        id
        content
        author {
          name
        }
      }
    }
  }
}
```

### Variables

```graphql
# Query with variables
query GetUser($userId: ID!, $includeEmail: Boolean = false) {
  user(id: $userId) {
    id
    name
    email @include(if: $includeEmail)
    posts(first: 10) {
      id
      title
    }
  }
}

# Variables (sent separately)
{
  "userId": "123",
  "includeEmail": true
}
```

### Mutations

```graphql
# Create mutation
mutation CreatePost($input: CreatePostInput!) {
  createPost(input: $input) {
    id
    title
    content
    author {
      id
      name
    }
    createdAt
  }
}

# Variables
{
  "input": {
    "title": "GraphQL Basics",
    "content": "Learning GraphQL...",
    "published": false
  }
}

# Update mutation
mutation UpdatePost($id: ID!, $input: UpdatePostInput!) {
  updatePost(id: $id, input: $input) {
    id
    title
    updatedAt
  }
}
```

### Subscriptions

```graphql
# Subscribe to new posts
subscription OnPostCreated {
  postCreated {
    id
    title
    author {
      name
    }
  }
}

# Subscribe with filter
subscription OnCommentAdded($postId: ID!) {
  commentAdded(postId: $postId) {
    id
    content
    author {
      name
    }
    createdAt
  }
}
```

### Fragments

```graphql
# Defining fragment
fragment UserFields on User {
  id
  name
  email
  role
}

# Using fragment
query GetUsers {
  users {
    ...UserFields
    posts {
      id
      title
    }
  }
}

# Inline fragments for unions/interfaces
query Search($query: String!) {
  search(query: $query) {
    ... on User {
      id
      name
      email
    }
    ... on Post {
      id
      title
      author {
        name
      }
    }
    ... on Comment {
      id
      content
    }
  }
}
```

### Directives

```graphql
# @include directive
query GetUser($includeEmail: Boolean!) {
  user(id: "123") {
    id
    name
    email @include(if: $includeEmail)
    posts {
      title
    }
  }
}

# @skip directive
query GetUser($skipPosts: Boolean!) {
  user(id: "123") {
    id
    name
    posts @skip(if: $skipPosts) {
      title
    }
  }
}

# Combining directives
query GetUser($includeEmail: Boolean!, $includePosts: Boolean!) {
  user(id: "123") {
    id
    name
    email @include(if: $includeEmail)
    posts @include(if: $includePosts) {
      title
      content
    }
  }
}

# Custom directives (server-defined)
query GetUser {
  user(id: "123") {
    id
    name
    email @auth(requires: ADMIN)
    posts @cacheControl(maxAge: 300) {
      title
    }
  }
}
```

### Advanced Mutations

```graphql
# Mutation with complex input
mutation CreateBlogPost($input: CreatePostInput!) {
  createPost(input: $input) {
    id
    title
    content
    status
    author {
      id
      name
    }
    tags
    createdAt
  }
}

# Variables
{
  "input": {
    "title": "GraphQL Best Practices",
    "content": "Here are some best practices...",
    "status": "DRAFT",
    "tags": ["graphql", "api", "best-practices"]
  }
}

# Multiple mutations in sequence
mutation CreatePostAndPublish($createInput: CreatePostInput!, $publishId: ID!) {
  createPost(input: $createInput) {
    id
    title
    status
  }
  
  publishPost(id: $publishId) {
    id
    title
    status
    publishedAt
  }
}

# Mutation with error handling pattern
mutation CreateUser($input: CreateUserInput!) {
  createUser(input: $input) {
    ... on User {
      id
      name
      email
    }
    ... on ValidationError {
      fields {
        field
        message
      }
    }
    ... on DuplicateEmailError {
      message
      existingUser {
        id
      }
    }
  }
}

# Optimistic updates pattern
mutation LikePost($postId: ID!) {
  likePost(postId: $postId) {
    id
    likeCount
    likes {
      id
      name
    }
  }
}
```

### Subscriptions

```graphql
# Basic subscription
subscription OnPostCreated {
  postCreated {
    id
    title
    content
    author {
      id
      name
    }
    createdAt
  }
}

# Subscription with filter
subscription OnCommentAdded($postId: ID!) {
  commentAdded(postId: $postId) {
    id
    content
    author {
      id
      name
      avatar
    }
    createdAt
  }
}

# Multiple subscriptions
subscription OnUserActivity($userId: ID!) {
  postCreated(authorId: $userId) {
    id
    title
  }
  
  commentAdded(authorId: $userId) {
    id
    content
    post {
      title
    }
  }
}

# Subscription with complex data
subscription OnNotification {
  notificationReceived {
    ... on CommentNotification {
      id
      message
      comment {
        content
        post {
          title
        }
      }
    }
    ... on LikeNotification {
      id
      message
      post {
        title
      }
      likedBy {
        name
      }
    }
    ... on FollowNotification {
      id
      message
      follower {
        name
        avatar
      }
    }
  }
}
```

### Query Optimization Techniques

```graphql
# Use fragments to reduce duplication
fragment PostSummary on Post {
  id
  title
  excerpt
  author {
    id
    name
  }
  createdAt
}

query GetPosts {
  recent: posts(first: 10, sortBy: CREATED_AT) {
    edges {
      node {
        ...PostSummary
      }
    }
  }
  
  trending: posts(first: 10, sortBy: VIEW_COUNT) {
    edges {
      node {
        ...PostSummary
        viewCount
      }
    }
  }
}

# Request only needed fields
query GetUserForDisplay {
  user(id: "123") {
    # Only fields needed for display
    name
    avatar
    # Not: email, password, etc.
  }
}

# Use pagination to limit data
query GetPosts($first: Int = 20, $after: String) {
  posts(first: $first, after: $after) {
    edges {
      node {
        id
        title
      }
      cursor
    }
    pageInfo {
      hasNextPage
      endCursor
    }
  }
}

# Batch related queries
query GetDashboard {
  me {
    id
    name
  }
  posts(first: 10) {
    edges {
      node {
        id
        title
      }
    }
  }
  notifications(unreadOnly: true) {
    id
    message
  }
}
```

### Query Complexity

```graphql
# Simple query - low complexity
query SimpleQuery {
  user(id: "123") {
    id
    name
  }
}

# Complex query - high complexity (may be rate limited)
query ComplexQuery {
  users(first: 100) {
    edges {
      node {
        id
        name
        posts(first: 100) {
          edges {
            node {
              id
              title
              comments(first: 100) {
                edges {
                  node {
                    id
                    content
                    author {
                      id
                      name
                      posts(first: 100) {
                        edges {
                          node {
                            id
                          }
                        }
                      }
                    }
                  }
                }
              }
            }
          }
        }
      }
    }
  }
}

# Better approach - paginate and fetch on demand
query BetterApproach($userId: ID!, $after: String) {
  user(id: $userId) {
    id
    name
    posts(first: 20, after: $after) {
      edges {
        node {
          id
          title
          commentCount
        }
      }
      pageInfo {
        hasNextPage
        endCursor
      }
    }
  }
}
```
