## Introduction

Input validation and sanitization are critical for API security. They protect against injection attacks, ensure data integrity, and provide clear error messages to API consumers. Proper validation should occur at multiple layers of your application.

### FluentValidation

```csharp
// Install: dotnet add package FluentValidation.AspNetCore

public class CreateProductRequestValidator : AbstractValidator<CreateProductRequest>
{
    public CreateProductRequestValidator()
    {
        RuleFor(x => x.Name)
            .NotEmpty().WithMessage("Product name is required")
            .Length(3, 100).WithMessage("Name must be between 3 and 100 characters")
            .Matches(@"^[a-zA-Z0-9\s\-]+$")
                .WithMessage("Name can only contain letters, numbers, spaces, and hyphens");
        
        RuleFor(x => x.Price)
            .GreaterThan(0).WithMessage("Price must be greater than zero")
            .LessThanOrEqualTo(10000).WithMessage("Price cannot exceed 10000");

            .EmailAddress().When(x => !string.IsNullOrEmpty(x.ContactEmail))
            .WithMessage("Invalid email format");
        
        RuleFor(x => x.Description)
            .MaximumLength(500).When(x => !string.IsNullOrEmpty(x.Description))
            .WithMessage("Description cannot exceed 500 characters");
        
        RuleFor(x => x.CategoryId)
            .MustAsync(CategoryExistsAsync)
            .WithMessage("Category does not exist");
    }
    
    private async Task<bool> CategoryExistsAsync(int categoryId, CancellationToken cancellationToken)
    {
        // Check if category exists in database
        return await _categoryRepository.ExistsAsync(categoryId);
    }
}
```

### Preventing SQL Injection

```csharp
// VULNERABLE - Never do this!
public async Task<Product?> GetProductUnsafe(string name)
{
    var query = $"SELECT * FROM Products WHERE Name = '{name}'";
    return await _context.Products.FromSqlRaw(query).FirstOrDefaultAsync();
}

// SAFE - Use parameterized queries
public async Task<Product?> GetProductSafe(string name)
{
    return await _context.Products
        .FromSqlInterpolated($"SELECT * FROM Products WHERE Name = {name}")
        .FirstOrDefaultAsync();
}

// BETTER - Use LINQ
public async Task<Product?> GetProductBest(string name)
{
    return await _context.Products
        .FirstOrDefaultAsync(p => p.Name == name);
}
```

```csharp
public class XssProtectionService

        if (string.IsNullOrEmpty(input))

    {
        if (string.IsNullOrEmpty(input))

    {
        if (string.IsNullOrEmpty(input))
            return input;
        
        // Simple regex to remove HTML tags

    }
```

### Custom Validation Attribute

```csharp
public class NoSqlInjectionAttribute : ValidationAttribute
{
    private static readonly Regex SqlInjectionRegex = new Regex(
        @"(\b(SELECT|INSERT|UPDATE|DELETE|DROP|CREATE|ALTER|EXEC|EXECUTE)\b)|([';])",
        RegexOptions.IgnoreCase | RegexOptions.Compiled);
    
    protected override ValidationResult? IsValid(object? value, ValidationContext validationContext)
    {
        if (value is string stringValue && !string.IsNullOrEmpty(stringValue))
        {
            if (SqlInjectionRegex.IsMatch(stringValue))
            {
                return new ValidationResult("Input contains potentially dangerous SQL keywords");
            }
        }
        
        return ValidationResult.Success;
    }
}

// Usage
public class SearchRequest
{
    [NoSqlInjection]
    [StringLength(100)]
    public string Query { get; set; } = string.Empty;
}
```

### Validation Filter

```csharp
public class ValidationFilter : IAsyncActionFilter
{
    public async Task OnActionExecutionAsync(
        ActionExecutingContext context,
        ActionExecutionDelegate next)
    {
        if (!context.ModelState.IsValid)
        {
            var errors = context.ModelState
                .Where(x => x.Value?.Errors.Count > 0)
                .Select(x => new
                {
                    Field = x.Key,
                    Errors = x.Value!.Errors.Select(e => e.ErrorMessage)
                });
            
            context.Result = new BadRequestObjectResult(new
            {
                Message = "Validation failed",
                Errors = errors
            });
            
            return;
        }
        
        await next();
    }
}

// Register globally
builder.Services.AddControllers(options =>
{
    options.Filters.Add<ValidationFilter>();
});
```

### File Upload Validation

```csharp
public class FileUploadValidator
{
    private static readonly string[] AllowedExtensions = { ".jpg", ".jpeg", ".png", ".pdf" };
    private const long MaxFileSize = 5 * 1024 * 1024; // 5 MB
    
    public static ValidationResult ValidateFile(IFormFile file)
    {
        if (file == null || file.Length == 0)
        {
            return ValidationResult.Error("File is required");
        }
        
        if (file.Length > MaxFileSize)
        {
            return ValidationResult.Error($"File size cannot exceed {MaxFileSize / 1024 / 1024} MB");
        }
        
        var extension = Path.GetExtension(file.FileName).ToLowerInvariant();
        if (!AllowedExtensions.Contains(extension))
        {
            return ValidationResult.Error(
                $"File type {extension} is not allowed. Allowed types: {string.Join(", ", AllowedExtensions)}");
        }
        
        // Check file signature (magic numbers)
        if (!IsValidFileSignature(file))
        {
            return ValidationResult.Error("File signature does not match extension");
        }
        
        return ValidationResult.Success();
    }
    
    private static bool IsValidFileSignature(IFormFile file)
    {
        var extension = Path.GetExtension(file.FileName).ToLowerInvariant();
        var signatures = GetFileSignatures(extension);
        
        using var reader = new BinaryReader(file.OpenReadStream());
        var headerBytes = reader.ReadBytes(signatures.Max(m => m.Length));
        
        return signatures.Any(signature =>
            headerBytes.Take(signature.Length).SequenceEqual(signature));
    }
    
    private static List<byte[]> GetFileSignatures(string extension)
    {
        return extension switch
        {
            ".jpg" or ".jpeg" => new List<byte[]>
            {
                new byte[] { 0xFF, 0xD8, 0xFF, 0xE0 },
                new byte[] { 0xFF, 0xD8, 0xFF, 0xE1 }
            },
            ".png" => new List<byte[]>
            {
                new byte[] { 0x89, 0x50, 0x4E, 0x47 }
            },
            ".pdf" => new List<byte[]>
            {
                new byte[] { 0x25, 0x50, 0x44, 0x46 }
            },
            _ => new List<byte[]>()
        };
    }
}
```

### Advanced FluentValidation Patterns

```csharp
public class CreateOrderRequestValidator : AbstractValidator<CreateOrderRequest>
{
    private readonly IProductRepository _productRepository;
    private readonly ICustomerRepository _customerRepository;
    
    public CreateOrderRequestValidator(
        IProductRepository productRepository,
        ICustomerRepository customerRepository)
    {
        _productRepository = productRepository;
        _customerRepository = customerRepository;
        
        // Basic validation
        RuleFor(x => x.CustomerId)
            .NotEmpty()
            .MustAsync(CustomerExistsAsync)
            .WithMessage("Customer does not exist");
        
        RuleFor(x => x.OrderDate)
            .NotEmpty()
            .LessThanOrEqualTo(DateTime.UtcNow)
            .WithMessage("Order date cannot be in the future");
        
        // Nested collection validation
        RuleFor(x => x.Items)
            .NotEmpty()
            .WithMessage("Order must contain at least one item");
        
        RuleForEach(x => x.Items)
            .SetValidator(new OrderItemValidator(_productRepository));
        
        // Custom validation
        RuleFor(x => x)
            .Must(HaveValidTotalAmount)
            .WithMessage("Order total does not match sum of items");
        
        // Async validation with dependencies
        RuleFor(x => x)
            .MustAsync(HaveAvailableStockAsync)
            .WithMessage("One or more items are out of stock");
    }
    
    private async Task<bool> CustomerExistsAsync(
        int customerId,
        CancellationToken cancellationToken)
    {
        return await _customerRepository.ExistsAsync(customerId, cancellationToken);
    }
    
    private bool HaveValidTotalAmount(CreateOrderRequest request)
    {
        var calculatedTotal = request.Items.Sum(i => i.Quantity * i.UnitPrice);
        return Math.Abs(request.TotalAmount - calculatedTotal) < 0.01m;
    }
    
    private async Task<bool> HaveAvailableStockAsync(
        CreateOrderRequest request,
        CancellationToken cancellationToken)
    {
        foreach (var item in request.Items)
        {
            var product = await _productRepository
                .GetByIdAsync(item.ProductId, cancellationToken);
            
            if (product == null || product.StockQuantity < item.Quantity)
            {
                return false;
            }
        }
        
        return true;
    }
}

public class OrderItemValidator : AbstractValidator<OrderItem>
{
    private readonly IProductRepository _productRepository;
    
    public OrderItemValidator(IProductRepository productRepository)
    {
        _productRepository = productRepository;
        
        RuleFor(x => x.ProductId)
            .NotEmpty()
            .MustAsync(ProductExistsAsync)
            .WithMessage("Product does not exist");
        
        RuleFor(x => x.Quantity)
            .GreaterThan(0)
            .WithMessage("Quantity must be greater than zero")
            .LessThanOrEqualTo(1000)
            .WithMessage("Quantity cannot exceed 1000");
        
        RuleFor(x => x.UnitPrice)
            .GreaterThan(0)
            .WithMessage("Unit price must be greater than zero");
    }
    
    private async Task<bool> ProductExistsAsync(
        int productId,
        CancellationToken cancellationToken)
    {
        return await _productRepository.ExistsAsync(productId, cancellationToken);
    }
}
```

### Preventing Command Injection

```csharp
public class CommandExecutionService
{
    // VULNERABLE - Never do this!
    public async Task<string> ExecuteCommandUnsafe(string filename)
    {
        var process = new Process
        {
            StartInfo = new ProcessStartInfo
            {
                FileName = "convert",
                Arguments = $"{filename} output.pdf", // User input in command
                RedirectStandardOutput = true
            }
        };
        process.Start();
        return await process.StandardOutput.ReadToEndAsync();
    }
    
    // SAFE - Validate and sanitize input
    public async Task<string> ExecuteCommandSafe(string filename)
    {
        // Whitelist allowed characters
        if (!Regex.IsMatch(filename, @"^[a-zA-Z0-9_\-\.]+$"))
        {
            throw new ArgumentException("Invalid filename");
        }
        
        // Validate file exists
        if (!File.Exists(filename))
        {
            throw new FileNotFoundException();
        }
        
        // Use argument list instead of string
        var process = new Process
        {
            StartInfo = new ProcessStartInfo
            {
                FileName = "convert",
                ArgumentList = { filename, "output.pdf" },
                RedirectStandardOutput = true
            }
        };
        process.Start();
        return await process.StandardOutput.ReadToEndAsync();
    }
}
```

### Path Traversal Prevention

```csharp
public class SecureFileService
{
    private readonly string _baseDirectory;
    
    public SecureFileService(IConfiguration configuration)
    {
        _baseDirectory = configuration["FileStorage:BasePath"]!;
    }
    
    // VULNERABLE
    public byte[] ReadFileUnsafe(string path)
    {
        // Attacker could use "../../../etc/passwd"
        return File.ReadAllBytes(Path.Combine(_baseDirectory, path));
    }
    
    // SAFE
    public byte[] ReadFileSafe(string path)
    {
        // Validate filename
        if (string.IsNullOrWhiteSpace(path) ||
            path.Contains("..") ||
            path.Contains("/") ||
            path.Contains("\\"))
        {
            throw new ArgumentException("Invalid path");
        }
        
        // Get full path and verify it's within base directory
        var fullPath = Path.GetFullPath(Path.Combine(_baseDirectory, path));
        
        if (!fullPath.StartsWith(_baseDirectory, StringComparison.OrdinalIgnoreCase))
        {
            throw new UnauthorizedAccessException("Path traversal attempt detected");
        }
        
        if (!File.Exists(fullPath))
        {
            throw new FileNotFoundException();
        }
        
        return File.ReadAllBytes(fullPath);
    }
}
```

### Regex Denial of Service (ReDoS) Prevention

```csharp
public class SafeRegexValidator
{
    // VULNERABLE - Can cause ReDoS with input like "aaaaaaaaaaaaaaaaaaaaaaaa!"
    private static readonly Regex UnsafeRegex = new Regex(@"^(a+)+$");
    
    // SAFE - Use timeout
    private static readonly Regex SafeRegex = new Regex(
        @"^[a-zA-Z0-9]+$",
        RegexOptions.None,
        TimeSpan.FromMilliseconds(100));
    
    public bool ValidateInput(string input)
    {
        try
        {
            return SafeRegex.IsMatch(input);
        }
        catch (RegexMatchTimeoutException)
        {
            // Log potential ReDoS attempt
            return false;
        }
    }
}
```

### JSON Deserialization Security

```csharp
// Configure JSON options for security
builder.Services.AddControllers()
    .AddJsonOptions(options =>
    {
        // Limit JSON depth to prevent stack overflow
        options.JsonSerializerOptions.MaxDepth = 32;
        
        // Disable type name handling to prevent deserialization attacks
        options.JsonSerializerOptions.TypeInfoResolver = null;
        
        // Strict validation
        options.JsonSerializerOptions.AllowTrailingCommas = false;
        options.JsonSerializerOptions.ReadCommentHandling = JsonCommentHandling.Disallow;
    });

// Safe deserialization with size limit
public class SafeJsonDeserializer
{
    private const int MaxJsonSize = 1024 * 1024; // 1 MB
    
    public async Task<T?> DeserializeAsync<T>(Stream stream)
    {
        if (stream.Length > MaxJsonSize)
        {
            throw new InvalidOperationException("JSON payload too large");
        }
        
        try
        {
            return await JsonSerializer.DeserializeAsync<T>(stream, new JsonSerializerOptions
            {
                MaxDepth = 32,
                PropertyNameCaseInsensitive = true
            });
        }
        catch (JsonException ex)
        {
            throw new ValidationException("Invalid JSON format", ex);
        }
    }
}
```

### Comprehensive Input Sanitization Service

```csharp
public class InputSanitizationService
{
    private readonly HtmlEncoder _htmlEncoder;
    private readonly JavaScriptEncoder _jsEncoder;
    private readonly UrlEncoder _urlEncoder;
    
    public string SanitizeHtml(string input)
    {
        if (string.IsNullOrEmpty(input))
            return input;
        
        // Encode HTML special characters
        var encoded = _htmlEncoder.Encode(input);
        
        // Remove any remaining scripts
        encoded = Regex.Replace(encoded, @"<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>", 
            string.Empty, RegexOptions.IgnoreCase);
        
        return encoded;
    }
    
    public string SanitizeForDatabase(string input)
    {
        if (string.IsNullOrEmpty(input))
            return input;
        
        // Remove null bytes
        input = input.Replace("\0", string.Empty);
        
        // Remove control characters except newline and tab
        input = Regex.Replace(input, @"[\x00-\x08\x0B\x0C\x0E-\x1F]", string.Empty);
        
        return input;
    }
    
    public string SanitizeFilename(string filename)
    {
        if (string.IsNullOrEmpty(filename))
            return filename;
        
        // Remove path separators
        filename = Path.GetFileName(filename);
        
        // Remove invalid characters
        var invalidChars = Path.GetInvalidFileNameChars();
        foreach (var c in invalidChars)
        {
            filename = filename.Replace(c.ToString(), string.Empty);
        }
        
        // Limit length
        if (filename.Length > 255)
        {
            var extension = Path.GetExtension(filename);
            filename = filename.Substring(0, 255 - extension.Length) + extension;
        }
        
        return filename;
    }
    
    public string SanitizeUrl(string url)
    {
        if (string.IsNullOrEmpty(url))
            return url;
        
        // Validate URL format
        if (!Uri.TryCreate(url, UriKind.Absolute, out var uri))
        {
            throw new ArgumentException("Invalid URL format");
        }
        
        // Only allow http and https
        if (uri.Scheme != "http" && uri.Scheme != "https")
        {
            throw new ArgumentException("Only HTTP and HTTPS protocols are allowed");
        }
        
        return uri.ToString();
    }
}
```

### Validation Pipeline

```csharp
public class ValidationPipeline<T>
{
    private readonly List<IValidator<T>> _validators = new();
    
    public ValidationPipeline<T> AddValidator(IValidator<T> validator)
    {
        _validators.Add(validator);
        return this;
    }
    
    public async Task<ValidationResult> ValidateAsync(T input)
    {
        var errors = new List<string>();
        
        foreach (var validator in _validators)
        {
            var result = await validator.ValidateAsync(input);
            if (!result.IsValid)
            {
                errors.AddRange(result.Errors.Select(e => e.ErrorMessage));
            }
        }
        
        return errors.Any()
            ? ValidationResult.Failure(errors)
            : ValidationResult.Success();
    }
}

// Usage
var pipeline = new ValidationPipeline<CreateProductRequest>()
    .AddValidator(new CreateProductRequestValidator())
    .AddValidator(new SecurityValidator())
    .AddValidator(new BusinessRuleValidator());

var result = await pipeline.ValidateAsync(request);
```
