# Lesson 3: Swagger UI and Interactive Documentation

## Learning Objectives

By the end of this lesson, you will be able to:
- Configure and customize Swagger UI
- Use Swagger UI for API testing
- Customize the UI theme and layout
- Add authentication to Swagger UI
- Deploy Swagger UI for production use

## Introduction

Swagger UI transforms your OpenAPI specification into an interactive web interface where developers can explore your API, read documentation, and test endpoints directly in their browser. It significantly improves developer experience and API adoption.

## Key Concepts Summary

### Basic Swagger UI Configuration

```csharp
// Program.cs
app.UseSwaggerUI(options =>
{
    // Serve at root
    options.RoutePrefix = string.Empty;
    options.SwaggerEndpoint("/swagger/v1/swagger.json", "API V1");
    
    // UI settings
    options.DocumentTitle = "Products API Documentation";
    options.DefaultModelsExpandDepth(2);
    options.DefaultModelRendering(ModelRendering.Example);
    options.DocExpansion(DocExpansion.None);
    options.EnableDeepLinking();
    options.DisplayRequestDuration();
    options.EnableFilter();
});
```

### Advanced UI Customization

```csharp
app.UseSwaggerUI(options =>
{
    // Endpoints
    options.SwaggerEndpoint("/swagger/v1/swagger.json", "API V1");
    options.SwaggerEndpoint("/swagger/v2/swagger.json", "API V2");
    
    // Routing
    options.RoutePrefix = "api-docs";  // Available at /api-docs
    
    // Display options
    options.DocumentTitle = "My API Documentation";
    options.DocExpansion(DocExpansion.None);  // None, List, or Full
    options.DefaultModelsExpandDepth(2);      // -1 to hide models
    options.DefaultModelRendering(ModelRendering.Example);  // Example or Model
    
    // Filtering
    options.EnableFilter();
    options.EnableDeepLinking();
    
    // Request/Response features
    options.DisplayRequestDuration();
    options.EnableValidator();
    options.ShowExtensions();
    
    // UI behavior
    options.EnableTryItOutByDefault();
    options.DisplayOperationId();
    options.ShowCommonExtensions();
    
    // Request snippets
    options.ConfigObject.AdditionalItems["syntaxHighlight"] = new Dictionary<string, object>
    {
        ["activated"] = true,
        ["theme"] = "monokai"
    };
    
    // Persist authorization
    options.ConfigObject.AdditionalItems["persistAuthorization"] = true;
});
```

### OAuth2 Integration

```csharp
// Swagger configuration
builder.Services.AddSwaggerGen(options =>
{
    options.AddSecurityDefinition("OAuth2", new OpenApiSecurityScheme
    {
        Type = SecuritySchemeType.OAuth2,
        Flows = new OpenApiOAuthFlows
        {
            AuthorizationCode = new OpenApiOAuthFlow
            {
                AuthorizationUrl = new Uri("https://auth.example.com/oauth/authorize"),
                TokenUrl = new Uri("https://auth.example.com/oauth/token"),
                Scopes = new Dictionary<string, string>
                {
                    ["read:products"] = "Read access to products",
                    ["write:products"] = "Write access to products",
                    ["read:orders"] = "Read access to orders",
                    ["write:orders"] = "Write access to orders"
                }
            }
        }
    });
    
    options.AddSecurityRequirement(new OpenApiSecurityRequirement
    {
        {
            new OpenApiSecurityScheme
            {
                Reference = new OpenApiReference
                {
                    Type = ReferenceType.SecurityScheme,

// UI configuration
app.UseSwaggerUI(options =>

        ["audience"] = "api.example.com"
    });
});
```

### Multiple Authentication Schemes

```csharp
builder.Services.AddSwaggerGen(options =>
{
    // Define multiple security schemes
    options.AddSecurityDefinition("Bearer", new OpenApiSecurityScheme
    {
        Type = SecuritySchemeType.Http,
        Scheme = "bearer",
        BearerFormat = "JWT",
        Description = "JWT Authorization header using Bearer scheme"
    });
    
    options.AddSecurityDefinition("ApiKey", new OpenApiSecurityScheme
    {
        Type = SecuritySchemeType.ApiKey,
        In = ParameterLocation.Header,
        Name = "X-API-Key",
        Description = "API Key authentication"
    });
    
    options.AddSecurityDefinition("Basic", new OpenApiSecurityScheme
    {
        Type = SecuritySchemeType.Http,
        Scheme = "basic",
        Description = "Basic authentication"
    });
    
    // Require at least one scheme
    options.AddSecurityRequirement(new OpenApiSecurityRequirement
    {
        {
            new OpenApiSecurityScheme
            {
                Reference = new OpenApiReference
                {
                    Type = ReferenceType.SecurityScheme,
                    Id = "Bearer"
                }
            },
            Array.Empty<string>()
        },
        {
            new OpenApiSecurityScheme
            {
                Reference = new OpenApiReference
                {
                    Type = ReferenceType.SecurityScheme,
                    Id = "ApiKey"
                }
            },
            Array.Empty<string>()
        }
    });
});

// Endpoints can specify which scheme they accept
[HttpGet]
[Authorize(AuthenticationSchemes = "Bearer,ApiKey")]
public ActionResult<IEnumerable<Product>> GetProducts() { }
```

### Environment-Based Configuration

```csharp
var builder = WebApplication.CreateBuilder(args);

builder.Services.AddSwaggerGen(options =>
{
    if (builder.Environment.IsDevelopment())
    {
        // Development: Include internal endpoints
        options.SwaggerDoc("v1", new OpenApiInfo
        {
            Title = "Products API (Development)",
            Version = "v1-dev"
        });
    }
    else
    {
        // Production: Public API only
        options.SwaggerDoc("v1", new OpenApiInfo
        {
            Title = "Products API",
            Version = "v1"
        });
    }
});

var app = builder.Build();

// Only expose Swagger UI in development by default
if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI(options =>
    {
        options.SwaggerEndpoint("/swagger/v1/swagger.json", "V1");
        options.RoutePrefix = string.Empty;  // Serve at root in dev
    });
}
else
{
    // Optional: Expose with authentication in production
    app.UseSwagger();
    app.UseSwaggerUI(options =>
    {
        options.SwaggerEndpoint("/swagger/v1/swagger.json", "V1");
        options.RoutePrefix = "docs";  // Serve at /docs in production
    });
    
    // Protect Swagger UI in production
    app.MapWhen(
        context => context.Request.Path.StartsWithSegments("/docs"),
        appBuilder =>
        {
            appBuilder.Use(async (context, next) =>
            {
                if (!context.User.Identity?.IsAuthenticated ?? true)
                {
                    context.Response.StatusCode = 401;
                    await context.Response.WriteAsync("Unauthorized");
                    return;
                }
                await next();
            });
        });
}
```

### Request Interceptors

```csharp
app.UseSwaggerUI(options =>
{
    options.SwaggerEndpoint("/swagger/v1/swagger.json", "V1");
    
    // Add custom headers to all requests
    options.UseRequestInterceptor(@"
        (request) => {
            // Add custom header
            request.headers['X-Custom-Header'] = 'swagger-ui';
            
            // Add correlation ID
            request.headers['X-Correlation-ID'] = Date.now().toString();
            
            // Log requests in console
            console.log('Request:', request.method, request.url);
            
            return request;
        }
    ");
    
    // Intercept responses
    options.UseResponseInterceptor(@"
        (response) => {
            console.log('Response:', response.status, response.url);
            return response;
        }
    ");
});
```

### Versioned Documentation

```csharp
// Configure multiple versions
builder.Services.AddSwaggerGen(options =>
{
    var versions = new[] { "v1", "v2", "v3" };
    
    foreach (var version in versions)
    {
        options.SwaggerDoc(version, new OpenApiInfo
        {
            Title = $"Products API {version.ToUpper()}",
            Version = version,
            Description = version switch
            {
                "v1" => "Legacy API - Deprecated",
                "v2" => "Current stable API",
                "v3" => "Beta API - Under development",
                _ => ""
            }
        });
    }
    
    // Filter endpoints by version
    options.DocInclusionPredicate((docName, apiDesc) =>
    {
        var versions = apiDesc.ActionDescriptor.EndpointMetadata
            .OfType<ApiVersionAttribute>()
            .SelectMany(attr => attr.Versions);
            
        return versions.Any(v => $"v{v}" == docName);
    });
});

app.UseSwaggerUI(options =>
{
    // Create dropdown for version selection
    options.SwaggerEndpoint("/swagger/v1/swagger.json", "V1 (Deprecated)");
    options.SwaggerEndpoint("/swagger/v2/swagger.json", "V2 (Stable)");
    options.SwaggerEndpoint("/swagger/v3/swagger.json", "V3 (Beta)");
    
    options.RoutePrefix = string.Empty;
    options.DocumentTitle = "Products API - All Versions";
});
```

### Rate Limiting Information

```csharp
// Display rate limiting info in UI
public class RateLimitOperationFilter : IOperationFilter
{
    public void Apply(OpenApiOperation operation, OperationFilterContext context)
    {
        operation.Extensions.Add("x-rate-limit", new OpenApiObject
        {
            ["limit"] = new OpenApiInteger(100),
            ["window"] = new OpenApiString("1 minute")
        });
        
        // Add to response headers
        foreach (var response in operation.Responses.Values)
        {
            response.Headers ??= new Dictionary<string, OpenApiHeader>();
            
            response.Headers["X-RateLimit-Limit"] = new OpenApiHeader
            {
                Description = "Request limit per window",
                Schema = new OpenApiSchema { Type = "integer" }
            };
            
            response.Headers["X-RateLimit-Remaining"] = new OpenApiHeader
            {
                Description = "Remaining requests in window",
                Schema = new OpenApiSchema { Type = "integer" }
            };
            
            response.Headers["X-RateLimit-Reset"] = new OpenApiHeader
            {
                Description = "Time when limit resets (Unix timestamp)",
                Schema = new OpenApiSchema { Type = "integer" }
            };
        }
    }
}
```

### Production Considerations

**Security:**
```csharp
// Option 1: Disable in production
if (!app.Environment.IsProduction())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

// Option 2: Protect with middleware
app.UseWhen(
    context => context.Request.Path.StartsWithSegments("/swagger"),
    appBuilder =>
    {
        appBuilder.UseMiddleware<SwaggerAuthenticationMiddleware>();
    });

public class SwaggerAuthenticationMiddleware
{
    private readonly RequestDelegate _next;
    
    public SwaggerAuthenticationMiddleware(RequestDelegate next)
    {
        _next = next;
    }
    
    public async Task InvokeAsync(HttpContext context)
    {
        // Check if user is authenticated
        if (!context.User.Identity?.IsAuthenticated ?? true)
        {
            // Check for API key
            if (!context.Request.Headers.TryGetValue("X-API-Key", out var apiKey) ||
                apiKey != "secret-swagger-key")
            {
                context.Response.StatusCode = 401;
                await context.Response.WriteAsync("Unauthorized");
                return;
            }
        }
        
        await _next(context);
    }
}

// Option 3: Use IP whitelist
app.UseWhen(
    context => context.Request.Path.StartsWithSegments("/swagger"),
    appBuilder =>
    {
        appBuilder.Use(async (context, next) =>
        {
            var allowedIps = new[] { "127.0.0.1", "::1" };
            var remoteIp = context.Connection.RemoteIpAddress?.ToString();
            
            if (!allowedIps.Contains(remoteIp))
            {
                context.Response.StatusCode = 403;
                await context.Response.WriteAsync("Forbidden");
                return;
            }
            
            await next();
        });
    });
```
