# Lesson 2: Generating OpenAPI from .NET Code

## Learning Objectives

By the end of this lesson, you will be able to:
- Configure Swashbuckle in ASP.NET Core
- Generate OpenAPI specifications automatically from controllers
- Enrich specifications with XML comments and attributes
- Customize the generated OpenAPI document
- Support multiple API versions in OpenAPI

---

## Introduction

Rather than writing OpenAPI specifications by hand, you can generate them automatically from your .NET code using tools like Swashbuckle. This code-first approach ensures your documentation stays in sync with implementation, reduces manual work, and leverages C# type information for accurate schemas.

### Installing and Configuring Swashbuckle

```csharp
// Install NuGet package:
// dotnet add package Swashbuckle.AspNetCore

// Program.cs
using Microsoft.OpenApi.Models;

var builder = WebApplication.CreateBuilder(args);

// Add services
builder.Services.AddControllers();

// Configure Swagger/OpenAPI
builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen(options =>
{
    options.SwaggerDoc("v1", new OpenApiInfo
    {
        Title = "Products API",
        Version = "v1",
        Description = "API for managing products",
        TermsOfService = new Uri("https://example.com/terms"),
        Contact = new OpenApiContact
        {
            Name = "API Support",
            Email = "support@example.com",
            Url = new Uri("https://example.com/support")
        },
        License = new OpenApiLicense
        {
            Name = "MIT",
            Url = new Uri("https://opensource.org/licenses/MIT")
        }
    });
});

var app = builder.Build();

// Configure middleware
if (app.Environment.IsDevelopment())
{
    app.UseSwagger();  // Serves OpenAPI JSON at /swagger/v1/swagger.json
    app.UseSwaggerUI(); // Serves UI at /swagger
}

app.UseHttpsRedirection();
app.UseAuthorization();
app.MapControllers();

app.Run();
```

### Basic Controller with Generated OpenAPI

```csharp
[ApiController]
[Route("api/[controller]")]
public class ProductsController : ControllerBase
{
    // Swagger automatically generates:
    // - Operation ID: GetProducts
    // - Response: 200 with IEnumerable<Product>
    [HttpGet]
    public ActionResult<IEnumerable<Product>> GetProducts()
    {
        var products = _repository.GetAll();
        return Ok(products);
    }
    
    // Swagger generates parameter from route
    [HttpGet("{id}")]
    public ActionResult<Product> GetProduct(int id)
    {
        var product = _repository.GetById(id);
        if (product == null)
            return NotFound();
        return Ok(product);
    }
    
    // Swagger generates request body from model
    [HttpPost]
    public ActionResult<Product> CreateProduct([FromBody] CreateProductRequest request)
    {
        var product = _repository.Create(request);
        return CreatedAtAction(nameof(GetProduct), new { id = product.Id }, product);
    }
}

public class Product
{
    public int Id { get; set; }
    public string Name { get; set; }
    public decimal Price { get; set; }
}

public class CreateProductRequest
{
    public string Name { get; set; }
    public decimal Price { get; set; }
}
```

### Operation Filters

// Custom operation filter
public class AddResponseHeadersFilter : IOperationFilter
{
    public void Apply(OpenApiOperation operation, OperationFilterContext context)
    {
        if (operation.Responses.ContainsKey("200"))
        {
            operation.Responses["200"].Headers = new Dictionary<string, OpenApiHeader>
            {
                ["X-Rate-Limit-Remaining"] = new OpenApiHeader
                {
                    Description = "Number of requests remaining",
                    Schema = new OpenApiSchema { Type = "integer" }
                },
                ["X-Rate-Limit-Reset"] = new OpenApiHeader
                {
                    Description = "Time when rate limit resets",
                    Schema = new OpenApiSchema { Type = "integer", Format = "int64" }
                }
            };
        }
    }
}

// Register filter

### Handling Polymorphic Responses

```csharp
// Base type
[JsonDerivedType(typeof(SuccessResult), "success")]
[JsonDerivedType(typeof(ErrorResult), "error")]
public abstract class OperationResult
{
    public string Type { get; set; }
}

public class SuccessResult : OperationResult
{
    public SuccessResult() => Type = "success";
    public object Data { get; set; }
}

public class ErrorResult : OperationResult
{
    public ErrorResult() => Type = "error";
    public string Message { get; set; }
    public List<string> Errors { get; set; }
}

[HttpPost("process")]
[ProducesResponseType(typeof(OperationResult), StatusCodes.Status200OK)]
public ActionResult<OperationResult> Process([FromBody] ProcessRequest request)
{
    if (request.IsValid)
        return new SuccessResult { Data = new { id = 123 } };
    else
        return new ErrorResult 
        { 
            Message = "Validation failed",
            Errors = new List<string> { "Invalid data" }
        };
}

// Configure polymorphic serialization
builder.Services.AddSwaggerGen(options =>
{
    options.UseOneOfForPolymorphism();
    options.SelectDiscriminatorNameUsing(type => "type");
    options.SelectDiscriminatorValueUsing(subType => subType.Name.ToLower());
});
```

// Configure tag ordering
builder.Services.AddSwaggerGen(options =>
{
    options.TagActionsBy(api =>
    {
        if (api.GroupName != null)
            return new[] { api.GroupName };
            
        if (api.ActionDescriptor is ControllerActionDescriptor controllerAction)
            return new[] { controllerAction.ControllerName };
            
        throw new InvalidOperationException("Unable to determine tag for endpoint.");
    });
    
    options.DocInclusionPredicate((name, api) => true);
});

### Advanced Response Headers

```csharp
/// <summary>
/// Get product by ID with caching support
/// </summary>
[HttpGet("{id}")]
[ProducesResponseType(StatusCodes.Status200OK)]
[ProducesResponseType(StatusCodes.Status304NotModified)]
public ActionResult<Product> GetProduct(
    int id,
    [FromHeader(Name = "If-None-Match")] string? etag = null)
{
    var product = _repository.GetById(id);
    if (product == null)
        return NotFound();
        
    var currentEtag = GenerateEtag(product);
    
    if (etag == currentEtag)
        return StatusCode(304);
        
    Response.Headers.Append("ETag", currentEtag);
    Response.Headers.Append("Cache-Control", "max-age=3600");
    
    return Ok(product);
}

// Document headers with filter
public class ETagOperationFilter : IOperationFilter
{
    public void Apply(OpenApiOperation operation, OperationFilterContext context)
    {
        if (context.MethodInfo.Name.StartsWith("Get"))
        {
            operation.Responses["200"].Headers.Add("ETag", new OpenApiHeader
            {
                Description = "Entity tag for cache validation",
                Schema = new OpenApiSchema { Type = "string" }
            });
        }
    }
}
```

### Conditional Schema Generation

```csharp
// Different schemas for different environments
builder.Services.AddSwaggerGen(options =>
{
    if (builder.Environment.IsDevelopment())
    {
        // Include internal endpoints in dev
        options.DocInclusionPredicate((docName, apiDesc) => true);
    }
    else
    {
        // Exclude internal endpoints in production
        options.DocInclusionPredicate((docName, apiDesc) =>
        {
            var groupName = apiDesc.GroupName ?? apiDesc.ActionDescriptor.RouteValues["controller"];
            return !groupName?.StartsWith("Internal") ?? true;
        });
    }
});
```
