## Learning Objectives

By the end of this lesson, you will be able to:

- Handle performance metrics

---

## Introduction

Interceptors are gRPC's middleware mechanism for implementing cross-cutting concerns like logging, authentication, monitoring, and error handling. They can intercept and modify RPC calls on both client and server sides.

## Key Concepts Summary

### Server Interceptor

```csharp
public class LoggingInterceptor : Interceptor
{
    private readonly ILogger<LoggingInterceptor> _logger;
    
    public LoggingInterceptor(ILogger<LoggingInterceptor> logger)
    {
        _logger = logger;
    }
    
    public override async Task<TResponse> UnaryServerHandler<TRequest, TResponse>(
        TRequest request,
        ServerCallContext context,
        UnaryServerMethod<TRequest, TResponse> continuation)
    {
        _logger.LogInformation(
            "Starting {Method}",
            context.Method);
        
        var stopwatch = Stopwatch.StartNew();
        
        try
        {
            var response = await continuation(request, context);
            
            _logger.LogInformation(
                "Completed {Method} in {Duration}ms",
                context.Method,
                stopwatch.ElapsedMilliseconds);
            
            return response;
        }
        catch (Exception ex)
        {
            _logger.LogError(
                ex,
                "Error in {Method}",
                context.Method);
            throw;
        }
    }
}

// Register interceptor
builder.Services.AddGrpc(options =>
{
    options.Interceptors.Add<LoggingInterceptor>();
});
```

### Client Interceptor

```csharp
public class AuthInterceptor : Interceptor
{
    private readonly ITokenProvider _tokenProvider;
    
    public override AsyncUnaryCall<TResponse> AsyncUnaryCall<TRequest, TResponse>(
        TRequest request,
        ClientInterceptorContext<TRequest, TResponse> context,
        AsyncUnaryCallContinuation<TRequest, TResponse> continuation)
    {
        var headers = new Metadata();
        var token = await _tokenProvider.GetTokenAsync();
        headers.Add("Authorization", $"Bearer {token}");
        
        var newContext = new ClientInterceptorContext<TRequest, TResponse>(
            context.Method,
            context.Host,
            context.Options.WithHeaders(headers));
        
        return continuation(request, newContext);
    }
}

// Use interceptor
var channel = GrpcChannel.ForAddress("https://localhost:5001");
var client = new ProductService.ProductServiceClient(
    channel.Intercept(new AuthInterceptor()));
```

### Advanced Interceptor Patterns

**Performance Monitoring Interceptor:**

```csharp
public class PerformanceInterceptor : Interceptor
{
    private readonly ILogger<PerformanceInterceptor> _logger;
    private readonly IMetricsCollector _metrics;
    
    public PerformanceInterceptor(
        ILogger<PerformanceInterceptor> logger,
        IMetricsCollector metrics)
    {
        _logger = logger;
        _metrics = metrics;
    }
    
    public override async Task<TResponse> UnaryServerHandler<TRequest, TResponse>(
        TRequest request,
        ServerCallContext context,
        UnaryServerMethod<TRequest, TResponse> continuation)
    {
        var stopwatch = Stopwatch.StartNew();
        var method = context.Method;
        
        try
        {
            var response = await continuation(request, context);
            
            stopwatch.Stop();
            
            _metrics.RecordDuration(method, stopwatch.ElapsedMilliseconds);
            _metrics.IncrementCounter($"{method}.success");
            
            _logger.LogInformation(
                "Method {Method} completed in {Duration}ms",
                method,
                stopwatch.ElapsedMilliseconds);
            
            return response;
        }
        catch (Exception ex)
        {
            stopwatch.Stop();
            
            _metrics.RecordDuration(method, stopwatch.ElapsedMilliseconds);
            _metrics.IncrementCounter($"{method}.error");
            
            _logger.LogError(
                ex,
                "Method {Method} failed after {Duration}ms",
                method,
                stopwatch.ElapsedMilliseconds);
            
            throw;
        }
    }
    
    // Handle streaming methods
    public override async Task ServerStreamingServerHandler<TRequest, TResponse>(
        TRequest request,
        IServerStreamWriter<TResponse> responseStream,
        ServerCallContext context,
        ServerStreamingServerMethod<TRequest, TResponse> continuation)
    {
        var stopwatch = Stopwatch.StartNew();
        var method = context.Method;
        var itemCount = 0;
        
        try
        {
            // Wrap response stream to count items
            var wrappedStream = new MetricsStreamWriter<TResponse>(
                responseStream,
                () => itemCount++);
            
            await continuation(request, wrappedStream, context);
            
            stopwatch.Stop();
            
            _metrics.RecordDuration(method, stopwatch.ElapsedMilliseconds);
            _metrics.RecordValue($"{method}.items", itemCount);
            
            _logger.LogInformation(
                "Streaming method {Method} completed {Count} items in {Duration}ms",
                method,
                itemCount,
                stopwatch.ElapsedMilliseconds);
        }
        catch (Exception ex)
        {
            stopwatch.Stop();
            
            _logger.LogError(
                ex,
                "Streaming method {Method} failed after {Count} items in {Duration}ms",
                method,
                itemCount,
                stopwatch.ElapsedMilliseconds);
            
            throw;
        }
    }
}

// Wrapper to count streamed items
public class MetricsStreamWriter<T> : IServerStreamWriter<T>
{
    private readonly IServerStreamWriter<T> _inner;
    private readonly Action _onWrite;
    
    public MetricsStreamWriter(IServerStreamWriter<T> inner, Action onWrite)
    {
        _inner = inner;
        _onWrite = onWrite;
    }
    
    public WriteOptions WriteOptions
    {
        get => _inner.WriteOptions;
        set => _inner.WriteOptions = value;
    }
    
    public async Task WriteAsync(T message)
    {
        _onWrite();
        await _inner.WriteAsync(message);
    }
}
```

**Authentication and Authorization Interceptor:**

```csharp
public class AuthenticationInterceptor : Interceptor
{
    private readonly IAuthenticationService _authService;
    private readonly ILogger<AuthenticationInterceptor> _logger;
    
    public AuthenticationInterceptor(
        IAuthenticationService authService,
        ILogger<AuthenticationInterceptor> logger)
    {
        _authService = authService;
        _logger = logger;
    }
    
    public override async Task<TResponse> UnaryServerHandler<TRequest, TResponse>(
        TRequest request,
        ServerCallContext context,
        UnaryServerMethod<TRequest, TResponse> continuation)
    {
        // Extract token from metadata
        var authHeader = context.RequestHeaders.GetValue("authorization");
        
        if (string.IsNullOrEmpty(authHeader))
        {
            throw new RpcException(new Status(
                StatusCode.Unauthenticated,
                "Missing authentication token"));
        }
        
        if (!authHeader.StartsWith("Bearer ", StringComparison.OrdinalIgnoreCase))
        {
            throw new RpcException(new Status(
                StatusCode.Unauthenticated,
                "Invalid authentication scheme"));
        }
        
        var token = authHeader.Substring("Bearer ".Length).Trim();
        
        try
        {
            // Validate token
            var principal = await _authService.ValidateTokenAsync(token);
            
            if (principal == null)
            {
                throw new RpcException(new Status(
                    StatusCode.Unauthenticated,
                    "Invalid or expired token"));
            }
            
            // Set user in HTTP context
            context.GetHttpContext().User = principal;
            
            _logger.LogInformation(
                "User {UserId} authenticated for {Method}",
                principal.Identity.Name,
                context.Method);
            
            // Continue with authenticated context
            return await continuation(request, context);
        }
        catch (SecurityTokenException ex)
        {
            _logger.LogWarning(ex, "Authentication failed");
            
            throw new RpcException(new Status(
                StatusCode.Unauthenticated,
                "Authentication failed"));
        }
    }
}

// Authorization interceptor (checks permissions)
public class AuthorizationInterceptor : Interceptor
{
    private readonly IAuthorizationService _authzService;
    
    public override async Task<TResponse> UnaryServerHandler<TRequest, TResponse>(
        TRequest request,
        ServerCallContext context,
        UnaryServerMethod<TRequest, TResponse> continuation)
    {
        var user = context.GetHttpContext().User;
        
        if (user?.Identity?.IsAuthenticated != true)
        {
            throw new RpcException(new Status(
                StatusCode.Unauthenticated,
                "User must be authenticated"));
        }
        
        // Check method-level permissions
        var requiredPermission = GetRequiredPermission(context.Method);
        
        if (!string.IsNullOrEmpty(requiredPermission))
        {
            if (!await _authzService.HasPermissionAsync(user, requiredPermission))
            {
                throw new RpcException(new Status(
                    StatusCode.PermissionDenied,
                    $"User lacks required permission: {requiredPermission}"));
            }
        }
        
        return await continuation(request, context);
    }
    
    private string GetRequiredPermission(string method)
    {
        // Map gRPC methods to permissions
        return method switch
        {
            "/products.v1.ProductService/CreateProduct" => "products:create",
            "/products.v1.ProductService/UpdateProduct" => "products:update",
            "/products.v1.ProductService/DeleteProduct" => "products:delete",
            _ => null
        };
    }
}
```

**Request/Response Logging Interceptor:**

```csharp
public class LoggingInterceptor : Interceptor
{
    private readonly ILogger<LoggingInterceptor> _logger;
    private readonly bool _logPayloads;
    
    public LoggingInterceptor(
        ILogger<LoggingInterceptor> logger,
        bool logPayloads = false)
    {
        _logger = logger;
        _logPayloads = logPayloads;
    }
    
    public override async Task<TResponse> UnaryServerHandler<TRequest, TResponse>(
        TRequest request,
        ServerCallContext context,
        UnaryServerMethod<TRequest, TResponse> continuation)
    {
        var correlationId = Guid.NewGuid().ToString();
        
        using var scope = _logger.BeginScope(new Dictionary<string, object>
        {
            ["CorrelationId"] = correlationId,
            ["Method"] = context.Method,
            ["Peer"] = context.Peer
        });
        
        _logger.LogInformation(
            "Request started: {Method}",
            context.Method);
        
        if (_logPayloads && request is IMessage protoRequest)
        {
            _logger.LogDebug(
                "Request payload: {Payload}",
                JsonFormatter.Default.Format(protoRequest));
        }
        
        var stopwatch = Stopwatch.StartNew();
        
        try
        {
            var response = await continuation(request, context);
            
            stopwatch.Stop();
            
            _logger.LogInformation(
                "Request completed: {Method} in {Duration}ms",
                context.Method,
                stopwatch.ElapsedMilliseconds);
            
            if (_logPayloads && response is IMessage protoResponse)
            {
                _logger.LogDebug(
                    "Response payload: {Payload}",
                    JsonFormatter.Default.Format(protoResponse));
            }
            
            return response;
        }
        catch (Exception ex)
        {
            stopwatch.Stop();
            
            _logger.LogError(
                ex,
                "Request failed: {Method} after {Duration}ms",
                context.Method,
                stopwatch.ElapsedMilliseconds);
            
            throw;
        }
    }
}
```

**Rate Limiting Interceptor:**

```csharp
using System.Threading.RateLimiting;

public class RateLimitingInterceptor : Interceptor
{
    private readonly RateLimiter _rateLimiter;
    private readonly ILogger<RateLimitingInterceptor> _logger;
    
    public RateLimitingInterceptor(
        RateLimiter rateLimiter,
        ILogger<RateLimitingInterceptor> logger)
    {
        _rateLimiter = rateLimiter;
        _logger = logger;
    }
    
    public override async Task<TResponse> UnaryServerHandler<TRequest, TResponse>(
        TRequest request,
        ServerCallContext context,
        UnaryServerMethod<TRequest, TResponse> continuation)
    {
        // Get user identifier
        var userId = context.GetHttpContext().User?.Identity?.Name ?? "anonymous";
        
        // Try to acquire lease
        using var lease = await _rateLimiter.AcquireAsync(
            permitCount: 1,
            context.CancellationToken);
        
        if (!lease.IsAcquired)
        {
            _logger.LogWarning(
                "Rate limit exceeded for user {UserId} on {Method}",
                userId,
                context.Method);
            
            // Add retry-after header
            var retryAfter = lease.TryGetMetadata(
                MetadataName.RetryAfter,
                out var retryAfterMetadata) 
                    ? retryAfterMetadata.TotalSeconds 
                    : 60;
            
            context.ResponseTrailers.Add(
                "retry-after",
                retryAfter.ToString());
            
            throw new RpcException(new Status(
                StatusCode.ResourceExhausted,
                "Rate limit exceeded"));
        }
        
        return await continuation(request, context);
    }
}

// Configure rate limiter
builder.Services.AddSingleton<RateLimiter>(_ =>
{
    return new SlidingWindowRateLimiter(new SlidingWindowRateLimiterOptions
    {
        PermitLimit = 100,
        Window = TimeSpan.FromMinutes(1),
        SegmentsPerWindow = 6
    });
});
```

**Retry Interceptor (Client-side):**

```csharp
public class RetryInterceptor : Interceptor
{
    private readonly int _maxRetries;
    private readonly ILogger<RetryInterceptor> _logger;
    
    public RetryInterceptor(int maxRetries, ILogger<RetryInterceptor> logger)
    {
        _maxRetries = maxRetries;
        _logger = logger;
    }
    
    public override AsyncUnaryCall<TResponse> AsyncUnaryCall<TRequest, TResponse>(
        TRequest request,
        ClientInterceptorContext<TRequest, TResponse> context,
        AsyncUnaryCallContinuation<TRequest, TResponse> continuation)
    {
        var call = continuation(request, context);
        
        return new AsyncUnaryCall<TResponse>(
            HandleUnaryCall(call),
            call.ResponseHeadersAsync,
            call.GetStatus,
            call.GetTrailers,
            call.Dispose);
    }
    
    private async Task<TResponse> HandleUnaryCall<TResponse>(
        AsyncUnaryCall<TResponse> call)
    {
        var retryCount = 0;
        var delay = TimeSpan.FromMilliseconds(100);
        
        while (true)
        {
            try
            {
                return await call.ResponseAsync;
            }
            catch (RpcException ex) when (
                IsRetryableError(ex) && retryCount < _maxRetries)
            {
                retryCount++;
                
                _logger.LogWarning(
                    "Retry {Count}/{Max} after error: {Status}",
                    retryCount,
                    _maxRetries,
                    ex.StatusCode);
                
                await Task.Delay(delay);
                delay = TimeSpan.FromMilliseconds(delay.TotalMilliseconds * 2);
                
                // Recreate call for retry
                // Note: In production, use Polly or gRPC retry policies
            }
        }
    }
    
    private static bool IsRetryableError(RpcException ex)
    {
        return ex.StatusCode == StatusCode.Unavailable ||
               ex.StatusCode == StatusCode.DeadlineExceeded ||
               ex.StatusCode == StatusCode.ResourceExhausted;
    }
}
```

**Caching Interceptor:**

```csharp
using Microsoft.Extensions.Caching.Memory;

public class CachingInterceptor : Interceptor
{
    private readonly IMemoryCache _cache;
    private readonly ILogger<CachingInterceptor> _logger;
    
    public CachingInterceptor(
        IMemoryCache cache,
        ILogger<CachingInterceptor> logger)
    {
        _cache = cache;
        _logger = logger;
    }
    
    public override async Task<TResponse> UnaryServerHandler<TRequest, TResponse>(
        TRequest request,
        ServerCallContext context,
        UnaryServerMethod<TRequest, TResponse> continuation)
    {
        // Only cache GET-style methods
        if (!IsCacheable(context.Method))
        {
            return await continuation(request, context);
        }
        
        // Generate cache key
        var cacheKey = GenerateCacheKey(context.Method, request);
        
        // Try get from cache
        if (_cache.TryGetValue<TResponse>(cacheKey, out var cachedResponse))
        {
            _logger.LogInformation(
                "Cache hit for {Method}",
                context.Method);
            
            context.ResponseTrailers.Add("x-cache", "HIT");
            
            return cachedResponse;
        }
        
        _logger.LogInformation(
            "Cache miss for {Method}",
            context.Method);
        
        // Execute and cache
        var response = await continuation(request, context);
        
        var cacheOptions = new MemoryCacheEntryOptions
        {
            AbsoluteExpirationRelativeToNow = TimeSpan.FromMinutes(5)
        };
        
        _cache.Set(cacheKey, response, cacheOptions);
        
        context.ResponseTrailers.Add("x-cache", "MISS");
        
        return response;
    }
    
    private static bool IsCacheable(string method)
    {
        return method.Contains("/Get") || method.Contains("/List");
    }
    
    private static string GenerateCacheKey<TRequest>(string method, TRequest request)
    {
        if (request is IMessage protoRequest)
        {
            var json = JsonFormatter.Default.Format(protoRequest);
            var hash = Convert.ToBase64String(
                SHA256.HashData(Encoding.UTF8.GetBytes(json)));
            return $"{method}:{hash}";
        }
        
        return $"{method}:{request?.GetHashCode()}";
    }
}
```

**Correlation ID Interceptor:**

```csharp
public class CorrelationIdInterceptor : Interceptor
{
    private const string CorrelationIdHeader = "x-correlation-id";
    
    public override async Task<TResponse> UnaryServerHandler<TRequest, TResponse>(
        TRequest request,
        ServerCallContext context,
        UnaryServerMethod<TRequest, TResponse> continuation)
    {
        // Get or generate correlation ID
        var correlationId = context.RequestHeaders.GetValue(CorrelationIdHeader)
            ?? Guid.NewGuid().ToString();
        
        // Add to response headers
        context.ResponseTrailers.Add(CorrelationIdHeader, correlationId);
        
        // Store in async local for access throughout the call
        CorrelationContext.Current = correlationId;
        
        try
        {
            return await continuation(request, context);
        }
        finally
        {
            CorrelationContext.Current = null;
        }
    }
}

// Correlation context
public static class CorrelationContext
{
    private static readonly AsyncLocal<string> _correlationId = new();
    
    public static string Current
    {
        get => _correlationId.Value;
        set => _correlationId.Value = value;
    }
}

// Client-side: Add correlation ID
public class CorrelationIdClientInterceptor : Interceptor
{
    public override AsyncUnaryCall<TResponse> AsyncUnaryCall<TRequest, TResponse>(
        TRequest request,
        ClientInterceptorContext<TRequest, TResponse> context,
        AsyncUnaryCallContinuation<TRequest, TResponse> continuation)
    {
        var headers = context.Options.Headers ?? new Metadata();
        
        // Propagate or generate correlation ID
        var correlationId = CorrelationContext.Current 
            ?? Guid.NewGuid().ToString();
        
        headers.Add("x-correlation-id", correlationId);
        
        var newContext = new ClientInterceptorContext<TRequest, TResponse>(
            context.Method,
            context.Host,
            context.Options.WithHeaders(headers));
        
        return continuation(request, newContext);
    }
}
```

**Chaining Multiple Interceptors:**

```csharp
// Register interceptors in order
builder.Services.AddGrpc(options =>
{
    // Order matters! Interceptors wrap each other
    options.Interceptors.Add<CorrelationIdInterceptor>();
    options.Interceptors.Add<LoggingInterceptor>();
    options.Interceptors.Add<AuthenticationInterceptor>();
    options.Interceptors.Add<AuthorizationInterceptor>();
    options.Interceptors.Add<RateLimitingInterceptor>();
    options.Interceptors.Add<CachingInterceptor>();
    options.Interceptors.Add<PerformanceInterceptor>();
    options.Interceptors.Add<ExceptionInterceptor>();
});

// Client-side interceptor chain
var channel = GrpcChannel.ForAddress("https://localhost:5001");

var client = new ProductService.ProductServiceClient(
    channel
        .Intercept(new CorrelationIdClientInterceptor())
        .Intercept(new AuthInterceptor(tokenProvider))
        .Intercept(new RetryInterceptor(maxRetries: 3, logger)));
```

```csharp
// Apply interceptor to specific services
```
