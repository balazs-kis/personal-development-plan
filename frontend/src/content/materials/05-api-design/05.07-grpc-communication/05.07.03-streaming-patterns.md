## Learning Objectives

By the end of this lesson, you will be able to:

## Introduction

gRPC supports four types of RPC patterns: unary (request-response), server streaming, client streaming, and bidirectional streaming. Streaming enables efficient handling of large datasets and real-time communication scenarios.

## Key Concepts Summary

### Server Streaming

```csharp
// Server implementation
public override async Task ListProducts(
    ListProductsRequest request,
    IServerStreamWriter<Product> responseStream,
    ServerCallContext context)
{
    var products = await _repository.GetAllAsync();
    
    foreach (var product in products)
    {
        // Check for cancellation
        context.CancellationToken.ThrowIfCancellationRequested();
        
        await responseStream.WriteAsync(new Product
        {
            Id = product.Id,
            Name = product.Name,
            Price = product.Price
        });
        
        // Optional: Add delay for backpressure
        await Task.Delay(100);
    }
}

// Client usage
var call = client.ListProducts(new ListProductsRequest());

await foreach (var product in call.ResponseStream.ReadAllAsync())
{
    Console.WriteLine($"{product.Name}: ${product.Price}");
}
```

### Client Streaming

```csharp
// Server implementation
public override async Task<CreateProductsResponse> CreateProducts(
    IAsyncStreamReader<CreateProductRequest> requestStream,
    ServerCallContext context)
{
    int count = 0;
    
    await foreach (var request in requestStream.ReadAllAsync())
    {
        await _repository.CreateAsync(new Product
        {
            Name = request.Name,
            Price = request.Price
        });
        count++;
    }
    
    return new CreateProductsResponse { CreatedCount = count };
}

// Client usage
using var call = client.CreateProducts();

foreach (var product in products)
{
    await call.RequestStream.WriteAsync(new CreateProductRequest
    {
        Name = product.Name,
        Price = product.Price
    });
}

await call.RequestStream.CompleteAsync();
var response = await call.ResponseAsync;
```

### Bidirectional Streaming

```csharp
// Server implementation
public override async Task SyncProducts(
    IAsyncStreamReader<ProductUpdate> requestStream,
    IServerStreamWriter<ProductUpdate> responseStream,
    ServerCallContext context)
{
    await foreach (var update in requestStream.ReadAllAsync())
    {
        // Process update
        var result = await _repository.UpdateAsync(update);
        
        // Send back confirmation
        await responseStream.WriteAsync(new ProductUpdate
        {
            Id = result.Id,
            Status = "Updated"
        });
    }
}

// Client usage
using var call = client.SyncProducts();

// Start reading responses
var readTask = Task.Run(async () =>
{
    await foreach (var response in call.ResponseStream.ReadAllAsync())
    {
        Console.WriteLine($"Received: {response.Status}");
    }
});

// Send requests
foreach (var update in updates)
{
    await call.RequestStream.WriteAsync(update);
}

await call.RequestStream.CompleteAsync();
await readTask;
```

### Advanced Streaming Patterns

**Batch Processing with Server Streaming:**

```protobuf
syntax = "proto3";

package products.v1;

service ProductService {
  // Export large dataset
  rpc ExportProducts(ExportProductsRequest) returns (stream ProductBatch);
  
  // Process and stream results
  rpc ProcessOrders(stream ProcessOrderRequest) returns (stream ProcessOrderResponse);
}

message ExportProductsRequest {
  string filter = 1;
  int32 batch_size = 2;
}

message ProductBatch {
  repeated Product products = 1;
  int32 batch_number = 2;
  int32 total_batches = 3;
  bool is_final = 4;
}

message ProcessOrderRequest {
  int64 order_id = 1;
  repeated OrderItem items = 2;
}

message ProcessOrderResponse {
  int64 order_id = 1;
  OrderStatus status = 2;
  string message = 3;
}
```

```csharp
// Server: Export with batching
public override async Task ExportProducts(
    ExportProductsRequest request,
    IServerStreamWriter<ProductBatch> responseStream,
    ServerCallContext context)
{
    var batchSize = request.BatchSize > 0 ? request.BatchSize : 100;
    var products = await _repository.GetAllAsync(request.Filter);
    
    var batches = products
        .Select((product, index) => new { product, index })
        .GroupBy(x => x.index / batchSize)
        .Select(g => g.Select(x => x.product).ToList())
        .ToList();
    
    var totalBatches = batches.Count;
    
    for (int i = 0; i < batches.Count; i++)
    {
        context.CancellationToken.ThrowIfCancellationRequested();
        
        var batch = new ProductBatch
        {
            BatchNumber = i + 1,
            TotalBatches = totalBatches,
            IsFinal = i == batches.Count - 1
        };
        
        batch.Products.AddRange(batches[i].Select(MapToProto));
        
        await responseStream.WriteAsync(batch);
        
        // Optional: throttle to avoid overwhelming client
        await Task.Delay(100, context.CancellationToken);
    }
}

// Client: Process batches
var request = new ExportProductsRequest
{
    Filter = "category=electronics",
    BatchSize = 50
};

var call = client.ExportProducts(request);
var allProducts = new List<Product>();

await foreach (var batch in call.ResponseStream.ReadAllAsync())
{
    Console.WriteLine($"Received batch {batch.BatchNumber}/{batch.TotalBatches}");
    allProducts.AddRange(batch.Products);
    
    if (batch.IsFinal)
    {
        Console.WriteLine($"Export complete: {allProducts.Count} products");
    }
}
```

**Real-Time Notifications with Server Streaming:**

```protobuf
service NotificationService {
  // Subscribe to real-time notifications
  rpc Subscribe(SubscribeRequest) returns (stream Notification);
}

message SubscribeRequest {
  repeated string topics = 1;
  string user_id = 2;
}

message Notification {
  string id = 1;
  string topic = 2;
  string title = 3;
  string message = 4;
  google.protobuf.Timestamp timestamp = 5;
  map<string, string> metadata = 6;
}
```

```csharp
// Server: Push notifications
public override async Task Subscribe(
    SubscribeRequest request,
    IServerStreamWriter<Notification> responseStream,
    ServerCallContext context)
{
    var channel = Channel.CreateUnbounded<Notification>();
    
    // Subscribe to notification system
    await _notificationService.SubscribeAsync(
        request.UserId,
        request.Topics.ToList(),
        notification => channel.Writer.TryWrite(notification));
    
    try
    {
        // Stream notifications until cancelled
        await foreach (var notification in channel.Reader.ReadAllAsync(
            context.CancellationToken))
        {
            await responseStream.WriteAsync(notification);
        }
    }
    finally
    {
        // Cleanup
        await _notificationService.UnsubscribeAsync(request.UserId);
        channel.Writer.Complete();
    }
}

// Client: Receive notifications
var request = new SubscribeRequest
{
    UserId = "user123",
    Topics = { "orders", "messages", "alerts" }
};

var cts = new CancellationTokenSource();

var call = client.Subscribe(request, cancellationToken: cts.Token);

try
{
    await foreach (var notification in call.ResponseStream.ReadAllAsync())
    {
        Console.WriteLine($"[{notification.Topic}] {notification.Title}");
        Console.WriteLine($"  {notification.Message}");
        Console.WriteLine($"  Time: {notification.Timestamp}");
    }
}
catch (RpcException ex) when (ex.StatusCode == StatusCode.Cancelled)
{
    Console.WriteLine("Subscription cancelled");
}
```

**Bulk Upload with Client Streaming:**

```protobuf
service ProductService {
  // Upload products in bulk
  rpc BulkUpload(stream UploadProductRequest) returns (BulkUploadResponse);
}

message UploadProductRequest {
  oneof data {
    UploadMetadata metadata = 1;
    ProductData product = 2;
  }
}

message UploadMetadata {
  string upload_id = 1;
  int32 total_count = 2;
  string user_id = 3;
}

message ProductData {
  string name = 1;
  double price = 2;
  string description = 3;
  repeated string tags = 4;
}

message BulkUploadResponse {
  string upload_id = 1;
  int32 successful_count = 2;
  int32 failed_count = 3;
  repeated string errors = 4;
}
```

```csharp
// Server: Process bulk upload
public override async Task<BulkUploadResponse> BulkUpload(
    IAsyncStreamReader<UploadProductRequest> requestStream,
    ServerCallContext context)
{
    string uploadId = null;
    var successCount = 0;
    var failCount = 0;
    var errors = new List<string>();
    
    await foreach (var request in requestStream.ReadAllAsync())
    {
        switch (request.DataCase)
        {
            case UploadProductRequest.DataOneofCase.Metadata:
                uploadId = request.Metadata.UploadId;
                _logger.LogInformation(
                    "Starting upload {UploadId} with {Count} products",
                    uploadId,
                    request.Metadata.TotalCount);
                break;
                
            case UploadProductRequest.DataOneofCase.Product:
                try
                {
                    var product = new ProductEntity
                    {
                        Name = request.Product.Name,
                        Price = request.Product.Price,
                        Description = request.Product.Description,
                        Tags = request.Product.Tags.ToList()
                    };
                    
                    await _repository.CreateAsync(product);
                    successCount++;
                }
                catch (Exception ex)
                {
                    failCount++;
                    errors.Add($"Failed to create {request.Product.Name}: {ex.Message}");
                    _logger.LogError(ex, "Failed to create product");
                }
                break;
        }
    }
    
    _logger.LogInformation(
        "Upload {UploadId} complete: {Success} successful, {Failed} failed",
        uploadId,
        successCount,
        failCount);
    
    return new BulkUploadResponse
    {
        UploadId = uploadId,
        SuccessfulCount = successCount,
        FailedCount = failCount,
        Errors = { errors }
    };
}

// Client: Upload products
var products = await LoadProductsFromFile("products.csv");

using var call = client.BulkUpload();

// Send metadata first
await call.RequestStream.WriteAsync(new UploadProductRequest
{
    Metadata = new UploadMetadata
    {
        UploadId = Guid.NewGuid().ToString(),
        TotalCount = products.Count,
        UserId = "user123"
    }
});

// Send products
foreach (var product in products)
{
    await call.RequestStream.WriteAsync(new UploadProductRequest
    {
        Product = new ProductData
        {
            Name = product.Name,
            Price = product.Price,
            Description = product.Description,
            Tags = { product.Tags }
        }
    });
}

// Complete and get response
await call.RequestStream.CompleteAsync();
var response = await call.ResponseAsync;

Console.WriteLine($"Upload complete:");
Console.WriteLine($"  Successful: {response.SuccessfulCount}");
Console.WriteLine($"  Failed: {response.FailedCount}");

if (response.Errors.Any())
{
    Console.WriteLine("Errors:");
    foreach (var error in response.Errors)
    {
        Console.WriteLine($"  - {error}");
    }
}
```

**Chat System with Bidirectional Streaming:**

```protobuf
service ChatService {
  // Real-time chat
  rpc Chat(stream ChatMessage) returns (stream ChatMessage);
}

message ChatMessage {
  string id = 1;
  string user_id = 2;
  string username = 3;
  string room_id = 4;
  string content = 5;
  google.protobuf.Timestamp timestamp = 6;
  MessageType type = 7;
}

enum MessageType {
  MESSAGE_TYPE_UNSPECIFIED = 0;
  TEXT = 1;
  JOIN = 2;
  LEAVE = 3;
  TYPING = 4;
}
```

```csharp
// Server: Chat room management
public class ChatService : Chat.ChatServiceBase
{
    private readonly ConcurrentDictionary<string, List<IServerStreamWriter<ChatMessage>>> 
        _rooms = new();
    
    public override async Task Chat(
        IAsyncStreamReader<ChatMessage> requestStream,
        IServerStreamWriter<ChatMessage> responseStream,
        ServerCallContext context)
    {
        string currentRoom = null;
        
        try
        {
            await foreach (var message in requestStream.ReadAllAsync())
            {
                // Handle room join/leave
                if (message.Type == MessageType.Join)
                {
                    currentRoom = message.RoomId;
                    
                    var writers = _rooms.GetOrAdd(
                        currentRoom,
                        _ => new List<IServerStreamWriter<ChatMessage>>());
                    
                    lock (writers)
                    {
                        writers.Add(responseStream);
                    }
                    
                    // Broadcast join message
                    await BroadcastToRoom(currentRoom, message);
                }
                else if (message.Type == MessageType.Leave)
                {
                    await BroadcastToRoom(currentRoom, message);
                    
                    if (_rooms.TryGetValue(currentRoom, out var writers))
                    {
                        lock (writers)
                        {
                            writers.Remove(responseStream);
                        }
                    }
                    
                    currentRoom = null;
                }
                else if (!string.IsNullOrEmpty(currentRoom))
                {
                    // Broadcast message to room
                    await BroadcastToRoom(currentRoom, message);
                }
            }
        }
        finally
        {
            // Cleanup on disconnect
            if (currentRoom != null && _rooms.TryGetValue(currentRoom, out var writers))
            {
                lock (writers)
                {
                    writers.Remove(responseStream);
                }
            }
        }
    }
    
    private async Task BroadcastToRoom(string roomId, ChatMessage message)
    {
        if (!_rooms.TryGetValue(roomId, out var writers))
            return;
        
        List<IServerStreamWriter<ChatMessage>> snapshot;
        lock (writers)
        {
            snapshot = writers.ToList();
        }
        
        var tasks = snapshot.Select(async writer =>
        {
            try
            {
                await writer.WriteAsync(message);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to send message to client");
            }
        });
        
        await Task.WhenAll(tasks);
    }
}

// Client: Chat participant
var client = new ChatService.ChatServiceClient(channel);
var call = client.Chat();

// Start receiving messages
var receiveTask = Task.Run(async () =>
{
    try
    {
        await foreach (var message in call.ResponseStream.ReadAllAsync())
        {
            switch (message.Type)
            {
                case MessageType.Join:
                    Console.WriteLine($"{message.Username} joined");
                    break;
                case MessageType.Leave:
                    Console.WriteLine($"{message.Username} left");
                    break;
                case MessageType.Text:
                    Console.WriteLine($"{message.Username}: {message.Content}");
                    break;
                case MessageType.Typing:
                    Console.WriteLine($"{message.Username} is typing...");
                    break;
            }
        }
    }
    catch (RpcException ex) when (ex.StatusCode == StatusCode.Cancelled)
    {
        Console.WriteLine("Chat ended");
    }
});

// Join room
await call.RequestStream.WriteAsync(new ChatMessage
{
    Id = Guid.NewGuid().ToString(),
    UserId = userId,
    Username = username,
    RoomId = roomId,
    Type = MessageType.Join,
    Timestamp = Timestamp.FromDateTime(DateTime.UtcNow)
});

// Send messages
while (true)
{
    var input = Console.ReadLine();
    
    if (input == "/quit")
        break;
    
    await call.RequestStream.WriteAsync(new ChatMessage
    {
        Id = Guid.NewGuid().ToString(),
        UserId = userId,
        Username = username,
        RoomId = roomId,
        Content = input,
        Type = MessageType.Text,
        Timestamp = Timestamp.FromDateTime(DateTime.UtcNow)
    });
}

// Leave room
await call.RequestStream.WriteAsync(new ChatMessage
{
    Id = Guid.NewGuid().ToString(),
    UserId = userId,
    Username = username,
    RoomId = roomId,
    Type = MessageType.Leave,
    Timestamp = Timestamp.FromDateTime(DateTime.UtcNow)
});

await call.RequestStream.CompleteAsync();
await receiveTask;
```

### Handling Backpressure

```csharp
// Server: Apply backpressure
public override async Task StreamData(
    StreamDataRequest request,
    IServerStreamWriter<DataChunk> responseStream,
    ServerCallContext context)
{
    var semaphore = new SemaphoreSlim(10); // Max 10 in-flight
    
    await foreach (var chunk in GetDataChunks())
    {
        await semaphore.WaitAsync(context.CancellationToken);
        
        // Write with backpressure
        _ = Task.Run(async () =>
        {
            try
            {
                await responseStream.WriteAsync(chunk);
            }
            finally
            {
                semaphore.Release();
            }
        });
    }
    
    // Wait for all writes to complete
    for (int i = 0; i < 10; i++)
    {
        await semaphore.WaitAsync();
    }
}

// Client: Control flow
var call = client.StreamData(request);
var channel = Channel.CreateBounded<DataChunk>(new BoundedChannelOptions(100)
{
    FullMode = BoundedChannelFullMode.Wait
});

// Read into channel (applies backpressure)
var readTask = Task.Run(async () =>
{
    try
    {
        await foreach (var chunk in call.ResponseStream.ReadAllAsync())
        {
            await channel.Writer.WriteAsync(chunk); // Blocks when full
        }
    }
    finally
    {
        channel.Writer.Complete();
    }
});

// Process at controlled rate
await foreach (var chunk in channel.Reader.ReadAllAsync())
{
    await ProcessChunk(chunk);
    await Task.Delay(100); // Simulated processing time
}

await readTask;
```

### Cancellation and Timeouts

```csharp
// Server: Respect cancellation
public override async Task StreamData(
    StreamDataRequest request,
    IServerStreamWriter<DataChunk> responseStream,
    ServerCallContext context)
{
    try
    {
        while (!context.CancellationToken.IsCancellationRequested)
        {
            var chunk = await GetNextChunk(context.CancellationToken);
            await responseStream.WriteAsync(chunk, context.CancellationToken);
        }
    }
    catch (OperationCanceledException)
    {
        _logger.LogInformation("Stream cancelled by client");
    }
}

// Client: Cancel stream
var cts = new CancellationTokenSource();

var call = client.StreamData(request, cancellationToken: cts.Token);

// Cancel after 30 seconds or on user action
cts.CancelAfter(TimeSpan.FromSeconds(30));

try
{
    await foreach (var chunk in call.ResponseStream.ReadAllAsync())
    {
        ProcessChunk(chunk);
    }
}
catch (RpcException ex) when (ex.StatusCode == StatusCode.Cancelled)
{
    Console.WriteLine("Stream cancelled");
}

// Client: Set deadline
var deadline = DateTime.UtcNow.AddSeconds(30);
var call = client.StreamData(request, deadline: deadline);
```
