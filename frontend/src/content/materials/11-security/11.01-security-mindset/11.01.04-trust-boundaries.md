---

## Introduction

Trust boundaries are critical security concepts that separate components with different levels of trust. Traditional security models assumed everything inside a network perimeter was trustworthy, but modern threats require a different approach: Zero Trust. This model assumes breach, verifies every request, and minimizes blast radius regardless of where the request originates.

## Key Concepts Summary

### Understanding Trust Boundaries

```csharp
// Trust boundary example: External input vs. internal processing

// UNTRUSTED ZONE: Data from external sources
public class OrderController : ControllerBase
{
    [HttpPost]
    public async Task<IActionResult> CreateOrder([FromBody] CreateOrderRequest request)
    {
        // ⚠️ TRUST BOUNDARY CROSSING HERE ⚠️
        // Data crosses from untrusted (HTTP request) to trusted (application)
        
        // MUST validate at the boundary
        if (!ModelState.IsValid)
            return BadRequest(ModelState);
        
        // MUST sanitize and validate
        var sanitizedRequest = _validator.ValidateAndSanitize(request);
        
        // NOW entering TRUSTED ZONE
        var order = await _orderService.CreateOrderAsync(sanitizedRequest);
        
        return CreatedAtAction(nameof(GetOrder), new { id = order.Id }, order);
    }
}

// Trust boundaries exist at:
// 1. User input → Application
// 2. Application → Database
// 3. External API → Your service
// 4. Frontend → Backend
// 5. Public network → Private network
```

### Three Pillars of Zero Trust

```csharp
// 1. VERIFY EXPLICITLY
// Always authenticate and authorize based on all available data points

public class ZeroTrustAuthenticationMiddleware
{
    public async Task InvokeAsync(HttpContext context)
    {
        // Verify identity
        if (!context.User.Identity?.IsAuthenticated ?? true)
        {
            context.Response.StatusCode = 401;
            return;
        }
        
        // Verify device
        var deviceId = context.Request.Headers["X-Device-Id"];
        if (!await _deviceService.IsDeviceRegistered(deviceId))
        {
            await LogSuspiciousActivity(context);
            context.Response.StatusCode = 403;
            return;
        }
        
        // Verify location/context
        var ipAddress = context.Connection.RemoteIpAddress;
        if (await _threatService.IsHighRiskLocation(ipAddress))
        {
            // Require additional verification
            await RequireMfaChallenge(context);
        }
        
        await _next(context);
    }
}

// 2. USE LEAST PRIVILEGE ACCESS
// Grant minimum necessary permissions for each request

public class ResourceAuthorizationHandler
{
    public async Task<bool> AuthorizeAsync(
        ClaimsPrincipal user, 
        string resource, 
        string action)
    {
        // Check specific permission for this exact action
        var hasPermission = await _permissionService.HasPermissionAsync(
            user.GetUserId(), 
            resource, 
            action
        );
        
        // Time-limited permissions
        if (hasPermission)
        {
            var permission = await _permissionService.GetPermissionAsync(
                user.GetUserId(), resource);
            
            if (permission.ExpiresAt < DateTime.UtcNow)
                return false;
        }
        
        return hasPermission;
    }
}

// 3. ASSUME BREACH
// Minimize blast radius, segment access, encrypt everything

public class DataAccessService
{
    public async Task<CustomerData> GetCustomerDataAsync(int customerId)
    {
        // Encrypt data at rest
        var encryptedData = await _repository.GetEncryptedDataAsync(customerId);
        
        // Decrypt only what's needed
        var decryptedData = _encryption.Decrypt(encryptedData);
        
        // Log access for monitoring
        await _auditLog.LogDataAccessAsync(
            userId: _currentUser.Id,
            customerId: customerId,
            action: "Read",
            timestamp: DateTime.UtcNow
        );
        
        // Return only necessary fields (minimize exposure)
        return new CustomerData
        {
            Id = decryptedData.Id,
            Name = decryptedData.Name,
            // Don't return SSN, credit card, etc. unless explicitly needed
        };
    }
}
```

### Validating at Trust Boundaries

```csharp
// Always validate when crossing trust boundaries
public class TrustBoundaryValidator
{
    // Boundary 1: External input → Application
    public Result<CreateUserCommand> ValidateExternalInput(CreateUserRequest request)
    {
        var errors = new List<string>();
        
        // Validate format
        if (string.IsNullOrWhiteSpace(request.Email))
            errors.Add("Email is required");
        else if (!IsValidEmail(request.Email))
            errors.Add("Invalid email format");
        
        // Validate length (prevent DoS)
        if (request.Name?.Length > 100)
            errors.Add("Name too long");
        
        // Sanitize (prevent injection)
        var sanitizedName = _sanitizer.SanitizeInput(request.Name);
        
        if (errors.Any())
            return Result<CreateUserCommand>.Failure(errors);
        
        return Result<CreateUserCommand>.Success(new CreateUserCommand
        {
            Email = request.Email.ToLowerInvariant(),
            Name = sanitizedName
        });
    }
    
    // Boundary 2: Application → Database
    public string CreateSafeQuery(int userId)
    {
        // Use parameterized queries (never string concatenation)
        return "SELECT * FROM Users WHERE Id = @UserId";
    }
    
    // Boundary 3: Database → Application
    public User ValidateDatabaseOutput(dynamic dbResult)
    {
        // Validate data integrity
        if (dbResult == null)
            throw new DataIntegrityException("User not found");
        
        // Type safety
        return new User
        {
            Id = Convert.ToInt32(dbResult.Id),
            Email = Convert.ToString(dbResult.Email) ?? "",
            Name = Convert.ToString(dbResult.Name) ?? ""
        };
    }
}
```

### Network Segmentation

```
┌─────────────────────────────────────────────┐
│ Public Internet (UNTRUSTED)                 │
└─────────────────────────────────────────────┘
                    ↓
        [Trust Boundary - Firewall]
                    ↓
┌─────────────────────────────────────────────┐
│ DMZ - Web Tier (LIMITED TRUST)              │
│ - Web servers                                │
│ - Load balancers                             │
└─────────────────────────────────────────────┘
                    ↓
        [Trust Boundary - Network Segment]
                    ↓
┌─────────────────────────────────────────────┐
│ Application Tier (MORE TRUSTED)             │
│ - API servers                                │
│ - Business logic                             │
└─────────────────────────────────────────────┘
                    ↓
        [Trust Boundary - Network Segment]
                    ↓
┌─────────────────────────────────────────────┐
│ Data Tier (MOST TRUSTED)                    │
│ - Databases                                  │
│ - Data stores                                │
└─────────────────────────────────────────────┘
```
