## Introduction

Fail-safe defaults is a security principle that ensures systems default to a secure state rather than an unsafe state when errors or unexpected conditions occur. Unless an entity is explicitly granted access to a resource, it should be denied by default. This principle is fundamental to building resilient security architectures that remain secure even when things go wrong.

## Key Concepts Summary

### Fail-Safe Defaults in Action

```csharp
// Bad: Defaults to allowing access
public class InsecureAuthorizationService
{
    public bool CanUserAccessResource(User user, Resource resource)
    {
        try
        {
            // If an exception occurs, this returns false (default bool value)
            // but the calling code might not check it properly
            return _permissionService.HasPermission(user.Id, resource.Id);
        }
        catch
        {
            // Silent failure - caller might assume success
            return false;
        }
    }
}

// Good: Explicitly denies by default, fails securely
public class SecureAuthorizationService
{
    public AuthorizationResult CanUserAccessResource(User user, Resource resource)
    {
        if (user == null || resource == null)
            return AuthorizationResult.Denied("Invalid user or resource");
        
        try
        {
            var hasPermission = _permissionService.HasPermission(user.Id, resource.Id);
            
            return hasPermission 
                ? AuthorizationResult.Allowed()
                : AuthorizationResult.Denied("User lacks required permissions");
        }
        catch (Exception ex)
        {
            // Log the error but deny access
            _logger.LogError(ex, "Authorization check failed for user {UserId}", user.Id);
            return AuthorizationResult.Denied("Authorization system error");
        }
    }
}

public class AuthorizationResult
{
    public bool IsAllowed { get; }
    public string Reason { get; }
    
    private AuthorizationResult(bool isAllowed, string reason)
    {
        IsAllowed = isAllowed;
        Reason = reason;
    }
    
    public static AuthorizationResult Allowed() => new(true, "Access granted");
    public static AuthorizationResult Denied(string reason) => new(false, reason);
}
```

### Secure Configuration Defaults

```csharp
// Bad: Insecure defaults that require manual hardening
public class ApiConfiguration
{
    public bool EnableCors { get; set; } = true;  // Allows all origins!
    public bool RequireHttps { get; set; } = false;
    public int MaxRequestSize { get; set; } = int.MaxValue;
    public bool EnableDetailedErrors { get; set; } = true;
}

// Good: Secure by default
public class SecureApiConfiguration
{
    public bool EnableCors { get; set; } = false;  // Disabled by default
    public bool RequireHttps { get; set; } = true;
    public int MaxRequestSize { get; set; } = 1024 * 1024; // 1 MB default
    public bool EnableDetailedErrors { get; set; } = false;
    public List<string> AllowedOrigins { get; set; } = new();
    
    public void EnableCorsForOrigins(params string[] origins)
    {
        EnableCors = true;
        AllowedOrigins.AddRange(origins);
    }
}
```

### Deny by Default Access Control

```csharp
// Middleware that denies all requests unless explicitly allowed
public class DefaultDenyAuthorizationMiddleware
{
    private readonly RequestDelegate _next;
    private readonly HashSet<string> _publicEndpoints;
    
    public DefaultDenyAuthorizationMiddleware(
        RequestDelegate next, 
        IConfiguration config)
    {
        _next = next;
        // Only these endpoints are accessible without authentication
        _publicEndpoints = config.GetSection("PublicEndpoints")
            .Get<HashSet<string>>() ?? new HashSet<string>();
    }
    
    public async Task InvokeAsync(HttpContext context)
    {
        var path = context.Request.Path.Value?.ToLower() ?? "";
        
        // Public endpoints explicitly listed
        if (_publicEndpoints.Contains(path))
        {
            await _next(context);
            return;
        }
        
        // Everything else requires authentication
        if (!context.User.Identity?.IsAuthenticated ?? true)
        {
            context.Response.StatusCode = 401;
            await context.Response.WriteAsync("Authentication required");
            return;
        }
        
        await _next(context);
    }
}
```

### Remove Unnecessary Features

```csharp
// Configure ASP.NET Core to be secure by default
public class Startup
{
    public void ConfigureServices(IServiceCollection services)
    {
        services.AddControllers(options =>
        {
            // Disable unnecessary HTTP methods
            options.Filters.Add(new HttpMethodsFilter(
                allowedMethods: new[] { "GET", "POST", "PUT", "DELETE" }
            ));
            
            // Require HTTPS
            options.Filters.Add(new RequireHttpsAttribute());
            
            // Set reasonable size limits
            options.MaxModelBindingCollectionSize = 100;
        });
        
        // Remove server header
        services.Configure<KestrelServerOptions>(options =>
        {
            options.AddServerHeader = false;
        });
        
        // Disable detailed error pages in production
        services.AddProblemDetails(options =>
        {
            options.IncludeExceptionDetails = (ctx, ex) => 
                ctx.RequestServices.GetRequiredService<IHostEnvironment>()
                    .IsDevelopment();
        });
    }
}
```

### Key Principles

1. **Deny by Default:** Access is denied unless explicitly granted
2. **Fail Securely:** Errors should result in denied access, not granted access
3. **Minimal Surface Area:** Remove or disable unnecessary features
4. **Secure Configurations:** Default settings should be the most secure option
5. **Explicit Allow-lists:** Use allow-lists instead of deny-lists

After completing the materials, try this exercise (do not spend more than 10 minutes):

Review this code and identify the fail-safe violations:

```csharp
public class UserService
{
    public async Task<User> GetUserProfile(int userId, int requesterId)
    {
        try
        {
            var user = await _repository.GetUserAsync(userId);
            
            // Users can view any profile by default
            if (requesterId == userId || true) // Always true!
                return user;
            
            return null;
        }
        catch
        {
            return await _repository.GetUserAsync(userId); // Try again?
        }
    }
}
```

What should be fixed to make this fail-safe?
