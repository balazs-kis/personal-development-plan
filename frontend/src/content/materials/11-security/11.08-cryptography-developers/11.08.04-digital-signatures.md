## Introduction

Digital signatures use asymmetric cryptography to prove that data came from a specific sender and hasn't been modified. Unlike encryption (which provides confidentiality), signatures provide authentication, integrity, and non-repudiation. Digital signatures are used everywhere: code signing, document signing, SSL/TLS certificates, and API authentication.

**Key principle:** Sign hashes, not raw data. Verify signatures before trusting data. Understand certificate chains and trust models.

## Key Concepts Summary

### How Digital Signatures Work

```
1. Sender creates a hash of the message
2. Sender encrypts the hash with their PRIVATE key (signature)
3. Sender sends: message + signature
4. Receiver decrypts signature with sender's PUBLIC key (gets hash)
5. Receiver computes hash of received message
6. If both hashes match: signature is valid!

This proves:
- Authentication: Message came from sender (only they have the private key)
- Integrity: Message wasn't modified (hash would change)
- Non-repudiation: Sender can't deny sending it (their key signed it)
```

### Creating and Verifying RSA Signatures

```csharp
using System.Security.Cryptography;
using System.Text;

public class DigitalSignatureService
{
    // Sign data with private key
    public static byte[] SignData(byte[] data, RSA privateKey)
    {
        // Sign using SHA-256 hash
        return privateKey.SignData(
            data, 
            HashAlgorithmName.SHA256, 
            RSASignaturePadding.Pkcs1);
    }
    
    // Verify signature with public key
    public static bool VerifySignature(byte[] data, byte[] signature, RSA publicKey)
    {
        return publicKey.VerifyData(
            data, 
            signature, 
            HashAlgorithmName.SHA256, 
            RSASignaturePadding.Pkcs1);
    }
}

// Usage
using var rsa = RSA.Create(2048);

// Sender signs the data
string message = "This is an important contract";
byte[] data = Encoding.UTF8.GetBytes(message);
byte[] signature = DigitalSignatureService.SignData(data, rsa);

Console.WriteLine($"Message: {message}");
Console.WriteLine($"Signature: {Convert.ToBase64String(signature)}");

// Receiver verifies the signature
bool isValid = DigitalSignatureService.VerifySignature(data, signature, rsa);
Console.WriteLine($"Signature valid: {isValid}");

// Try tampering with the message
byte[] tamperedData = Encoding.UTF8.GetBytes("This is a tampered contract");
bool isStillValid = DigitalSignatureService.VerifySignature(tamperedData, signature, rsa);
Console.WriteLine($"Tampered signature valid: {isStillValid}"); // False!
```

### Working with X.509 Certificates

```csharp
using System.Security.Cryptography.X509Certificates;

public class CertificateSignatureService
{
    // Load certificate from file
    public static X509Certificate2 LoadCertificate(string pfxPath, string password)
    {
        return new X509Certificate2(pfxPath, password);
    }
    
    // Sign with certificate
    public static byte[] SignWithCertificate(byte[] data, X509Certificate2 certificate)
    {
        using var rsa = certificate.GetRSAPrivateKey();
        if (rsa == null)
            throw new InvalidOperationException("Certificate does not have a private key");
        
        return rsa.SignData(data, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
    }
    
    // Verify with certificate
    public static bool VerifyWithCertificate(
        byte[] data, 
        byte[] signature, 
        X509Certificate2 certificate)
    {
        using var rsa = certificate.GetRSAPublicKey();
        if (rsa == null)
            throw new InvalidOperationException("Certificate does not have a public key");
        
        return rsa.VerifyData(data, signature, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);
    }
    
    // Verify certificate chain and validity
    public static bool ValidateCertificate(X509Certificate2 certificate)
    {
        using var chain = new X509Chain();
        chain.ChainPolicy.RevocationMode = X509RevocationMode.Online;
        chain.ChainPolicy.RevocationFlag = X509RevocationFlag.ExcludeRoot;
        
        bool isValid = chain.Build(certificate);
        
        if (!isValid)
        {
            foreach (var element in chain.ChainElements)
            {
                foreach (var status in element.ChainElementStatus)
                {
                    Console.WriteLine($"Chain error: {status.StatusInformation}");
                }
            }
        }
        
        return isValid;
    }
}

// Usage
var cert = CertificateSignatureService.LoadCertificate("mycert.pfx", "password");

// Check certificate validity
if (CertificateSignatureService.ValidateCertificate(cert))
{
    byte[] data = Encoding.UTF8.GetBytes("Signed document");
    byte[] signature = CertificateSignatureService.SignWithCertificate(data, cert);
    
    bool isValid = CertificateSignatureService.VerifyWithCertificate(data, signature, cert);
    Console.WriteLine($"Signature valid: {isValid}");
}
```

### Signing XML Documents

```csharp
using System.Security.Cryptography.Xml;
using System.Xml;

public class XmlSignatureService
{
    // Sign an XML document
    public static void SignXml(XmlDocument doc, RSA key)
    {
        // Create SignedXml object
        var signedXml = new SignedXml(doc);
        signedXml.SigningKey = key;
        
        // Create reference to sign entire document
        var reference = new Reference("");
        reference.AddTransform(new XmlDsigEnvelopedSignatureTransform());
        signedXml.AddReference(reference);
        
        // Compute signature
        signedXml.ComputeSignature();
        
        // Get XML representation of signature
        XmlElement xmlSignature = signedXml.GetXml();
        
        // Append signature to document
        doc.DocumentElement.AppendChild(doc.ImportNode(xmlSignature, true));
    }
    
    // Verify XML signature
    public static bool VerifyXml(XmlDocument doc, RSA key)
    {
        // Find Signature element
        XmlNodeList nodeList = doc.GetElementsByTagName("Signature");
        if (nodeList.Count == 0)
            return false;
        
        // Load signature
        var signedXml = new SignedXml(doc);
        signedXml.LoadXml((XmlElement)nodeList[0]);
        
        // Verify signature
        return signedXml.CheckSignature(key);
    }
}

// Usage
var xmlDoc = new XmlDocument();
xmlDoc.LoadXml(@"<document>
    <data>Important content</data>
</document>");

using var rsa = RSA.Create(2048);

// Sign
XmlSignatureService.SignXml(xmlDoc, rsa);
Console.WriteLine("Signed XML:");
Console.WriteLine(xmlDoc.OuterXml);

// Verify
bool isValid = XmlSignatureService.VerifyXml(xmlDoc, rsa);
Console.WriteLine($"Signature valid: {isValid}");
```

### Code Signing Scenario

```csharp
public class CodeSigningService
{
    private readonly X509Certificate2 _signingCertificate;
    
    public CodeSigningService(X509Certificate2 certificate)
    {
        _signingCertificate = certificate;
    }
    
    // Sign an assembly or file
    public SignedPackage SignFile(string filePath)
    {
        byte[] fileData = File.ReadAllBytes(filePath);
        byte[] fileHash = SHA256.HashData(fileData);
        
        using var rsa = _signingCertificate.GetRSAPrivateKey();
        byte[] signature = rsa.SignHash(
            fileHash, 
            HashAlgorithmName.SHA256, 
            RSASignaturePadding.Pkcs1);
        
        return new SignedPackage
        {
            FileName = Path.GetFileName(filePath),
            FileData = fileData,
            Signature = signature,
            Certificate = _signingCertificate.Export(X509ContentType.Cert),
            SignedAt = DateTime.UtcNow,
            HashAlgorithm = "SHA256"
        };
    }
    
    // Verify signed file
    public bool VerifySignedFile(SignedPackage package)
    {
        // Load certificate from package
        var cert = new X509Certificate2(package.Certificate);
        
        // Verify certificate is trusted
        if (!CertificateSignatureService.ValidateCertificate(cert))
        {
            Console.WriteLine("Certificate is not trusted");
            return false;
        }
        
        // Compute hash of file data
        byte[] fileHash = SHA256.HashData(package.FileData);
        
        // Verify signature
        using var rsa = cert.GetRSAPublicKey();
        return rsa.VerifyHash(
            fileHash, 
            package.Signature, 
            HashAlgorithmName.SHA256, 
            RSASignaturePadding.Pkcs1);
    }
}

public class SignedPackage
{
    public string FileName { get; set; }
    public byte[] FileData { get; set; }
    public byte[] Signature { get; set; }
    public byte[] Certificate { get; set; }
    public DateTime SignedAt { get; set; }
    public string HashAlgorithm { get; set; }
}
```

### JWT Signature (Simplified)

```csharp
public class JwtSignatureExample
{
    // Sign JWT payload (simplified - use a library in production)
    public static string CreateSignedJwt(object payload, RSA privateKey)
    {
        // Header
        var header = new { alg = "RS256", typ = "JWT" };
        string headerJson = JsonSerializer.Serialize(header);
        string headerBase64 = Base64UrlEncode(Encoding.UTF8.GetBytes(headerJson));
        
        // Payload
        string payloadJson = JsonSerializer.Serialize(payload);
        string payloadBase64 = Base64UrlEncode(Encoding.UTF8.GetBytes(payloadJson));
        
        // Sign header.payload
        string dataToSign = $"{headerBase64}.{payloadBase64}";
        byte[] signature = privateKey.SignData(
            Encoding.UTF8.GetBytes(dataToSign),
            HashAlgorithmName.SHA256,
            RSASignaturePadding.Pkcs1);
        
        string signatureBase64 = Base64UrlEncode(signature);
        
        return $"{dataToSign}.{signatureBase64}";
    }
    
    // Verify JWT signature
    public static bool VerifyJwt(string jwt, RSA publicKey)
    {
        string[] parts = jwt.Split('.');
        if (parts.Length != 3)
            return false;
        
        string dataToVerify = $"{parts[0]}.{parts[1]}";
        byte[] signature = Base64UrlDecode(parts[2]);
        
        return publicKey.VerifyData(
            Encoding.UTF8.GetBytes(dataToVerify),
            signature,
            HashAlgorithmName.SHA256,
            RSASignaturePadding.Pkcs1);
    }
    
    private static string Base64UrlEncode(byte[] input)
    {
        return Convert.ToBase64String(input)
            .TrimEnd('=')
            .Replace('+', '-')
            .Replace('/', '_');
    }
    
    private static byte[] Base64UrlDecode(string input)
    {
        string base64 = input.Replace('-', '+').Replace('_', '/');
        switch (input.Length % 4)
        {
            case 2: base64 += "=="; break;
            case 3: base64 += "="; break;
        }
        return Convert.FromBase64String(base64);
    }
}
```

### Certificate Store Access

```csharp
public class CertificateStoreService
{
    // Find certificate in Windows certificate store
    public static X509Certificate2 FindCertificate(
        string thumbprint, 
        StoreName storeName = StoreName.My,
        StoreLocation storeLocation = StoreLocation.CurrentUser)
    {
        using var store = new X509Store(storeName, storeLocation);
        store.Open(OpenFlags.ReadOnly);
        
        var certificates = store.Certificates.Find(
            X509FindType.FindByThumbprint,
            thumbprint,
            validOnly: true);
        
        if (certificates.Count == 0)
            throw new InvalidOperationException($"Certificate with thumbprint {thumbprint} not found");
        
        return certificates[0];
    }
    
    // List all certificates in store
    public static void ListCertificates()
    {
        using var store = new X509Store(StoreName.My, StoreLocation.CurrentUser);
        store.Open(OpenFlags.ReadOnly);
        
        foreach (var cert in store.Certificates)
        {
            Console.WriteLine($"Subject: {cert.Subject}");
            Console.WriteLine($"Thumbprint: {cert.Thumbprint}");
            Console.WriteLine($"Valid: {cert.NotBefore} to {cert.NotAfter}");
            Console.WriteLine($"Has Private Key: {cert.HasPrivateKey}");
            Console.WriteLine();
        }
    }
}
```

### Best Practices

```csharp
// ✅ DO: Sign hashes for large data
byte[] hash = SHA256.HashData(largeFile);
byte[] signature = rsa.SignHash(hash, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);

// ✅ DO: Verify certificate chain before trusting signatures
if (!ValidateCertificate(cert))
    throw new SecurityException("Certificate not trusted");

// ✅ DO: Use SHA-256 or stronger for signatures
rsa.SignData(data, HashAlgorithmName.SHA256, RSASignaturePadding.Pkcs1);

// ✅ DO: Check certificate expiration
if (cert.NotBefore > DateTime.Now || cert.NotAfter < DateTime.Now)
    throw new SecurityException("Certificate expired or not yet valid");

// ❌ DON'T: Use weak hash algorithms
// rsa.SignData(data, HashAlgorithmName.SHA1, ...); // WEAK!

// ❌ DON'T: Skip certificate validation
// Just because a signature is cryptographically valid doesn't mean
// you should trust it - verify the certificate chain!

// ❌ DON'T: Use the same key for encryption and signing
// Use separate key pairs for different purposes

// ✅ DO: Include timestamps in signatures
public class TimestampedSignature
{
    public byte[] Signature { get; set; }
    public DateTime SignedAt { get; set; }
    public string SignerIdentity { get; set; }
}
```
