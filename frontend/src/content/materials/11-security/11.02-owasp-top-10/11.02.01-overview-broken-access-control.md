## Introduction

The OWASP Top 10 is the industry-standard awareness document for web application security risks. First published in 2003 and updated regularly, it represents the most critical security risks based on real-world data from security professionals worldwide. The 2025 edition reflects current threat landscapes with broken access control remaining the #1 vulnerability, affecting 100% of tested applications.

### What Changed in 2025?

- **Broken Access Control** remains #1 (moved from #5 in 2017)
- **Security Misconfiguration** jumped to #2 (100% of apps have some misconfiguration)
- **Software Supply Chain Failures** is a new category (evolved from "Vulnerable Components")
- Focus shifted to root causes over symptoms

### Broken Access Control Deep Dive

```csharp
// BAD: Insecure Direct Object Reference (IDOR)
[HttpGet("orders/{orderId}")]
public async Task<IActionResult> GetOrder(int orderId)
{
    // No authorization check - any authenticated user can access any order!
    var order = await _orderService.GetOrderAsync(orderId);
    return Ok(order);
}

// GOOD: Proper authorization check
[HttpGet("orders/{orderId}")]
public async Task<IActionResult> GetOrder(int orderId)
{
    var userId = User.GetUserId();
    var order = await _orderService.GetOrderAsync(orderId);
    
    // Verify user owns this order
    if (order.UserId != userId && !User.IsInRole("Admin"))
    {
        return Forbid(); // 403 Forbidden
    }
    
    return Ok(order);
}
```

### Common Access Control Vulnerabilities

```csharp
// 1. URL Manipulation
// User changes URL parameter to access others' data
// /api/users/123 → /api/users/456

// 2. Privilege Escalation
// BAD: Client-side role check
public class UserDto
{
    public int Id { get; set; }
    public string Name { get; set; }
    public bool IsAdmin { get; set; } // Never trust client!
}

[HttpPost("delete-user")]
public async Task<IActionResult> DeleteUser([FromBody] DeleteUserRequest request)
{
    // Attacker can set IsAdmin=true in the request!
    if (request.IsAdmin)
    {
        await _userService.DeleteAsync(request.UserId);
    }
    return Ok();
}

// GOOD: Server-side authorization
[HttpPost("delete-user")]
[Authorize(Roles = "Admin")]
public async Task<IActionResult> DeleteUser([FromBody] DeleteUserRequest request)
{
    // Role verified by [Authorize] attribute on server
    await _userService.DeleteAsync(request.UserId);
    return Ok();
}

// 3. Missing Function-Level Access Control
// BAD: Admin endpoint without protection
[HttpGet("admin/users")]
public async Task<IActionResult> GetAllUsers()
{
    // No authorization check - anyone can call this!
    var users = await _userService.GetAllUsersAsync();
    return Ok(users);
}

// GOOD: Protected admin endpoint
[HttpGet("admin/users")]
[Authorize(Roles = "Admin")]
public async Task<IActionResult> GetAllUsers()
{
    var users = await _userService.GetAllUsersAsync();
    return Ok(users);
}
```

### Preventing Broken Access Control

```csharp
// Principle 1: Deny by Default
public class Startup
{
    public void ConfigureServices(IServiceCollection services)
    {
        services.AddAuthorization(options =>
        {
            // Require authentication by default
            options.FallbackPolicy = new AuthorizationPolicyBuilder()
                .RequireAuthenticatedUser()
                .Build();
        });
    }
}

// Principle 2: Centralized Authorization
public class ResourceAuthorizationHandler : 
    AuthorizationHandler<OperationAuthorizationRequirement, Order>
{
    protected override Task HandleRequirementAsync(
        AuthorizationHandlerContext context,
        OperationAuthorizationRequirement requirement,
        Order resource)
    {
        var userId = context.User.GetUserId();
        
        // Centralized logic for order access
        if (resource.UserId == userId || 
            context.User.IsInRole("Admin") ||
            context.User.HasClaim("Department", resource.Department))
        {
            context.Succeed(requirement);
        }
        
        return Task.CompletedTask;
    }
}

// Usage
[HttpGet("orders/{orderId}")]
public async Task<IActionResult> GetOrder(int orderId)
{
    var order = await _orderService.GetOrderAsync(orderId);
    
    var authResult = await _authorizationService.AuthorizeAsync(
        User, order, Operations.Read);
    
    if (!authResult.Succeeded)
        return Forbid();
    
    return Ok(order);
}

// Principle 3: Log Access Control Failures
public class AuditLoggingMiddleware
{
    public async Task InvokeAsync(HttpContext context)
    {
        await _next(context);
        
        // Log 403 Forbidden responses
        if (context.Response.StatusCode == 403)
        {
            await _auditLog.LogAsync(new AccessDeniedEvent
            {
                UserId = context.User.GetUserId(),
                Path = context.Request.Path,
                Method = context.Request.Method,
                Timestamp = DateTime.UtcNow,
                IpAddress = context.Connection.RemoteIpAddress?.ToString()
            });
        }
    }
}

// Principle 4: Disable Directory Listing
public class Startup
{
    public void Configure(IApplicationBuilder app)
    {
        // Don't use app.UseDirectoryBrowser()
        // Don't use app.UseStaticFiles() for sensitive directories
        
        app.UseStaticFiles(new StaticFileOptions
        {
            ServeUnknownFileTypes = false,
            DefaultContentType = "application/octet-stream"
        });
    }
}
```

### Rate Limiting for APIs

```csharp
// Prevent brute force access attempts
public class RateLimitingMiddleware
{
    private readonly IMemoryCache _cache;
    
    public async Task InvokeAsync(HttpContext context)
    {
        var key = $"rate_limit_{context.Connection.RemoteIpAddress}";
        
        if (!_cache.TryGetValue(key, out int requestCount))
        {
            requestCount = 0;
        }
        
        if (requestCount > 100) // 100 requests per minute
        {
            context.Response.StatusCode = 429; // Too Many Requests
            await context.Response.WriteAsync("Rate limit exceeded");
            return;
        }
        
        _cache.Set(key, requestCount + 1, TimeSpan.FromMinutes(1));
        await _next(context);
    }
}
```
