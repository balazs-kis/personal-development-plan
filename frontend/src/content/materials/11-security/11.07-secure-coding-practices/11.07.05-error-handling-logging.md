## Learning Objectives

By the end of this lesson, you will be able to:
- Implement global exception handling in ASP.NET Core
- Prevent information disclosure through error messages
- Log security events effectively without exposing sensitive data
- Use structured logging for better security monitoring
- Apply different error handling strategies for development vs production
- Implement correlation IDs for request tracing

---

## Introduction

Error handling and logging are critical for both security and operations. Poor error handling can leak sensitive information to attackers (stack traces, connection strings, internal paths), while inadequate logging makes it impossible to detect and respond to security incidents. This lesson teaches you how to handle errors securely and log security-relevant events properly.

**Key principle:** Fail securely. Show generic error messages to users while logging detailed information for developers. Never expose sensitive data in logs or error messages.

### Global Exception Handling Middleware

```csharp
// Program.cs
var builder = WebApplication.CreateBuilder(args);
var app = builder.Build();

if (app.Environment.IsDevelopment())
{
    // Detailed errors for developers
    app.UseDeveloperExceptionPage();
}
else
{
    // Generic errors for production
    app.UseExceptionHandler("/error");
    app.UseHsts();
}

// Error handling endpoint
app.MapGet("/error", (HttpContext context) =>
{
    // Log the original exception
    var exceptionFeature = context.Features.Get<IExceptionHandlerFeature>();
    var exception = exceptionFeature?.Error;
    
    var logger = context.RequestServices.GetRequiredService<ILogger<Program>>();
    logger.LogError(exception, "Unhandled exception occurred");
    
    // Return generic error to user
    return Results.Problem(
        title: "An error occurred",
        statusCode: StatusCodes.Status500InternalServerError);
});

app.Run();
```

### Custom Exception Handling Middleware

```csharp
public class GlobalExceptionHandlerMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ILogger<GlobalExceptionHandlerMiddleware> _logger;
    
    public GlobalExceptionHandlerMiddleware(
        RequestDelegate next,
        ILogger<GlobalExceptionHandlerMiddleware> logger)
    {
        _next = next;
        _logger = logger;
    }
    
    public async Task InvokeAsync(HttpContext context)
    {
        try
        {
            await _next(context);
        }
        catch (Exception ex)
        {
            await HandleExceptionAsync(context, ex);
        }
    }
    
    private async Task HandleExceptionAsync(HttpContext context, Exception exception)
    {
        var correlationId = context.TraceIdentifier;
        
        // Log with correlation ID
        _logger.LogError(exception, 
            "Unhandled exception. CorrelationId: {CorrelationId}", 
            correlationId);
        
        // Determine status code and message based on exception type
        var (statusCode, message) = exception switch
        {
            UnauthorizedAccessException => (StatusCodes.Status403Forbidden, "Access denied"),
            ArgumentException => (StatusCodes.Status400BadRequest, "Invalid request"),
            KeyNotFoundException => (StatusCodes.Status404NotFound, "Resource not found"),
            _ => (StatusCodes.Status500InternalServerError, "An error occurred processing your request")
        };
        
        context.Response.StatusCode = statusCode;
        context.Response.ContentType = "application/json";
        
        var response = new
        {
            error = message,
            correlationId = correlationId
            // Never include: exception.Message, exception.StackTrace
        };
        
        await context.Response.WriteAsJsonAsync(response);
    }
}

// Register middleware
app.UseMiddleware<GlobalExceptionHandlerMiddleware>();
```

### What NOT to Expose in Production

```csharp
// ❌ BAD: Exposing detailed error information
public IActionResult ProcessPayment(PaymentRequest request)
{
    try
    {
        _paymentService.Process(request);
        return Ok();
    }
    catch (Exception ex)
    {
        // NEVER do this in production!
        return BadRequest(new
        {
            error = ex.Message,  // May contain sensitive info
            stackTrace = ex.StackTrace,  // Reveals internal structure
            innerException = ex.InnerException?.Message,
            connectionString = _configuration["ConnectionString"]  // Disaster!
        });
    }
}

// ✅ GOOD: Generic error with correlation ID
public IActionResult ProcessPayment(PaymentRequest request)
{
    try
    {
        _paymentService.Process(request);
        return Ok();
    }
    catch (Exception ex)
    {
        var correlationId = HttpContext.TraceIdentifier;
        _logger.LogError(ex, "Payment processing failed. CorrelationId: {CorrelationId}", 
            correlationId);
        
        return StatusCode(500, new
        {
            error = "Payment processing failed",
            correlationId = correlationId  // User can provide this to support
        });
    }
}
```

using Serilog;

var builder = WebApplication.CreateBuilder(args);

Log.Logger = new LoggerConfiguration()
    .MinimumLevel.Information()
    .MinimumLevel.Override("Microsoft", LogEventLevel.Warning)
    .Enrich.FromLogContext()
    .Enrich.WithMachineName()
    .Enrich.WithProperty("Application", "MyApp")

        retainedFileCountLimit: 30)
    .CreateLogger();

app.Run();

### Logging Security Events

```csharp
public class AuthenticationService
{
    private readonly ILogger<AuthenticationService> _logger;
    
    public async Task<LoginResult> LoginAsync(string username, string password)
    {
        var ipAddress = _httpContextAccessor.HttpContext?.Connection.RemoteIpAddress?.ToString();
        
        var user = await _userRepository.GetByUsernameAsync(username);
        
        if (user == null)
        {
            // Log failed attempt - but don't reveal whether user exists
            _logger.LogWarning(
                "Login attempt failed for username: {Username}, IP: {IpAddress}, Reason: Invalid credentials",
                username, ipAddress);
            
            return LoginResult.Failed("Invalid username or password");
        }
        
        if (!_passwordHasher.VerifyPassword(password, user.PasswordHash))
        {
            user.FailedLoginAttempts++;
            await _userRepository.UpdateAsync(user);
            
            _logger.LogWarning(
                "Login attempt failed for user: {UserId}, Username: {Username}, IP: {IpAddress}, " +
                "Failed attempts: {FailedAttempts}, Reason: Invalid password",
                user.Id, username, ipAddress, user.FailedLoginAttempts);
            
            if (user.FailedLoginAttempts >= 5)
            {
                _logger.LogError(
                    "Account locked due to multiple failed login attempts. " +
                    "UserId: {UserId}, Username: {Username}, IP: {IpAddress}",
                    user.Id, username, ipAddress);
                
                await LockAccountAsync(user);
            }
            
            return LoginResult.Failed("Invalid username or password");
        }
        
        // Successful login
        user.FailedLoginAttempts = 0;
        user.LastLoginDate = DateTime.UtcNow;
        user.LastLoginIp = ipAddress;
        await _userRepository.UpdateAsync(user);
        
        _logger.LogInformation(
            "User logged in successfully. UserId: {UserId}, Username: {Username}, IP: {IpAddress}",
            user.Id, username, ipAddress);
        
        return LoginResult.Success(user);
    }
}
```

### What to Log and What Not to Log

```csharp
public class SecureLoggingExample
{
    private readonly ILogger<SecureLoggingExample> _logger;
    
    public async Task ProcessUserRegistration(RegisterRequest request)
    {
        // ✅ DO: Log security-relevant events
        _logger.LogInformation(
            "User registration initiated. Email: {Email}, IP: {IpAddress}",
            request.Email, GetClientIp());
        
        // ❌ DON'T: Log sensitive data
        // _logger.LogInformation("Password: {Password}", request.Password);
        // _logger.LogInformation("SSN: {SSN}", request.SocialSecurityNumber);
        // _logger.LogInformation("CreditCard: {Card}", request.CreditCard);
        
        try
        {
            var user = await _userService.CreateUserAsync(request);
            
            // ✅ DO: Log successful operations with identifiers
            _logger.LogInformation(
                "User created successfully. UserId: {UserId}, Email: {Email}",
                user.Id, user.Email);
        }
        catch (DuplicateEmailException ex)
        {
            // ✅ DO: Log security-relevant failures
            _logger.LogWarning(ex,
                "Registration failed - duplicate email. Email: {Email}, IP: {IpAddress}",
                request.Email, GetClientIp());
            
            throw;
        }
        catch (Exception ex)
        {
            // ✅ DO: Log unexpected errors with context
            _logger.LogError(ex,
                "User registration failed unexpectedly. Email: {Email}",
                request.Email);
            
            throw;
        }
    }
}
```

### Log Levels for Security Events

```csharp
public class SecurityEventLogger
{
    private readonly ILogger<SecurityEventLogger> _logger;
    
    public void LogSecurityEvent(SecurityEventType eventType, string userId, string details)
    {
        switch (eventType)
        {
            case SecurityEventType.SuccessfulLogin:
            case SecurityEventType.SuccessfulLogout:
            case SecurityEventType.PasswordChanged:
                // Normal security events
                _logger.LogInformation(
                    "Security Event: {EventType}, UserId: {UserId}, Details: {Details}",
                    eventType, userId, details);
                break;
                
            case SecurityEventType.FailedLogin:
            case SecurityEventType.InvalidToken:
            case SecurityEventType.SuspiciousActivity:
                // Potential security issues
                _logger.LogWarning(
                    "Security Warning: {EventType}, UserId: {UserId}, Details: {Details}",
                    eventType, userId, details);
                break;
                
            case SecurityEventType.UnauthorizedAccess:
            case SecurityEventType.AccountLocked:
            case SecurityEventType.PrivilegeEscalationAttempt:
                // Serious security violations
                _logger.LogError(
                    "Security Violation: {EventType}, UserId: {UserId}, Details: {Details}",
                    eventType, userId, details);
                break;
                
            case SecurityEventType.PossibleBreach:
            case SecurityEventType.DataExfiltration:
                // Critical security incidents
                _logger.LogCritical(
                    "SECURITY INCIDENT: {EventType}, UserId: {UserId}, Details: {Details}",
                    eventType, userId, details);
                break;
        }
    }
}
```

### Correlation IDs for Request Tracing

```csharp
// Middleware to ensure correlation ID
public class CorrelationIdMiddleware
{
    private readonly RequestDelegate _next;
    private const string CorrelationIdHeader = "X-Correlation-ID";
    
    public CorrelationIdMiddleware(RequestDelegate next)
    {
        _next = next;
    }
    
    public async Task InvokeAsync(HttpContext context)
    {
        var correlationId = context.Request.Headers[CorrelationIdHeader].FirstOrDefault() 
            ?? Guid.NewGuid().ToString();
        
        context.TraceIdentifier = correlationId;
        context.Response.Headers.Add(CorrelationIdHeader, correlationId);
        
        using (LogContext.PushProperty("CorrelationId", correlationId))
        {
            await _next(context);
        }
    }
}

// Usage in logging
_logger.LogInformation("Processing request. CorrelationId is automatically included");
```

### Sanitizing Log Output

```csharp
public class LogSanitizer
{
    private static readonly Regex EmailRegex = new Regex(@"\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b");
    private static readonly Regex CreditCardRegex = new Regex(@"\b\d{4}[\s-]?\d{4}[\s-]?\d{4}[\s-]?\d{4}\b");
    private static readonly Regex SsnRegex = new Regex(@"\b\d{3}-\d{2}-\d{4}\b");
    
    public static string Sanitize(string message)
    {
        if (string.IsNullOrEmpty(message))
            return message;
        
        // Mask email (keep domain visible for debugging)
        message = EmailRegex.Replace(message, m => 
            $"{m.Value[0]}***@{m.Value.Split('@')[1]}");
        
        // Mask credit cards (keep last 4 digits)
        message = CreditCardRegex.Replace(message, m => 
            $"****-****-****-{m.Value.Substring(m.Value.Length - 4)}");
        
        // Completely mask SSN
        message = SsnRegex.Replace(message, "***-**-****");
        
        return message;
    }
}
```
