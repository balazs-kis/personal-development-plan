By the end of this lesson, you will be able to:

## Introduction

k6 is a modern, developer-friendly load testing tool that uses JavaScript for test scripts. It's designed for performance testing of APIs, microservices, and websites, with excellent support for CI/CD integration and realistic load simulation.

### Your First k6 Test

Create `test.js`:

```javascript
import http from 'k6/http';
import { sleep, check } from 'k6';

export default function () {
  const res = http.get('https://test.k6.io');
  
  check(res, {
    'status is 200': (r) => r.status === 200,
    'response time < 500ms': (r) => r.timings.duration < 500,
  });
  
  sleep(1);
}
```

Run it:
```bash
k6 run test.js
```

### Test Options and Configuration

#### Configure Load via Options

```javascript
import http from 'k6/http';
import { sleep } from 'k6';

export const options = {
  vus: 10,              // 10 virtual users
  duration: '30s',      // for 30 seconds
};

export default function () {
  http.get('https://test.k6.io');
  sleep(1);
}
```

#### Command-Line Options

```bash
# Override options from command line
k6 run --vus 10 --duration 30s test.js

# Specify number of iterations
k6 run --iterations 100 test.js

# Run with stages (ramp up/down)
k6 run --stage 5s:10,10s:20,5s:0 test.js
```

### Stages: Ramping Load

```javascript
export const options = {
  stages: [
    { duration: '30s', target: 20 },  // Ramp up to 20 users over 30s
    { duration: '1m',  target: 20 },  // Stay at 20 users for 1 minute
    { duration: '30s', target: 0  },  // Ramp down to 0 users
  ],
};

export default function () {
  http.get('https://test.k6.io');
  sleep(1);
}
```

### Checks: Validating Responses

```javascript
import { check } from 'k6';
import http from 'k6/http';

export default function () {
  const res = http.get('https://test.k6.io');
  
  check(res, {
    'is status 200': (r) => r.status === 200,
    'body contains text': (r) => r.body.includes('Collection of simple web-pages'),
    'response time OK': (r) => r.timings.duration < 200,
  });
}
```

Checks don't stop the test if they fail - they just record the failure rate.

### Thresholds: Pass/Fail Criteria

```javascript
export const options = {
  thresholds: {
    http_req_duration: ['p(95)<500'],      // 95% of requests must be < 500ms
    http_req_failed: ['rate<0.01'],         // Error rate must be < 1%
    checks: ['rate>0.95'],                  // 95% of checks must pass
  },
};
```

If thresholds fail, k6 exits with non-zero code (useful for CI/CD).

### Using Test Data

#### From Array

```javascript
const users = [
  { username: 'user1', password: 'pass1' },
  { username: 'user2', password: 'pass2' },
  { username: 'user3', password: 'pass3' },
];

export default function () {
  const user = users[Math.floor(Math.random() * users.length)];
  http.post('https://api.example.com/login', JSON.stringify(user));
}
```

#### From CSV File

```javascript
import { SharedArray } from 'k6/data';
import papaparse from 'https://jslib.k6.io/papaparse/5.1.1/index.js';

const csvData = new SharedArray('users', function () {
  return papaparse.parse(open('./users.csv'), { header: true }).data;
});

export default function () {
  const user = csvData[Math.floor(Math.random() * csvData.length)];
  http.post('https://api.example.com/login', JSON.stringify(user));
}
```

### Different Test Types

#### Smoke Test

Minimal load to verify the script works:

```javascript
export const options = {
  vus: 1,
  duration: '1m',
};
```

#### Load Test

Expected normal/peak load:

```javascript
export const options = {
  stages: [
    { duration: '5m', target: 100 },   // Ramp-up
    { duration: '10m', target: 100 },  // Stay at peak
    { duration: '5m', target: 0 },     // Ramp-down
  ],
};
```

#### Stress Test

Push beyond limits to find breaking point:

```javascript
export const options = {
  stages: [
    { duration: '2m', target: 100 },   // Below normal load
    { duration: '5m', target: 100 },
    { duration: '2m', target: 200 },   // Normal load
    { duration: '5m', target: 200 },
    { duration: '2m', target: 300 },   // Around breaking point
    { duration: '5m', target: 300 },
    { duration: '2m', target: 400 },   // Beyond breaking point
    { duration: '5m', target: 400 },
    { duration: '10m', target: 0 },    // Recovery
  ],
};
```

#### Spike Test

Sudden surge in traffic:

```javascript
export const options = {
  stages: [
    { duration: '10s', target: 100 },  // Normal load
    { duration: '1m', target: 100 },
    { duration: '10s', target: 1400 }, // Spike to 1400 users
    { duration: '3m', target: 1400 },  // Maintain spike
    { duration: '10s', target: 100 },  // Return to normal
    { duration: '3m', target: 100 },
    { duration: '10s', target: 0 },
  ],
};
```

#### Soak Test

Extended duration at normal load:

```javascript
export const options = {
  stages: [
    { duration: '5m', target: 200 },   // Ramp up
    { duration: '4h', target: 200 },   // Stay for 4 hours
    { duration: '5m', target: 0 },     // Ramp down
  ],
};
```

## Complete Example: E-Commerce API Test

```javascript
import http from 'k6/http';
import { check, sleep, group } from 'k6';
import { Rate } from 'k6/metrics';

// Custom metric for successful purchases
const purchaseSuccessRate = new Rate('purchase_success');

export const options = {
  stages: [
    { duration: '1m', target: 50 },   // Ramp up
    { duration: '3m', target: 50 },   // Stay at load
    { duration: '1m', target: 0 },    // Ramp down
  ],
  thresholds: {
    http_req_duration: ['p(95)<500'],
    http_req_failed: ['rate<0.01'],
    'purchase_success': ['rate>0.95'],
  },
};

const BASE_URL = 'https://api.example.com';

export default function () {
  // 60% browsers, 30% shoppers, 10% buyers
  const rand = Math.random();
  
  if (rand < 0.6) {
    browserFlow();
  } else if (rand < 0.9) {
    shopperFlow();
  } else {
    buyerFlow();
  }
}

function browserFlow() {
  group('Browser Flow', () => {
    // Browse products
    let res = http.get(`${BASE_URL}/products`);
    check(res, { 'products loaded': (r) => r.status === 200 });
    sleep(randomBetween(5, 8));
    
    // View product details
    const productId = Math.floor(Math.random() * 100) + 1;
    res = http.get(`${BASE_URL}/products/${productId}`);
    check(res, { 'product details loaded': (r) => r.status === 200 });
    sleep(randomBetween(10, 15));
  });
}

function shopperFlow() {
  group('Shopper Flow', () => {
    // Browse and add to cart
    let res = http.get(`${BASE_URL}/products`);
    sleep(randomBetween(5, 8));
    
    const productId = Math.floor(Math.random() * 100) + 1;
    res = http.get(`${BASE_URL}/products/${productId}`);
    sleep(randomBetween(8, 12));
    
    // Add to cart
    res = http.post(
      `${BASE_URL}/cart/items`,
      JSON.stringify({ productId, quantity: 1 }),
      { headers: { 'Content-Type': 'application/json' } }
    );
    check(res, { 'added to cart': (r) => r.status === 201 });
    sleep(randomBetween(3, 5));
    
    // View cart and abandon
    res = http.get(`${BASE_URL}/cart`);
    sleep(randomBetween(5, 10));
  });
}

function buyerFlow() {
  group('Buyer Flow', () => {
    // Login
    const loginPayload = JSON.stringify({
      email: `user${Math.floor(Math.random() * 1000)}@example.com`,
      password: 'password123',
    });
    
    let res = http.post(
      `${BASE_URL}/auth/login`,
      loginPayload,
      { headers: { 'Content-Type': 'application/json' } }
    );
    
    const authToken = res.json('token');
    const headers = {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${authToken}`,
    };
    
    sleep(randomBetween(2, 3));
    
    // Browse
    res = http.get(`${BASE_URL}/products`);
    sleep(randomBetween(5, 8));
    
    // Add to cart
    const productId = Math.floor(Math.random() * 100) + 1;
    res = http.post(
      `${BASE_URL}/cart/items`,
      JSON.stringify({ productId, quantity: 1 }),
      { headers }
    );
    sleep(randomBetween(2, 3));
    
    // Checkout
    res = http.post(
      `${BASE_URL}/orders`,
      JSON.stringify({
        paymentMethod: 'credit_card',
        shippingAddress: '123 Main St',
      }),
      { headers }
    );
    
    const purchaseSuccess = check(res, {
      'order created': (r) => r.status === 201,
    });
    purchaseSuccessRate.add(purchaseSuccess);
    
    sleep(randomBetween(15, 20));
  });
}

function randomBetween(min, max) {
  return Math.random() * (max - min) + min;
}
```

## Best Practices

✅ **Start with smoke tests** - Verify your script works with minimal load first

✅ **Use meaningful checks** - Validate response status and content

✅ **Set thresholds** - Define pass/fail criteria for CI/CD

✅ **Implement realistic think times** - Use `sleep()` between requests

✅ **Organize with groups** - Structure related requests together

✅ **Parameterize test data** - Don't hardcode values

✅ **Use SharedArray for large datasets** - Saves memory across VUs

✅ **Monitor during tests** - Watch server metrics, not just k6 output

✅ **Run from multiple locations** - Distributed testing for realistic network conditions
