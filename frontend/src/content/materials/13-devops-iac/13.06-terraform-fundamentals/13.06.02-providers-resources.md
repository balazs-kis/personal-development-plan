## Introduction

Providers are the foundation of Terraform's extensibility. They act as plugins that enable Terraform to interact with cloud platforms, SaaS providers, and other APIs. Resources are the most important element in Terraform—they represent infrastructure objects like virtual networks, compute instances, or DNS records that providers manage.

## Key Concepts Summary

### What are Providers?

**Providers** are plugins that implement resource types. They handle:
- API authentication
- API calls to cloud platforms
- Resource creation, reading, updating, and deletion
- Translating Terraform configuration to API calls

```hcl
# Provider configuration
terraform {
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }
}

provider "aws" {
  region = "us-west-2"
}
```

---

## Provider Sources

Providers are distributed through the Terraform Registry:

```hcl
terraform {
  required_providers {
    # Format: namespace/name
    aws = {
      source  = "hashicorp/aws"    # Official HashiCorp provider
      version = "~> 5.0"
    }
    
    datadog = {
      source  = "datadog/datadog"  # Third-party provider
      version = "~> 3.0"
    }
    
    custom = {
      source  = "mycompany.com/internal/custom"  # Private provider
      version = "~> 1.0"
    }
  }
}
```

**Provider tiers:**
- **Official:** Maintained by HashiCorp
- **Partner:** Maintained by technology companies
- **Community:** Maintained by the community

## Provider Configuration

### Basic Configuration

```hcl
provider "aws" {
  region = "us-west-2"
}

provider "azurerm" {
  features {}
  subscription_id = var.azure_subscription_id
}

provider "google" {
  project = var.gcp_project_id
  region  = var.gcp_region
}
```

### Provider Configuration Options

```hcl
provider "aws" {
  # Region
  region = "us-west-2"
  
  # Authentication (multiple methods)
  access_key = var.aws_access_key
  secret_key = var.aws_secret_key
  
  # Or use profile from ~/.aws/credentials
  profile = "production"
  
  # Or use assume role
  assume_role {
    role_arn = "arn:aws:iam::123456789:role/TerraformRole"
  }
  
  # Default tags applied to all resources
  default_tags {
    tags = {
      ManagedBy   = "Terraform"
      Environment = var.environment
    }
  }
}
```

### Provider Aliases (Multiple Configurations)

```hcl
# Default provider
provider "aws" {
  region = "us-west-2"
}

# Alternative provider with alias
provider "aws" {
  alias  = "east"
  region = "us-east-1"
}

provider "aws" {
  alias  = "backup"
  region = "us-west-1"
}

# Use the aliased provider
resource "aws_instance" "west" {
  # Uses default provider (us-west-2)
  ami           = "ami-abc123"
  instance_type = "t2.micro"
}

resource "aws_instance" "east" {
  # Explicitly use the east provider
  provider      = aws.east
  ami           = "ami-def456"
  instance_type = "t2.micro"
}

resource "aws_s3_bucket" "backup" {
  # Use the backup provider
  provider = aws.backup
  bucket   = "my-backup-bucket"
}
```

## Version Constraints

### Specifying Provider Versions

```hcl
terraform {
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      
      # Exact version
      version = "5.0.0"
      
      # Pessimistic constraint (recommended)
      version = "~> 5.0"      # >= 5.0, < 6.0
      version = "~> 5.1.0"    # >= 5.1.0, < 5.2.0
      
      # Comparison operators
      version = ">= 5.0"      # 5.0 or newer
      version = ">= 5.0, < 6.0"  # Between 5.0 and 6.0
    }
  }
}
```

**Best practice:** Use pessimistic constraints (`~>`) to allow patch and minor updates while preventing breaking changes.

### Lock File

```bash
# Create/update lock file
terraform init

# Upgrade providers within constraints
terraform init -upgrade
```

The `.terraform.lock.hcl` file records exact provider versions:

```hcl
# .terraform.lock.hcl
provider "registry.terraform.io/hashicorp/aws" {
  version     = "5.31.0"
  constraints = "~> 5.0"
  hashes = [
    "h1:abc123...",
    "zh:def456...",
  ]
}
```

**Always commit** `.terraform.lock.hcl` to version control!

## Resources

### Resource Syntax

```hcl
resource "<PROVIDER>_<TYPE>" "<NAME>" {
  # Configuration arguments
  argument1 = value1
  argument2 = value2
  
  # Nested blocks
  nested_block {
    nested_argument = nested_value
  }
}
```

### Example Resources

```hcl
# AWS VPC
resource "aws_vpc" "main" {
  cidr_block           = "10.0.0.0/16"
  enable_dns_hostnames = true
  enable_dns_support   = true
  
  tags = {
    Name = "main-vpc"
  }
}

# AWS Subnet
resource "aws_subnet" "public" {
  vpc_id                  = aws_vpc.main.id
  cidr_block              = "10.0.1.0/24"
  availability_zone       = "us-west-2a"
  map_public_ip_on_launch = true
  
  tags = {
    Name = "public-subnet"
    Type = "public"
  }
}

# AWS Instance
resource "aws_instance" "web" {
  ami           = "ami-0c55b159cbfafe1f0"
  instance_type = "t2.micro"
  subnet_id     = aws_subnet.public.id
  
  user_data = <<-EOF
              #!/bin/bash
              apt-get update
              apt-get install -y nginx
              EOF
  
  tags = {
    Name = "web-server"
  }
}
```

## Resource Meta-Arguments

Meta-arguments change resource behavior and are available for all resources.

### depends_on

```hcl
# Explicit dependency
resource "aws_iam_role" "example" {
  name = "example-role"
  # ...
}

resource "aws_instance" "example" {
  ami           = "ami-abc123"
  instance_type = "t2.micro"
  
  # Wait for IAM role even though we don't reference it
  depends_on = [aws_iam_role.example]
}
```

### count

```hcl
# Create multiple instances
resource "aws_instance" "server" {
  count = 3  # Creates 3 instances
  
  ami           = "ami-abc123"
  instance_type = "t2.micro"
  
  tags = {
    Name = "server-${count.index}"  # server-0, server-1, server-2
  }
}

# Reference specific instance
output "first_server_id" {
  value = aws_instance.server[0].id
}

# Reference all instances
output "all_server_ids" {
  value = aws_instance.server[*].id
}
```

### for_each

```hcl
# Create resources from a map
variable "instances" {
  type = map(object({
    instance_type = string
    ami           = string
  }))
  
  default = {
    web = {
      instance_type = "t2.micro"
      ami           = "ami-abc123"
    }
    api = {
      instance_type = "t2.small"
      ami           = "ami-def456"
    }
  }
}

resource "aws_instance" "servers" {
  for_each = var.instances
  
  ami           = each.value.ami
  instance_type = each.value.instance_type
  
  tags = {
    Name = each.key  # "web" or "api"
  }
}

# Reference specific instance
output "web_server_id" {
  value = aws_instance.servers["web"].id
}
```

### provider

```hcl
# Use specific provider alias
resource "aws_instance" "east_server" {
  provider      = aws.east  # Use aliased provider
  ami           = "ami-abc123"
  instance_type = "t2.micro"
}
```

### lifecycle

```hcl
resource "aws_instance" "example" {
  ami           = "ami-abc123"
  instance_type = "t2.micro"
  
  lifecycle {
    # Create new resource before destroying old one
    create_before_destroy = true
    
    # Prevent resource deletion
    prevent_destroy = true
    
    # Ignore changes to specific attributes
    ignore_changes = [
      tags,
      user_data,
    ]
    
    # Replace resource if any of these change
    replace_triggered_by = [
      aws_instance.other.id
    ]
  }
}
```

## Resource Behavior

### Resource Creation

```bash
# Terraform creates resources that don't exist
terraform apply

# Output:
# + resource "aws_instance" "web" {
#     + ami           = "ami-abc123"
#     + instance_type = "t2.micro"
#     ...
#   }
```

### Resource Updates

```hcl
# Change instance type
resource "aws_instance" "web" {
  ami           = "ami-abc123"
  instance_type = "t2.small"  # Changed from t2.micro
}
```

```bash
terraform apply

# Output:
# ~ resource "aws_instance" "web" {
#     ~ instance_type = "t2.micro" -> "t2.small"
#   }
```

### Resource Replacement

Some changes require replacement:

```hcl
resource "aws_instance" "web" {
  ami           = "ami-xyz789"  # AMI change requires replacement
  instance_type = "t2.micro"
}
```

```bash
terraform apply

# Output:
# -/+ resource "aws_instance" "web" {
#     ~ ami           = "ami-abc123" -> "ami-xyz789" # forces replacement
#     }
```

### Resource Deletion

```bash
# Remove resource from configuration, then:
terraform apply

# Output:
# - resource "aws_instance" "web" {
#     - ami           = "ami-abc123"
#     ...
#   }
```

## Provider Authentication

### AWS Authentication Methods

```hcl
# 1. Static credentials (not recommended)
provider "aws" {
  access_key = var.aws_access_key
  secret_key = var.aws_secret_key
  region     = "us-west-2"
}

# 2. Shared credentials file
provider "aws" {
  profile = "production"  # Uses ~/.aws/credentials
  region  = "us-west-2"
}

# 3. Environment variables (recommended)
# AWS_ACCESS_KEY_ID
# AWS_SECRET_ACCESS_KEY
# AWS_DEFAULT_REGION
provider "aws" {
  # Automatically uses env vars
  region = "us-west-2"
}

# 4. IAM role (best for CI/CD)
provider "aws" {
  assume_role {
    role_arn = "arn:aws:iam::123456789:role/TerraformRole"
  }
  region = "us-west-2"
}
```

### Azure Authentication

```hcl
provider "azurerm" {
  features {}
  
  # Service principal
  subscription_id = var.subscription_id
  client_id       = var.client_id
  client_secret   = var.client_secret
  tenant_id       = var.tenant_id
}
```

### Google Cloud Authentication

```hcl
provider "google" {
  project     = var.project_id
  region      = var.region
  credentials = file("service-account-key.json")
}
```

## Multi-Cloud Example

```hcl
# Configure multiple cloud providers
terraform {
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
    azurerm = {
      source  = "hashicorp/azurerm"
      version = "~> 3.0"
    }
    google = {
      source  = "hashicorp/google"
      version = "~> 5.0"
    }
  }
}

provider "aws" {
  region = "us-west-2"
}

provider "azurerm" {
  features {}
}

provider "google" {
  project = "my-project"
  region  = "us-central1"
}

# AWS resources
resource "aws_s3_bucket" "data" {
  bucket = "my-data-bucket"
}

# Azure resources
resource "azurerm_resource_group" "example" {
  name     = "example-resources"
  location = "East US"
}

# Google Cloud resources
resource "google_storage_bucket" "data" {
  name     = "my-data-bucket-gcp"
  location = "US"
}
```

## Common Mistakes to Avoid

| Mistake | Problem | Solution |
|---------|---------|----------|
| No version constraints | Breaking changes | Use `~>` constraints |
| Hardcoded credentials | Security risk | Use env vars or IAM roles |
| Not using lock file | Version inconsistencies | Commit `.terraform.lock.hcl` |
| Missing provider configuration | Resources fail | Configure provider before resources |
| Using `count` with changing lists | Resource recreation | Use `for_each` with stable keys |
| Ignoring provider deprecation warnings | Future breakage | Update deprecated syntax |
