# Lesson 2: Data Builders and Factories with Bogus

## Learning Objectives

By the end of this lesson, you will be able to:
- Understand the Test Data Builder pattern and its benefits
- Use Bogus to generate realistic fake data for .NET applications
- Create reusable data builders and factories
- Compare Bogus with AutoFixture and choose the right tool
- Implement maintainable test data generation strategies

---

## Introduction

Manually creating test data is tedious, error-prone, and creates brittle tests that break when models change. The Test Data Builder pattern combined with libraries like Bogus provides a maintainable, fluent way to generate realistic test data. This lesson teaches you how to create test data that's both easy to write and easy to maintain.

**Key principle:** Make it easy to create test data with only the relevant properties specified, while sensible defaults handle everything else.

```csharp
        .Build();
    
    var result = _orderProcessor.Process(order);
    
    Assert.True(result.Success);
}
```

### Implementing a Simple Builder

```csharp
// Simple Test Data Builder implementation
public class OrderBuilder
{
    private int _id = 1;
    private int _customerId = 1;
    private DateTime _orderDate = DateTime.UtcNow;
    private OrderStatus _status = OrderStatus.Pending;
    private List<OrderItem> _items = new()
    {
        new OrderItem { Id = 1, ProductId = 1, Quantity = 1, Price = 10.00m }
    };
    
    public OrderBuilder WithId(int id)
    {
        _id = id;
        return this;
    }
    
    public OrderBuilder WithCustomerId(int customerId)
    {
        _customerId = customerId;
        return this;
    }
    
    public OrderBuilder WithOrderDate(DateTime orderDate)
    {
        _orderDate = orderDate;
        return this;
    }
    
    public OrderBuilder WithStatus(OrderStatus status)
    {
        _status = status;
        return this;
    }
    
    public OrderBuilder WithItems(params OrderItem[] items)
    {
        _items = items.ToList();
        return this;
    }
    
    public OrderBuilder WithoutItems()
    {
        _items = new List<OrderItem>();
        return this;
    }
    
    public Order Build()
    {
        return new Order
        {
            Id = _id,
            CustomerId = _customerId,
            OrderDate = _orderDate,
            Status = _status,
            Items = _items
        };
    }
}

// Usage examples
var defaultOrder = new OrderBuilder().Build();

var emptyOrder = new OrderBuilder()
    .WithoutItems()
    .Build();

var cancelledOrder = new OrderBuilder()
    .WithStatus(OrderStatus.Cancelled)
    .WithOrderDate(DateTime.UtcNow.AddDays(-30))
    .Build();
```

### Builder Composition

```csharp
// Builders compose beautifully for complex object graphs
public class CustomerBuilder
{
    private int _id = 1;
    private string _firstName = "John";
    private string _lastName = "Doe";
    private string _email = "john.doe@example.com";
    private Address _address = new AddressBuilder().Build();
    
    public CustomerBuilder WithAddress(Address address)
    {
        _address = address;
        return this;
    }
    
    public Customer Build()
    {
        return new Customer
        {
            Id = _id,
            FirstName = _firstName,
            LastName = _lastName,
            Email = _email,
            Address = _address
        };
    }
}

public class AddressBuilder
{
    private string _street = "123 Main St";
    private string _city = "Springfield";
    private string _state = "IL";
    private string _zipCode = "62701";
    
    public AddressBuilder InCity(string city)
    {
        _city = city;
        return this;
    }
    
    public Address Build()
    {
        return new Address
        {
            Street = _street,
            City = _city,
            State = _state,
            ZipCode = _zipCode
        };
    }
}

// Compose builders together
var customer = new CustomerBuilder()
    .WithAddress(new AddressBuilder()
        .InCity("Chicago")
        .Build())
    .Build();
```

### Bogus with Generic Types

```csharp
// Define rules for generating objects
var orderFaker = new Faker<Order>()
    .RuleFor(o => o.Id, f => f.Random.Int(1, 10000))
    .RuleFor(o => o.CustomerId, f => f.Random.Int(1, 1000))
    .RuleFor(o => o.OrderDate, f => f.Date.Past(1))
    .RuleFor(o => o.Status, f => f.PickRandom<OrderStatus>())
    .RuleFor(o => o.Total, f => f.Finance.Amount(10, 1000));

// Generate a single order
var order = orderFaker.Generate();

// Generate multiple orders
var orders = orderFaker.Generate(100);

// Generate with specific overrides
var specificOrder = orderFaker
    .RuleFor(o => o.Status, _ => OrderStatus.Completed)
    .Generate();
```

### Bogus Rules in Detail

```csharp
public class CustomerFakerExamples
{
    public static Faker<Customer> CreateCustomerFaker()
    {
        return new Faker<Customer>()
            // Simple property mapping
            .RuleFor(c => c.Id, f => f.Random.Int(1, 10000))
            
            // Using specialized generators
            .RuleFor(c => c.FirstName, f => f.Name.FirstName())
            .RuleFor(c => c.LastName, f => f.Name.LastName())
            
            // Using multiple values for consistency
            .RuleFor(c => c.Email, (f, c) => 
                f.Internet.Email(c.FirstName, c.LastName))
            
            // Pick from enum
            .RuleFor(c => c.Status, f => f.PickRandom<CustomerStatus>())
            
            // Complex nested objects
            .RuleFor(c => c.Address, f => new Address
            {
                Street = f.Address.StreetAddress(),
                City = f.Address.City(),
                State = f.Address.State(),
                ZipCode = f.Address.ZipCode()
            })
            
            // Collections
            .RuleFor(c => c.PhoneNumbers, f => 
                f.Make(2, () => f.Phone.PhoneNumber()))
            
            // Conditional rules
            .RuleFor(c => c.Discount, (f, c) => 
                c.Status == CustomerStatus.Premium ? 0.15m : 0.0m)
            
            // Ignore certain properties (leave default)
            .Ignore(c => c.CreatedAt)
            
            // Set value directly without faker
            .RuleFor(c => c.IsActive, _ => true);
    }
}

// Usage
var customerFaker = CustomerFakerExamples.CreateCustomerFaker();
var customer = customerFaker.Generate();
var customers = customerFaker.Generate(50);
```

### Combining Builders with Bogus

```csharp
// Best of both worlds: Bogus for defaults, Builder for control
public class CustomerBuilder
{
    private static readonly Faker<Customer> _faker = new Faker<Customer>()
        .RuleFor(c => c.Id, f => f.Random.Int(1, 10000))
        .RuleFor(c => c.FirstName, f => f.Name.FirstName())
        .RuleFor(c => c.LastName, f => f.Name.LastName())
        .RuleFor(c => c.Email, (f, c) => f.Internet.Email(c.FirstName, c.LastName))
        .RuleFor(c => c.Phone, f => f.Phone.PhoneNumber())
        .RuleFor(c => c.Address, f => new Address
        {
            Street = f.Address.StreetAddress(),
            City = f.Address.City(),
            State = f.Address.State(),
            ZipCode = f.Address.ZipCode()
        })
        .RuleFor(c => c.CreatedAt, f => f.Date.Past(2))
        .RuleFor(c => c.IsActive, _ => true);
    
    private Customer _customer;
    
    public CustomerBuilder()
    {
        // Start with realistic defaults from Bogus
        _customer = _faker.Generate();
    }
    
    // Provide explicit overrides when needed
    public CustomerBuilder WithEmail(string email)
    {
        _customer.Email = email;
        return this;
    }
    
    public CustomerBuilder WithInactiveStatus()
    {
        _customer.IsActive = false;
        return this;
    }
    
    public CustomerBuilder WithAddress(Address address)
    {
        _customer.Address = address;
        return this;
    }
    
    public Customer Build() => _customer;
}

// Usage: Realistic defaults with specific overrides
var customer = new CustomerBuilder()
    .WithEmail("specific@example.com")
    .Build();
```

### Factory Pattern for Common Scenarios

```csharp
// Create factories for common test scenarios
public static class TestDataFactory
{
    private static readonly Faker<Customer> CustomerFaker = new Faker<Customer>()
        .RuleFor(c => c.Id, f => f.Random.Int(1, 10000))
        .RuleFor(c => c.FirstName, f => f.Name.FirstName())
        .RuleFor(c => c.LastName, f => f.Name.LastName())
        .RuleFor(c => c.Email, (f, c) => f.Internet.Email(c.FirstName, c.LastName))
        .RuleFor(c => c.IsActive, _ => true);
    
    private static readonly Faker<Order> OrderFaker = new Faker<Order>()
        .RuleFor(o => o.Id, f => f.Random.Int(1, 10000))
        .RuleFor(o => o.CustomerId, f => f.Random.Int(1, 1000))
        .RuleFor(o => o.OrderDate, f => f.Date.Past(1))
        .RuleFor(o => o.Status, f => f.PickRandom<OrderStatus>());
    
    // Named factory methods for common scenarios
    public static Customer CreateActiveCustomer() => 
        CustomerFaker.Clone()
            .RuleFor(c => c.IsActive, _ => true)
            .Generate();
    
    public static Customer CreateInactiveCustomer() => 
        CustomerFaker.Clone()
            .RuleFor(c => c.IsActive, _ => false)
            .Generate();
    
    public static Customer CreatePremiumCustomer() => 
        CustomerFaker.Clone()
            .RuleFor(c => c.IsActive, _ => true)
            .RuleFor(c => c.Status, _ => CustomerStatus.Premium)
            .Generate();
    
    public static Order CreatePendingOrder() => 
        OrderFaker.Clone()
            .RuleFor(o => o.Status, _ => OrderStatus.Pending)
            .Generate();
    
    public static Order CreateCompletedOrder() => 
        OrderFaker.Clone()
            .RuleFor(o => o.Status, _ => OrderStatus.Completed)
            .RuleFor(o => o.CompletedAt, f => f.Date.Past())
            .Generate();
    
    // Bulk generation for performance tests
    public static List<Customer> CreateCustomers(int count) => 
        CustomerFaker.Generate(count);
    
    public static List<Order> CreateOrdersForCustomer(int customerId, int count) =>
        OrderFaker.Clone()
            .RuleFor(o => o.CustomerId, _ => customerId)
            .Generate(count);
}

// Usage in tests
[Fact]
public void ProcessOrder_WithPendingOrder_UpdatesStatus()
{
    var order = TestDataFactory.CreatePendingOrder();
    
    _orderProcessor.Process(order);
    
    Assert.Equal(OrderStatus.Processing, order.Status);
}
```

### Deterministic Random Data

```csharp
// Use seeds for reproducible test data
public class SeededTestData
{
    // Same seed = same data every time
    private static Faker<Customer> CreateSeededFaker(int seed)
    {
        return new Faker<Customer>()
            .UseSeed(seed)
            .RuleFor(c => c.FirstName, f => f.Name.FirstName())
            .RuleFor(c => c.LastName, f => f.Name.LastName())
            .RuleFor(c => c.Email, (f, c) => f.Internet.Email(c.FirstName, c.LastName));
    }
    
    [Fact]
    public void Test_WithReproducibleData()
    {
        // This will always generate the same customer
        var faker = CreateSeededFaker(seed: 12345);
        var customer1 = faker.Generate();
        
        // Reset with same seed
        var faker2 = CreateSeededFaker(seed: 12345);
        var customer2 = faker2.Generate();
        
        // They're identical
        Assert.Equal(customer1.FirstName, customer2.FirstName);
        Assert.Equal(customer1.Email, customer2.Email);
    }
}
```

### Bogus vs AutoFixture

```csharp
// AutoFixture: Less setup, less realistic
[Fact]
public void Test_WithAutoFixture()
{
    var fixture = new Fixture();
    
    // Automatically fills all properties with generated values
    var customer = fixture.Create<Customer>();
    
    // email might be: "emailf35a7e4b-d50e-4b1e-8e6e-1234567890ab"
    // Not very realistic, but fast to set up
}

// Bogus: More setup, more realistic
[Fact]
public void Test_WithBogus()
{
    var faker = new Faker<Customer>()
        .RuleFor(c => c.FirstName, f => f.Name.FirstName())
        .RuleFor(c => c.LastName, f => f.Name.LastName())
        .RuleFor(c => c.Email, (f, c) => f.Internet.Email(c.FirstName, c.LastName));
    
    var customer = faker.Generate();
    
    // email might be: "john.doe@example.com"
    // Much more realistic
}

// When to use each:
// AutoFixture: Unit tests, quick setup, don't care about realism
// Bogus: Integration tests, E2E tests, database seeding, demos
```
