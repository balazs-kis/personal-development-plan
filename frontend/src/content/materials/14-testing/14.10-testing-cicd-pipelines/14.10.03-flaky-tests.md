- Understand what makes tests flaky and why they're problematic
- Identify common causes of flaky tests
- Use built-in and third-party tools to detect flaky tests
- Implement retry strategies to handle flaky tests
- Design tests that are deterministic and reliable

---

## Introduction

Flaky tests are the silent killers of CI/CD pipelines. They pass sometimes and fail other times with the same code, eroding trust in your test suite. Developers start ignoring test failures, or worse, re-running pipelines until tests pass. This lesson teaches you how to detect, manage, and ultimately eliminate flaky tests.

**Key principle:** A flaky test is worse than no test. It wastes time, destroys confidence, and trains developers to ignore failures.

## Key Concepts Summary

### What Makes a Test Flaky?

```
Definition:
A flaky test is one that produces inconsistent results when run 
multiple times against the same code without any environmental changes.

Symptoms:
✅ Pass → ❌ Fail → ✅ Pass (without code changes)
✅ Passes locally, ❌ fails in CI
✅ Passes when run alone, ❌ fails when run with others
✅ Passes most of the time, ❌ occasionally fails

Impact:
- Wastes developer time (investigating false failures)
- Delays deployments (re-running pipelines)
- Erodes confidence in test suite
- Trains developers to ignore failures
- Increases CI/CD costs (more pipeline runs)
```

### Common Causes of Flaky Tests

```csharp
// 1. TIMING AND RACE CONDITIONS
// ❌ BAD: Hard-coded sleep
[Fact]
public async Task LoadData_ReturnsData()
{
    await _service.StartLoadingAsync();
    Thread.Sleep(1000);  // Flaky! Might not be enough time
    
    var data = await _service.GetDataAsync();
    Assert.NotEmpty(data);
}

// ✅ GOOD: Poll with timeout
[Fact]
public async Task LoadData_ReturnsData()
{
    await _service.StartLoadingAsync();
    
    // Poll for data with timeout
    var data = await PollAsync(
        async () => await _service.GetDataAsync(),
        data => data.Any(),
        timeout: TimeSpan.FromSeconds(5)
    );
    
    Assert.NotEmpty(data);
}

private async Task<T> PollAsync<T>(
    Func<Task<T>> getValue,
    Func<T, bool> condition,
    TimeSpan timeout)
{
    var stopwatch = Stopwatch.StartNew();
    
    while (stopwatch.Elapsed < timeout)
    {
        var value = await getValue();
        if (condition(value))
            return value;
        
        await Task.Delay(100);
    }
    
    throw new TimeoutException($"Condition not met within {timeout}");
}

// 2. NON-DETERMINISTIC TEST DATA
// ❌ BAD: Random data without seed
[Fact]
public void SortCustomers_SortsCorrectly()
{
    var random = new Random();  // Flaky! Different each run
    var customers = Enumerable.Range(0, 100)
        .Select(_ => new Customer { Age = random.Next(100) })
        .ToList();
    
    var sorted = customers.OrderBy(c => c.Age).ToList();
    
    Assert.Equal(sorted.First().Age, customers.Min(c => c.Age));
}

// ✅ GOOD: Seeded random data
[Fact]
public void SortCustomers_SortsCorrectly()
{
    var random = new Random(12345);  // Deterministic
    var customers = Enumerable.Range(0, 100)
        .Select(_ => new Customer { Age = random.Next(100) })
        .ToList();
    
    var sorted = customers.OrderBy(c => c.Age).ToList();
    
    Assert.Equal(sorted.First().Age, customers.Min(c => c.Age));
}

// 3. TEST ORDER DEPENDENCIES
// ❌ BAD: Tests depend on execution order
public class OrderDependentTests
{
    private static int _counter = 0;  // Shared state!
    
    [Fact]
    public void Test1()
    {
        _counter++;
        Assert.Equal(1, _counter);  // Flaky! Depends on execution order
    }
    
    [Fact]
    public void Test2()
    {
        _counter++;
        Assert.Equal(2, _counter);  // Flaky!
    }
}

// ✅ GOOD: Independent tests
public class IndependentTests
{
    private int _counter;  // Instance field, reset per test
    
    public IndependentTests()
    {
        _counter = 0;  // Initialize in constructor
    }
    
    [Fact]
    public void Test1()
    {
        _counter++;
        Assert.Equal(1, _counter);  // Always works
    }
    
    [Fact]
    public void Test2()
    {
        _counter++;
        Assert.Equal(1, _counter);  // Always works
    }
}

// 4. SHARED RESOURCES
// ❌ BAD: Shared database without cleanup
[Fact]
public async Task AddCustomer_IncreasesCount()
{
    await _repository.AddAsync(new Customer { Name = "John" });
    
    var count = await _repository.CountAsync();
    Assert.Equal(1, count);  // Flaky! Other tests might add customers
}

// ✅ GOOD: Isolated database per test
public class CustomerRepositoryTests : IAsyncLifetime
{
    private readonly TestDatabase _database;
    
    public CustomerRepositoryTests()
    {
        _database = new TestDatabase();
    }
    
    public async Task InitializeAsync()
    {
        await _database.ResetAsync();  // Clean state
    }
    
    [Fact]
    public async Task AddCustomer_IncreasesCount()
    {
        await _repository.AddAsync(new Customer { Name = "John" });
        
        var count = await _repository.CountAsync();
        Assert.Equal(1, count);  // Reliable
    }
    
    public Task DisposeAsync() => _database.DisposeAsync();
}

// 5. DATETIME DEPENDENCIES
// ❌ BAD: Depends on current time
[Fact]
public void IsExpired_ReturnsTrueForPastDates()
{
    var subscription = new Subscription
    {
        ExpiresAt = DateTime.UtcNow.AddDays(-1)  // Flaky if run at midnight!
    };
    
    Assert.True(subscription.IsExpired);
}

// ✅ GOOD: Use time provider or explicit dates
[Fact]
public void IsExpired_ReturnsTrueForPastDates()
{
    var subscription = new Subscription
    {
        ExpiresAt = new DateTime(2024, 1, 1)  // Fixed date
    };
    
    var result = subscription.IsExpired(new DateTime(2024, 1, 2));
    
    Assert.True(result);
}

// Or use TimeProvider (.NET 8+)
public class SubscriptionService
{
    private readonly TimeProvider _timeProvider;
    
    public SubscriptionService(TimeProvider timeProvider)
    {
        _timeProvider = timeProvider;
    }
    
    public bool IsExpired(Subscription subscription)
    {
        return subscription.ExpiresAt < _timeProvider.GetUtcNow();
    }
}

[Fact]
public void IsExpired_ReturnsTrueForPastDates()
{
    var fakeTime = new FakeTimeProvider();
    fakeTime.SetUtcNow(new DateTime(2024, 1, 2));
    
    var service = new SubscriptionService(fakeTime);
    var subscription = new Subscription { ExpiresAt = new DateTime(2024, 1, 1) };
    
    Assert.True(service.IsExpired(subscription));
}

// 6. EXTERNAL DEPENDENCIES
// ❌ BAD: Depends on external API
[Fact]
public async Task GetWeather_ReturnsData()
{
    var weather = await _client.GetWeatherAsync("London");
    
    Assert.NotNull(weather);  // Flaky! API might be down or rate-limited
}

// ✅ GOOD: Mock external dependencies
[Fact]
public async Task GetWeather_ReturnsData()
{
    var mockClient = new Mock<IWeatherClient>();
    mockClient.Setup(c => c.GetWeatherAsync("London"))
        .ReturnsAsync(new Weather { Temperature = 20 });
    
    var service = new WeatherService(mockClient.Object);
    var weather = await service.GetWeatherAsync("London");
    
    Assert.NotNull(weather);
    Assert.Equal(20, weather.Temperature);
}

// 7. RESOURCE CONTENTION
// ❌ BAD: Multiple tests using same port
[Fact]
public async Task StartServer_ListensOnPort()
{
    var server = new TestServer(port: 5000);  // Flaky! Port might be in use
    await server.StartAsync();
    
    Assert.True(server.IsListening);
}

// ✅ GOOD: Use dynamic port allocation
[Fact]
public async Task StartServer_ListensOnPort()
{
    var server = new TestServer(port: 0);  // OS assigns random available port
    await server.StartAsync();
    
    Assert.True(server.IsListening);
    Assert.True(server.Port > 0);
}
```

### NUnit Retry Attribute

```csharp
// NUnit has built-in retry support
using NUnit.Framework;

public class RetryTests
{
    // Retry test up to 3 times (3 total attempts)
    [Test, Retry(3)]
    public async Task FlakyApiCall_EventuallySucceeds()
    {
        var result = await _apiClient.CallAsync();
        Assert.IsNotNull(result);
    }
    
    // Retry only for specific exceptions (.NET 4.5.0+)
    [Test]
    [Retry(5, ExceptionTypes = new[] { typeof(TimeoutException), typeof(HttpRequestException) })]
    public async Task ExternalService_HandlesTransientFailures()
    {
        var result = await _externalService.CallAsync();
        Assert.IsNotNull(result);
    }
}

// Note: Retry is a band-aid, not a cure!
// Use it for genuinely flaky external dependencies
// Don't use it to hide problems in your tests
```

### PowerShell Script for Test Retry

```powershell
# retry-tests.ps1
# Retry failed tests up to N times

param(
    [int]$MaxRetries = 3,
    [string]$TestProject = "MyApp.Tests",
    [int]$PassPercentageThreshold = 100
)

$attempt = 1
$allTestsPassed = $false
$failedTests = @()

while ($attempt -le $MaxRetries -and -not $allTestsPassed) {
    Write-Host "Test attempt $attempt of $MaxRetries" -ForegroundColor Cyan
    
    if ($attempt -eq 1) {
        # First run: all tests
        dotnet test $TestProject `
            --logger "trx;LogFileName=TestResults_$attempt.trx" `
            --results-directory TestResults
    } else {
        # Subsequent runs: only failed tests
        $filter = ($failedTests -join "|")
        Write-Host "Retrying failed tests: $filter" -ForegroundColor Yellow
        
        dotnet test $TestProject `
            --filter "FullyQualifiedName~$filter" `
            --logger "trx;LogFileName=TestResults_$attempt.trx" `
            --results-directory TestResults
    }
    
    # Parse TRX to find failed tests
    $trxFile = Get-ChildItem -Path TestResults -Filter "TestResults_$attempt.trx" | Select-Object -First 1
    [xml]$trx = Get-Content $trxFile.FullName
    
    $failed = $trx.TestRun.Results.UnitTestResult | Where-Object { $_.outcome -eq "Failed" }
    
    if ($failed.Count -eq 0) {
        $allTestsPassed = $true
        Write-Host "All tests passed!" -ForegroundColor Green
    } else {
        $failedTests = $failed | ForEach-Object { $_.testName }
        Write-Host "Failed tests: $($failedTests.Count)" -ForegroundColor Red
        $attempt++
    }
}

if (-not $allTestsPassed) {
    Write-Host "Tests failed after $MaxRetries attempts" -ForegroundColor Red
    Write-Host "Failed tests:" -ForegroundColor Red
    $failedTests | ForEach-Object { Write-Host "  - $_" }
    exit 1
}

exit 0
```

### Quarantine Strategy

```csharp
// Quarantine flaky tests instead of deleting them
// Mark with attribute and exclude from regular runs

public class QuarantineAttribute : FactAttribute
{
    public QuarantineAttribute(string reason)
    {
        if (!IsQuarantineEnabled())
        {
            Skip = $"Test is quarantined: {reason}";
        }
    }
    
    private static bool IsQuarantineEnabled()
    {
        // Only run quarantined tests when explicitly requested
        return Environment.GetEnvironmentVariable("RUN_QUARANTINED_TESTS") == "true";
    }
}

// Usage
[Quarantine("Flaky due to timing issues - under investigation")]
public async Task FlakyTest_ToBeFixed()
{
    // Test code that's currently unreliable
}

// Run normally (skips quarantined tests):
// dotnet test

// Run including quarantined tests:
// RUN_QUARANTINED_TESTS=true dotnet test
```

```yaml
# GitHub Actions: Separate job for quarantined tests
name: CI

jobs:
  main-tests:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-dotnet@v4
        with:
          dotnet-version: '8.0.x'
      - name: Run Stable Tests

  quarantined-tests:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-dotnet@v4
        with:
          dotnet-version: '8.0.x'
      - name: Run Quarantined Tests

          RUN_QUARANTINED_TESTS: 'true'
        run: dotnet test  # Runs quarantined tests
```

### Flaky Test Metrics

```csharp
// Track flakiness metrics
public class FlakyTestTracker
{
    public class TestResult
    {
        public string TestName { get; set; }
        public DateTime Timestamp { get; set; }
        public bool Passed { get; set; }
        public string CommitSha { get; set; }
    }
    
    public async Task<FlakinessReport> AnalyzeTestFlakiness(
        string testName,
        int lookbackDays = 30)
    {
        var results = await GetTestResults(testName, lookbackDays);
        
        var totalRuns = results.Count;
        var failures = results.Count(r => !r.Passed);
        var passRate = (totalRuns - failures) / (double)totalRuns * 100;
        
        // Detect flakiness: test has both passes and failures
        var hasFailures = failures > 0;
        var hasPasses = totalRuns - failures > 0;
        var isFlaky = hasFailures && hasPasses;
        
        // Check if failures are for same commit (true flakiness)
        var flakyCommits = results
            .GroupBy(r => r.CommitSha)
            .Where(g => g.Any(r => r.Passed) && g.Any(r => !r.Passed))
            .Select(g => g.Key)
            .ToList();
        
        return new FlakinessReport
        {
            TestName = testName,
            TotalRuns = totalRuns,
            Failures = failures,
            PassRate = passRate,
            IsFlaky = flakyCommits.Any(),
            FlakyCommits = flakyCommits
        };
    }
}
```
