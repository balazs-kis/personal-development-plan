## Introduction

While in-memory fakes are great for fast unit tests, integration tests with real message brokers provide the highest confidence. Testcontainers makes it easy to run real RabbitMQ, Kafka, and other brokers in Docker containers during tests.

### Setting Up RabbitMQ Tests

**Install NuGet Package:**
```bash
dotnet add package Testcontainers.RabbitMq
dotnet add package RabbitMQ.Client
dotnet add package xunit
```

**Basic RabbitMQ Test:**
```csharp
using Testcontainers.RabbitMq;
using RabbitMQ.Client;
using Xunit;

public class RabbitMqTests : IAsyncLifetime
{
    private readonly RabbitMqContainer _rabbitMqContainer = new RabbitMqBuilder()
        .WithImage("rabbitmq:3.12-management")
        .WithPortBinding(5672, true)
        .WithPortBinding(15672, true)
        .Build();

    public async Task InitializeAsync()
    {
        await _rabbitMqContainer.StartAsync();
    }

    public async Task DisposeAsync()
    {
        await _rabbitMqContainer.DisposeAsync();
    }

    [Fact]
    public async Task PublishMessage_ConsumesSuccessfully()
    {
        // Arrange
        var connectionString = _rabbitMqContainer.GetConnectionString();
        var factory = new ConnectionFactory
        {
            Uri = new Uri(connectionString)
        };

        using var connection = factory.CreateConnection();
        using var channel = connection.CreateModel();

        var queueName = "test-queue";
        channel.QueueDeclare(
            queue: queueName,
            durable: false,
            exclusive: false,
            autoDelete: false);

        var receivedMessages = new List<string>();
        var consumer = new EventingBasicConsumer(channel);
        consumer.Received += (sender, args) =>
        {
            var message = Encoding.UTF8.GetString(args.Body.ToArray());
            receivedMessages.Add(message);
        };

        channel.BasicConsume(
            queue: queueName,
            autoAck: true,
            consumer: consumer);

        // Act
        var messageBody = Encoding.UTF8.GetBytes("Hello, RabbitMQ!");
        channel.BasicPublish(
            exchange: string.Empty,
            routingKey: queueName,
            body: messageBody);

        // Assert
        await Task.Delay(500); // Wait for message processing
        Assert.Single(receivedMessages);
        Assert.Equal("Hello, RabbitMQ!", receivedMessages[0]);
    }
}
```

### Testing Message Publisher

```csharp
public interface IMessagePublisher
{
    Task PublishAsync<T>(string routingKey, T message);
}

public class RabbitMqPublisher : IMessagePublisher
{
    private readonly IConnection _connection;
    private readonly IModel _channel;

    public RabbitMqPublisher(string connectionString)
    {
        var factory = new ConnectionFactory { Uri = new Uri(connectionString) };
        _connection = factory.CreateConnection();
        _channel = _connection.CreateModel();
    }

    public async Task PublishAsync<T>(string routingKey, T message)
    {
        var json = JsonSerializer.Serialize(message);
        var body = Encoding.UTF8.GetBytes(json);

        _channel.BasicPublish(
            exchange: string.Empty,
            routingKey: routingKey,
            body: body);

        await Task.CompletedTask;
    }

    public void Dispose()
    {
        _channel?.Dispose();
        _connection?.Dispose();
    }
}

[Fact]
public async Task PublishOrder_MessageAppearsInQueue()
{
    // Arrange
    var connectionString = _rabbitMqContainer.GetConnectionString();
    var publisher = new RabbitMqPublisher(connectionString);

    var queueName = "orders";
    using var factory = new ConnectionFactory { Uri = new Uri(connectionString) };
    using var connection = factory.CreateConnection();
    using var channel = connection.CreateModel();
    
    channel.QueueDeclare(queueName, false, false, false);

    var order = new OrderCreatedMessage
    {
        OrderId = Guid.NewGuid(),
        CustomerId = Guid.NewGuid(),
        CreatedAt = DateTime.UtcNow
    };

    // Act
    await publisher.PublishAsync(queueName, order);

    // Assert
    await Task.Delay(100);
    var result = channel.BasicGet(queueName, autoAck: true);
    
    Assert.NotNull(result);
    var json = Encoding.UTF8.GetString(result.Body.ToArray());
    var received = JsonSerializer.Deserialize<OrderCreatedMessage>(json);
    
    Assert.Equal(order.OrderId, received.OrderId);
}
```

### Testing Message Consumer

```csharp
public class OrderConsumer : BackgroundService
{
    private readonly string _connectionString;
    private readonly string _queueName;
    private readonly IOrderService _orderService;
    private IConnection _connection;
    private IModel _channel;

    public OrderConsumer(
        string connectionString,
        string queueName,
        IOrderService orderService)
    {
        _connectionString = connectionString;
        _queueName = queueName;
        _orderService = orderService;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        var factory = new ConnectionFactory { Uri = new Uri(_connectionString) };
        _connection = factory.CreateConnection();
        _channel = _connection.CreateModel();

        _channel.QueueDeclare(_queueName, false, false, false);

        var consumer = new EventingBasicConsumer(_channel);
        consumer.Received += async (sender, args) =>
        {
            var json = Encoding.UTF8.GetString(args.Body.ToArray());
            var message = JsonSerializer.Deserialize<OrderCreatedMessage>(json);
            
            await _orderService.ProcessOrderAsync(message);
            
            _channel.BasicAck(args.DeliveryTag, false);
        };

        _channel.BasicConsume(_queueName, false, consumer);

        await Task.Delay(Timeout.Infinite, stoppingToken);
    }

    public override void Dispose()
    {
        _channel?.Dispose();
        _connection?.Dispose();
        base.Dispose();
    }
}

[Fact]
public async Task ConsumeMessage_ProcessesOrder()
{
    // Arrange
    var connectionString = _rabbitMqContainer.GetConnectionString();
    var orderService = new Mock<IOrderService>();
    
    var consumer = new OrderConsumer(
        connectionString,
        "orders",
        orderService.Object);

    await consumer.StartAsync(CancellationToken.None);

    using var factory = new ConnectionFactory { Uri = new Uri(connectionString) };
    using var connection = factory.CreateConnection();
    using var channel = connection.CreateModel();
    
    channel.QueueDeclare("orders", false, false, false);

    var order = new OrderCreatedMessage
    {
        OrderId = Guid.NewGuid(),
        CustomerId = Guid.NewGuid()
    };

    // Act
    var json = JsonSerializer.Serialize(order);
    var body = Encoding.UTF8.GetBytes(json);
    channel.BasicPublish(string.Empty, "orders", body);

    // Assert
    await Task.Delay(1000); // Wait for consumption

    orderService.Verify(x => x.ProcessOrderAsync(
        It.Is<OrderCreatedMessage>(m => m.OrderId == order.OrderId)),
        Times.Once);

    await consumer.StopAsync(CancellationToken.None);
}
```

```csharp
using Testcontainers.Kafka;
using Confluent.Kafka;

public class KafkaTests : IAsyncLifetime
{
    private readonly KafkaContainer _kafkaContainer = new KafkaBuilder()
        .WithImage("confluentinc/cp-kafka:7.5.0")
        .Build();

    public async Task InitializeAsync()
    {
        await _kafkaContainer.StartAsync();
    }

    public async Task DisposeAsync()
    {
        await _kafkaContainer.DisposeAsync();
    }

    [Fact]
    public async Task ProduceMessage_ConsumesSuccessfully()
    {
        // Arrange
        var bootstrapServers = _kafkaContainer.GetBootstrapAddress();
        var topic = "test-topic";

        var producerConfig = new ProducerConfig
        {
            BootstrapServers = bootstrapServers
        };

        var consumerConfig = new ConsumerConfig
        {
            BootstrapServers = bootstrapServers,
            GroupId = "test-group",
            AutoOffsetReset = AutoOffsetReset.Earliest
        };

        // Act
        using var producer = new ProducerBuilder<string, string>(producerConfig).Build();
        await producer.ProduceAsync(topic, new Message<string, string>
        {
            Key = "test-key",
            Value = "Hello, Kafka!"
        });

        // Assert
        using var consumer = new ConsumerBuilder<string, string>(consumerConfig).Build();
        consumer.Subscribe(topic);

        var consumeResult = consumer.Consume(TimeSpan.FromSeconds(10));
        
        Assert.NotNull(consumeResult);
        Assert.Equal("test-key", consumeResult.Message.Key);
        Assert.Equal("Hello, Kafka!", consumeResult.Message.Value);
    }
}
```

### Testing Kafka Producer

```csharp
public class KafkaOrderProducer
{
    private readonly IProducer<string, string> _producer;

    public KafkaOrderProducer(string bootstrapServers)
    {
        var config = new ProducerConfig
        {
            BootstrapServers = bootstrapServers
        };
        _producer = new ProducerBuilder<string, string>(config).Build();
    }

    public async Task PublishOrderAsync(OrderCreatedMessage order)
    {
        var json = JsonSerializer.Serialize(order);
        await _producer.ProduceAsync("orders", new Message<string, string>
        {
            Key = order.OrderId.ToString(),
            Value = json
        });
    }

    public void Dispose()
    {
        _producer?.Dispose();
    }
}

[Fact]
public async Task PublishOrder_MessageAvailableInKafka()
{
    // Arrange
    var bootstrapServers = _kafkaContainer.GetBootstrapAddress();
    var producer = new KafkaOrderProducer(bootstrapServers);

    var order = new OrderCreatedMessage
    {
        OrderId = Guid.NewGuid(),
        CustomerId = Guid.NewGuid()
    };

    // Act
    await producer.PublishOrderAsync(order);

    // Assert
    using var consumer = new ConsumerBuilder<string, string>(
        new ConsumerConfig
        {
            BootstrapServers = bootstrapServers,
            GroupId = "test-group",
            AutoOffsetReset = AutoOffsetReset.Earliest
        }).Build();

    consumer.Subscribe("orders");
    var result = consumer.Consume(TimeSpan.FromSeconds(10));

    Assert.NotNull(result);
    var received = JsonSerializer.Deserialize<OrderCreatedMessage>(
        result.Message.Value);
    Assert.Equal(order.OrderId, received.OrderId);
}
```

### Testing with MassTransit

```csharp
public class MassTransitRabbitMqTests : IAsyncLifetime
{
    private readonly RabbitMqContainer _rabbitMqContainer = new RabbitMqBuilder()
        .WithImage("rabbitmq:3.12")
        .Build();

    private IBusControl _bus;

    public async Task InitializeAsync()
    {
        await _rabbitMqContainer.StartAsync();

        _bus = Bus.Factory.CreateUsingRabbitMq(cfg =>
        {
            cfg.Host(_rabbitMqContainer.GetConnectionString());
        });

        await _bus.StartAsync();
    }

    public async Task DisposeAsync()
    {
        await _bus.StopAsync();
        await _rabbitMqContainer.DisposeAsync();
    }

    [Fact]
    public async Task PublishMessage_ConsumesSuccessfully()
    {
        var receivedMessages = new List<OrderCreatedMessage>();
        var received = new TaskCompletionSource<bool>();

        _bus.ConnectReceiveEndpoint("test-queue", cfg =>
        {
            cfg.Handler<OrderCreatedMessage>(async context =>
            {
                receivedMessages.Add(context.Message);
                received.TrySetResult(true);
            });
        });

        var message = new OrderCreatedMessage
        {
            OrderId = Guid.NewGuid()
        };

        await _bus.Publish(message);

        await received.Task.WaitAsync(TimeSpan.FromSeconds(10));

        Assert.Single(receivedMessages);
        Assert.Equal(message.OrderId, receivedMessages[0].OrderId);
    }
}
```

### Best Practices

**1. Use IAsyncLifetime for Lifecycle:**
```csharp
// ✓ Good - Proper lifecycle management
public class Tests : IAsyncLifetime
{
    public async Task InitializeAsync()
    {
        await _container.StartAsync();
    }

    public async Task DisposeAsync()
    {
        await _container.DisposeAsync();
    }
}
```

**2. Share Containers When Possible:**
```csharp
// ✓ Good - Share across test class
[Collection("RabbitMQ")]
public class OrderTests { }

// ✗ Bad - Start container per test
[Fact]
public async Task Test()
{
    var container = new RabbitMqBuilder().Build();
    await container.StartAsync();
}
```

**3. Use Appropriate Timeouts:**
```csharp
// ✓ Good - Reasonable timeout
var result = consumer.Consume(TimeSpan.FromSeconds(10));

// ✗ Bad - Infinite wait
var result = consumer.Consume();
```

**4. Clean Up Between Tests:**
```csharp
// ✓ Good - Delete queue after test
[Fact]
public async Task Test()
{
    try
    {
        channel.QueueDeclare("test-queue", ...);
        // Test code
    }
    finally
    {
        channel.QueueDelete("test-queue");
    }
}
```

**5. Use Specific Images:**
```csharp
// ✓ Good - Specific version
.WithImage("rabbitmq:3.12-management")

// ✗ Bad - Latest tag
.WithImage("rabbitmq:latest")
```
