## Introduction

k6 is a modern, developer-centric load testing tool that uses JavaScript for test scripting. Built in Go for performance, k6 can generate significant load while providing an excellent developer experience. It's particularly well-suited for testing APIs and integrating performance testing into CI/CD pipelines.

### Basic Test Structure

#### Test Lifecycle

k6 tests have four phases:

```javascript
// 1. init code - runs once per VU (Virtual User)
import http from 'k6/http';
import { check, sleep } from 'k6';

export const options = {
  vus: 10,
  duration: '30s',
};

// 2. setup code - runs once before test
export function setup() {
  const res = http.get('https://api.example.com/setup');
  return { data: res.json() };
}

// 3. VU code - runs repeatedly for each VU
export default function (data) {
  const res = http.get('https://api.example.com/users');
  
  check(res, {
    'status is 200': (r) => r.status === 200,
    'response time < 200ms': (r) => r.timings.duration < 200,
  });
  
  sleep(1);
}

// 4. teardown code - runs once after test
export function teardown(data) {
  // Cleanup logic
}
```

### POST Request with JSON Payload

```javascript
import http from 'k6/http';
import { check } from 'k6';

export default function () {
  const url = 'https://api.example.com/users';
  
  const payload = JSON.stringify({
    name: 'Test User',
    email: 'test@example.com',
  });
  
  const params = {
    headers: {
      'Content-Type': 'application/json',
    },
  };
  
  const res = http.post(url, payload, params);
  
  check(res, {
    'status is 201': (r) => r.status === 201,
    'has user id': (r) => r.json().id !== undefined,
  });
}
```

### Load Patterns with Scenarios

#### Constant VUs
```javascript
export const options = {
  scenarios: {
    constant_load: {
      executor: 'constant-vus',
      vus: 50,
      duration: '5m',
    },
  },
};
```

#### Ramping VUs (Load Test Pattern)
```javascript
export const options = {
  scenarios: {
    load_test: {
      executor: 'ramping-vus',
      startVUs: 0,
      stages: [
        { duration: '2m', target: 50 },  // Ramp up to 50 VUs
        { duration: '5m', target: 50 },  // Stay at 50 VUs
        { duration: '2m', target: 0 },   // Ramp down to 0 VUs
      ],
      gracefulRampDown: '30s',
    },
  },
};
```

#### Stress Test Pattern
```javascript
export const options = {
  scenarios: {
    stress_test: {
      executor: 'ramping-vus',
      startVUs: 0,
      stages: [
        { duration: '2m', target: 100 },
        { duration: '5m', target: 100 },
        { duration: '2m', target: 200 },
        { duration: '5m', target: 200 },
        { duration: '2m', target: 300 },
        { duration: '5m', target: 300 },
        { duration: '5m', target: 0 },
      ],
    },
  },
};
```

#### Spike Test Pattern
```javascript
export const options = {
  scenarios: {
    spike_test: {
      executor: 'ramping-vus',
      startVUs: 0,
      stages: [
        { duration: '10s', target: 100 },  // Quick ramp up
        { duration: '1m', target: 100 },   // Stay at spike
        { duration: '10s', target: 0 },    // Quick ramp down
      ],
    },
  },
};
```

#### Constant Arrival Rate (Open Model)
```javascript
export const options = {
  scenarios: {
    open_model: {
      executor: 'constant-arrival-rate',
      rate: 100,                    // 100 iterations per timeUnit
      timeUnit: '1s',               // iterations per second
      duration: '5m',
      preAllocatedVUs: 50,          // Initial VU pool
      maxVUs: 100,                  // Maximum VU pool
    },
  },
};
```

### Checks: Validating Responses

Checks validate that responses meet expectations but don't stop the test if they fail:

```javascript
import { check } from 'k6';
import http from 'k6/http';

export default function () {
  const res = http.get('https://api.example.com/users/1');
  
  check(res, {
    'status is 200': (r) => r.status === 200,
    'response time < 500ms': (r) => r.timings.duration < 500,
    'has correct content type': (r) => 
      r.headers['Content-Type'] === 'application/json',
    'body has user data': (r) => {
      const body = r.json();
      return body.id && body.name && body.email;
    },
    'user email is valid': (r) => {
      const email = r.json().email;
      return email && email.includes('@');
    },
  });
}
```

### Thresholds: Pass/Fail Criteria

Thresholds define performance requirements and fail the test if not met:

```javascript
export const options = {
  thresholds: {
    // 95% of requests must complete within 500ms
    'http_req_duration': ['p(95)<500'],
    
    // 99% of requests must complete within 1s
    'http_req_duration': ['p(99)<1000'],
    
    // Error rate must be below 1%
    'http_req_failed': ['rate<0.01'],
    
    // 95% of checks must pass
    'checks': ['rate>0.95'],
    
    // Minimum throughput: 100 RPS
    'http_reqs': ['rate>100'],
  },
  
  scenarios: {
    load_test: {
      executor: 'constant-vus',
      vus: 50,
      duration: '5m',
    },
  },
};
```

**Example with multiple thresholds:**
```javascript
export const options = {
  thresholds: {
    'http_req_duration': [
      'p(50)<200',   // median < 200ms
      'p(90)<400',   // 90th percentile < 400ms
      'p(95)<500',   // 95th percentile < 500ms
      'p(99)<1000',  // 99th percentile < 1s
    ],
    'http_req_failed': ['rate<0.01'],        // Error rate < 1%
    'http_req_duration{status:200}': ['p(95)<300'],  // Tagged threshold
  },
};
```

### Tagging and Grouping

**Tags** add metadata to metrics:
```javascript
import http from 'k6/http';

export default function () {
  http.get('https://api.example.com/users', {
    tags: { name: 'UsersList' },
  });
  
  http.get('https://api.example.com/products', {
    tags: { name: 'ProductsList' },
  });
}
```

**Groups** organize test logic:
```javascript
import { group } from 'k6';
import http from 'k6/http';

export default function () {
  group('User Operations', () => {
    http.get('https://api.example.com/users');
    http.post('https://api.example.com/users', JSON.stringify({
      name: 'Test User',
    }));
  });
  
  group('Product Operations', () => {
    http.get('https://api.example.com/products');
  });
}
```

### Complete Example: E-Commerce API Test

```javascript
import http from 'k6/http';
import { check, group, sleep } from 'k6';

export const options = {
  scenarios: {
    load_test: {
      executor: 'ramping-vus',
      startVUs: 0,
      stages: [
        { duration: '1m', target: 20 },
        { duration: '3m', target: 20 },
        { duration: '1m', target: 0 },
      ],
    },
  },
  thresholds: {
    'http_req_duration': ['p(95)<500'],
    'http_req_failed': ['rate<0.01'],
    'checks': ['rate>0.95'],
  },
};

const BASE_URL = 'https://api.example.com';

export default function () {
  group('Browse Products', () => {
    const productRes = http.get(`${BASE_URL}/api/products`);
    
    check(productRes, {
      'products loaded': (r) => r.status === 200,
      'has products': (r) => r.json().length > 0,
    });
    
    sleep(1);
    
    // Get specific product
    if (productRes.status === 200) {
      const products = productRes.json();
      const productId = products[0].id;
      
      const detailRes = http.get(`${BASE_URL}/api/products/${productId}`);
      
      check(detailRes, {
        'product detail loaded': (r) => r.status === 200,
      });
    }
  });
  
  sleep(2);
  
  group('Add to Cart', () => {
    const payload = JSON.stringify({
      productId: 123,
      quantity: 1,
    });
    
    const params = {
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer fake-token',
      },
    };
    
    const cartRes = http.post(`${BASE_URL}/api/cart`, payload, params);
    
    check(cartRes, {
      'added to cart': (r) => r.status === 201,
    });
  });
  
  sleep(1);
}
```
