---

## Introduction

Database integration tests verify that your application correctly interacts with a real database. Using Testcontainers, you can spin up actual database instances (PostgreSQL, SQL Server, MySQL, etc.) in Docker containers, apply migrations, seed data, and run tests in complete isolation—all without maintaining shared test databases or complex infrastructure.

## Key Concepts Summary

### Setting Up Database Containers

Install the appropriate database module:
```bash
dotnet add package Testcontainers.PostgreSql
# Or for SQL Server:
dotnet add package Testcontainers.MsSql
```

Basic PostgreSQL container setup:
```csharp
private readonly PostgreSqlContainer _dbContainer = new PostgreSqlBuilder()
    .WithImage("postgres:16-alpine")
    .WithDatabase("testdb")
    .WithUsername("postgres")
    .WithPassword("postgres")
    .Build();
```

### Custom WebApplicationFactory

Override the default database configuration in your tests:
```csharp
public class IntegrationTestWebAppFactory : WebApplicationFactory<Program>
{
    private readonly PostgreSqlContainer _dbContainer;

    public IntegrationTestWebAppFactory(PostgreSqlContainer dbContainer)
    {
        _dbContainer = dbContainer;
    }

    protected override void ConfigureWebHost(IWebHostBuilder builder)
    {
        builder.ConfigureTestServices(services =>
        {
            // Remove the existing DbContext registration
            var descriptor = services.SingleOrDefault(
                d => d.ServiceType == typeof(DbContextOptions<AppDbContext>));
            
            if (descriptor != null)
            {
                services.Remove(descriptor);
            }

            // Add the test database
            services.AddDbContext<AppDbContext>(options =>
            {
                options.UseNpgsql(_dbContainer.GetConnectionString());
            });
        });
    }
}
```

### Database Migration Strategies

**Option 1: Apply Migrations at Test Startup**
```csharp
public async Task InitializeAsync()
{
    await _dbContainer.StartAsync();
    
    // Apply migrations
    using var scope = _factory.Services.CreateScope();
    var dbContext = scope.ServiceProvider.GetRequiredService<AppDbContext>();
    await dbContext.Database.MigrateAsync();
}
```

**Option 2: Create Database from Script**
```csharp
private readonly PostgreSqlContainer _dbContainer = new PostgreSqlBuilder()
    .WithImage("postgres:16-alpine")
    .WithResourceMapping("init-db.sql", "/docker-entrypoint-initdb.d/")
    .Build();
```

**Option 3: Use Respawn for Database Cleanup**
```bash
dotnet add package Respawn
```

```csharp
private Respawner _respawner = null!;

public async Task InitializeAsync()
{
    await _dbContainer.StartAsync();
    await InitializeDatabase();
    
    _respawner = await Respawner.CreateAsync(
        _dbContainer.GetConnectionString(),
        new RespawnerOptions
        {
            DbAdapter = DbAdapter.Postgres
        });
}

public async Task ResetDatabaseAsync()
{
    await _respawner.ResetAsync(_dbContainer.GetConnectionString());
}
```

### Test Data Seeding

**Seed Data in Tests:**
```csharp
private async Task SeedTestData(AppDbContext context)
{
    var products = new List<Product>
    {
        new() { Name = "Product 1", Price = 10.00m },
        new() { Name = "Product 2", Price = 20.00m }
    };
    
    context.Products.AddRange(products);
    await context.SaveChangesAsync();
}
```

**Use Object Mother or Builder Pattern:**
```csharp
public static class ProductFactory
{
    public static Product CreateTestProduct(
        string name = "Test Product",
        decimal price = 10.00m)
    {
        return new Product
        {
            Name = name,
            Price = price,
            CreatedAt = DateTime.UtcNow
        };
    }
}
```

### Complete Integration Test Example

```csharp
public class ProductTests : IClassFixture<IntegrationTestWebAppFactory>, IAsyncLifetime
{
    private readonly IntegrationTestWebAppFactory _factory;
    private readonly HttpClient _client;
    private readonly AppDbContext _dbContext;

    public ProductTests(IntegrationTestWebAppFactory factory)
    {
        _factory = factory;
        _client = factory.CreateClient();
        
        var scope = factory.Services.CreateScope();
        _dbContext = scope.ServiceProvider.GetRequiredService<AppDbContext>();
    }

    [Fact]
    public async Task CreateProduct_ReturnsCreated_WhenProductIsValid()
    {
        // Arrange
        var request = new CreateProductRequest("Test Product", 25.99m);
        
        // Act
        var response = await _client.PostAsJsonAsync("/api/products", request);
        
        // Assert
        response.StatusCode.Should().Be(HttpStatusCode.Created);
        
        var product = await _dbContext.Products
            .FirstOrDefaultAsync(p => p.Name == "Test Product");
        
        product.Should().NotBeNull();
        product!.Price.Should().Be(25.99m);
    }

    public Task InitializeAsync() => Task.CompletedTask;

    public async Task DisposeAsync()
    {
        // Clean up test data
        await _factory.ResetDatabaseAsync();
    }
}
```
