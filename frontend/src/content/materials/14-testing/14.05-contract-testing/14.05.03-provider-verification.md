## Learning Objectives

By the end of this lesson, you will be able to:

## Introduction

Provider verification is the process of replaying consumer contracts against the actual provider implementation to ensure the provider satisfies all consumer expectations. This lesson covers setting up verification tests and implementing provider states.

## Key Concepts Summary

### Provider Verification Architecture

```
┌─────────────────┐
│   Pact File     │ Contract from consumer
│   (Contract)    │
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│ Pact Verifier   │ Reads contract, replays requests
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│ Provider States │ Sets up required data
│    Handler      │
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│ Running Provider│ Actual API implementation
│      API        │
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│ Verification    │ Pass/Fail results
│    Results      │
└─────────────────┘
```

### Basic Provider Verification Setup

```csharp
using PactNet;
using PactNet.Infrastructure.Outputters;
using Xunit;
using Xunit.Abstractions;

public class ProductApiProviderTests : IDisposable
{
    private readonly ITestOutputHelper _outputHelper;
    private readonly WebApplication _app;
    private readonly string _providerUri = "http://localhost:9001";

    public ProductApiProviderTests(ITestOutputHelper output)
    {
        _outputHelper = output;
        
        // Start the provider API
        _app = CreateWebApplication();
        _app.RunAsync(_providerUri);
    }

    [Fact]
    public void EnsureProviderHonorsConsumerContract()
    {
        // Arrange
        var config = new PactVerifierConfig
        {
            Outputters = new[]
            {
                new XUnitOutput(_outputHelper)
            }
        };

        // Act & Assert
        new PactVerifier(config)
            .WithHttpEndpoint(new Uri(_providerUri))
            .WithFileSource(new FileInfo("../../../pacts/ProductConsumer-ProductAPI.json"))
            .WithProviderStateUrl(new Uri($"{_providerUri}/provider-states"))
            .Verify();
    }

    public void Dispose()
    {
        _app?.DisposeAsync();
    }

    private WebApplication CreateWebApplication()
    {
        var builder = WebApplication.CreateBuilder();
        // Configure your API services
        var app = builder.Build();
        // Configure your API endpoints
        return app;
    }
}
```

### Implementing Provider States

**Provider State Endpoint:**
```csharp
// In your API startup/configuration
app.MapPost("/provider-states", async (ProviderState state) =>
{
    await SetupProviderState(state.State, state.Params);
    return Results.Ok();
});

app.MapDelete("/provider-states/{state}", async (string state) =>
{
    await TeardownProviderState(state);
    return Results.Ok();
});

// Provider state models
public record ProviderState(string State, Dictionary<string, object>? Params);
```

**State Handler Implementation:**
```csharp
public class ProviderStateHandler
{
    private readonly ProductDbContext _dbContext;

    public ProviderStateHandler(ProductDbContext dbContext)
    {
        _dbContext = dbContext;
    }

    public async Task SetupState(string state, Dictionary<string, object>? parameters)
    {
        switch (state)
        {
            case "product with ID 10 exists":
                await SetupProductExists(10);
                break;

            case "product with ID 999 does not exist":
                await EnsureProductDoesNotExist(999);
                break;

            case "no products exist":
                await ClearAllProducts();
                break;

            case "products exist":
                await SetupMultipleProducts();
                break;

            case "user is authenticated":
                // No data setup needed, auth middleware will handle
                break;

            default:
                throw new NotImplementedException($"Provider state '{state}' not implemented");
        }
    }

    private async Task SetupProductExists(int productId)
    {
        // Clean up first
        var existing = await _dbContext.Products.FindAsync(productId);
        if (existing != null)
        {
            _dbContext.Products.Remove(existing);
        }

        // Create test product
        _dbContext.Products.Add(new Product
        {
            Id = productId,
            Name = "Widget",
            Price = 19.99m,
            Category = "Tools",
            InStock = true,
            CreatedAt = DateTime.UtcNow
        });

        await _dbContext.SaveChangesAsync();
    }

    private async Task EnsureProductDoesNotExist(int productId)
    {
        var product = await _dbContext.Products.FindAsync(productId);
        if (product != null)
        {
            _dbContext.Products.Remove(product);
            await _dbContext.SaveChangesAsync();
        }
    }

    private async Task ClearAllProducts()
    {
        _dbContext.Products.RemoveRange(_dbContext.Products);
        await _dbContext.SaveChangesAsync();
    }

    private async Task SetupMultipleProducts()
    {
        await ClearAllProducts();

        _dbContext.Products.AddRange(
            new Product { Id = 1, Name = "Widget", Price = 19.99m, Category = "Tools" },
            new Product { Id = 2, Name = "Gadget", Price = 29.99m, Category = "Electronics" },
            new Product { Id = 3, Name = "Gizmo", Price = 39.99m, Category = "Tools" }
        );

        await _dbContext.SaveChangesAsync();
    }

    public async Task TeardownState(string state)
    {
        // Clean up after test
        // Often clearing all data is sufficient
        await ClearAllProducts();
    }
}
```

### Using In-Memory Database for Testing

```csharp
public class ProductApiProviderTests
{
    private WebApplication CreateWebApplication()
    {
        var builder = WebApplication.CreateBuilder();

        // Use in-memory database for tests
        builder.Services.AddDbContext<ProductDbContext>(options =>
            options.UseInMemoryDatabase("ProviderTests"));

        // Register state handler
        builder.Services.AddScoped<ProviderStateHandler>();

        builder.Services.AddControllers();

        var app = builder.Build();

        // Configure provider state endpoints
        app.MapPost("/provider-states", async (
            ProviderState state,
            ProviderStateHandler handler) =>
        {
            await handler.SetupState(state.State, state.Params);
            return Results.Ok();
        });

        // Configure your API endpoints
        app.MapGet("/api/products/{id}", async (
            int id,
            ProductDbContext db) =>
        {
            var product = await db.Products.FindAsync(id);
            return product is not null ? Results.Ok(product) : Results.NotFound();
        });

        app.MapGet("/api/products", async (
            ProductDbContext db,
            string? category = null) =>
        {
            var query = db.Products.AsQueryable();
            
            if (!string.IsNullOrEmpty(category))
                query = query.Where(p => p.Category == category);

            var products = await query.ToListAsync();
            return Results.Ok(new { products });
        });

        app.MapPost("/api/products", async (
            ProductRequest request,
            ProductDbContext db) =>
        {
            var product = new Product
            {
                Name = request.Name,
                Price = request.Price,
                Category = request.Category,
                CreatedAt = DateTime.UtcNow
            };

            db.Products.Add(product);
            await db.SaveChangesAsync();

            return Results.Created($"/api/products/{product.Id}", product);
        });

        return app;
    }
}
```

```csharp
public async Task GetProduct_WithSpecificProduct_ReturnsProduct()

            id = 42,
            name = Type("Special Widget")
        });

    await _pactBuilder.VerifyAsync(async ctx =>
    {
        var client = new ProductApiClient(_mockServerUri);
        var product = await client.GetProductAsync(42);
        
        Assert.Equal(42, product.Id);
    });
}
```

**Provider State Handler with Parameters:**
```csharp
public async Task SetupState(string state, Dictionary<string, object>? parameters)
{
    switch (state)
    {
        case "product exists":
            if (parameters != null)
            {
                var id = Convert.ToInt32(parameters["productId"]);
                var name = parameters["productName"].ToString();
                await SetupProduct(id, name);
            }
            break;

        // ... other states
    }
}

private async Task SetupProduct(int id, string name)
{
    _dbContext.Products.Add(new Product
    {
        Id = id,
        Name = name,
        Price = 19.99m,
        Category = "Tools"
    });

    await _dbContext.SaveChangesAsync();
}
```

### Verifying from Pact Broker

```csharp
[Fact]
public void EnsureProviderHonorsConsumerContractsFromBroker()
{
    var config = new PactVerifierConfig
    {
        Outputters = new[] { new XUnitOutput(_outputHelper) }
    };

    new PactVerifier(config)
        .WithHttpEndpoint(new Uri(_providerUri))
        .WithPactBrokerSource(new Uri("https://your-pact-broker.com"), options =>
        {
            options
                .TokenAuthentication("your-token")
                .ConsumerVersionSelectors(
                    new ConsumerVersionSelector { Branch = "main" },
                    new ConsumerVersionSelector { DeployedOrReleased = true }
                )
                .ProviderVersion("1.2.3")
                .PublishResults(publishResults: true);
        })
        .WithProviderStateUrl(new Uri($"{_providerUri}/provider-states"))
        .Verify();
}
```

### Debugging Verification Failures

**Common Failure Scenarios:**

**1. Response Body Mismatch:**
```
Failure: Response body does not match
Expected: { "id": 10, "name": "Widget" }
Actual:   { "id": 10, "productName": "Widget" }

Fix: Ensure provider returns field names matching consumer expectations
```

**2. Provider State Not Implemented:**
```
Failure: Provider state 'product with ID 10 exists' not found

Fix: Implement the missing provider state in state handler
```

**3. Wrong Status Code:**
```
Failure: Expected status 200, got 404

Fix: Ensure provider state correctly sets up data
```

**4. Provider Not Running:**
```
Failure: Connection refused to http://localhost:9001

Fix: Ensure provider API is started before verification
```

### Best Practices

**Isolate Tests:**
```csharp
// ✓ Good - Clean state before each test
public async Task SetupState(string state, Dictionary<string, object>? parameters)
{
    // Always clean up first
    await _dbContext.Database.EnsureDeletedAsync();
    await _dbContext.Database.EnsureCreatedAsync();

    // Then set up specific state
    switch (state)
    {
        // ... state setup
    }
}

// ✗ Bad - Tests can interfere with each other
public async Task SetupState(string state, Dictionary<string, object>? parameters)
{
    // Just adding data without cleanup
    _dbContext.Products.Add(new Product { ... });
}
```

**Stub External Dependencies:**
```csharp
// ✓ Good - Stub external services
builder.Services.AddSingleton<IPaymentGateway>(
    new StubPaymentGateway());

// ✗ Bad - Calling real external services
builder.Services.AddScoped<IPaymentGateway>(
    sp => new StripePaymentGateway(apiKey));
```

**Local Verification Only:**
```csharp
// ✓ Good - Run against local instance
new PactVerifier(config)
    .WithHttpEndpoint(new Uri("http://localhost:9001"))

// ✗ Bad - Verifying against deployed environment
new PactVerifier(config)
    .WithHttpEndpoint(new Uri("https://production-api.com"))
```

**Descriptive State Names:**
```csharp
// ✓ Good - Clear, descriptive names
"product with ID 10 exists"
"user is authenticated with admin role"
"shopping cart contains 3 items"

// ✗ Bad - Vague or unclear
"setup1"
"data exists"
"test state"
```

### Testing Provider States Independently

```csharp
// Test that provider states work correctly
[Fact]
public async Task ProviderState_ProductExists_CreatesProduct()
{
    var handler = new ProviderStateHandler(_dbContext);
    
    await handler.SetupState("product with ID 10 exists", null);
    
    var product = await _dbContext.Products.FindAsync(10);
    Assert.NotNull(product);
    Assert.Equal("Widget", product.Name);
}

[Fact]
public async Task ProviderState_NoProductsExist_ClearsAllProducts()
{
    // Arrange - Add some products
    _dbContext.Products.Add(new Product { Id = 1, Name = "Test" });
    await _dbContext.SaveChangesAsync();
    
    var handler = new ProviderStateHandler(_dbContext);
    
    // Act
    await handler.SetupState("no products exist", null);
    
    // Assert
    var count = await _dbContext.Products.CountAsync();
    Assert.Equal(0, count);
}
```
