## Key Concepts Summary

### Publish/Subscribe (Pub/Sub)

Pub/Sub enables real-time messaging between components without direct connections.

#### Basic Pub/Sub

```csharp
using StackExchange.Redis;

var redis = ConnectionMultiplexer.Connect("localhost:6379");
var subscriber = redis.GetSubscriber();

// Subscribe to a channel
await subscriber.SubscribeAsync("notifications", (channel, message) =>
{
    Console.WriteLine($"Received on {channel}: {message}");
});

// Publish a message
var publisher = redis.GetSubscriber();
await publisher.PublishAsync("notifications", "Hello, World!");

// Unsubscribe
await subscriber.UnsubscribeAsync("notifications");
```

#### Pattern-Based Subscriptions

```csharp
// Subscribe to multiple channels with pattern
await subscriber.SubscribeAsync(
    new RedisChannel("user:*:notifications", RedisChannel.PatternMode.Pattern),
    (channel, message) =>
    {
        Console.WriteLine($"Pattern match on {channel}: {message}");
    }
);

// This will trigger the pattern subscriber
await publisher.PublishAsync("user:123:notifications", "You have a new message");
await publisher.PublishAsync("user:456:notifications", "Friend request received");
```

### Real-Time Notifications System

```csharp
// Notification service
public class NotificationService
{
    private readonly IConnectionMultiplexer _redis;
    private readonly ISubscriber _subscriber;
    
    public NotificationService(IConnectionMultiplexer redis)
    {
        _redis = redis;
        _subscriber = redis.GetSubscriber();
    }
    
    public async Task SendNotificationAsync(string userId, string message)
    {
        var channel = $"user:{userId}:notifications";
        await _subscriber.PublishAsync(channel, message);
    }
    
    public async Task SubscribeToNotificationsAsync(
        string userId,
        Action<string> onNotification)
    {
        var channel = $"user:{userId}:notifications";
        await _subscriber.SubscribeAsync(channel, (ch, message) =>
        {
            onNotification(message.ToString());
        });
    }
    
    public async Task UnsubscribeFromNotificationsAsync(string userId)
    {
        var channel = $"user:{userId}:notifications";
        await _subscriber.UnsubscribeAsync(channel);
    }
}

// Usage in SignalR hub
public class NotificationHub : Hub
{
    private readonly NotificationService _notifications;
    
    public async Task Subscribe(string userId)
    {
        await _notifications.SubscribeToNotificationsAsync(userId, async (message) =>
        {
            await Clients.User(userId).SendAsync("ReceiveNotification", message);
        });
    }
}
```

### Cache Invalidation with Pub/Sub

```csharp
public class DistributedCacheInvalidator
{
    private readonly IConnectionMultiplexer _redis;
    private readonly IMemoryCache _localCache;
    
    public DistributedCacheInvalidator(
        IConnectionMultiplexer redis,
        IMemoryCache localCache)
    {
        _redis = redis;
        _localCache = localCache;
        
        // Subscribe to invalidation messages
        var subscriber = redis.GetSubscriber();
        subscriber.Subscribe("cache:invalidate", (channel, message) =>
        {
            _localCache.Remove(message.ToString());
        });
    }
    
    public async Task InvalidateCacheAsync(string cacheKey)
    {
        // Remove from local cache
        _localCache.Remove(cacheKey);
        
        // Notify other instances
        var subscriber = _redis.GetSubscriber();
        await subscriber.PublishAsync("cache:invalidate", cacheKey);
    }
}
```

### Keyspace Notifications

Listen to Redis events (key changes, expirations, etc.).

#### Enable Keyspace Notifications

```bash
# In redis.conf or via CONFIG SET
CONFIG SET notify-keyspace-events KEA

# K: Keyspace events
# E: Keyevent events  
# A: All events
```

#### Subscribe to Key Events

```csharp
public class KeyspaceNotificationService
{
    private readonly IConnectionMultiplexer _redis;
    
    public KeyspaceNotificationService(IConnectionMultiplexer redis)
    {
        _redis = redis;
    }
    
    public async Task ListenToKeyExpirations(Action<string> onExpire)
    {
        var subscriber = _redis.GetSubscriber();
        
        // Listen to expired events in database 0
        await subscriber.SubscribeAsync(
            "__keyevent@0__:expired",
            (channel, message) =>
            {
                onExpire(message.ToString());
            }
        );
    }
    
    public async Task ListenToKeyChanges(string pattern, Action<string, string> onChange)
    {
        var subscriber = _redis.GetSubscriber();
        
        // Listen to SET operations on keys matching pattern
        var channel = $"__keyspace@0__:{pattern}";
        await subscriber.SubscribeAsync(
            new RedisChannel(channel, RedisChannel.PatternMode.Pattern),
            (ch, message) =>
            {
                var key = ch.ToString().Replace("__keyspace@0__:", "");
                var operation = message.ToString();
                onChange(key, operation);
            }
        );
    }
}

// Usage: Session expiration handler
var service = new KeyspaceNotificationService(redis);
await service.ListenToKeyExpirations(async (key) =>
{
    if (key.StartsWith("session:"))
    {
        var userId = key.Replace("session:", "");
        await LogoutUserAsync(userId);
    }
});
```

### Redis Streams

Streams provide durable message queuing with consumer groups.

```csharp
public class RedisStreamService
{
    private readonly IDatabase _db;
    
    public RedisStreamService(IConnectionMultiplexer redis)
    {
        _db = redis.GetDatabase();
    }
    
    // Add message to stream
    public async Task<RedisValue> AddToStreamAsync(
        string streamKey,
        Dictionary<string, string> message)
    {
        var entries = message.Select(kvp => 
            new NameValueEntry(kvp.Key, kvp.Value)
        ).ToArray();
        
        return await _db.StreamAddAsync(streamKey, entries);
    }
    
    // Read from stream
    public async Task<StreamEntry[]> ReadStreamAsync(
        string streamKey,
        string startId = "0-0",
        int count = 10)
    {
        return await _db.StreamReadAsync(streamKey, startId, count);
    }
    
    // Create consumer group
    public async Task CreateConsumerGroupAsync(
        string streamKey,
        string groupName)
    {
        try
        {
            await _db.StreamCreateConsumerGroupAsync(
                streamKey,
                groupName,
                StreamPosition.Beginning
            );
        }
        catch (RedisServerException ex) when (ex.Message.Contains("BUSYGROUP"))
        {
            // Group already exists
        }
    }
    
    // Read as consumer group member
    public async Task<StreamEntry[]> ReadGroupAsync(
        string streamKey,
        string groupName,
        string consumerName,
        int count = 10)
    {
        var result = await _db.StreamReadGroupAsync(
            streamKey,
            groupName,
            consumerName,
            StreamPosition.NewMessages,
            count
        );
        
        return result;
    }
    
    // Acknowledge message
    public async Task AcknowledgeAsync(
        string streamKey,
        string groupName,
        RedisValue messageId)
    {
        await _db.StreamAcknowledgeAsync(streamKey, groupName, messageId);
    }
}

// Usage: Event processing
var streamService = new RedisStreamService(redis);

// Producer
await streamService.AddToStreamAsync("orders", new Dictionary<string, string>
{
    ["orderId"] = "12345",
    ["customerId"] = "user123",
    ["amount"] = "99.99"
});

// Consumer
await streamService.CreateConsumerGroupAsync("orders", "processors");

var messages = await streamService.ReadGroupAsync(
    "orders",
    "processors",
    "worker-1"
);

foreach (var message in messages)
{
    try
    {
        // Process message
        var orderId = message["orderId"];
        await ProcessOrderAsync(orderId);
        
        // Acknowledge
        await streamService.AcknowledgeAsync(
            "orders",
            "processors",
            message.Id
        );
    }
    catch (Exception ex)
    {
        // Handle error, message will be redelivered
        _logger.LogError(ex, "Failed to process message {MessageId}", message.Id);
    }
}
```

### Distributed Locking

Implement distributed locks for coordination across instances.

```csharp
public class RedisDistributedLock
{
    private readonly IDatabase _db;
    
    public RedisDistributedLock(IConnectionMultiplexer redis)
    {
        _db = redis.GetDatabase();
    }
    
    public async Task<bool> AcquireLockAsync(
        string lockKey,
        string lockValue,
        TimeSpan expiry)
    {
        return await _db.StringSetAsync(
            lockKey,
            lockValue,
            expiry,
            When.NotExists
        );
    }
    
    public async Task<bool> ReleaseLockAsync(string lockKey, string lockValue)
    {
        // Use Lua script to ensure atomic check-and-delete
        const string script = @"
            if redis.call('GET', KEYS[1]) == ARGV[1] then
                return redis.call('DEL', KEYS[1])
            else
                return 0
            end
        ";
        
        var result = await _db.ScriptEvaluateAsync(
            script,
            new RedisKey[] { lockKey },
            new RedisValue[] { lockValue }
        );
        
        return (int)result == 1;
    }
}

// Usage with helper
public class LockManager
{
    private readonly RedisDistributedLock _lock;
    
    public async Task<T?> ExecuteWithLockAsync<T>(
        string resourceId,
        Func<Task<T>> operation,
        TimeSpan? lockExpiry = null)
    {
        var lockKey = $"lock:{resourceId}";
        var lockValue = Guid.NewGuid().ToString();
        var expiry = lockExpiry ?? TimeSpan.FromSeconds(30);
        
        // Try to acquire lock
        if (!await _lock.AcquireLockAsync(lockKey, lockValue, expiry))
        {
            throw new InvalidOperationException("Could not acquire lock");
        }
        
        try
        {
            // Execute operation
            return await operation();
        }
        finally
        {
            // Release lock
            await _lock.ReleaseLockAsync(lockKey, lockValue);
        }
    }
}

// Usage
await lockManager.ExecuteWithLockAsync("order:12345", async () =>
{
    var order = await GetOrderAsync(12345);
    order.Status = OrderStatus.Processing;
    await SaveOrderAsync(order);
    return order;
});
```

### Rate Limiting

Implement rate limiting with Redis.

#### Fixed Window Rate Limiter

```csharp
public class FixedWindowRateLimiter
{
    private readonly IDatabase _db;
    
    public FixedWindowRateLimiter(IConnectionMultiplexer redis)
    {
        _db = redis.GetDatabase();
    }
    
    public async Task<bool> IsAllowedAsync(
        string identifier,
        int limit,
        TimeSpan window)
    {
        var key = $"rate:{identifier}:{DateTimeOffset.UtcNow.ToUnixTimeSeconds() / (int)window.TotalSeconds}";
        
        var count = await _db.StringIncrementAsync(key);
        
        if (count == 1)
        {
            await _db.KeyExpireAsync(key, window);
        }
        
        return count <= limit;
    }
}
```

#### Sliding Window Rate Limiter (more accurate)

```csharp
public class SlidingWindowRateLimiter
{
    private readonly IDatabase _db;
    
    public async Task<bool> IsAllowedAsync(
        string identifier,
        int limit,
        TimeSpan window)
    {
        var key = $"rate:{identifier}";
        var now = DateTimeOffset.UtcNow.ToUnixTimeMilliseconds();
        var windowStart = now - (long)window.TotalMilliseconds;
        
        // Remove old entries
        await _db.SortedSetRemoveRangeByScoreAsync(key, 0, windowStart);
        
        // Count current entries
        var count = await _db.SortedSetLengthAsync(key);
        
        if (count < limit)
        {
            // Add new entry
            await _db.SortedSetAddAsync(key, now.ToString(), now);
            await _db.KeyExpireAsync(key, window);
            return true;
        }
        
        return false;
    }
}

// Middleware usage
public class RateLimitingMiddleware
{
    private readonly RequestDelegate _next;
    private readonly SlidingWindowRateLimiter _rateLimiter;
    
    public async Task InvokeAsync(HttpContext context)
    {
        var identifier = context.User.Identity?.Name ?? context.Connection.RemoteIpAddress?.ToString();
        
        if (!await _rateLimiter.IsAllowedAsync(identifier!, 100, TimeSpan.FromMinutes(1)))
        {
            context.Response.StatusCode = 429; // Too Many Requests
            await context.Response.WriteAsync("Rate limit exceeded");
            return;
        }
        
        await _next(context);
    }
}
```

### RedLock (Advanced Distributed Locking)

For highly critical distributed locking across multiple Redis instances.

```csharp
// Install: dotnet add package RedLock.net

using RedLockNet.SERedis;
using RedLockNet.SERedis.Configuration;

var multiplexers = new[]
{
    ConnectionMultiplexer.Connect("redis1:6379"),
    ConnectionMultiplexer.Connect("redis2:6379"),
    ConnectionMultiplexer.Connect("redis3:6379")
};

var redlockFactory = RedLockFactory.Create(multiplexers);

var resource = "order:12345";
var expiry = TimeSpan.FromSeconds(30);

await using (var redLock = await redlockFactory.CreateLockAsync(resource, expiry))
{
    if (redLock.IsAcquired)
    {
        // Critical section - only one process can execute this
        await ProcessOrderAsync(12345);
    }
    else
    {
        throw new InvalidOperationException("Could not acquire lock");
    }
}
```

## Advanced Patterns

### Leaky Bucket Rate Limiter

```csharp
public class LeakyBucketRateLimiter
{
    private readonly IDatabase _db;
    
    public async Task<bool> IsAllowedAsync(
        string identifier,
        int capacity,
        int leakRate)
    {
        const string script = @"
            local key = KEYS[1]
            local capacity = tonumber(ARGV[1])
            local leak_rate = tonumber(ARGV[2])
            local now = tonumber(ARGV[3])
            
            local bucket = redis.call('HMGET', key, 'tokens', 'last_leak')
            local tokens = tonumber(bucket[1]) or capacity
            local last_leak = tonumber(bucket[2]) or now
            
            -- Leak tokens based on time elapsed
            local elapsed = now - last_leak
            local leaked = math.floor(elapsed * leak_rate)
            tokens = math.min(capacity, tokens + leaked)
            
            if tokens >= 1 then
                tokens = tokens - 1
                redis.call('HMSET', key, 'tokens', tokens, 'last_leak', now)
                redis.call('EXPIRE', key, capacity / leak_rate)
                return 1
            else
                return 0
            end
        ";
        
        var key = $"leaky_bucket:{identifier}";
        var now = DateTimeOffset.UtcNow.ToUnixTimeSeconds();
        
        var result = await _db.ScriptEvaluateAsync(
            script,
            new RedisKey[] { key },
            new RedisValue[] { capacity, leakRate, now }
        );
        
        return (int)result == 1;
    }
}
```

## Monitoring and Observability

```csharp
public class RedisMonitoringService
{
    private readonly IConnectionMultiplexer _redis;
    private readonly ILogger<RedisMonitoringService> _logger;
    
    public RedisMonitoringService(
        IConnectionMultiplexer redis,
        ILogger<RedisMonitoringService> logger)
    {
        _redis = redis;
        _logger = logger;
        
        // Subscribe to connection events
        _redis.ConnectionFailed += OnConnectionFailed;
        _redis.ConnectionRestored += OnConnectionRestored;
        _redis.ErrorMessage += OnErrorMessage;
        _redis.InternalError += OnInternalError;
    }
    
    private void OnConnectionFailed(object? sender, ConnectionFailedEventArgs e)
    {
        _logger.LogError(
            "Redis connection failed: {Reason} - {Exception}",
            e.FailureType,
            e.Exception?.Message
        );
    }
    
    private void OnConnectionRestored(object? sender, ConnectionFailedEventArgs e)
    {
        _logger.LogInformation("Redis connection restored");
    }
    
    private void OnErrorMessage(object? sender, RedisErrorEventArgs e)
    {
        _logger.LogError("Redis error: {Message}", e.Message);
    }
    
    private void OnInternalError(object? sender, InternalErrorEventArgs e)
    {
        _logger.LogError(e.Exception, "Redis internal error");
    }
    
    public async Task<Dictionary<string, string>> GetServerInfoAsync()
    {
        var server = _redis.GetServer(_redis.GetEndPoints().First());
        var info = await server.InfoAsync();
        
        return info.ToDictionary(
            group => group.Key,
            group => string.Join("; ", group.Select(kvp => $"{kvp.Key}={kvp.Value}"))
        );
    }
}
```

## Best Practices

1. **Pub/Sub Characteristics:**
   - Fire-and-forget (no message persistence)
   - Use for real-time, non-critical notifications
   - For durable messaging, use Streams

2. **Streams vs Pub/Sub:**
   - Use Streams when messages must not be lost
   - Use Pub/Sub for real-time, ephemeral notifications
   - Streams support consumer groups and message acknowledgment

3. **Distributed Locks:**
   - Always set expiration to prevent deadlocks
   - Use unique lock values for safe release
   - Consider RedLock for critical scenarios

4. **Rate Limiting:**
   - Choose algorithm based on requirements (fixed window vs sliding window)
   - Monitor rate limit hits
   - Provide clear error messages

5. **Production Readiness:**
   - Monitor connection health
   - Handle disconnections gracefully
   - Log errors for debugging
   - Test failure scenarios
