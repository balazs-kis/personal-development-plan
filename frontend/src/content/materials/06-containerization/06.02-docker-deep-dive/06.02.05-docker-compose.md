---

## Introduction

Docker Compose is a tool for defining and running multi-container Docker applications. Instead of running multiple `docker run` commands with complex arguments, you define your entire application stack in a YAML file and manage it with simple commands. It's essential for local development, testing, and small production deployments.

## Key Concepts Summary

### What is Docker Compose?

**Without Compose:**
```bash
# Create network
docker network create myapp

# Run database
docker run -d --name db --network myapp \
  -e POSTGRES_PASSWORD=secret postgres:15

# Run Redis
docker run -d --name cache --network myapp redis:7

# Run API
docker run -d --name api --network myapp \
  -e DB_HOST=db -e REDIS_HOST=cache \
  -p 3000:3000 myapi:latest

# Run frontend
docker run -d --name web --network myapp \
  -e API_URL=http://api:3000 \
  -p 8080:80 myweb:latest
```

**With Compose:**
```yaml
# docker-compose.yml
version: '3.8'

services:
  db:
    image: postgres:15
    environment:
      POSTGRES_PASSWORD: secret
  
  cache:
    image: redis:7
  
  api:
    image: myapi:latest
    environment:
      DB_HOST: db
      REDIS_HOST: cache
    ports:
      - "3000:3000"
    depends_on:
      - db
      - cache
  
  web:
    image: myweb:latest
    environment:
      API_URL: http://api:3000
    ports:
      - "8080:80"
    depends_on:
      - api
```

```bash
# Start everything
docker compose up

# Stop everything
docker compose down
```

### Basic docker-compose.yml Structure

```yaml
version: '3.8'  # Compose file version

services:       # Define containers
  service1:
    # Service configuration
  
  service2:
    # Service configuration

networks:       # Define networks (optional)
  network1:

volumes:        # Define volumes (optional)
  volume1:
```

### Service Definition Options

```yaml
services:
  myapp:
    # Build from Dockerfile
    build: ./app
    
    # Or use existing image
    image: nginx:alpine
    
    # Container name
    container_name: my-container
    
    # Environment variables
    environment:
      NODE_ENV: production
      API_KEY: ${API_KEY}
    
    # Or load from file
    env_file:
      - .env
    
    # Port mapping
    ports:
      - "8080:80"
      - "127.0.0.1:3000:3000"
    
    # Volumes
    volumes:
      - ./src:/app/src
      - node_modules:/app/node_modules
    
    # Networks
    networks:
      - frontend
      - backend
    
    # Startup order
    depends_on:
      - db
      - cache
    
    # Restart policy
    restart: unless-stopped
    
    # Command override
    command: npm run dev
    
    # Health check
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost/health"]
      interval: 30s
      timeout: 3s
      retries: 3
```

### Complete Example: Full-Stack Application

**docker-compose.yml:**
```yaml
version: '3.8'

services:
  # PostgreSQL Database
  db:
    image: postgres:15-alpine
    container_name: myapp-db
    environment:
      POSTGRES_DB: myapp
      POSTGRES_USER: admin
      POSTGRES_PASSWORD: ${DB_PASSWORD}
    volumes:
      - postgres-data:/var/lib/postgresql/data
      - ./init.sql:/docker-entrypoint-initdb.d/init.sql
    networks:
      - backend
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U admin"]
      interval: 10s
      timeout: 5s
      retries: 5

  # Redis Cache
  cache:
    image: redis:7-alpine
    container_name: myapp-cache
    command: redis-server --appendonly yes
    volumes:
      - redis-data:/data
    networks:
      - backend
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 3s
      retries: 3

  # Backend API
  api:
    build:
      context: ./api
      dockerfile: Dockerfile
      args:
        NODE_VERSION: 18
    container_name: myapp-api
    environment:
      NODE_ENV: production
      DATABASE_URL: postgres://admin:${DB_PASSWORD}@db:5432/myapp
      REDIS_URL: redis://cache:6379
      JWT_SECRET: ${JWT_SECRET}
    ports:
      - "3000:3000"
    volumes:
      - ./api/src:/app/src  # Development: live reload
      - /app/node_modules   # Don't overwrite
    networks:
      - frontend
      - backend
    depends_on:
      db:
        condition: service_healthy
      cache:
        condition: service_healthy
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:3000/health"]
      interval: 30s
      timeout: 5s
      retries: 3

  # Frontend
  web:
    build:
      context: ./web
      dockerfile: Dockerfile.prod
    container_name: myapp-web
    environment:
      REACT_APP_API_URL: http://localhost:3000
    ports:
      - "8080:80"
    networks:
      - frontend
    depends_on:
      - api
    restart: unless-stopped

networks:
  frontend:
    driver: bridge
  backend:
    driver: bridge
    internal: true  # No external access

volumes:
  postgres-data:
  redis-data:
```

**.env file:**
```env
DB_PASSWORD=supersecret
JWT_SECRET=your-jwt-secret-key
```

### Docker Compose Commands

```bash
# Start services
docker compose up                    # Foreground
docker compose up -d                 # Background (detached)
docker compose up --build            # Rebuild images
docker compose up --force-recreate   # Recreate containers

# Stop services
docker compose stop                  # Stop (can restart)
docker compose down                  # Stop and remove
docker compose down -v               # Also remove volumes
docker compose down --rmi all        # Also remove images

# View status
docker compose ps                    # List containers
docker compose ps -a                 # Include stopped
docker compose top                   # Show processes

# Logs
docker compose logs                  # All services
docker compose logs api              # Specific service
docker compose logs -f               # Follow
docker compose logs --tail=100 api   # Last 100 lines

# Execute commands
docker compose exec api sh           # Shell in api container
docker compose exec db psql -U admin # Run psql

# Scale services
docker compose up -d --scale api=3   # Run 3 API instances

# Restart services
docker compose restart               # All services
docker compose restart api           # Specific service

# Pull images
docker compose pull                  # Update all images

# Validate compose file
docker compose config                # Check syntax
docker compose config --services     # List services
```

### Build Configuration

```yaml
services:
  app:
    build:
      context: ./app        # Build context
      dockerfile: Dockerfile.prod
      args:                 # Build arguments
        VERSION: 1.0
        BUILD_DATE: ${BUILD_DATE}
      cache_from:           # Use cache from
        - myapp:latest
      target: production    # Multi-stage build target
      labels:
        version: "1.0"
```

### Networking in Compose

**Default behavior:**
- Compose creates a default network
- All services can reach each other by service name
- Services are isolated from other Compose apps

**Custom networks:**
```yaml
services:
  web:
    networks:
      - frontend
  
  api:
    networks:
      - frontend
      - backend
  
  db:
    networks:
      - backend

networks:
  frontend:
  backend:
    internal: true  # No internet access
```

**Network aliases:**
```yaml
services:
  app:
    networks:
      default:
        aliases:
          - myapp.local
          - api.local
```

### Volume Configuration

**Named volumes:**
```yaml
services:
  db:
    volumes:
      - postgres-data:/var/lib/postgresql/data

volumes:
  postgres-data:
    driver: local
    driver_opts:
      type: none
      o: bind
      device: /path/on/host
```

**Bind mounts:**
```yaml
services:
  app:
    volumes:
      - ./src:/app/src              # Relative path
      - /absolute/path:/app/data    # Absolute path
      - ~/config:/app/config        # Home directory
```

**Volume options:**
```yaml
volumes:
  - type: volume
    source: mydata
    target: /data
    volume:
      nocopy: true
  
  - type: bind
    source: ./static
    target: /app/static
    read_only: true
```

### Environment Variables

**Method 1: Inline**
```yaml
services:
  app:
    environment:
      NODE_ENV: production
      API_KEY: abc123
```

**Method 2: From .env file**
```yaml
services:
  app:
    env_file:
      - .env
      - .env.local
```

**Method 3: Variable substitution**
```yaml
services:
  app:
    image: myapp:${TAG:-latest}
    environment:
      DB_PASSWORD: ${DB_PASSWORD}
```

### Dependency Management

**Simple depends_on:**
```yaml
services:
  api:
    depends_on:
      - db
      - cache
```

**Wait for healthy:**
```yaml
services:
  api:
    depends_on:
      db:
        condition: service_healthy
      cache:
        condition: service_started
  
  db:
    healthcheck:
      test: ["CMD-SHELL", "pg_isready"]
      interval: 5s
```

### Development vs Production

**docker-compose.yml** (base):
```yaml
services:
  app:
    build: .
    ports:
      - "3000:3000"
```

**docker-compose.override.yml** (development, auto-loaded):
```yaml
services:
  app:
    volumes:
      - ./src:/app/src
    environment:
      NODE_ENV: development
    command: npm run dev
```

**docker-compose.prod.yml** (production):
```yaml
services:
  app:
    environment:
      NODE_ENV: production
    restart: always
    command: npm start
```

```bash
# Development (uses override automatically)
docker compose up

# Production
docker compose -f docker-compose.yml -f docker-compose.prod.yml up
```

### Real-World Example: .NET + PostgreSQL + Redis

**docker-compose.yml:**
```yaml
version: '3.8'

services:
  api:
    build:
      context: .
      dockerfile: src/Api/Dockerfile
    environment:
      - ASPNETCORE_ENVIRONMENT=Development
      - ConnectionStrings__DefaultConnection=Host=db;Database=myapp;Username=postgres;Password=postgres
      - ConnectionStrings__Redis=cache:6379
    ports:
      - "5000:8080"
    depends_on:
      - db
      - cache
    volumes:
      - ~/.aspnet/https:/https:ro

  db:
    image: postgres:15-alpine
    environment:
      POSTGRES_DB: myapp
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: postgres
    ports:
      - "5432:5432"
    volumes:
      - postgres-data:/var/lib/postgresql/data

  cache:
    image: redis:7-alpine
    ports:
      - "6379:6379"

  adminer:
    image: adminer
    ports:
      - "8080:8080"
    depends_on:
      - db

volumes:
  postgres-data:
```

### Best Practices

**1. Use version control:**
```bash
# Commit compose files
git add docker-compose.yml
git add docker-compose.override.yml

# Ignore environment files
echo ".env" >> .gitignore
echo ".env.local" >> .gitignore
```

**2. Use .env for secrets:**
```yaml
# docker-compose.yml
services:
  db:
    environment:
      POSTGRES_PASSWORD: ${DB_PASSWORD}
```

**3. Pin image versions:**
```yaml
# ✅ Good
image: postgres:15.2-alpine

# ❌ Bad
image: postgres:latest
```

**4. Use health checks:**
```yaml
services:
  db:
    healthcheck:
      test: pg_isready -U postgres
```

**5. Set restart policies:**
```yaml
services:
  app:
    restart: unless-stopped
```

**6. Use networks for isolation:**
```yaml
networks:
  frontend:
  backend:
    internal: true
```
