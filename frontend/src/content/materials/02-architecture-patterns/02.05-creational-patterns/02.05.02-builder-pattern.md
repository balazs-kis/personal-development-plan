# Lesson 2: Builder Pattern for Complex Objects

## Learning Objectives
- Implement Builder pattern in C#
- Use Fluent interfaces for builders
- Handle complex object construction
- Compare Builder vs Constructor patterns

## Introduction

The Builder pattern separates the construction of a complex object from its representation, allowing the same construction process to create different representations. It's especially useful for objects with many optional parameters.

## Key Concepts Summary

### Problem: Constructor Overload Hell

```csharp
// BAD - Too many constructors
public class HttpClient
{
    public HttpClient(string baseUrl) { }
    public HttpClient(string baseUrl, int timeout) { }
    public HttpClient(string baseUrl, int timeout, Dictionary<string, string> headers) { }
    public HttpClient(string baseUrl, int timeout, Dictionary<string, string> headers, bool followRedirects) { }
    public HttpClient(string baseUrl, int timeout, Dictionary<string, string> headers, bool followRedirects, IAuthenticator authenticator) { }
}

// Usage is confusing
var client = new HttpClient("https://api.com", 30, null, true, null);
```

### Solution: Builder Pattern

```csharp
// Product
public class HttpClient
{
    public string BaseUrl { get; internal set; }
    public int Timeout { get; internal set; } = 30;
    public Dictionary<string, string> Headers { get; internal set; } = new();
    public bool FollowRedirects { get; internal set; } = true;
    public IAuthenticator Authenticator { get; internal set; }
    
    // Private constructor - force use of builder
    internal HttpClient() { }
}

// Builder
public class HttpClientBuilder
{
    private readonly HttpClient _client = new();
    
    public HttpClientBuilder WithBaseUrl(string url)
    {
        _client.BaseUrl = url;
        return this;  // Fluent interface
    }
    
    public HttpClientBuilder WithTimeout(int seconds)
    {
        if (seconds <= 0)
            throw new ArgumentException("Timeout must be positive");
        
        _client.Timeout = seconds;
        return this;
    }
    
    public HttpClientBuilder AddHeader(string key, string value)
    {
        _client.Headers[key] = value;
        return this;
    }
    
    public HttpClientBuilder WithFollowRedirects(bool follow)
    {
        _client.FollowRedirects = follow;
        return this;
    }
    
    public HttpClientBuilder WithAuthenticator(IAuthenticator authenticator)
    {
        _client.Authenticator = authenticator;
        return this;
    }
    
    public HttpClient Build()
    {
        // Validation
        if (string.IsNullOrEmpty(_client.BaseUrl))
            throw new InvalidOperationException("BaseUrl is required");
        
        return _client;
    }
}

// Usage - clear and readable
var client = new HttpClientBuilder()
    .WithBaseUrl("https://api.example.com")
    .WithTimeout(60)
    .AddHeader("Authorization", "Bearer token")
    .AddHeader("Accept", "application/json")
    .WithFollowRedirects(true)
    .Build();
```

### Advanced Builder with Director

```csharp
// Complex product
public class Report
{
    public string Title { get; set; }
    public string Author { get; set; }
    public DateTime GeneratedDate { get; set; }
    public List<Section> Sections { get; set; } = new();
    public ReportFormat Format { get; set; }
    public bool IncludeTableOfContents { get; set; }
    public bool IncludePageNumbers { get; set; }
}

// Builder interface
public interface IReportBuilder
{
    IReportBuilder SetTitle(string title);
    IReportBuilder SetAuthor(string author);
    IReportBuilder AddSection(Section section);
    IReportBuilder SetFormat(ReportFormat format);
    IReportBuilder IncludeTableOfContents();
    IReportBuilder IncludePageNumbers();
    Report Build();
}

// Concrete builder
public class PdfReportBuilder : IReportBuilder
{
    private readonly Report _report = new();
    
    public IReportBuilder SetTitle(string title)
    {
        _report.Title = title;
        return this;
    }
    
    public IReportBuilder SetAuthor(string author)
    {
        _report.Author = author;
        return this;
    }
    
    public IReportBuilder AddSection(Section section)
    {
        _report.Sections.Add(section);
        return this;
    }
    
    public IReportBuilder SetFormat(ReportFormat format)
    {
        _report.Format = format;
        return this;
    }
    
    public IReportBuilder IncludeTableOfContents()
    {
        _report.IncludeTableOfContents = true;
        return this;
    }
    
    public IReportBuilder IncludePageNumbers()
    {
        _report.IncludePageNumbers = true;
        return this;
    }
    
    public Report Build()
    {
        _report.GeneratedDate = DateTime.UtcNow;
        return _report;
    }
}

// Director - encapsulates common construction sequences
public class ReportDirector
{
    private readonly IReportBuilder _builder;
    
    public ReportDirector(IReportBuilder builder)
    {
        _builder = builder;
    }
    
    public Report BuildStandardReport(string title, string author, List<Section> sections)
    {
        return _builder
            .SetTitle(title)
            .SetAuthor(author)
            .SetFormat(ReportFormat.Standard)
            .IncludePageNumbers()
            .Build();
    }
    
    public Report BuildDetailedReport(string title, string author, List<Section> sections)
    {
        var builder = _builder
            .SetTitle(title)
            .SetAuthor(author)
            .SetFormat(ReportFormat.Detailed)
            .IncludeTableOfContents()
            .IncludePageNumbers();
        
        foreach (var section in sections)
        {
            builder.AddSection(section);
        }
        
        return builder.Build();
    }
}

// Usage
var builder = new PdfReportBuilder();
var director = new ReportDirector(builder);

var standardReport = director.BuildStandardReport("Q4 Report", "John Doe", sections);
var detailedReport = director.BuildDetailedReport("Annual Report", "Jane Smith", sections);
```

### Step Builder (Type-Safe Builder)

```csharp
// Ensures required fields are set in correct order
public interface IEmailBuilderFrom
{
    IEmailBuilderTo From(string email);
}

public interface IEmailBuilderTo
{
    IEmailBuilderSubject To(string email);
}

public interface IEmailBuilderSubject
{
    IEmailBuilderBody Subject(string subject);
}

public interface IEmailBuilderBody
{
    IEmailBuilderBuild Body(string body);
}

public interface IEmailBuilderBuild
{
    IEmailBuilderBuild WithAttachment(Attachment attachment);
    IEmailBuilderBuild WithCc(string email);
    Email Build();
}

public class EmailBuilder : 
    IEmailBuilderFrom,
    IEmailBuilderTo,
    IEmailBuilderSubject,
    IEmailBuilderBody,
    IEmailBuilderBuild
{
    private readonly Email _email = new();
    
    public static IEmailBuilderFrom CreateEmail() => new EmailBuilder();
    
    public IEmailBuilderTo From(string email)
    {
        _email.From = email;
        return this;
    }
    
    public IEmailBuilderSubject To(string email)
    {
        _email.To = email;
        return this;
    }
    
    public IEmailBuilderBody Subject(string subject)
    {
        _email.Subject = subject;
        return this;
    }
    
    public IEmailBuilderBuild Body(string body)
    {
        _email.Body = body;
        return this;
    }
    
    public IEmailBuilderBuild WithAttachment(Attachment attachment)
    {
        _email.Attachments.Add(attachment);
        return this;
    }
    
    public IEmailBuilderBuild WithCc(string email)
    {
        _email.Cc.Add(email);
        return this;
    }
    
    public Email Build() => _email;
}

// Usage - compiler enforces order!
var email = EmailBuilder.CreateEmail()
    .From("sender@example.com")         // Must be first
    .To("recipient@example.com")        // Must be second
    .Subject("Important Message")       // Must be third
    .Body("This is the content")        // Must be fourth
    .WithAttachment(new Attachment())   // Optional
    .WithCc("cc@example.com")           // Optional
    .Build();
```

### Nested Builders for Complex Hierarchies

```csharp
// Building complex object graphs
public class EmailMessage
{
    public string From { get; set; }
    public List<string> To { get; set; } = new();
    public string Subject { get; set; }
    public EmailBody Body { get; set; }
    public List<Attachment> Attachments { get; set; } = new();
}

public class EmailBody
{
    public string HtmlContent { get; set; }
    public string TextContent { get; set; }
}

public class EmailBodyBuilder
{
    private readonly EmailBody _body = new();
    
    public EmailBodyBuilder WithHtml(string html)
    {
        _body.HtmlContent = html;
        return this;
    }
    
    public EmailBodyBuilder WithText(string text)
    {
        _body.TextContent = text;
        return this;
    }
    
    internal EmailBody Build() => _body;
}

public class EmailBuilder
{
    private readonly EmailMessage _message = new();
    
    public EmailBuilder From(string address)
    {
        _message.From = address;
        return this;
    }
    
    public EmailBuilder To(params string[] addresses)
    {
        _message.To.AddRange(addresses);
        return this;
    }
    
    public EmailBuilder Subject(string subject)
    {
        _message.Subject = subject;
        return this;
    }
    
    public EmailBuilder WithBody(Action<EmailBodyBuilder> configure)
    {
        var bodyBuilder = new EmailBodyBuilder();
        configure(bodyBuilder);
        _message.Body = bodyBuilder.Build();
        return this;
    }
    
    public EmailBuilder AddAttachment(Attachment attachment)
    {
        _message.Attachments.Add(attachment);
        return this;
    }
    
    public EmailMessage Build()
    {
        if (string.IsNullOrEmpty(_message.From))
            throw new InvalidOperationException("From address is required");
        if (!_message.To.Any())
            throw new InvalidOperationException("At least one recipient is required");
        
        return _message;
    }
}

// Usage
var email = new EmailBuilder()
    .From("sender@example.com")
    .To("recipient1@example.com", "recipient2@example.com")
    .Subject("Monthly Newsletter")
    .WithBody(body => body
        .WithHtml("<h1>Welcome</h1><p>Content here...</p>")
        .WithText("Welcome\n\nContent here..."))
    .AddAttachment(new Attachment("report.pdf", fileBytes))
    .Build();
```

### Builder with Validation Rules

```csharp
public class UserAccountBuilder
{
    private readonly User _user = new();
    private readonly List<string> _validationErrors = new();
    
    public UserAccountBuilder WithUsername(string username)
    {
        if (string.IsNullOrWhiteSpace(username))
            _validationErrors.Add("Username cannot be empty");
        else if (username.Length < 3)
            _validationErrors.Add("Username must be at least 3 characters");
        
        _user.Username = username;
        return this;
    }
    
    public UserAccountBuilder WithEmail(string email)
    {
        if (string.IsNullOrWhiteSpace(email))
            _validationErrors.Add("Email cannot be empty");
        else if (!Regex.IsMatch(email, @"^[^@]+@[^@]+\.[^@]+$"))
            _validationErrors.Add("Email format is invalid");
        
        _user.Email = email;
        return this;
    }
    
    public ValidationResult<User> Build()
    {
        if (string.IsNullOrEmpty(_user.Username))
            _validationErrors.Add("Username is required");
        if (string.IsNullOrEmpty(_user.Email))
            _validationErrors.Add("Email is required");
        
        if (_validationErrors.Any())
            return ValidationResult<User>.Failure(_validationErrors);
        
        return ValidationResult<User>.Success(_user);
    }
}

// Usage
var result = new UserAccountBuilder()
    .WithUsername("john_doe")
    .WithEmail("john@example.com")
    .Build();

if (result.IsValid)
{
    var user = result.Value;
}
else
{
    foreach (var error in result.Errors)
        Console.WriteLine($"Error: {error}");
}
```

### Builder Comparison

**Builder vs Constructor:**
```csharp
// Constructor - required parameters
public class User
{
    public User(string username, string email)
    {
        Username = username;
        Email = email;
    }
}

// Builder - many optional parameters
var user = new UserBuilder()
    .WithUsername("john")
    .WithEmail("john@example.com")
    .WithAge(25)
    .Build();
```

**Builder vs Factory:**
```csharp
// Factory - which class to instantiate
IPaymentProcessor processor = factory.Create(PaymentMethod.CreditCard);

// Builder - how to construct complex object
var order = new OrderBuilder()
    .AddItem("Product1", 2, 29.99m)
    .WithShipping(ShippingMethod.Express)
    .Build();
```

### Common Pitfalls

1. **Mutable Product:**
```csharp
// BAD - public setters
public class Product
{
    public string Name { get; set; }
}

// GOOD - internal setters
public class Product
{
    public string Name { get; internal set; }
    internal Product() { }
}
```

2. **Builder Reuse:**
```csharp
// BAD - shared state
var builder = new UserBuilder();
var user1 = builder.WithName("John").Build();
var user2 = builder.WithName("Jane").Build(); // user1 changed!

// GOOD - new instance
var user1 = new UserBuilder().WithName("John").Build();
var user2 = new UserBuilder().WithName("Jane").Build();
```

public interface IConnectionStringStep
{
    IDatabaseTypeStep WithConnectionString(string connectionString);

public interface IDatabaseTypeStep
{
    IOptionalStep WithDatabaseType(DatabaseType type);

public interface IOptionalStep
{

    IOptionalStep EnableLogging();
    DatabaseConnection Build();

public class DatabaseConnectionStepBuilder : 
    IConnectionStringStep,
    IDatabaseTypeStep,
    IOptionalStep
{
    private readonly DatabaseConnection _connection = new();

    private DatabaseConnectionStepBuilder() { }
    
    public IDatabaseTypeStep WithConnectionString(string connectionString)
    {
        _connection.ConnectionString = connectionString;
        return this;
    }
    
    public IOptionalStep WithDatabaseType(DatabaseType type)
    {
        _connection.DatabaseType = type;
        return this;
    }
    
    public IOptionalStep WithTimeout(int seconds)
    {
        _connection.Timeout = seconds;
        return this;
    }
    
    public IOptionalStep WithMaxRetries(int retries)
    {
        _connection.MaxRetries = retries;

## Real-World Scenario

**Complex API Request Builder**

Build an HTTP request builder for a REST API client that supports:
- Multiple HTTP methods
- Query parameters
- Headers
- Body content
- Authentication
- Retry policies
- Timeout configuration

```csharp
public class ApiRequest
{
    public string Url { get; internal set; }
    public HttpMethod Method { get; internal set; }
    public Dictionary<string, string> QueryParameters { get; internal set; } = new();
    public Dictionary<string, string> Headers { get; internal set; } = new();
    public object Body { get; internal set; }
    public TimeSpan Timeout { get; internal set; } = TimeSpan.FromSeconds(30);
    public RetryPolicy RetryPolicy { get; internal set; }
}

public class ApiRequestBuilder
{
    private readonly ApiRequest _request = new();
    
    public ApiRequestBuilder Get(string url)
    {
        _request.Url = url;
        _request.Method = HttpMethod.Get;
        return this;
    }
    
    public ApiRequestBuilder Post(string url)
    {
        _request.Url = url;
        _request.Method = HttpMethod.Post;
        return this;
    }
    
    public ApiRequestBuilder AddQueryParameter(string key, string value)
    {
        _request.QueryParameters[key] = value;
        return this;
    }
    
    public ApiRequestBuilder AddHeader(string key, string value)
    {
        _request.Headers[key] = value;
        return this;
    }
    
    public ApiRequestBuilder WithBearerToken(string token)
    {
        _request.Headers["Authorization"] = $"Bearer {token}";
        return this;
    }
    
    public ApiRequestBuilder WithBody(object body)
    {
        _request.Body = body;
        return this;
    }
    
    public ApiRequestBuilder WithTimeout(TimeSpan timeout)
    {
        _request.Timeout = timeout;
        return this;
    }
    
    public ApiRequestBuilder WithRetryPolicy(int maxRetries, TimeSpan delay)
    {
        _request.RetryPolicy = new RetryPolicy
        {
            MaxRetries = maxRetries,
            DelayBetweenRetries = delay
        };
        return this;
    }
    
    public ApiRequest Build()
    {
        if (string.IsNullOrEmpty(_request.Url))
            throw new InvalidOperationException("URL is required");
        
        return _request;
    }
}

// Usage
var request = new ApiRequestBuilder()
    .Post("https://api.example.com/users")
    .AddQueryParameter("page", "1")
    .AddQueryParameter("limit", "10")
    .WithBearerToken("eyJhbGciOiJIUzI1NiIs...")
    .WithBody(new { name = "John Doe", email = "john@example.com" })
    .WithTimeout(TimeSpan.FromSeconds(60))
    .WithRetryPolicy(maxRetries: 3, delay: TimeSpan.FromSeconds(2))
    .Build();
```

## Self-Check Questions

1. **When should you use the Builder pattern instead of a constructor?**
   - Many optional parameters (more than 3-4)
   - Complex object initialization logic
   - Need to construct object step-by-step
   - Want to enforce parameter order or validation

2. **What's the difference between a Builder and a Factory?**
   - Builder: Focuses on HOW to construct a complex object step-by-step
   - Factory: Focuses on WHICH class to instantiate

3. **What is a fluent interface and why is it useful?**
   - Methods return `this` to allow method chaining
   - Creates readable, self-documenting code
   - Guides developers through construction process

4. **What's the role of the Director in the Builder pattern?**
   - Encapsulates common construction sequences
   - Provides pre-configured builder recipes
   - Simplifies complex building processes for common use cases

5. **When is a Step Builder (type-safe builder) beneficial?**
   - When certain parameters must be set in a specific order
   - To enforce required fields at compile time
   - Complex APIs where order matters

6. **Should builders be reusable?**
   - Generally NO - each build should use a fresh builder
   - Builders maintain state that shouldn't be shared
   - Create new builder instances for each build operation
